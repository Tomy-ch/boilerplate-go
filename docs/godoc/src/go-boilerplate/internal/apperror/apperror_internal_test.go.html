    21  // TestAppErrorsCompleteness は、apperror パッケージで宣言された全 Err* センチネルが、
    22  // worker 分類センチネル（appErrors から意図的に除外）を除いて漏れなく appErrors に列挙されて
    23  // いることを機械検証します。HTTP taxonomy センチネルは var ブロックと appErrors スライスの
    24  // 2 箇所へ二重列挙されており、片方への追加漏れを検出する手段が他にないためです。
    25  func TestAppErrorsCompleteness(t *testing.T) {
    26  	t.Parallel()
    27  
    28  	// worker のメッセージ処理分類センチネル。HTTP taxonomy ではないため appErrors には含めない。
    29  	workerSentinels := []error{ErrRetryable, ErrPermanent, ErrFatal}
    30  
    31  	declared := declaredErrNames(t)
    32  
    33  	// 検出ロジックが命名変更等で空振りすると完全性検証が常に成功してしまう。既知の宣言数（14 app +
    34  	// 3 worker = 17）を下限として直接検証し、正規表現の陳腐化を検出する。
    35  	require.GreaterOrEqual(t, len(declared), 17,
    36  		"Err* 宣言の検出数が想定を下回る（errDeclRe の陳腐化を疑う）: %v", declared)
    37  
    38  	require.Len(t, appErrors, len(declared)-len(workerSentinels),
    39  		"appErrors の件数がソース宣言総数と worker センチネル数の差に一致しない（同期漏れを疑う）")
    40  
    41  	// appErrors に nil や重複が混入していないことを検証する。
    42  	for i, e := range appErrors {
    43  		require.Errorf(t, e, "appErrors[%d] が nil", i)
    44  	}
    45  	for i := range appErrors {
    46  		for j := i + 1; j < len(appErrors); j++ {
    47  			assert.Falsef(t, xerrors.Is(appErrors[i], appErrors[j]),
    48  				"appErrors に重複センチネルがある: index %d と %d", i, j)
    49  		}
    50  	}
    51  
    52  	// worker センチネルは HTTP taxonomy 対象外なので IsAppError は false でなければならない。
    53  	for _, s := range workerSentinels {
    54  		assert.Falsef(t, IsAppError(s), "worker センチネル %v が IsAppError=true（appErrors への誤混入を疑う）", s)
    55  	}
    56  }

    61  func declaredErrNames(t *testing.T) []string {
    62  	t.Helper()
    63  
    64  	_, thisFile, _, ok := runtime.Caller(0)
    65  	require.True(t, ok)
    66  	dir := filepath.Dir(thisFile)
    67  
    68  	paths, err := pkgfs.OS{}.Glob(filepath.Join(dir, "*.go"))
    69  	require.NoError(t, err)
    70  
    71  	var names []string
    72  	for _, path := range paths {
    73  		if strings.HasSuffix(path, "_test.go") {
    74  			continue
    75  		}
    76  		src, readErr := pkgfs.OS{}.ReadFile(path)
    77  		require.NoError(t, readErr)
    78  		for line := range strings.SplitSeq(string(src), "\n") {
    79  			if m := errDeclRe.FindStringSubmatch(line); m != nil {
    80  				names = append(names, "Err"+m[1])
    81  			}
    82  		}
    83  	}
    84  	return names
