    32  	t.Parallel()

    34  	issuedAt := time.Date(2026, time.September, 7, 0, 0, 0, 0, time.UTC)
    35  	expiresAt := time.Date(2026, time.December, 31, 0, 0, 0, 0, time.UTC)
    36  
    37  	// Idempotency-Key はこの操作では必須なので、admin のヘッダには必ず載せる。
    38  	// key はケースごとに変えて、リプレイが混ざらないようにする。
    39  	availableAdmin := func(t *testing.T, e *echo.Echo, key string) http.Header {
    40  		t.Helper()
    41  
    42  		headers := MakeAvailableUserID(t, e, uuidtestkit.NewTestFromSalt(t, "integration_bulk_issue_admin"))
    43  		headers.Set("Idempotency-Key", key)
    44  
    45  		return headers
    46  	}
    47  
    48  	// キーが必須になったことで冪等機構が必ず動く。claims / completes に期待回数を渡し、
    49  	// 「実際に claim → complete まで走ったか」を検証する。AnyTimes にすると、middleware の配線が
    50  	// 壊れて idempotency.Run が素通りしても 200 が返るため、退行を検出できない。
    51  	newIdempotencyDeps := func(t *testing.T, claims, completes int) idempotency.Deps {
    52  		t.Helper()
    53  
    54  		ctrl := gomock.NewController(t)
    55  		store := mock_idempotency.NewMockStore(ctrl)
    56  		store.EXPECT().Claim(gomock.Any(), gomock.Any()).Return(true, nil).Times(claims)
    57  		store.EXPECT().Complete(gomock.Any(), gomock.Any()).Return(nil).Times(completes)
    58  		txm := mock_tx.NewMockManager(ctrl)
    59  		txm.EXPECT().Do(gomock.Any(), gomock.Any()).DoAndReturn(
    60  			func(ctx context.Context, fn func(context.Context) error) error { return fn(ctx) },
    61  		).Times(claims)
    62  
    63  		return idempotency.Deps{
    64  			Txm:   txm,
    65  			Store: store,
    66  			Clock: clocktest.NewMockClock(t, issuedAt),
    67  		}
    68  	}

    70  	newBody := func() *couponsbulkissuegen.PostCouponsBulkIssueJSONRequestBody {

    73  			Scope:     couponsbulkissuegen.CouponScopeInput{Kind: couponsbulkissuegen.All},
    74  			ExpiresAt: expiresAt,
    75  		}
    76  	}

    82  			t.Parallel()
    83  
    84  			e := echo.New()
    85  			tf := observability.NewNoopTracerFactory(t)
    86  
    87  			var captured couponuc.IssuePromotionalCouponsParams
    88  			mockUC := mock_coupon.NewMockUsecase(gomock.NewController(t))
    89  			mockUC.EXPECT().IssuePromotionalCoupons(gomock.Any(), gomock.Any(), gomock.Any()).DoAndReturn(
    90  				func(
    91  					_ context.Context, authn *auth.Authn, params couponuc.IssuePromotionalCouponsParams,
    92  				) (couponuc.IssuePromotionalCouponsView, error) {
    93  					require.NotNil(t, authn)
    94  					captured = params
    95  
    96  					return couponuc.IssuePromotionalCouponsView{
    97  						IssuedAt:          issuedAt,
    98  						ExpiresAt:         expiresAt,
    99  						RecipientCount:    1204,
   100  						IssuedCouponCount: 1204,
   101  					}, nil
   102  				},
   103  			)
   104  
   105  			couponsbulkissue.BindHandler(e, tf, mockUC, newIdempotencyDeps(t, 1, 1))
   106  
   107  			actual := StartServer(t, e).DoJSON(http.MethodPost, couponsBulkIssuePath, newBody(), availableAdmin(t, e, "ok-basic"))
   108  			assert.Equal(t, http.StatusOK, actual.StatusCode)
   109  			var body couponsbulkissuegen.CouponBulkIssueResponse
   110  			require.NoError(t, json.NewDecoder(actual.Body).Decode(&body))
   111  
   112  			assert.Equal(t, issuedAt, body.IssuedAt)
   113  			assert.Equal(t, expiresAt, body.ExpiresAt)
   114  			assert.Equal(t, int64(1204), body.RecipientCount)
   115  			assert.Equal(t, int64(1204), body.IssuedCouponCount)
   116  			assert.Equal(t, "rate", captured.DiscountKind)
   117  			assert.Equal(t, "0.15", captured.DiscountValue.String())
   118  			assert.Equal(t, "all", captured.ScopeKind)
   119  			assert.Nil(t, captured.ScopeTargetID)
   120  			assert.Equal(t, expiresAt, captured.ExpiresAt)
   121  		})
