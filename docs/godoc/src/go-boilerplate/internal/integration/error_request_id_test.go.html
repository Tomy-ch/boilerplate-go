    21  // newRequestIDStackServer は、requestId を読み戻す 2 つ目の消費者（logging の res.After
    22  // フック）と、requestId を空にしていた secure_cookie を [UseAppErrorHandler] の配線へ加えた
    23  // Echo を起動する。
    24  // logger は logging ミドルウェアへ渡すもので、アクセスログの内容を検証する場合に観測可能な
    25  // Logger を指定する。
    26  func newRequestIDStackServer(t *testing.T, logger logging.Logger) *Server {
    27  	t.Helper()
    28  
    29  	secCookie := cookie.NewSecurityCookie(config.NewSecureCookieConfig(config.MockConfigForTest(t)))
    30  
    31  	e := echo.New()
    32  	UseAppErrorHandler(t, e,
    33  		instrumentation.LoggingMiddleware(logger, logging.NewTestLogFieldBuilder(t), redaction.Redactor{}).Middleware,
    34  		security.CookieMiddleware(secCookie).Middleware,
    35  	)
    36  
    37  	health.BindHandler(e, observability.NewNoopTracerFactory(t))
    38  
    39  	return StartServer(t, e)
    40  }
