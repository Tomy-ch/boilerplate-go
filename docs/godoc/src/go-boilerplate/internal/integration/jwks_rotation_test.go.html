   108  // loadProviderKey は provider の PEM 秘密鍵（PKCS#8）を embed から読み、golden の公開鍵に対応する署名鍵を返す。
   109  func loadProviderKey(t *testing.T, kid string) *rsa.PrivateKey {
   110  	t.Helper()
   111  	data, err := rotFS.ReadFile("testdata/keys/" + kid + ".pem")
   112  	require.NoError(t, err)
   113  	block, _ := pem.Decode(data)

   122  // mintRotToken は clk の現在時刻を基準に、指定 kid・provider 鍵で署名した有効な access token を発行する。
   123  func mintRotToken(t *testing.T, key *rsa.PrivateKey, kid string, clk *rotClock) string {
   124  	t.Helper()
   125  	now := clk.Now()
   126  	claims := jwtlib.MapClaims{
   127  		"iss": jwtTestIssuer,
   128  		"aud": jwtTestAudience,
   129  		"sub": jwtTestSubject,
   130  		"iat": jwtlib.NewNumericDate(now),
   131  		"nbf": jwtlib.NewNumericDate(now),
   132  		"exp": jwtlib.NewNumericDate(now.Add(time.Hour)),
   133  	}
   134  	token := jwtlib.NewWithClaims(jwtlib.SigningMethodRS256, claims)
   135  	token.Header["kid"] = kid
   136  	token.Header["typ"] = jwtAccessTokenType
   137  	signed, err := token.SignedString(key)
   138  	require.NoError(t, err)
   139  	return signed
   140  }

   158  func TestJWKSRotationIntegration(t *testing.T) {
   159  	t.Parallel()
   160  
   161  	keyA := loadProviderKey(t, rotKIDPrimary)
   162  	keyB := loadProviderKey(t, rotKIDRotation)
   163  	phase1, phase2, phase3 := loadGoldenJWKS(t, 1), loadGoldenJWKS(t, 2), loadGoldenJWKS(t, 3)
   164  
   165  	// get は Bearer トークンで GET /protected を叩き、HTTP レスポンスを返す。
   166  	get := func(srv *Server, token string) *http.Response {
   167  		return srv.DoJSON(http.MethodGet, "/protected", nil, bearerHeader(token))
   168  	}
   169  
   170  	t.Run("正常系", func(t *testing.T) {
   171  		t.Parallel()
   172  
   173  		t.Run("Phase 遷移で新旧 Token を検証し、退役鍵を拒否する", func(t *testing.T) {
   174  			t.Parallel()
   175  			clk := &rotClock{now: time.Now()}
   176  			src := &rotatingJWKS{body: phase1}
   177  			srv := startRotationServer(t, src, clk)
   178  
   179  			// Phase1: key-a 署名 Token を受理。初回の 1 回だけ JWKS を取得する。
   180  			require.Equal(t, http.StatusOK, get(srv, mintRotToken(t, keyA, rotKIDPrimary, clk)).StatusCode)
   181  			require.Equal(t, 1, src.count())
   182  
   183  			require.Equal(t, http.StatusOK, get(srv, mintRotToken(t, keyA, rotKIDPrimary, clk)).StatusCode)
   184  			require.Equal(t, 1, src.count(), "既知 kid の通常リクエストで再取得しない")
   185  
   186  			clk.advance(rotCooldown + time.Second)
   187  			src.setPhase(phase2)
   188  			require.Equal(t, http.StatusOK, get(srv, mintRotToken(t, keyB, rotKIDRotation, clk)).StatusCode)
   189  			require.Equal(t, 2, src.count(), "未知 kid の初回のみ再取得する")
   190  
   191  			require.Equal(t, http.StatusOK, get(srv, mintRotToken(t, keyA, rotKIDPrimary, clk)).StatusCode)
   192  			require.Equal(t, http.StatusOK, get(srv, mintRotToken(t, keyB, rotKIDRotation, clk)).StatusCode)
   193  			require.Equal(t, 2, src.count(), "移行期の新旧受理で再取得しない")
   194  
   195  			clk.advance(time.Hour + time.Minute)
   196  			src.setPhase(phase3)
   197  			AssertErrorResponse(t, get(srv, mintRotToken(t, keyA, rotKIDPrimary, clk)), http.StatusUnauthorized)
   198  			require.Equal(t, 3, src.count(), "cacheTTL 経過後に再取得して退役を反映する")
   199  			require.Equal(t, http.StatusOK, get(srv, mintRotToken(t, keyB, rotKIDRotation, clk)).StatusCode)
   200  		})
   201  
   202  		t.Run("同時の未知 kid 検証は JWKS 更新を 1 回に抑制する（受入条件 5）", func(t *testing.T) {
   203  			t.Parallel()
   204  			clk := &rotClock{now: time.Now()}
   205  			src := &rotatingJWKS{body: phase1}
   206  			srv := startRotationServer(t, src, clk)
   207  
   208  			var wg sync.WaitGroup
   209  			for range 8 {
   210  				wg.Go(func() {
   211  					_ = get(srv, mintRotToken(t, keyB, rotKIDRotation, clk)).Body.Close()
   212  				})
   213  			}
   214  			wg.Wait()
   215  
   216  			assert.Equal(t, 1, src.count(), "同時 JWKS 更新は singleflight で 1 回に抑制される")
   217  		})
   218  	})

   236  func (s *blockingJWKS) Do(ctx context.Context, _ *httpclient.Request) (*httpclient.Response, error) {
   237  	s.mu.Lock()
   238  	s.calls++
   239  	first := s.calls == 1
   240  	s.mu.Unlock()
   241  
   242  	if first {
   243  		close(s.entered)
   244  	}
   245  	select {
   246  	case <-ctx.Done():
   247  		// 実 substrate が返す分類に揃える（httpclient.normalizeTransportError と同じ写像）。
   248  		return nil, xerrors.Wrap(apperror.ErrCanceled, "canceled")
   249  	case <-s.release:
   250  		return &httpclient.Response{StatusCode: 200, Body: s.body}, nil
   251  	}
   252  }

   260  func TestJWKSRefreshDetachIntegration(t *testing.T) {
   261  	t.Parallel()
   262  
   263  	t.Run("正常系", func(t *testing.T) {
   264  		t.Parallel()
   265  
   266  		t.Run("クライアントが切断しても JWKS 取得は完走し cooldown が次の取得を抑える", func(t *testing.T) {
   267  			t.Parallel()
   268  			keyA := loadProviderKey(t, rotKIDPrimary)
   269  			clk := &rotClock{now: time.Now()}
   270  			src := &blockingJWKS{body: loadGoldenJWKS(t, 1), entered: make(chan struct{}), release: make(chan struct{})}
   271  			srv := startRotationServer(t, src, clk)
   272  
   273  			// 実クライアントが、JWKS 取得の最中に接続を切る。
   274  			ctx, cancel := context.WithCancel(t.Context())
   275  			req, err := http.NewRequestWithContext(ctx, http.MethodGet, srv.baseURL+"/protected", nil)
   276  			require.NoError(t, err)
   277  			req.Header.Set("Authorization", "Bearer "+mintRotToken(t, keyA, rotKIDPrimary, clk))
   278  
   279  			aborted := make(chan struct{})
   280  			go func() {
   281  				defer close(aborted)
   282  				res, doErr := srv.client.Do(req)
   283  				assert.Error(t, doErr, "切断したリクエストは応答を受け取らない")
   284  				assert.Nil(t, res)
   285  			}()
   286  
   287  			<-src.entered
   288  			cancel()
   289  			<-aborted
   290  
   291  			// 切断がサーバへ伝わる猶予。取得が呼び出し元の ctx を継いでいれば、この間に打ち切られる。
   292  			time.Sleep(200 * time.Millisecond)
   293  			close(src.release)
   294  
   295  			// 切断された取得が完走していれば鍵集合はキャッシュ済みで、後続は追加取得なしで検証できる。
   296  			// このリクエストは取得権の獲得で先行取得の完了を待つため、判定は待ち合わせ済みの状態で行われる。
   297  			res := srv.DoJSON(http.MethodGet, "/protected", nil, bearerHeader(mintRotToken(t, keyA, rotKIDPrimary, clk)))
   298  			require.Equal(t, http.StatusOK, res.StatusCode, "切断された取得の結果が次のリクエストで使える")
   299  			assert.Equal(t, 1, src.count(), "cooldown が起動しており再取得しない")
   300  		})
   301  	})
