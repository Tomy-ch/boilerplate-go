   153  // assertDeclaredArraysNotNull は、typ 内でスライスとして宣言されたフィールドが raw の JSON 上で
   154  // null にシリアライズされていないことを再帰的に検証する。
   155  //
   156  // 違反となるのはキーが存在して値が null の場合のみで、キー自体が無い（omitempty で absent）場合は
   157  // 違反としない。[]byte は base64 文字列にシリアライズされるため対象外。
   158  func assertDeclaredArraysNotNull(t *testing.T, raw json.RawMessage, typ reflect.Type, path string) {
   159  	t.Helper()
   160  
   161  	if isJSONNull(raw) {
   162  		return
   163  	}
   164  	for typ.Kind() == reflect.Pointer {
   165  		typ = typ.Elem()
   166  	}
   167  
   168  	switch typ.Kind() {
   169  	case reflect.Struct:
   170  		assertStructArraysNotNull(t, raw, typ, path)
   171  	case reflect.Slice:
   172  		assertSliceArraysNotNull(t, raw, typ, path)
   173  	default:
   174  		// スカラー等、配列を含み得ない型は検査対象外。
   175  	}
   176  }

   178  // assertStructArraysNotNull は、struct 型 typ の各フィールドを JSON と突き合わせて検査する。
   179  func assertStructArraysNotNull(t *testing.T, raw json.RawMessage, typ reflect.Type, path string) {
   180  	t.Helper()
   181  
   182  	var fields map[string]json.RawMessage
   183  	if err := json.Unmarshal(raw, &fields); err != nil {
   184  		// time.Time など JSON オブジェクト以外へシリアライズされる struct は対象外。
   185  		return
   186  	}
   187  	// VisibleFields はタグ無し埋め込みフィールドを昇格済みの形で列挙するため、
   188  	// 埋め込みのインライン化を個別に扱う必要がない。
   189  	for _, f := range reflect.VisibleFields(typ) {
   190  		if f.Anonymous || !f.IsExported() {
   191  			continue
   192  		}
   193  		name, ok := jsonFieldName(f)
   194  		if !ok {
   195  			continue
   196  		}
   197  		fieldRaw, present := fields[name]
   198  		if !present {
   199  			continue
   200  		}
   201  		fieldPath := name
   202  		if path != "" {
   203  			fieldPath = path + "." + name
   204  		}
   205  		if isDeclaredArray(f.Type) && isJSONNull(fieldRaw) {
   206  			assert.Failf(t, "配列フィールドが null",
   207  				"配列フィールド %q が JSON で null にシリアライズされています（[] を返す契約違反）", fieldPath)
   208  			continue
   209  		}
   210  		assertDeclaredArraysNotNull(t, fieldRaw, f.Type, fieldPath)
   211  	}
   212  }

   239  // jsonFieldName は、struct フィールドの JSON キー名を解決する。`json:"-"` のときは ok=false。
   240  func jsonFieldName(f reflect.StructField) (string, bool) {
   241  	tag, tagged := f.Tag.Lookup("json")
   242  	if !tagged {
   243  		return f.Name, true
   244  	}
   245  	if tag == "-" {
   246  		return "", false
   247  	}
   248  	name, _, _ := strings.Cut(tag, ",")
   249  	if name == "" {
   250  		return f.Name, true
   251  	}
   252  	return name, true
   253  }
