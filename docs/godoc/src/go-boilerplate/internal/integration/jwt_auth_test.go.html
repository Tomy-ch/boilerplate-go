    93  // signAccessToken は kid を付け、指定の署名鍵・typ・クレームでトークンを署名する。
    94  func signAccessToken(t *testing.T, key *rsa.PrivateKey, method jwtlib.SigningMethod, typ string, claims jwtlib.MapClaims) string {
    95  	t.Helper()
    96  	token := jwtlib.NewWithClaims(method, claims)
    97  	token.Header["kid"] = jwtTestKID
    98  	if typ != "" {
    99  		token.Header["typ"] = typ
   100  	}
   101  
   102  	if method == jwtlib.SigningMethodNone {
   103  		signed, err := token.SignedString(jwtlib.UnsafeAllowNoneSignatureType)
   104  		require.NoError(t, err)
   105  		return signed
   106  	}
   107  
   108  	signed, err := token.SignedString(key)
   109  	require.NoError(t, err)
   110  	return signed
   111  }

   121  // JWKS は httpclient substrate のモックから供給し、Bearer 抽出 → Credential → Authenticate → JWKS 解決 → 検証 → 401/200 の HTTP 境界経路を実検証する。
   122  func startJWTAuthServer(t *testing.T, key *rsa.PrivateKey) *Server {
   123  	t.Helper()
   124  
   125  	ctrl := gomock.NewController(t)
   126  	client := mock_httpclient.NewMockClient(ctrl)
   127  	client.EXPECT().Do(gomock.Any(), gomock.Any()).
   128  		Return(&httpclient.Response{StatusCode: 200, Body: jwksJSON(t, &key.PublicKey, jwtTestKID)}, nil).AnyTimes()
   129  
   130  	authenticator, err := authjwt.NewJWKS(authjwt.JWKSParams{
   131  		Issuer:           jwtTestIssuer,
   132  		Audience:         jwtTestAudience,
   133  		ExpectedType:     jwtAccessTokenType,
   134  		Clock:            system.NewClock(),
   135  		JWKSURL:          "http://jwks.example.test/keys.json",
   136  		AllowInsecureURL: true,
   137  	}, client)
   138  	require.NoError(t, err)
   139  
   140  	return newProtectedServer(t, authenticator)
   141  }

   143  // newProtectedServer は、与えた authenticator を auth ミドルウェアへ配線した保護付き Echo（GET /protected）を起動する。
   144  // Bearer 抽出 → Credential → Authenticate → IdentityResolver → 401/200 の HTTP 境界経路を実検証するための土台。
   145  func newProtectedServer(t *testing.T, authenticator authbd.Authenticator) *Server {
   146  	t.Helper()
   147  
   148  	resolvedUserID, err := uuid.Parse(jwtTestResolvedUserID)
   149  	require.NoError(t, err)
   150  	authFunc, err := auth.NewAuthenticator(authenticator, stubIdentityResolver{userID: resolvedUserID}, nil)
   151  	require.NoError(t, err)
   152  
   153  	e := echo.New()
   154  	UseAppErrorHandler(t, e)
   155  	e.Use(func(next echo.HandlerFunc) echo.HandlerFunc {
   156  		return func(c *echo.Context) error {
   157  			req := c.Request()
   158  			req = req.WithContext(ctxhelper.WithAuthn(req.Context()))
   159  			c.SetRequest(req)
   160  
   161  			input := &openapi3filter.AuthenticationInput{
   162  				RequestValidationInput: &openapi3filter.RequestValidationInput{Request: req},
   163  			}
   164  			if err := authFunc(req.Context(), input); err != nil {
   165  				return err
   166  			}
   167  			return next(c)
   168  		}
   169  	})
   170  	// 解決済み UserID がハンドラまで伝播することを検証するため、Authn.UserID をレスポンスに載せる。
   171  	e.GET("/protected", func(c *echo.Context) error {
   172  		authn, ok := ctxhelper.GetAuthn(c.Request().Context())
   173  		if !ok {
   174  			return c.JSON(http.StatusInternalServerError, map[string]string{"error": "authn missing"})
   175  		}
   176  		userID, err := authn.UserID()
   177  		if err != nil {
   178  			return c.JSON(http.StatusInternalServerError, map[string]string{"error": "user id unresolved"})
   179  		}
   180  		return c.JSON(http.StatusOK, map[string]string{"user_id": userID.String()})
   181  	})
   182  
   183  	return StartServer(t, e)
   184  }
