    58  // connectSSEWithHeaders は、追加のヘッダ（再接続の Last-Event-ID など）を付けて 1 本繋ぎます。
    59  func connectSSEWithHeaders(
    60  	t *testing.T, srv *Server, path string, headers http.Header,
    61  ) (*sseClient, *http.Response) {
    62  	t.Helper()
    63  
    64  	req, err := http.NewRequestWithContext(context.Background(), http.MethodGet, srv.baseURL+path, nil)
    65  	require.NoError(t, err)
    66  
    67  	for k, vs := range headers {
    68  		for _, v := range vs {
    69  			req.Header.Add(k, v)
    70  		}
    71  	}
    72  
    73  	res, err := (&http.Client{}).Do(req)
    74  	require.NoError(t, err)
    75  
    76  	c := &sseClient{t: t, res: res, r: bufio.NewReader(res.Body)}
    77  	t.Cleanup(c.close)
    78  
    79  	return c, res
    80  }

    87  // next は、次のフレームを 1 つ受け取り、契約どおりに状態を進めます。届かなければテストを落とします。
    88  func (c *sseClient) next() sseFrame {
    89  	c.t.Helper()
    90  
    91  	type result struct {
    92  		frame sseFrame
    93  		err   error
    94  	}
    95  	ch := make(chan result, 1)
    96  	go func() {
    97  		f, err := c.readFrame()
    98  		ch <- result{frame: f, err: err}
    99  	}()
   100  
   101  	select {
   102  	case got := <-ch:
   103  		require.NoError(c.t, got.err)
   104  		c.apply(got.frame)
   105  
   106  		return got.frame
   107  	case <-time.After(sseReadTimeout):
   108  		c.t.Fatal("SSE のフレームが届かなかった")
   109  
   110  		return sseFrame{}
   111  	}
   112  }

   114  // apply は、受け取ったフレームを契約に沿って処理します。
   115  func (c *sseClient) apply(f sseFrame) {
   116  	c.t.Helper()
   117  
   118  	if f.id != "" {
   119  		c.lastEventID = f.id
   120  	}
   121  
   122  	if f.event != "control" {
   123  		return
   124  	}
   125  
   126  	var ev gen.ControlEvent
   127  	require.NoError(c.t, json.Unmarshal([]byte(f.data), &ev))
   128  	c.lastControl = &ev
   129  
   130  	if mustCloseOnControl(ev.Action) {
   131  		c.closedBySelf = true
   132  		c.close()
   133  	}
   134  }

   136  // readFrame は、空行までを 1 フレームとして読みます。
   137  func (c *sseClient) readFrame() (sseFrame, error) {
   138  	var f sseFrame
   139  
   140  	for {
   141  		line, err := c.r.ReadString('\n')
   142  		if err != nil {
   143  			return f, err
   144  		}
   145  
   146  		line = strings.TrimSuffix(line, "\n")
   147  		switch {
   148  		case line == "":
   149  			return f, nil
   150  		case strings.HasPrefix(line, ":"):
   151  			f.comment = true
   152  		case strings.HasPrefix(line, "id: "):
   153  			f.id = strings.TrimPrefix(line, "id: ")
   154  		case strings.HasPrefix(line, "event: "):
   155  			f.event = strings.TrimPrefix(line, "event: ")
   156  		case strings.HasPrefix(line, "data: "):
   157  			f.data = strings.TrimPrefix(line, "data: ")
   158  		}
   159  	}
   160  }
