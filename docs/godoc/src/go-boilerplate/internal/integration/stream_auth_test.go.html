    53  func (stubTicketVerifier) Verify(_ context.Context, value string, destination rt.StreamID) (rt.StreamGrant, error) {
    54  	if value != streamTestTicket || destination != streamTestDestination {
    55  		return rt.StreamGrant{}, ucrealtime.ErrTicketInvalid
    56  	}
    57  	return rt.StreamGrant{
    58  		Subject: "subject-1", Destination: streamTestDestination, Scope: "read", InitialCursor: streamTestInitialCursor,
    59  	}, nil
    60  }

    62  func (v stubCursorValidator) Validate(_ context.Context, _ rt.StreamID, cursor rt.Sequence) error {
    63  	if v.unavailable {
    64  		return xerrors.Wrap(apperror.ErrUnavailable, "event log unreachable")
    65  	}
    66  	if cursor < streamTestFloor {
    67  		return xerrors.Wrap(ucrealtime.ErrCursorExpired, "below the replay floor")
    68  	}
    69  	return nil
    70  }

    78  // newStreamServer は、本物の OpenAPI validator（実 spec）・StreamTicket scheme・stream handler・アクセスログ・
    79  // エラーハンドラを本番と同じ順で配線した Echo を起動する。ticket の検証と replay floor だけを stub に置き換える。
    80  func newStreamServer(t *testing.T, logger logging.Logger, cursors ucrealtime.CursorValidator) *Server {
    81  	t.Helper()
    82  
    83  	spec, err := validator.GetValidator()
    84  	require.NoError(t, err)
    85  
    86  	bearer := mock_auth.NewMockAuthenticator(gomock.NewController(t))
    87  	authFunc, err := auth.NewAuthenticator(bearer, stubIdentityResolver{}, []auth.SchemeAuthenticator{streamauth.New(stubTicketVerifier{})})
    88  	require.NoError(t, err)
    89  
    90  	e := echo.New()
    91  	UseAppErrorHandlerWithLogger(t, e, logger,
    92  		instrumentation.LoggingMiddleware(logger, logging.NewTestLogFieldBuilder(t), redaction.FromSpec(spec)).Middleware,
    93  		inbound.OpenAPIMiddleware(spec, skipper.New(), authFunc).Middleware,
    94  	)
    95  	stream.BindHandler(e, observability.NewNoopTracerFactory(t), cursors, stubStreamer{})
    96  
    97  	return StartServer(t, e)

   107  func TestStreamAuth_Integration(t *testing.T) {
