    66  // Sleep は、待ち時間の帯で tick の種類を見分け、catch-up だけテストの合図を待ちます。
    67  func (s *sseSleeper) Sleep(ctx context.Context, d time.Duration) error {
    68  	if d <= sseHeartbeatCeiling || d > sseCatchUpCeiling {
    69  		<-ctx.Done()
    70  
    71  		return ctx.Err()
    72  	}
    73  
    74  	select {
    75  	case <-s.catchUp:
    76  		return nil
    77  	case <-ctx.Done():
    78  		return ctx.Err()
    79  	}
    80  }

   128  // newSSEServer は、stream_auth_test.go と同じ配線に**本物の connection registry** を Streamer として差した
   129  // サーバーを起動します。差し替えるのは ticket の検証（stub）と EventLog（in-memory fake）だけで、
   130  // OpenAPI validator・security scheme・cursor の失効判定・replay は本物が走ります。
   131  func newSSEServer(t *testing.T, set stream.Settings) *sseFixture {
   132  	t.Helper()
   133  
   134  	spec, err := validator.GetValidator()
   135  	require.NoError(t, err)
   136  
   137  	bearer := mock_auth.NewMockAuthenticator(gomock.NewController(t))
   138  	authFunc, err := auth.NewAuthenticator(
   139  		bearer, stubIdentityResolver{}, []auth.SchemeAuthenticator{streamauth.New(stubTicketVerifier{})},
   140  	)
   141  	require.NoError(t, err)
   142  
   143  	cursorLog, replayLog := rttestkit.NewEventLog(), rttestkit.NewEventLog()
   144  	sleeper := newSSESleeper()
   145  	tf := observability.NewNoopTracerFactory(t)
   146  	logger := logging.NewTestLogger(t)
   147  
   148  	cursors := ucrealtime.NewCursorValidator(cursorLog, clocktestkit.NewMockClock(t, sseNow), tf)
   149  	registry := stream.NewRegistry(ucrealtime.NewReplayer(replayLog, tf), sleeper, logger, tf,
   150  		observability.NewNoopRealtimeMetrics(t), ucrealtime.NewHealth(replayLog), set)
   151  
   152  	e := echo.New()
   153  	UseAppErrorHandlerWithLogger(t, e, logger,
   154  		instrumentation.LoggingMiddleware(logger, logging.NewTestLogFieldBuilder(t), redaction.FromSpec(spec)).Middleware,
   155  		inbound.OpenAPIMiddleware(spec, skipper.New(), authFunc).Middleware,
   156  	)
   157  	stream.BindHandler(e, tf, cursors, registry)
   158  
   159  	return &sseFixture{
   160  		srv: StartServer(t, e), cursorLog: cursorLog, replayLog: replayLog, sleeper: sleeper, registry: registry,
   161  	}
