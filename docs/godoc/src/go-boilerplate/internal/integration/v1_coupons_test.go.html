    34  	t.Parallel()

    36  	issuedAt := time.Date(2026, time.September, 7, 0, 0, 0, 0, time.UTC)

    38  	recipientID := uuidtestkit.NewTestFromSalt(t, "integration_issue_recipient")

    40  	// Idempotency-Key はこの操作では必須なので、admin のヘッダには必ず載せる。
    41  	// key はケースごとに変えて、リプレイが混ざらないようにする。
    42  	availableAdmin := func(t *testing.T, e *echo.Echo, key string) http.Header {
    43  		t.Helper()
    44  
    45  		headers := MakeAvailableUserID(t, e, uuidtestkit.NewTestFromSalt(t, "integration_issue_admin"))
    46  		headers.Set("Idempotency-Key", key)
    47  
    48  		return headers
    49  	}
    50  
    51  	// claims / completes に期待回数を渡し、実際に claim → complete まで走ったかを検証する。
    52  	// AnyTimes にすると middleware の配線が壊れて idempotency.Run が素通りしても 201 が
    53  	// 返るため、退行を検出できない。
    54  	newIdempotencyDeps := func(t *testing.T, claims, completes int) idempotency.Deps {
    55  		t.Helper()
    56  
    57  		ctrl := gomock.NewController(t)
    58  		store := mock_idempotency.NewMockStore(ctrl)
    59  		store.EXPECT().Claim(gomock.Any(), gomock.Any()).Return(true, nil).Times(claims)
    60  		store.EXPECT().Complete(gomock.Any(), gomock.Any()).Return(nil).Times(completes)
    61  		txm := mock_tx.NewMockManager(ctrl)
    62  		txm.EXPECT().Do(gomock.Any(), gomock.Any()).DoAndReturn(
    63  			func(ctx context.Context, fn func(context.Context) error) error { return fn(ctx) },
    64  		).Times(claims)
    65  
    66  		return idempotency.Deps{
    67  			Txm:   txm,
    68  			Store: store,
    69  			Clock: clocktest.NewMockClock(t, issuedAt),
    70  		}
    71  	}

   102  			t.Parallel()

   125  			require.NoError(t, json.NewDecoder(actual.Body).Decode(&body))
   126

   128  			assert.Equal(t, "500", body.Discount.Value)
   129  			assert.Equal(t, couponsgen.CouponScopeKind("all"), body.Scope.Kind)
   130  			assert.Nil(t, body.Scope.TargetId)
   131  			assert.Nil(t, body.UsedAt)
   132  			assert.Equal(t, expiresAt, body.ExpiresAt)
   133  			assert.Equal(t, issuedAt, body.IssuedAt)
   134  			assert.Equal(t, recipientID, captured.UserID)
