    55  func newRevocationServer(t *testing.T) *revocationFixture {
    56  	t.Helper()
    57  
    58  	spec, err := validator.GetValidator()
    59  	require.NoError(t, err)
    60  
    61  	tickets := rttestkit.NewStreamTicketStore()
    62  	clk := clocktestkit.NewMockClock(t, sseNow)
    63  	tf := observability.NewNoopTracerFactory(t)
    64  	logger := logging.NewTestLogger(t)
    65  
    66  	verifier := ucrealtime.NewTicketVerifier(tickets, clk, tf)
    67  	bearer := mock_auth.NewMockAuthenticator(gomock.NewController(t))
    68  	authFunc, err := auth.NewAuthenticator(
    69  		bearer, stubIdentityResolver{}, []auth.SchemeAuthenticator{streamauth.New(verifier)},
    70  	)
    71  	require.NoError(t, err)
    72  
    73  	replayLog := rttestkit.NewEventLog()
    74  	seedEvents(replayLog, 1)
    75  	cursorLog := rttestkit.NewEventLog()
    76  	seedEvents(cursorLog, 1)
    77  
    78  	registry := stream.NewRegistry(ucrealtime.NewReplayer(replayLog, tf), newSSESleeper(), logger, tf,
    79  		observability.NewNoopRealtimeMetrics(t), ucrealtime.NewHealth(replayLog), stream.Settings{})
    80  
    81  	e := echo.New()
    82  	UseAppErrorHandlerWithLogger(t, e, logger,
    83  		instrumentation.LoggingMiddleware(logger, logging.NewTestLogFieldBuilder(t), redaction.FromSpec(spec)).Middleware,
    84  		inbound.OpenAPIMiddleware(spec, skipper.New(), authFunc).Middleware,
    85  	)
    86  	stream.BindHandler(e, tf, ucrealtime.NewCursorValidator(cursorLog, clk, tf), registry)
    87  
    88  	return &revocationFixture{
    89  		srv:      StartServer(t, e),
    90  		tickets:  tickets,
    91  		issuer:   ucrealtime.NewTicketIssuer(tickets, realtimesecret.New(), clk, tf),
    92  		revoker:  ucrealtime.NewAccessRevoker(tickets, fanoutToRegistry{registry: registry}, tf),
    93  		registry: registry,
    94  	}

   109  func TestStreamRevocation_Integration(t *testing.T) {
   110  	t.Parallel()
   111  
   112  	t.Run("正常系", func(t *testing.T) {
   113  		t.Parallel()
   114  
   115  		t.Run("失効させると接続が STOP で閉じ、同じ ticket では繋ぎ直せない", func(t *testing.T) {
   116  			t.Parallel()
   117  
   118  			f := newRevocationServer(t)
   119  			value := f.issue(t, "subject-1")
   120  			path := "/v1/streams/" + string(streamTestDestination) + "?ticket=" + value
   121  
   122  			c, res := connectSSE(t, f.srv, path)
   123  			require.Equal(t, http.StatusOK, res.StatusCode)
   124  			require.Equal(t, "1", c.next().id)
   125  
   126  			require.NoError(t, f.revoker.Revoke(context.Background(), "subject-1", streamTestDestination))
   127  
   128  			frame := c.next()
   129  			assert.Equal(t, "control", frame.event)
   130  			require.NotNil(t, c.lastControl)
   131  			assert.Equal(t, "STOP", string(c.lastControl.Action))
   132  			assert.True(t, c.closedBySelf, "STOP を受けた client は自ら閉じること")
   133  
   134  			// STOP を無視する client が同じ ticket で戻ってこられないこと。無効化が効いていなければここが 200 になる。
   135  			_, again := connectSSE(t, f.srv, path)
   136  			assert.Equal(t, http.StatusUnauthorized, again.StatusCode)
   137  			assert.Zero(t, f.tickets.Len(), "失効した ticket は store に残らないこと")
   138  		})
   139  
   140  		t.Run("失効は subject と destination の組にだけ効く", func(t *testing.T) {
   141  			t.Parallel()
   142  
   143  			f := newRevocationServer(t)
   144  			mine := f.issue(t, "subject-1")
   145  			other := f.issue(t, "subject-2")
   146  
   147  			require.NoError(t, f.revoker.Revoke(context.Background(), "subject-2", streamTestDestination))
   148  
   149  			_, res := connectSSE(t, f.srv, "/v1/streams/"+string(streamTestDestination)+"?ticket="+mine)
   150  			assert.Equal(t, http.StatusOK, res.StatusCode, "他人の失効で自分の ticket が巻き添えにならないこと")
   151  			_, revoked := connectSSE(t, f.srv, "/v1/streams/"+string(streamTestDestination)+"?ticket="+other)
   152  			assert.Equal(t, http.StatusUnauthorized, revoked.StatusCode)
   153  		})
   154  	})
