   138  // Resolve は、git 文脈と .gobp-db-slot から解決済みの値を組み立てます。
   139  // git リポジトリなのに構成を読めなかった場合だけエラーにします（inspectGit を参照）。
   140  func (r *Resolver) Resolve(ctx context.Context) (Values, error) {
   141  	gitCtx, err := r.inspectGit(ctx)
   142  	if err != nil {
   143  		return Values{}, err
   144  	}
   145  
   146  	values := Values{
   147  		Git:        gitCtx,
   148  		DBLocal:    defaultDBLocal,
   149  		DBTest:     defaultDBTest,
   150  		AppProject: "gobp-app-" + filepath.Base(r.cfg.Root),
   151  		AuthIssuer: mockAuthIssuer(r.cfg.MockAuthBase),
   152  
   153  		RealtimeTableSuffix: r.cfg.Realtime.TableSuffix,
   154  		RealtimeQueuePrefix: r.cfg.Realtime.QueuePrefix,
   155  		RealtimeTopic:       r.cfg.Realtime.Topic,
   156  	}
   157  
   158  	// リースを保持しているスロットの値だけを、その番号から導く
   159  	// （.gobp-db-slot の他フィールドを読まない理由は README.md「Resolved values」）。
   160  	if slot, held := r.heldSlot(); held {
   161  		num := strconv.Itoa(slot)
   162  		values.SlotHeld = true
   163  		values.DBLocal = dbLocal(slot)
   164  		values.DBTest = dbTest(slot)
   165  		values.AppProject = serveProject(slot)
   166  		values.AuthIssuer = mockAuthIssuer(r.cfg.MockAuthBase + slot)
   167  		values.RealtimeTableSuffix = realtimeName(r.cfg.Realtime.TableSuffix, num, slotInfixTable)
   168  		values.RealtimeQueuePrefix = realtimeName(r.cfg.Realtime.QueuePrefix, num, slotInfixName)
   169  		values.RealtimeTopic = realtimeName(r.cfg.Realtime.Topic, num, slotInfixName)
   170  	}
   171  
   172  	// 共有インフラを奪い合う相手が居るのはリンク worktree のときだけなので、単一 checkout では空にします。
   173  	if gitCtx == GitLinkedWorktree {
   174  		values.InfraNoRecreate = noRecreateFlag
   175  	}
   176  
   177  	return values, nil
   178  }

   180  // RequireOwner は、所有データベースを持たない状態（リンク worktree かつスロット未取得）を
   181  // 検出してエラーにします。主 checkout・ツールランナー内・CI は素通りします。
   182  //
   183  // 判定は三値です。「git 実行ファイルが無い」「リポジトリでない」は素通り、
   184  // 「リポジトリなのに読めない」は失敗（理由は docs/maintenance/db-worktree-pool.md）。
   185  func (r *Resolver) RequireOwner(ctx context.Context) error {
   186  	values, err := r.Resolve(ctx)
   187  	if err != nil {
   188  		return err
   189  	}
   190  
   191  	if values.Git != GitLinkedWorktree || values.SlotHeld {
   192  		return nil
   193  	}
   194  
   195  	_, _ = fmt.Fprint(r.out, "❌ この worktree は DB スロットを取得していない（または取得したスロットが\n"+
   196  		"   他 worktree に回収されている）ため、所有するデータベースがありません。\n"+
   197  		"   1 つのデータベースを複数の checkout から触らないよう、既定の local / test へは\n"+
   198  		"   フォールバックしません。\n"+
   199  		"   → make slot-acquire でスロットを取得してください（make slot-status で空きを確認）。\n")
   200  
   201  	return errNoDatabaseOwner
   202  }

   229  // heldSlot は、.gobp-db-slot が宣言するスロットのうち、自 worktree がリースを保持しているものを返します。
   230  // 宣言が壊れている（スロット番号や DB 名が読めない）場合と、リースを確認できない場合は保持なしとします。
   231  func (r *Resolver) heldSlot() (int, bool) {
   232  	slot := readSlotFile(filepath.Join(r.cfg.Root, ".gobp-db-slot"))
   233  	if slot["DB_NAME_LOCAL"] == "" || r.lease.OwnedBySelf == nil {
   234  		return 0, false
   235  	}
   236  
   237  	n, err := strconv.Atoi(slot["SLOT"])
   238  	if err != nil {
   239  		return 0, false
   240  	}
   241  
   242  	return n, r.lease.OwnedBySelf(n)
   243  }

   245  // inspectGit は、git 文脈を判定します。
   246  // git を引けなかったときは、.git の有無で「リポジトリでない」と「リポジトリなのに読めない」を
   247  // 分けます。git のエラーメッセージはロケールで変わるため、メッセージは判定材料にしません。
   248  func (r *Resolver) inspectGit(ctx context.Context) (GitContext, error) {
   249  	if err := r.probe.LookGit(); err != nil {
   250  		return GitUnavailable, nil //nolint:nilerr // git が無いのは異常ではなく「判定できない文脈」であり、素通りが正しい
   251  	}
   252  
   253  	out, err := r.probe.GitDirs(ctx, r.cfg.Root)
   254  	if err != nil {
   255  		if r.probe.HasGitEntry(r.cfg.Root) {
   256  			return 0, xerrors.Wrap(errGitLayoutUnreadable, err.Error())
   257  		}
   258  
   259  		return GitNoRepository, nil
   260  	}
   261  
   262  	fields := strings.Fields(out)
   263  	if len(fields) != gitDirsFields {
   264  		return 0, xerrors.Wrap(errGitLayoutUnreadable, "unexpected `git rev-parse` output: "+strconv.Quote(out))
   265  	}
   266  
   267  	if r.absDir(fields[0]) == r.absDir(fields[1]) {
   268  		return GitMainCheckout, nil
   269  	}
   270  
   271  	return GitLinkedWorktree, nil
   272  }

   283  // RenderEnv は、解決済みの値を make のレシピが eval できる KEY=VALUE へ整形します。
   284  func RenderEnv(v Values) string {
   285  	pairs := [][2]string{
   286  		{"GOBP_GIT_CONTEXT", v.Git.String()},
   287  		{"DB_LOCAL", v.DBLocal},
   288  		{"DB_TEST", v.DBTest},
   289  		{"APP_PROJECT", v.AppProject},
   290  		{"AUTH_ISSUER", v.AuthIssuer},
   291  		{"INFRA_NO_RECREATE", v.InfraNoRecreate},
   292  		{"REALTIME_TABLE_SUFFIX", v.RealtimeTableSuffix},
   293  		{"REALTIME_QUEUE_PREFIX", v.RealtimeQueuePrefix},
   294  		{"REALTIME_TOPIC", v.RealtimeTopic},
   295  	}
   296  
   297  	var sb strings.Builder
   298  	for _, kv := range pairs {
   299  		fmt.Fprintf(&sb, "%s='%s'\n", kv[0], strings.ReplaceAll(kv[1], "'", `'\''`))

   302  	return sb.String()

   325  func realGitProbe() GitProbe {
   326  	return GitProbe{
   327  		LookGit: func() error {
   328  			_, err := exec.LookPath("git")
   329  
   330  			return err
   331  		},
   332  		GitDirs: func(ctx context.Context, root string) (string, error) {
   333  			cmd := exec.CommandContext(ctx, "git", "rev-parse", "--git-dir", "--git-common-dir")
   334  			cmd.Dir = root
   335  			out, err := cmd.Output()
   336  
   337  			return string(out), err
   338  		},
   339  		HasGitEntry: hasGitEntry,
   340  	}
   341  }

   343  // hasGitEntry は、dir か その祖先に .git（ディレクトリ、またはリンク worktree の .git ファイル）が
   344  // あるかを返します。
   345  func hasGitEntry(dir string) bool {
   346  	for {
   347  		if _, err := os.Lstat(filepath.Join(dir, ".git")); err == nil {
   348  			return true
   349  		}
   350  
   351  		parent := filepath.Dir(dir)
   352  		if parent == dir {
   353  			return false
   354  		}
   355  
   356  		dir = parent
   357  	}
   358  }

   360  // readSlotFile は、.gobp-db-slot の KEY=VALUE を読み取ります。
   361  // ファイルが無い状態は「スロット未取得」という正常な状態なので、空の map を返します。
   362  func readSlotFile(path string) map[string]string {
   363  	values := make(map[string]string)
   364  
   365  	b, err := os.ReadFile(path) //nolint:gosec // 読むのは自 worktree の .gobp-db-slot のみ
   366  	if err != nil {
   367  		return values
   368  	}
   369  
   370  	for line := range strings.SplitSeq(string(b), "\n") {
   371  		key, value, found := strings.Cut(strings.TrimSpace(line), "=")
   372  		if !found {
   373  			continue
   374  		}
   375  
   376  		values[key] = value
   377  	}
   378  
   379  	return values
   380  }
