    34  // UpSharedDB は `docker compose --profile database up -d --wait --no-recreate database` を実行します。
    35  // --no-recreate を外すと、他の checkout が使っている稼働中コンテナを作り直します

    38  	return c.run(ctx, project, "--profile", "database", "up", "-d", "--wait", "--no-recreate", "database")

    41  // DownServe は、app 層の compose プロジェクトを down します。
    42  // docker compose down はコンテナ不在でも exit 0 のため、返るエラーは docker 未起動・権限不足などの実失敗のみです。
    43  // app 層の override（docker-compose.attach.yaml）は重ねません（理由は README.md「Compose」）。

    45  	return c.run(ctx, project, "--profile", appProfile, "-f", "docker-compose.yaml", "down")

    74  // newComposeCmd は、固定プロジェクトを環境変数で与えた docker compose コマンドを組み立てます。
    75  func newComposeCmd(ctx context.Context, project string, stdout io.Writer, args ...string) *exec.Cmd {
    76  	// 引数は本パッケージ内の固定文字列のみ（ユーザー入力を渡さない）。
    77  	cmd := exec.CommandContext(ctx, "docker", append([]string{"compose"}, args...)...) //nolint:gosec // 固定引数
    78  	cmd.Env = append(os.Environ(), "COMPOSE_PROJECT_NAME="+project)
    79  	cmd.Stdout = stdout
    80  	cmd.Stderr = os.Stderr
    81  
    82  	return cmd
