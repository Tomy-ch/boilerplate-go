    49  // Acquire は、空きスロットをリースし共有 DB 上に自 worktree の DB を用意して .gobp-db-slot を書き出します。
    50  func (p *Pool) Acquire(ctx context.Context) error {
    51  	if err := p.ensureLocalEnv(); err != nil {
    52  		return err
    53  	}
    54  	if err := p.reg.EnsureDir(); err != nil {
    55  		return err
    56  	}
    57  	if err := p.comp.UpSharedDB(ctx, p.cfg.SharedProject); err != nil {
    58  		return err
    59  	}
    60  
    61  	// 既に自分が保持するスロットがあれば heartbeat 更新して再利用（冪等）。
    62  	if slot, ok := p.heldSlot(); ok && p.reg.OwnedBySelf(slot) {
    63  		return p.finishAcquire(ctx, slot, "reuse")
    64  	}
    65  
    66  	// flock で走査を直列化し二重リースを防ぐ。
    67  	unlock, err := p.reg.Lock()
    68  	if err != nil {
    69  		return err
    70  	}
    71  	defer unlock()
    72  
    73  	for slot := 1; slot <= p.reg.MaxSlots(); slot++ {
    74  		switch {
    75  		case p.reg.TryAcquireFresh(slot):
    76  		case p.reg.IsStale(slot):
    77  			// heartbeat は make serve 時にしか打たれないため、起動しっぱなしの app を持つスロットも
    78  			// TTL 超過で stale になる。DB を作り直す前に serve 中でないことを確かめる。
    79  			busy, err := p.slotInUse(ctx, slot)
    80  			if err != nil {
    81  				return err
    82  			}
    83  			if busy {
    84  				continue
    85  			}
    86  			if !p.reg.TryReclaim(slot) {
    87  				p.logf("slot %d reclaim lost to another process, skip", slot)
    88  				continue
    89  			}
    90  			p.logf("reclaim stale slot %d", slot)
    91  		default:
    92  			continue
    93  		}
    94  		return p.finishAcquire(ctx, slot, "acquired")
    95  	}
    96  	return xerrors.Wrap(errNoFreeSlot, fmt.Sprintf("all %d slots in use", p.reg.MaxSlots()))
    97  }

    99  // Release は、serve コンテナを停止し、リースと .gobp-db-slot を解放します（DB は warm 保持）。
   100  func (p *Pool) Release(ctx context.Context) error {
   101  	slot, ok := p.heldSlot()
   102  	if !ok {
   103  		p.logf("no slot held by this worktree")
   104  		return nil
   105  	}
   106  	// serve した app コンテナを停止する（放置すると再割当て・reinit 後の DB を孤児が掴む）。
   107  	// 停止失敗（docker 未起動・権限不足など）は孤児コンテナ検知のためログへ可視化し、リース解放自体は続行する。
   108  	if err := p.comp.DownServe(ctx, serveProject(slot)); err != nil {
   109  		p.logf("failed to stop serve containers for slot %d: %v", slot, err)
   110  	}
   111  	p.reg.Release(slot)
   112  	_ = os.Remove(p.slotFilePath())
   113  	p.logf("released slot %d (databases left warm for reuse)", slot)
   114  	return nil
   115  }

   120  	if !ok || !p.reg.OwnedBySelf(slot) {

   126  // Status は、スロットの占有状況を表示します。
   127  func (p *Pool) Status() error {
   128  	if err := p.ensureLocalEnv(); err != nil {
   129  		return err
   130  	}
   131  	if err := p.reg.EnsureDir(); err != nil {
   132  		return err
   133  	}
   134  	_, _ = fmt.Fprintf(p.out, "%-5s %-14s %-14s %-6s %-9s %-9s %s\n",
   135  		"SLOT", "DB_LOCAL", "DB_TEST", "API", "STATE", "AGE", "OWNER")
   136  	for slot := 1; slot <= p.reg.MaxSlots(); slot++ {
   137  		state, age, owner := "free", "-", "-"
   138  		if p.reg.Exists(slot) {
   139  			m, _ := p.reg.ReadMeta(slot)
   140  			owner = m.Owner
   141  			age = strconv.FormatInt(p.reg.AgeSeconds(slot), 10)
   142  			if p.reg.IsStale(slot) {
   143  				state = "stale"
   144  			} else {
   145  				state = "in-use"
   146  			}
   147  		}
   148  		_, _ = fmt.Fprintf(p.out, "%-5d %-14s %-14s %-6d %-9s %-9s %s\n",
   149  			slot, dbLocal(slot), dbTest(slot), p.cfg.APIBasePort+slot, state, age, owner)
   150  	}
   151  	return nil
   152  }

   169  // slotInUse は、stale なスロットが実際にはまだ使われているかを返します。
   170  // app コンテナの稼働と DB への接続をそれぞれ確認します。接続プールはアイドルで空になるため、
   171  // 接続数だけでは serve 中の worktree を見落とします。
   172  func (p *Pool) slotInUse(ctx context.Context, slot int) (bool, error) {
   173  	running, err := p.comp.RunningContainers(ctx, serveProject(slot))
   174  	if err != nil {
   175  		return false, err
   176  	}
   177  	if running > 0 {
   178  		p.logf("slot %d is stale but %s has running containers, skip", slot, serveProject(slot))
   179  		return true, nil
   180  	}
   181  
   182  	conns, err := p.admin.ActiveConnections(ctx, dbLocal(slot), dbTest(slot))
   183  	if err != nil {
   184  		return false, err
   185  	}
   186  	if conns > 0 {
   187  		p.logf("slot %d is stale but has active DB connections, skip", slot)
   188  		return true, nil
   189  	}
   190  
   191  	return false, nil
   192  }

   194  // finishAcquire は、取得済みスロットに meta / DB / .gobp-db-slot を確定させます。
   195  func (p *Pool) finishAcquire(ctx context.Context, slot int, verb string) error {
   196  	if err := p.reg.WriteMeta(slot); err != nil {
   197  		return err
   198  	}
   199  	if err := p.ensureSlotDBs(ctx, slot); err != nil {
   200  		return err
   201  	}
   202  	if err := p.writeSlotFile(slot); err != nil {
   203  		return err
   204  	}
   205  	p.logf("%s slot %d (db %s / %s)", verb, slot, dbLocal(slot), dbTest(slot))
   206  	return p.printSlotFile()
   207  }

   209  // ensureSlotDBs は、wt<N>_local / wt<N>_test を作成し拡張を設定します。
   210  func (p *Pool) ensureSlotDBs(ctx context.Context, slot int) error {
   211  	for _, name := range []string{dbLocal(slot), dbTest(slot)} {
   212  		if err := p.admin.EnsureDatabase(ctx, name); err != nil {
   213  			return err
   214  		}
   215  		if err := p.admin.SetupDatabase(ctx, name); err != nil {
   216  			return err
   217  		}
   218  	}
   219  	return nil
   220  }

   222  // writeSlotFile は、.gobp-db-slot（make が -include で読む KEY=VALUE）を書き出します。
   223  // app コンテナのホスト公開ポートは全てスロット番号で相対化し、並列 serve が衝突しないようにします。
   224  func (p *Pool) writeSlotFile(slot int) error {
   225  	content := fmt.Sprintf(
   226  		"SLOT=%d\nDB_NAME_LOCAL=%s\nDB_NAME_TEST=%s\n"+
   227  			"API_HOST_PORT=%d\nMOCK_AUTH_HOST_PORT=%d\nDLV_HOST_PORT=%d\nPPROF_HOST_PORT=%d\n"+
   228  			"COMPOSE_PROJECT_NAME=%s\nSERVE_PROJECT=%s\n",
   229  		slot, dbLocal(slot), dbTest(slot),
   230  		p.cfg.APIBasePort+slot, p.cfg.MockAuthBase+slot, p.cfg.DlvBase+slot, p.cfg.PprofBase+slot,
   231  		p.cfg.SharedProject, serveProject(slot),
   232  	)
   233  	if err := os.WriteFile(p.slotFilePath(), []byte(content), filePerm); err != nil {
   234  		return xerrors.Wrap(err, "failed to write .gobp-db-slot")
   235  	}
   236  	return nil
   237  }

   248  // heldSlot は、.gobp-db-slot に記録された保持スロット番号を返します。
   249  func (p *Pool) heldSlot() (int, bool) {
   250  	b, err := os.ReadFile(p.slotFilePath())
   251  	if err != nil {
   252  		return 0, false
   253  	}
   254  	for line := range strings.SplitSeq(string(b), "\n") {
   255  		if v, ok := strings.CutPrefix(line, "SLOT="); ok {
   256  			n, err := strconv.Atoi(strings.TrimSpace(v))
   257  			return n, err == nil
   258  		}
   259  	}
   260  	return 0, false
   261  }
