    51  // Acquire は、空きスロットをリースし共有 DB 上に自 worktree の DB を用意して .gobp-db-slot を書き出します。
    52  func (p *Pool) Acquire(ctx context.Context) error {
    53  	if err := p.ensureLocalEnv(); err != nil {
    54  		return err
    55  	}
    56  	if err := p.reg.EnsureDir(); err != nil {
    57  		return err
    58  	}
    59  	if err := p.comp.UpSharedDB(ctx, p.cfg.SharedProject); err != nil {
    60  		return err
    61  	}
    62  
    63  	// 既に自分が保持するスロットがあれば heartbeat 更新して再利用（冪等）。
    64  	if slot, ok := p.heldSlot(); ok && p.reg.OwnedBySelf(slot) {
    65  		return p.finishAcquire(ctx, slot, "reuse")
    66  	}
    67  
    68  	// flock で走査を直列化し二重リースを防ぐ。
    69  	unlock, err := p.reg.Lock()
    70  	if err != nil {
    71  		return err
    72  	}
    73  	defer unlock()
    74  
    75  	for slot := 1; slot <= p.reg.MaxSlots(); slot++ {
    76  		switch {
    77  		case p.reg.TryAcquireFresh(slot):
    78  		case p.reg.IsStale(slot):
    79  			// DB を作り直す前に serve 中でないことを確かめる
    80  			// （起動中でも stale になる理由は docs/maintenance/db-worktree-pool.md）。
    81  			busy, err := p.slotInUse(ctx, slot)
    82  			if err != nil {
    83  				return err
    84  			}
    85  			if busy {
    86  				continue
    87  			}
    88  			if !p.reg.TryReclaim(slot) {
    89  				p.logf("slot %d reclaim lost to another process, skip", slot)
    90  				continue
    91  			}
    92  			p.logf("reclaim stale slot %d", slot)
    93  		default:
    94  			continue
    95  		}
    96  		return p.finishAcquire(ctx, slot, "acquired")
    97  	}
    98  	return xerrors.Wrap(errNoFreeSlot, fmt.Sprintf("all %d slots in use", p.reg.MaxSlots()))
    99  }

   101  // Release は、serve コンテナを停止し、リースと .gobp-db-slot を解放します（DB は warm 保持）。
   102  func (p *Pool) Release(ctx context.Context) error {
   103  	slot, ok := p.heldSlot()
   104  	if !ok {
   105  		p.logf("no slot held by this worktree")
   106  		return nil
   107  	}
   108  	// 確認から down までを acquire の走査と直列化し、他 worktree の stale 回収と競合して
   109  	// 現保持者のコンテナを落とすことを防ぐ。ロックを取れなくても、孤児を残さないため実行する。
   110  	if unlock, err := p.reg.Lock(); err != nil {
   111  		p.logf("failed to take the scan lock for slot %d; releasing without it: %v", slot, err)
   112  	} else {
   113  		defer unlock()
   114  	}
   115  	// リースが stale 回収で他 worktree へ渡っていれば、そのスロットの app コンテナは現保持者のもの。
   116  	// down すれば他人の serve を落とすため compose には触れず、失効した .gobp-db-slot だけを片付ける。
   117  	if p.reg.HeldByOther(slot) {
   118  		_ = os.Remove(p.slotFilePath())
   119  		p.logf("slot %d is held by another worktree; dropped the stale slot file only", slot)
   120  		return nil
   121  	}
   122  	// serve した app コンテナを停止する（放置すると再割当て・reinit 後の DB を孤児が掴む）。
   123  	// 停止失敗（docker 未起動・権限不足など）は孤児コンテナ検知のためログへ可視化し、リース解放自体は続行する。
   124  	if err := p.comp.DownServe(ctx, serveProject(slot)); err != nil {
   125  		p.logf("failed to stop serve containers for slot %d: %v", slot, err)
   126  	}
   127  	released := p.reg.Release(slot)
   128  	_ = os.Remove(p.slotFilePath())
   129  	if !released {
   130  		p.logf("slot %d lease could not be confirmed as ours; left it for stale reclaim", slot)
   131  		return nil
   132  	}
   133  	p.logf("released slot %d (databases left warm for reuse)", slot)
   134  
   135  	return nil
   136  }

   139  func (p *Pool) Heartbeat() error {

   141  	if !ok || !p.reg.OwnedBySelf(slot) {

   147  // Status は、スロットの占有状況を表示します。
   148  func (p *Pool) Status() error {
   149  	if err := p.ensureLocalEnv(); err != nil {
   150  		return err
   151  	}
   152  	if err := p.reg.EnsureDir(); err != nil {
   153  		return err
   154  	}
   155  	_, _ = fmt.Fprintf(p.out, "%-5s %-14s %-14s %-6s %-9s %-9s %s\n",
   156  		"SLOT", "DB_LOCAL", "DB_TEST", "API", "STATE", "AGE", "OWNER")
   157  	for slot := 1; slot <= p.reg.MaxSlots(); slot++ {
   158  		state, age, owner := "free", "-", "-"
   159  		if p.reg.Exists(slot) {
   160  			m, _ := p.reg.ReadMeta(slot)
   161  			owner = m.Owner
   162  			age = strconv.FormatInt(p.reg.AgeSeconds(slot), 10)
   163  			if p.reg.IsStale(slot) {
   164  				state = "stale"
   165  			} else {
   166  				state = "in-use"
   167  			}
   168  		}
   169  		_, _ = fmt.Fprintf(p.out, "%-5d %-14s %-14s %-6d %-9s %-9s %s\n",
   170  			slot, dbLocal(slot), dbTest(slot), p.cfg.APIBasePort+slot, state, age, owner)
   171  	}
   172  	return nil
   173  }

   190  // slotInUse は、stale なスロットが実際にはまだ使われているかを返します
   191  // （稼働と接続の両方を見る理由は README.md「In-use detection」）。
   192  func (p *Pool) slotInUse(ctx context.Context, slot int) (bool, error) {
   193  	running, err := p.comp.RunningContainers(ctx, serveProject(slot))
   194  	if err != nil {
   195  		return false, err
   196  	}
   197  	if running > 0 {
   198  		p.logf("slot %d is stale but %s has running containers, skip", slot, serveProject(slot))
   199  		return true, nil
   200  	}
   201  
   202  	conns, err := p.admin.ActiveConnections(ctx, dbLocal(slot), dbTest(slot))
   203  	if err != nil {
   204  		return false, err
   205  	}
   206  	if conns > 0 {
   207  		p.logf("slot %d is stale but has active DB connections, skip", slot)
   208  		return true, nil
   209  	}
   210  
   211  	return false, nil
   212  }

   214  // finishAcquire は、取得済みスロットに meta / DB / .gobp-db-slot を確定させます。
   215  func (p *Pool) finishAcquire(ctx context.Context, slot int, verb string) error {
   216  	if err := p.reg.WriteMeta(slot); err != nil {
   217  		return err
   218  	}
   219  	if err := p.ensureSlotDBs(ctx, slot); err != nil {
   220  		return err
   221  	}
   222  	if err := p.writeSlotFile(slot); err != nil {
   223  		return err
   224  	}
   225  	p.logf("%s slot %d (db %s / %s)", verb, slot, dbLocal(slot), dbTest(slot))
   226  	return p.printSlotFile()
   227  }

   230  func (p *Pool) ensureSlotDBs(ctx context.Context, slot int) error {
   231  	for _, name := range []string{dbLocal(slot), dbTest(slot)} {
   232  		if err := p.admin.EnsureDatabase(ctx, name); err != nil {
   233  			return err
   234  		}
   235  		if err := p.admin.SetupDatabase(ctx, name); err != nil {
   236  			return err
   237  		}
   238  	}
   239  	return nil
   240  }

   242  // writeSlotFile は、.gobp-db-slot（make が -include で読む KEY=VALUE）を書き出します。
   243  // app コンテナのホスト公開ポートは全てスロット番号で相対化し、並列 serve が衝突しないようにします。
   244  func (p *Pool) writeSlotFile(slot int) error {
   245  	content := fmt.Sprintf(
   246  		"SLOT=%d\nDB_NAME_LOCAL=%s\nDB_NAME_TEST=%s\n"+
   247  			"API_HOST_PORT=%d\nMOCK_AUTH_HOST_PORT=%d\nDLV_HOST_PORT=%d\nPPROF_HOST_PORT=%d\n"+
   248  			"COMPOSE_PROJECT_NAME=%s\nSERVE_PROJECT=%s\n",
   249  		slot, dbLocal(slot), dbTest(slot),
   250  		p.cfg.APIBasePort+slot, p.cfg.MockAuthBase+slot, p.cfg.DlvBase+slot, p.cfg.PprofBase+slot,
   251  		p.cfg.SharedProject, serveProject(slot),
   252  	)
   253  	if err := os.WriteFile(p.slotFilePath(), []byte(content), filePerm); err != nil {
   254  		return xerrors.Wrap(err, "failed to write .gobp-db-slot")
   255  	}
   256  	return nil
   257  }

   268  // heldSlot は、.gobp-db-slot に記録された保持スロット番号を返します。
   269  func (p *Pool) heldSlot() (int, bool) {
   270  	b, err := os.ReadFile(p.slotFilePath())
   271  	if err != nil {
   272  		return 0, false
   273  	}
   274  	for line := range strings.SplitSeq(string(b), "\n") {
   275  		if v, ok := strings.CutPrefix(line, "SLOT="); ok {
   276  			n, err := strconv.Atoi(strings.TrimSpace(v))
   277  			return n, err == nil
   278  		}
   279  	}
   280  	return 0, false
   281  }
