    37  // EnsureDatabase は、対象 DB が無ければ CREATE DATABASE します。
    38  func (a *PgxAdmin) EnsureDatabase(ctx context.Context, name string) error {
    39  	conn, err := a.connect(ctx, a.maintenanceDB)
    40  	if err != nil {
    41  		return err
    42  	}
    43  	defer func() { _ = conn.Close(ctx) }()
    44  
    45  	var exists bool
    46  	if err := conn.QueryRow(ctx,
    47  		"SELECT EXISTS(SELECT 1 FROM pg_database WHERE datname=$1)", name).Scan(&exists); err != nil {
    48  		return xerrors.Wrap(err, "failed to check database existence")
    49  	}
    50  	if exists {
    51  		return nil
    52  	}
    53  	// CREATE DATABASE は識別子をパラメータ化できないため安全に引用する。
    54  	if _, err := conn.Exec(ctx, "CREATE DATABASE "+pgx.Identifier{name}.Sanitize()); err != nil {
    55  		return xerrors.Wrap(err, "failed to create database "+name)
    56  	}
    57  	return nil
    58  }

    76  func (a *PgxAdmin) ActiveConnections(ctx context.Context, names ...string) (int, error) {
    77  	if len(names) == 0 {
    78  		return 0, nil
    79  	}
    80  	conn, err := a.connect(ctx, a.maintenanceDB)
    81  	if err != nil {
    82  		return 0, err
    83  	}
    84  	defer func() { _ = conn.Close(ctx) }()
    85  
    86  	var count int
    87  	if err := conn.QueryRow(ctx,
    88  		"SELECT count(*) FROM pg_stat_activity WHERE datname = ANY($1)", names).Scan(&count); err != nil {
    89  		return 0, xerrors.Wrap(err, "failed to count active connections")
    90  	}
    91  	return count, nil
    92  }
