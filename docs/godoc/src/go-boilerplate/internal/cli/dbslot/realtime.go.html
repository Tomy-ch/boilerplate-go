    35  	b, err := root.FS.ReadFile(embeddedEnvFile)

    50  func realtimeBaseFrom(kv map[string]string) (RealtimeBase, error) {
    51  	base := RealtimeBase{
    52  		TableSuffix: kv[realtimeTableSuffixKey],
    53  		QueuePrefix: kv[realtimeQueuePrefixKey],
    54  		Topic:       kv[realtimeTopicKey],
    55  	}
    56  
    57  	// 複数欠けたときに報告するキーが揺れないよう、順序を固定して見る。
    58  	for _, declared := range [][2]string{
    59  		{realtimeTableSuffixKey, base.TableSuffix},
    60  		{realtimeQueuePrefixKey, base.QueuePrefix},
    61  		{realtimeTopicKey, base.Topic},
    62  	} {
    63  		if declared[1] == "" {
    64  			return RealtimeBase{}, xerrors.Wrap(ErrRealtimeBaseMissing, declared[0])
    65  		}
    66  	}
    67  
    68  	return base, nil
