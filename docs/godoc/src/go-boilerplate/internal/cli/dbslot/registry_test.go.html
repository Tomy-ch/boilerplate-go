   122  func TestRegistry_TryReclaim(t *testing.T) {
   123  	t.Parallel()
   124  
   125  	t.Run("正常系", func(t *testing.T) {
   126  		t.Parallel()
   127  
   128  		t.Run("stale スロットを並行回収しても占有権を得るのは1つだけ", func(t *testing.T) {
   129  			t.Parallel()
   130  
   131  			now := time.Unix(10_000, 0)
   132  			// 他 worktree が古い heartbeat で保持中の状態を作る。
   133  			owner := newTestRegistry(t, "/w/other", now)
   134  			require.True(t, owner.TryAcquireFresh(1))
   135  			require.NoError(t, owner.WriteMeta(1))
   136  
   137  			late := now.Add(time.Hour)
   138  			const n = 20
   139  			var wg sync.WaitGroup
   140  			var mu sync.Mutex
   141  			wins := 0
   142  			wg.Add(n)
   143  			for i := range n {
   144  				id := i
   145  				go func() {
   146  					defer wg.Done()
   147  					// 各 goroutine は別 worktree（別 owner）としてオーケストレータと同じ scan lock + claim +
   148  					// WriteMeta を行う。flock で直列化され、最初の1つだけが占有権を得る。
   149  					ownerPath := "/w/me/" + strconv.Itoa(id)
   150  					r := NewRegistry(owner.dir, ownerPath, "b", 30*time.Minute, 8, func() time.Time { return late })
   151  					unlock, err := r.Lock()
   152  					if err != nil {
   153  						return
   154  					}
   155  					defer unlock()
   156  					if r.IsStale(1) && r.TryReclaim(1) {
   157  						_ = r.WriteMeta(1)
   158  						mu.Lock()
   159  						wins++
   160  						mu.Unlock()
   161  					}
   162  				}()
   163  			}
   164  			wg.Wait()
   165  
   166  			assert.Equal(t, 1, wins)
   167  		})
   168  	})
   169  
   170  	t.Run("異常系", func(t *testing.T) {
   171  		t.Parallel()
   172  
   173  		t.Run("対象スロットが存在しなければ false（先取り/消失）", func(t *testing.T) {
   174  			t.Parallel()
   175  
   176  			r := newTestRegistry(t, "/w/a", time.Unix(1000, 0))
   177  			assert.False(t, r.TryReclaim(7))
   178  		})
   179  	})
