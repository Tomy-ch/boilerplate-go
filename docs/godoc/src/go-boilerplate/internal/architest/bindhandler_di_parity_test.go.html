    59  // TestBindHandlerDIParity は、ハンドラの DI 配線が漏れていないことを機械検証する。
    60  //
    61  // 突き合わせるのは 3 つの集合で、いずれもモジュールルートからの相対パッケージディレクトリで表す。
    62  //
    63  //	X: func BindHandler を宣言するパッケージ（ハンドラツリー internal/controller/handler と
    64  //	   機構 transport internal/controller/stream の走査）
    65  //	Y: DI module の fx.Invoke が列挙する BindHandler の所属パッケージ（ControllerModule() の controller.go と、
    66  //	   機構 transport を所有する realtimeModule() の realtime.go の走査）
    67  //	Z: gen/server.gen.go に RegisterHandlers を持つパッケージ（同じ走査）
    68  //
    69  // 主張する契約は X == Y と Z ⊆ X で、それぞれ次の書き忘れを検出する。
    70  //
    71  //   - X ⊆ Y: ハンドラを実装したが fx.Invoke へ足し忘れた状態。実行時に Echo へルートが生えず 404 に
    72  //     なるが、TestRouteSpecParity は spec から機械生成される gen/server.gen.go のルート登録を見るため
    73  //     緑のまま通る。本テストの主目的。
    74  //   - Y ⊆ X: fx.Invoke が呼ぶ BindHandler の宣言を走査で見つけられない状態。通常はコンパイラが
    75  //     保証する向きで、ここで主張する意味は、走査条件が実態からずれて X が黙って縮んだときに
    76  //     loud に落とすこと。
    77  //   - Z ⊆ X: make gen-api までは済ませたがハンドラ実装（BindHandler）を書いていない状態。生成物は
    78  //     単独でコンパイルでき、TestRouteSpecParity も生成物のルート登録を見て緑になるため、
    79  //     この向きが無いと実行時 404 に気付けない。
    80  //
    81  // Z == X としないのは、生成コードを持たないハンドラが規約上ありうるため。/metrics は応答を
    82  // サードパーティのハンドラが作るため oapi-codegen を通さず、BindHandler の中で echo.HandlerFunc を
    83  // 直接登録する（internal/controller/handler/README.md の carve-out）。Z ⊆ X なら、この形は許可リスト
    84  // 無しでそのまま通る。
    85  //
    86  // 検出しないもの: BindHandler が配線されていて中身が誤ったルートを生やす状態は各ハンドラの
    87  // TestBindHandler が、fx.Invoke に書かれた依存が解決できない状態は internal/di/module の
    88  // TestControllerModule_GraphIsValid が担う。
    89  //
    90  // 許可リストは持たない。サンプル API を削除するとハンドラディレクトリと controller.go の
    91  // sample-api マーカー行が同時に消えるため、3 集合は歩調を合わせて縮む。

   657  // collectHandlerImports は、DI module の import 節から ident → controller 層のパッケージディレクトリ
   658  // （モジュールルートからの相対）の対応を返します。
   659  //
   660  // エイリアスの無い import が束縛する ident はパッケージが名乗る名前であって、ディレクトリ名では
   661  // ありません。ハンドラのディレクトリ名は HTTP リソース名に合わせる規約（docs/rules.md）のため
   662  // 両者は実際に食い違い（categories は package productcategories）、ディレクトリ名で代用すると
   663  // 正しく配線されたハンドラを解決できずに落とします。そこで走査で得た packages を先に引き、
   664  // 走査対象外のディレクトリに限りパスの最終セグメントで近似します。
   665  func collectHandlerImports(src string, packages map[string]string) map[string]string {
   666  	out := map[string]string{}
   667  	for line := range strings.SplitSeq(src, "\n") {
   668  		m := handlerImportRe.FindStringSubmatch(line)
   669  		if m == nil {
   670  			continue
   671  		}
   672  		dir := controllerTreeDir + strings.TrimPrefix(m[2], handlerImportPrefix)
   673  		ident := m[1]
   674  		if ident == "" {
   675  			ident = packages[dir]
   676  		}
   677  		if ident == "" {
   678  			ident = path.Base(dir)
   679  		}
   680  		out[ident] = dir
   681  	}
   682  	return out
   683  }
