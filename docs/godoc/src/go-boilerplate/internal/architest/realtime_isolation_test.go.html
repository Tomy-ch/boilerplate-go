   113  // collectRealtimeFeatureImports は、dirs 配下の production code が持つ feature 向けの import を `file -> import` の形で返す。
   114  func collectRealtimeFeatureImports(root string, dirs []string) ([]string, error) {
   115  	var violations []string
   116  	for _, dir := range dirs {
   117  		abs := filepath.Join(root, dir)
   118  		err := walkProductionGoFiles(abs, func(path string, content []byte) {
   119  			rel, _ := filepath.Rel(root, path)
   120  			violations = append(violations, featureImportViolations(filepath.ToSlash(rel), content)...)
   121  		})
   122  		if err != nil {
   123  			return nil, err
   124  		}
   125  	}
   126  
   127  	return violations, nil
   128  }

   130  // featureImportViolations は、1 ファイルの内容から feature 向けの import を `rel -> import` の形で返す。
   131  func featureImportViolations(rel string, content []byte) []string {
   132  	var out []string
   133  	for _, m := range internalImportRe.FindAllStringSubmatch(string(content), -1) {
   134  		if isFeatureImport(m[1]) {
   135  			out = append(out, rel+" -> "+m[1])
   136  		}
   137  	}
   138  
   139  	return out
   140  }

   142  // isFeatureImport は、feature の domain / usecase を指す import かを判定する。boundary と realtime の usecase は機構側。
   143  func isFeatureImport(imp string) bool {
   144  	switch {
   145  	case strings.HasPrefix(imp, "go-boilerplate/internal/domain/"):
   146  		return true
   147  	case strings.HasPrefix(imp, "go-boilerplate/internal/usecase/boundary/"):
   148  		return false
   149  	case imp == "go-boilerplate/internal/usecase/realtime",
   150  		strings.HasPrefix(imp, "go-boilerplate/internal/usecase/realtime/"):
   151  		return false
   152  	case strings.HasPrefix(imp, "go-boilerplate/internal/usecase/"):
   153  		return true
   154  	default:
   155  		return false
   156  	}
   157  }

   173  // usesInstanceLeaseStore は、内容が InstanceLeaseStore を参照しているかを判定する。
   174  func usesInstanceLeaseStore(content []byte) bool {

   178  // isInstanceLeaseStoreAllowed は、rel（repo root 相対）が InstanceLeaseStore を参照してよい場所かを判定する。
   179  // 許可は列挙せず、規約 1 が宣言する機構側 package と、その機構を結線する DI module という構造から導く
   180  // （allowlist それ自体がドリフト源になるため。方針は internal/architest/README.md）。
   181  func isInstanceLeaseStoreAllowed(rel string) bool {
   182  	for _, dir := range realtimeMechanismDirs {
   183  		if strings.HasPrefix(rel, dir+"/") {
   184  			return true
   185  		}
   186  	}
   187  
   188  	return isRealtimeDIModule(rel)
   189  }
   190  
   191  // isRealtimeDIModule は、rel が Realtime を結線する DI module かを判定する。
   192  // module ファイルは結線する機構の名前を持つので、その命名が carve-out の構造そのものになる。
   193  func isRealtimeDIModule(rel string) bool {
   194  	dir, file := path.Split(rel)
   195  
   196  	return dir == "internal/di/module/" && strings.HasPrefix(file, "realtime")
   197  }

   199  // walkProductionGoFiles は、dir 配下の production code（テスト・生成物・mock を除く .go）を fn へ渡す。
   200  // dir が無ければ何もしない（まだ実装されていない package を許すため）。
   201  func walkProductionGoFiles(dir string, fn func(path string, content []byte)) error {
   202  	matches, err := pkgfs.OS{}.Glob(dir)
   203  	if err != nil || len(matches) == 0 {
   204  		return err
   205  	}
   206  
   207  	return filepath.WalkDir(dir, func(path string, d fs.DirEntry, werr error) error {
   208  		if werr != nil {
   209  			return werr
   210  		}
   211  
   212  		if d.IsDir() {
   213  			if d.Name() == "mock" || d.Name() == "node_modules" {
   214  				return filepath.SkipDir
   215  			}
   216  
   217  			return nil
   218  		}
   219  
   220  		name := d.Name()
   221  		if !strings.HasSuffix(name, ".go") || strings.HasSuffix(name, "_test.go") || strings.HasSuffix(name, ".gen.go") {
   222  			return nil
   223  		}
   224  
   225  		content, err := pkgfs.OS{}.ReadFile(path)
   226  		if err != nil {
   227  			return err
   228  		}
   229  
   230  		fn(path, content)
   231  
   232  		return nil
   233  	})
   234  }
