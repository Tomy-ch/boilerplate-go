    61  // UnmarshalYAML は、スカラ形と `{omit: ...}` 形の両方を受け取ります。
    62  func (r *payloadFieldRule) UnmarshalYAML(value *yaml.Node) error {
    63  	if value.Kind == yaml.ScalarNode {
    64  		r.Carried = value.Value
    65  
    66  		return nil
    67  	}
    68  
    69  	var m struct {
    70  		Omit string `yaml:"omit"`
    71  	}
    72  	if err := value.Decode(&m); err != nil {
    73  		return err
    74  	}
    75  	r.Omit = m.Omit
    76  
    77  	return nil
    78  }

    80  // TestOutboxPayloadParity は、outbox の payload と、その写し元となる集約との対応が
    81  // 宣言されていることを機械検証します。
    82  //
    83  // 集約にフィールドが増えたとき、payload 側は黙って古いままになれます。#1473 では
    84  // 課税の基礎を値引き後へ変えたのに purchase.created.v1 が値引き額を運ばず、購読側で
    85  // subtotal + tax + shipping が total と一致しない payload が出荷されました。これを
    86  // 落とすのがこの検査です。
    87  //
    88  // 「自己完結」を名乗るだけでは判別式になりません。8 payload はすべてそう名乗りますが、
    89  // 状態を運ぶもの（snapshot）と、起きた事実と識別子だけを運ぶもの（notification）が
    90  // 混ざっており、後者が集約の全フィールドを持たないのは正しい姿です。そこで各 payload に
    91  // 種別を宣言させ、snapshot に限って集約の全フィールドが「運ぶ / 運ばない（理由つき）」に
    92  // 分類されていることを課します（ADR-0113 (outbox-payload-kinds-and-parity-declaration)）。
    93  //
    94  // 突き合わせる集合と、各方向が捕まえる間違いは次のとおりです。
    95  //
    96  //	X: internal/usecase/**/event/ で Build* を持つディレクトリ
    97  //	Y: payload_parity.yaml を持つディレクトリ
    98  //	  X ⊆ Y … event パッケージを書いたが宣言していない（この検査が在る理由）
    99  //	  Y ⊆ X … 宣言だけが孤立している（走査が黙って縮んだときに落ちる側）
   100  //	P: Build* から導く payload 型名 / D: 宣言のキー
   101  //	  P ⊆ D … payload を足したが宣言していない
   102  //	  D ⊆ P … 消えた payload の宣言が残っている
   103  //	F: snapshot の写し元 struct のフィールド / K: 宣言の分類キー
   104  //	  F ⊆ K … 集約が増えたのに分類されていない（#1473 の形）
   105  //	  K ⊆ F … 集約から消えたフィールドの宣言が残っている
   106  //
   107  // 対象 0 件は許容します（sample API 撤去後は event パッケージごと消えるため、
   108  // ここに canary は置けません。internal/architest/README.md の Notes を参照）。
   109  // この許容の代償として、**実ツリーの走査が縮退して 0 件になった場合と、本当に
   110  // 対象が無い場合を、このテストは区別しません。** 下の陽性対照と Test_collect* が

   373  // collectOutboxPayloadParityViolations は、root 配下の event パッケージを走査して違反を集めます。
   374  // 戻り値はソート済みで、同じツリーからは常に同じ並びになります。
   375  func collectOutboxPayloadParityViolations(root string) ([]string, error) {
   376  	dirs, err := collectEventPackages(root)
   377  	if err != nil {
   378  		return nil, err
   379  	}
   380  
   381  	violations := []string{}
   382  	for _, dir := range dirs {
   383  		found, verr := verifyEventPackage(root, dir)
   384  		if verr != nil {
   385  			return nil, verr
   386  		}
   387  		violations = append(violations, found...)
   388  	}
   389  	sort.Strings(violations)
   390  
   391  	return violations, nil
   392  }

   395  func collectEventPackages(root string) ([]string, error) {
   396  	base := filepath.Join(root, usecaseRoot)
   397  	dirs := map[string]struct{}{}
   398  
   399  	err := filepath.WalkDir(base, func(path string, d fs.DirEntry, walkErr error) error {
   400  		if walkErr != nil {
   401  			if isNotExist(walkErr) {
   402  				return filepath.SkipAll
   403  			}
   404  
   405  			return walkErr
   406  		}
   407  		if d.IsDir() || !isPlainGoFile(path) || filepath.Base(filepath.Dir(path)) != "event" {
   408  			return nil

   412  		if rerr != nil {
   413  			return rerr

   417  		}
   418  
   419  		return nil
   420  	})
   421  	if err != nil {
   422  		return nil, err
   423  	}
   424  
   425  	out := make([]string, 0, len(dirs))
   426  	for d := range dirs {
   427  		out = append(out, d)
   428  	}
   429  	sort.Strings(out)
   430  
   431  	return out, nil
   432  }

   457  // 同じファイルに <Name> の非公開形が在ればそちらを、無ければ <Name> をそのまま採ります。
   458  func payloadNamesFromLines(lines []string) []string {
   459  	declared := map[string]struct{}{}
   460  	for _, line := range lines {
   461  		if m := structOpenRe.FindStringSubmatch(line); m != nil {
   462  			declared[m[1]] = struct{}{}
   463  		}
   464  	}
   465  
   466  	names := []string{}
   467  	for _, line := range lines {
   468  		m := buildFuncRe.FindStringSubmatch(line)
   469  		if m == nil {
   470  			continue
   471  		}
   472  		lower := lowerFirst(m[1])
   473  		if _, ok := declared[lower]; ok {
   474  			names = append(names, lower)
   475  
   476  			continue
   477  		}
   478  		names = append(names, m[1])
   479  	}
   480  
   481  	return names
   482  }

   503  // verifyEventPackage は、1 つの event パッケージについて宣言と実物を突き合わせます。
   504  func verifyEventPackage(root, dir string) ([]string, error) {
   505  	rel := relPath(root, dir)
   506  
   507  	payloads, err := collectDirPayloadNames(dir)
   508  	if err != nil {
   509  		return nil, err
   510  	}

   514  		return nil, err
   515  	}

   575  func readParityDoc(dir string) (payloadParityDoc, bool, error) {

   578  		if isNotExist(err) {
   579  			return payloadParityDoc{}, false, nil

   593  // diffPayloadNames は、実在する payload と宣言されている payload の双方向の差を違反にします。
   594  func diffPayloadNames(rel string, payloads []string, doc payloadParityDoc) []string {
   595  	declared := map[string]struct{}{}
   596  	for name := range doc.Payloads {
   597  		declared[name] = struct{}{}
   598  	}
   599  
   600  	violations := []string{}
   601  	for _, name := range payloads {
   602  		if _, ok := declared[name]; !ok {
   603  			violations = append(violations, fmt.Sprintf("%s/%s: payload %s が宣言されていない", rel, parityFileName, name))
   604  		}
   605  		delete(declared, name)
   606  	}
   607  	for name := range declared {
   608  		violations = append(violations, fmt.Sprintf("%s/%s: 宣言 %s に対応する Build 関数が無い", rel, parityFileName, name))
   609  	}
   610  
   611  	return violations

   614  // verifyPayloadEntry は、宣言 1 件を実物へ突き合わせます。
   615  func verifyPayloadEntry(root, dir, name string, entry payloadParityEntry) ([]string, error) {
   616  	rel := relPath(root, dir) + "/" + parityFileName
   617  
   618  	fields, ok, err := lookupAggregateFields(root, entry.Of)
   619  	if err != nil {
   620  		return nil, err
   621  	}
   622  	if !ok {
   623  		return []string{fmt.Sprintf("%s: %s: 写し元 %s が見つからない", rel, name, entry.Of)}, nil
   624  	}
   625  
   626  	switch entry.Kind {
   627  	case "notification":
   628  		return nil, nil
   629  	case "snapshot":
   630  		tags, terr := collectPayloadJSONTags(dir, name)
   631  		if terr != nil {
   632  			return nil, terr
   633  		}
   634  
   635  		return verifySnapshotFields(rel, name, entry, fields, tags), nil
   636  	default:
   637  		msg := fmt.Sprintf("%s: %s: kind は snapshot か notification のどちらかにすること（実際: %q）", rel, name, entry.Kind)
   638  
   639  		return []string{msg}, nil
   640  	}
   641  }

   643  // verifySnapshotFields は、snapshot の分類が集約の全フィールドを覆い、
   644  // 運ぶと宣言した JSON 名が payload に実在することを確かめます。
   645  func verifySnapshotFields(rel, name string, entry payloadParityEntry, fields, tags []string) []string {
   646  	rules := entry.Fields
   647  	violations := []string{}
   648  
   649  	tagSet := map[string]struct{}{}
   650  	for _, tag := range tags {
   651  		tagSet[tag] = struct{}{}
   652  	}
   653  
   654  	remaining := map[string]struct{}{}
   655  	for field := range rules {
   656  		remaining[field] = struct{}{}
   657  	}
   658  
   659  	for _, field := range fields {
   660  		rule, declared := rules[field]
   661  		if !declared {
   662  			violations = append(violations, fmt.Sprintf("%s: %s: 集約 %s のフィールド %s が未分類", rel, name, entry.Of, field))
   663  
   664  			continue
   665  		}
   666  		delete(remaining, field)
   667  		violations = append(violations, verifyFieldRule(rel, name, field, rule, tagSet)...)
   668  	}
   669  
   670  	for field := range remaining {
   671  		violations = append(violations, fmt.Sprintf("%s: %s: 宣言 %s は集約 %s に無い", rel, name, field, entry.Of))
   672  	}
   673  
   674  	return violations
   675  }

   677  // verifyFieldRule は、1 フィールドの宣言が実物と噛み合っているかを確かめます。
   678  func verifyFieldRule(rel, name, field string, rule payloadFieldRule, tagSet map[string]struct{}) []string {
   679  	if rule.Carried == "" && strings.TrimSpace(rule.Omit) == "" {
   680  		return []string{fmt.Sprintf("%s: %s: %s は運ぶ JSON 名か omit の理由のどちらかを書くこと", rel, name, field)}
   681  	}
   682  	if rule.Carried == "" {
   683  		return nil
   684  	}
   685  	if _, ok := tagSet[rule.Carried]; !ok {
   686  		return []string{fmt.Sprintf("%s: %s: 運ぶと宣言した %s が payload の JSON タグに無い", rel, name, rule.Carried)}
   687  	}
   688  
   689  	return nil
   690  }

   692  // lookupAggregateFields は、`<pkg>.<Struct>` 形の写し元から internal/domain 配下の struct を探し、
   693  // gofmt 済み前提でトップレベルのフィールド名を宣言順に返します。
   694  func lookupAggregateFields(root, of string) ([]string, bool, error) {
   695  	pkg, name, ok := strings.Cut(of, ".")
   696  	if !ok {
   697  		return nil, false, nil

   700  	paths, err := plainGoFiles(filepath.Join(root, "internal", "domain", pkg))
   701  	if err != nil {
   702  		return nil, false, err
   703  	}
   704  
   705  	for _, path := range paths {
   706  		lines, rerr := readLines(path)
   707  		if rerr != nil {
   708  			return nil, false, rerr
   709  		}
   710  		if fields, found := collectStructFields(lines, name); found {
   711  			return fields, true, nil
   712  		}
   713  	}
   714  
   715  	return nil, false, nil
   716  }

   718  // collectStructFields は、`type <name> struct {` から `}` までのフィールド名を返します。
   719  // 埋め込みや複数名宣言のように解析できない行は、黙って読み飛ばさず擬似フィールド名として返し、
   720  // 呼び出し側で未分類として loud に落とします。
   721  func collectStructFields(lines []string, name string) ([]string, bool) {
   722  	open := "type " + name + " struct {"
   723  	fields := []string{}
   724  	inside := false
   725  
   726  	for _, line := range lines {
   727  		if !inside {
   728  			inside = line == open
   729  
   730  			continue
   731  		}
   732  		if line == "}" {
   733  			return fields, true
   734  		}
   735  		if trimmed := strings.TrimSpace(line); trimmed == "" || strings.HasPrefix(trimmed, "//") {
   736  			continue
   737  		}
   738  		if m := structFieldRe.FindStringSubmatch(line); m != nil {
   739  			fields = append(fields, m[1])
   740  
   741  			continue
   742  		}
   743  		fields = append(fields, "<解析できない行: "+strings.TrimSpace(line)+">")
   744  	}
   745  
   746  	return nil, false
   747  }
