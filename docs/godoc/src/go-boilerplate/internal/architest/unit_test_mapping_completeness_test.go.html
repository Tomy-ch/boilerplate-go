    17  // 命名規約: メソッドは `Test<Type>_<Method>`（Type が非公開なら `Test_<type>_<Method>`）、
    18  // 自由関数は `Test<Func>`（非公開なら `Test_<func>`）。
    19  var (
    20  	// 型パラメータ節（`[T any]` 等）を許容する。レシーバ型・自由関数名の直後に挟まりうるため、
    21  	// 省略可能な `(?:\[.*\])?` を挟んでからメソッド引数の `(` を要求する。
    22  	// レシーバ変数名は省略可能（`func (*T) M()` は Go として合法で、名前を省くだけで検証を逃れられないようにする）。
    23  	methodDeclRe = regexp.MustCompile(`^func \(\s*(?:\w+\s+)?\*?(\w+)(?:\[.*\])?\) (\w+)\(`)
    24  	freeFuncRe   = regexp.MustCompile(`^func (\w+)(?:\[.*\])?\(`)
    25  	testFuncRe   = regexp.MustCompile(`^func (Test\w+)\(`)
    26  	genHeaderRe  = regexp.MustCompile(`^// Code generated .* DO NOT EDIT\.$`)
    27  )

    35  // TestUnitTestMappingCompleteness は、production code の全 func / method が、命名規約に一致する
    36  // TestXxx を同一パッケージに持つことを機械検証する。テスト対象と TestXxx の 1:1 を強制し、
    37  // レビュー頼みの見落とし（例: keyFunc の防御分岐が呼び出し側テストに埋もれて専用テストが
    38  // 無い状態、ルート登録だけを行う分岐なしの BindHandler が未テストのまま残る状態）を
    39  // loud な失敗に変える。
    40  //
    41  // 「1:1 を採らない」ことは許さない。対象が検証不可能であるために到達できない場合に限り、
    42  // 命名規約どおりの TestXxx を宣言し中で t.Skip("なぜ検証不可能か") することで意図を明示する
    43  // （allowlist は持たない。理由がコード上に残る）。「他のテストでカバー済み」は skip の
    44  // 理由にならない。テスト可能な対象は実テストを書く。
    45  //
    46  // 候補名が複数あるのは非公開シンボルの命名揺れを許容するためであって、両方を並べてよい
    47  // という意味ではないため、候補名の TestXxx を 2 つ以上持つ subject も違反とする。
    48  // 逆方向（候補名に一致しない TestXxx が同じ subject を分担している形）は検知しない。
    49  // 理由と、その形を誰が受け持つかは docs/testing-conventions.md §1 が定める。
    50  func TestUnitTestMappingCompleteness(t *testing.T) {
    51  	t.Parallel()
    52  
    53  	prodByDir, testsByDir := scanPackages(t)
    54  	violations := collectViolations(prodByDir, testsByDir)
    55  	sort.Strings(violations)
    56  	for _, v := range violations {
    57  		t.Log("1:1 違反: " + v)
    58  	}
    59  	t.Logf("total 1:1 violations: %d", len(violations))
    60  
    61  	require.Empty(t, violations,
    62  		"production code の関数/メソッドと TestXxx の 1:1 が崩れている。"+
    63  			"対応する TestXxx が無い場合は専用テストを追加すること。検証不可能であるために到達できない"+
    64  			"対象に限り、命名規約どおりの TestXxx を宣言し、なぜ検証不可能かを t.Skip の理由文に書いて"+
    65  			"明示できる。候補名の TestXxx が重複している場合は、どちらか 1 本へケースを統合すること。")
    66  }

    68  // scanPackages は internal / pkg / scripts 配下を走査し、ディレクトリ単位で
    69  // production code の subject と TestXxx 名を収集して返す。
    70  //
    71  // scripts を含めるのは、そこに置かれた Go ツールがゲート（供給網ピン・連番検査・カバレッジ
    72  // 閾値・負荷帯の解決）であり、壊れ方が「何も検査しなくなる」方向に出るためである。
    73  // カバレッジ母数からは GO_TEST_EXCLUDE で外れているので、1:1 を欠くと対応の有無すら誰も見ない。
    74  func scanPackages(t *testing.T) (map[string][]prodSubject, map[string]map[string]struct{}) {
    75  	t.Helper()
    76  	prodByDir := map[string][]prodSubject{}
    77  	testsByDir := map[string]map[string]struct{}{}
    78  
    79  	for _, root := range moduleSubdirs(t, "internal", "pkg", "scripts") {
    80  		err := filepath.WalkDir(root, func(path string, d fs.DirEntry, werr error) error {
    81  			if werr != nil {
    82  				return werr
    83  			}
    84  			if serr := skipDependencyTree(d); serr != nil {
    85  				return serr
    86  			}
    87  			if d.IsDir() || !strings.HasSuffix(path, ".go") {
    88  				return nil
    89  			}
    90  			src, rerr := pkgfs.OS{}.ReadFile(path)
    91  			if rerr != nil {
    92  				return rerr
    93  			}
    94  			indexGoFile(path, strings.Split(string(src), "\n"), prodByDir, testsByDir)
    95  			return nil
    96  		})
    97  		require.NoError(t, err)
    98  	}
    99  	return prodByDir, testsByDir
   100  }

   102  // indexGoFile は 1 ファイルを分類する。_test.go は TestXxx 名を、production code は分岐持ち subject を収集する。
   103  func indexGoFile(path string, lines []string, prodByDir map[string][]prodSubject, testsByDir map[string]map[string]struct{}) {
   104  	dir := filepath.Dir(path)
   105  	if strings.HasSuffix(path, "_test.go") {
   106  		if testsByDir[dir] == nil {
   107  			testsByDir[dir] = map[string]struct{}{}
   108  		}
   109  		for _, l := range lines {
   110  			if m := testFuncRe.FindStringSubmatch(l); m != nil {
   111  				testsByDir[dir][m[1]] = struct{}{}
   112  			}
   113  		}
   114  		return
   115  	}
   116  	if isGeneratedGo(path, lines) {
   117  		return
   118  	}
   119  	prodByDir[dir] = append(prodByDir[dir], collectSubjects(lines, path)...)
   120  }

   122  // collectViolations は、対応する TestXxx を持たない subject と、候補名の TestXxx を
   123  // 2 つ以上持つ subject を違反として列挙する。
   124  func collectViolations(prodByDir map[string][]prodSubject, testsByDir map[string]map[string]struct{}) []string {
   125  	var violations []string
   126  	for dir, subs := range prodByDir {
   127  		tests := testsByDir[dir]
   128  		for _, s := range subs {
   129  			switch matched := matchingTests(tests, s.candidates); len(matched) {
   130  			case 0:
   131  				violations = append(violations, s.file+": "+s.name+" → 期待テスト名: "+strings.Join(s.candidates, " / "))
   132  			case 1:
   133  			default:
   134  				violations = append(violations, s.file+": "+s.name+" → 候補名の TestXxx が重複: "+strings.Join(matched, " / "))
   135  			}
   136  		}
   137  	}
   138  	return violations
   139  }

   141  // collectSubjects は、宣言されている全ての func / method を subject 化して返す。
   142  // 分岐の有無・body の行数では絞らない（getter / 単純ラッパ / ルート登録のみの関数も
   143  // 契約を持ちうるため）。除外するのは main / init のみ。
   144  func collectSubjects(lines []string, file string) []prodSubject {
   145  	var subs []prodSubject
   146  	for _, line := range lines {
   147  		var name string
   148  		var candidates []string
   149  
   150  		if m := methodDeclRe.FindStringSubmatch(line); m != nil {
   151  			typ, method := m[1], m[2]
   152  			name = typ + "." + method
   153  			candidates = []string{"Test" + typ + "_" + method, "Test_" + typ + "_" + method}
   154  		} else if m := freeFuncRe.FindStringSubmatch(line); m != nil {
   155  			fn := m[1]
   156  			if fn == "main" || fn == "init" {
   157  				continue
   158  			}
   159  			name = fn
   160  			candidates = []string{"Test" + fn, "Test_" + fn}
   161  		} else {
   162  			continue
   163  		}
   164  
   165  		subs = append(subs, prodSubject{name: name, file: file, candidates: candidates})
   166  	}
   167  	return subs
   168  }

   170  // matchingTests は、candidates のうち実在する TestXxx 名を宣言順で返す。
   171  func matchingTests(tests map[string]struct{}, candidates []string) []string {
   172  	var matched []string
   173  	for _, c := range candidates {
   174  		if _, ok := tests[c]; ok {
   175  			matched = append(matched, c)
   176  		}
   177  	}
   178  	return matched
   179  }

   181  // isGeneratedGo は、生成物（ファイル名 or `// Code generated ... DO NOT EDIT.` ヘッダ）を判定する。
   182  func isGeneratedGo(path string, lines []string) bool {
   183  	base := filepath.Base(path)
   184  	if strings.HasSuffix(base, ".gen.go") || strings.HasSuffix(base, ".sql.go") ||
   185  		strings.HasSuffix(base, "_mock.go") || strings.HasSuffix(base, ".pb.go") {
   186  		return true
   187  	}
   188  	for k := 0; k < len(lines) && k < 5; k++ {
   189  		if genHeaderRe.MatchString(lines[k]) {
   190  			return true
   191  		}
   192  	}
   193  	return false
   194  }
