    27  // TestIdempotencyCompleteness は、OpenAPI で Idempotency-Key を宣言した操作
    28  // （= 生成コードの <Op>Params 型が IdempotencyKey フィールドを持つ操作）のハンドラが、
    29  // 必ず idempotency.Run 経由で処理していることを機械検証する完全性テストです。
    30  // OpenAPI のヘッダ宣言を source of truth とし、Run 呼び忘れのような silent な dedup 欠落を loud な失敗に変える。
    31  //
    32  // 対象操作は 0 件でも許容する（サンプル API 削除後は該当が無くなり得る）が、検出ロジックの空振りを
    33  // 防ぐため、param 検出用の正規表現が既知の生成形にマッチすることを別途 assert する。
    34  func TestIdempotencyCompleteness(t *testing.T) {
    35  	t.Parallel()
    36  
    37  	root := handlerRoot(t)
    38  
    39  	marked := map[string]string{}    // operationID -> 宣言元ファイル（Idempotency-Key を宣言する操作）
    40  	wrapped := map[string]struct{}{} // idempotency.Run を呼ぶハンドラメソッド名
    41  	totalParamsSeen := 0             // 走査中に見た `<Op>Params struct` の総数（検出健全性の指標）
    42  
    43  	err := filepath.WalkDir(root, func(path string, d fs.DirEntry, walkErr error) error {
    44  		if walkErr != nil {
    45  			return walkErr
    46  		}
    47  		if d.IsDir() || !strings.HasSuffix(path, ".go") || strings.HasSuffix(path, "_test.go") {
    48  			return nil
    49  		}
    50  
    51  		src, readErr := pkgfs.OS{}.ReadFile(path)
    52  		if readErr != nil {
    53  			return readErr
    54  		}
    55  		lines := strings.Split(string(src), "\n")
    56  
    57  		if strings.HasSuffix(path, ".gen.go") {
    58  			totalParamsSeen += collectMarkedParams(lines, path, marked)
    59  		} else {
    60  			collectRunWrapped(lines, wrapped)
    61  		}
    62  		return nil
    63  	})
    64  	require.NoError(t, err)
    65  
    66  	// マーカー検出ロジックが生成コードの命名変更等で陳腐化して空振りすると、完全性検証が
    67  	// 常に成功してしまう。これを防ぐため、param 検出の正規表現が既知の生成形にマッチすることを
    68  	// 直接検証する。サンプル API を全削除すると param を持つ操作は 0 件になり得る（totalParamsSeen==0）
    69  	// ため、リポジトリ走査の件数ではなく正規表現の自己検査で陳腐化を検出する。
    70  	t.Logf("param 構造体の検出数: %d（サンプル全削除後は 0 になり得る）", totalParamsSeen)
    71  	require.NotEmpty(t, paramsStructRe.FindStringSubmatch("type XxxParams struct {"),
    72  		"paramsStructRe が `<Op>Params struct {` 形にマッチしない（正規表現の陳腐化を疑う）")
    73  
    74  	violations := collectUnwrappedOperations(marked, wrapped)
    75  	sort.Strings(violations)
    76  	for _, v := range violations {
    77  		t.Log("idempotency 未包含: " + v)
    78  	}
    79  
    80  	require.Empty(t, violations,
    81  		"Idempotency-Key を宣言している操作のハンドラが idempotency.Run を呼んでいない。"+
    82  			"ハンドラ本体を idempotency.Run 経由に変えること。")
    83  }

    85  // collectUnwrappedOperations は、Idempotency-Key を宣言していながら idempotency.Run に包まれていない
    86  // 操作を `<operationID>（宣言元ファイル）` 形式で列挙します。
    87  func collectUnwrappedOperations(marked map[string]string, wrapped map[string]struct{}) []string {
    88  	var violations []string
    89  	for op, file := range marked {
    90  		if _, ok := wrapped[op]; ok {
    91  			continue
    92  		}
    93  		violations = append(violations, fmt.Sprintf("%s（%s）", op, file))
    94  	}
    95  	return violations
    96  }

    99  func handlerRoot(t *testing.T) string {
   100  	t.Helper()
   101  	_, thisFile, _, ok := runtime.Caller(0)

   103  	return filepath.Join(filepath.Dir(thisFile), "..", "controller", "handler")

   106  // collectMarkedParams は、`type <Op>Params struct {` ブロックが IdempotencyKey フィールドを
   107  // 含むとき Op を marked へ加えます。gofmt 済みソースでは閉じ括弧が行頭 `}` に現れる前提です。
   108  // 戻り値は、IdempotencyKey の有無に依らず走査中に検出した `<Op>Params struct` の件数で、
   109  // 呼び出し側の検出健全性チェック（正規表現の陳腐化検出）に用います。
   110  func collectMarkedParams(lines []string, file string, marked map[string]string) int {
   111  	seen := 0
   112  	for i, line := range lines {
   113  		m := paramsStructRe.FindStringSubmatch(line)
   114  		if m == nil {
   115  			continue
   116  		}
   117  		seen++
   118  		op := m[1]
   119  		for _, body := range lines[i+1:] {
   120  			if body == "}" {
   121  				break
   122  			}
   123  			if strings.Contains(body, "IdempotencyKey") {
   124  				marked[op] = file
   125  				break
   126  			}
   127  		}
   128  	}
   129  	return seen
   130  }

   132  // collectRunWrapped は、メソッド宣言ブロック内に idempotency.Run 呼び出しがあるとき、その
   133  // メソッド名を wrapped へ加えます。ブロックは行頭 `func ` から次の行頭 `func ` までとします。
   134  //
   135  // 検出は import 別名なしの直接呼び出し `idempotency.Run(` に限る。別名 import（`idem.Run(` 等）や
   136  // 変数へ束ねてからの間接呼び出しは検出できない。現行ハンドラは直接呼び出しのみのため問題ないが、
   137  // この規約から外れると完全性検証が偽陰性になりうる点に注意。
   138  func collectRunWrapped(lines []string, wrapped map[string]struct{}) {
   139  	currentMethod := ""
   140  	for _, line := range lines {
   141  		if m := handlerMethodRe.FindStringSubmatch(line); m != nil {
   142  			currentMethod = m[1]
   143  			continue
   144  		}
   145  		if strings.HasPrefix(line, "func ") {
   146  			currentMethod = "" // メソッド以外の関数宣言に入ったらリセット
   147  			continue
   148  		}
   149  		if currentMethod != "" && strings.Contains(line, "idempotency.Run(") {
   150  			wrapped[currentMethod] = struct{}{}
   151  		}
   152  	}
   153  }

   155  func Test_collectMarkedParams(t *testing.T) {
   156  	t.Parallel()
   157  
   158  	t.Run("正常系", func(t *testing.T) {
   159  		t.Parallel()
   160  
   161  		t.Run("IdempotencyKeyを持つParams構造体のOpを収集する", func(t *testing.T) {
