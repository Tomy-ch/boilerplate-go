    35  // targetStatusCodesPolicy は、OBS_TARGET_STATUS_CODES の環境別ポリシーの宣言です。
    36  // 上の段ほど広く監視し、下の段ほど本番の監視ノイズを避けて絞る単調な絞り込みです。
    37  //
    38  // 各段の期待値は 1 つ上の段の実値から excluded を引いて導出するため、環境間で共通のステータスコードは
    39  // この宣言に写経されません。新しいコードを一部の env ファイルにだけ足す伝播漏れは、導出した期待値との
    40  // 差として失敗します。特定の環境から意図的に外す場合に限り excluded の更新が要り、環境別ポリシーの
    41  // 変更が人手の確認を経ることになります。
    42  var targetStatusCodesPolicy = []statusCodeTier{
    43  	{files: []string{"env/.env", "env/.env.ci", "env/.env.dast"}},
    44  	{files: []string{"env/.env.dev", "env/.env.stg"}, excluded: []string{"429"}},
    45  	{files: []string{"env/.env.prd"}, excluded: []string{"403", "404", "405", "410"}},
    46  }

    56  // TestEnvTargetStatusCodesPolicy は、OBS_TARGET_STATUS_CODES の値が env ファイル間で宣言された
    57  // 環境別ポリシーどおりに分岐していることを機械検証します。
    58  // env ファイルは互いに独立した手書きテキストで、キーの値が環境ごとに違ってよいのか揃うべきなのかを
    59  // 表現する場所が無く、伝播漏れと意図的なポリシーが同じ見た目になります。本テストがその区別を担い、
    60  // 「一部の env にだけ足した」も「揃えるべきでない環境を揃えた」も loud な失敗に変えます。
    61  func TestEnvTargetStatusCodesPolicy(t *testing.T) {
    62  	t.Parallel()
    63  
    64  	root := moduleRoot(t)
    65  	requireEnvValueFilesCoverDir(t, root)
    66  	require.ElementsMatch(t, envValueFiles, policyEnvFiles(),
    67  		"環境別ポリシーの段が挙げる env ファイルが envValueFiles と一致せず、対象から漏れた env ファイルがある")
    68  
    69  	var upper []string
    70  	for i, tier := range targetStatusCodesPolicy {
    71  		want := upper
    72  		if i > 0 {
    73  			want = excludeStatusCodes(t, upper, tier.excluded)
    74  		}
    75  
    76  		for _, file := range tier.files {
    77  			got := readStatusCodes(t, root, file)
    78  			if i == 0 {
    79  				require.NotEmptyf(t, got, "%s の %s が空で、以降の段の検証が空振りする", file, targetStatusCodesKey)
    80  				continue
    81  			}
    82  			assert.Equalf(t, want, got,
    83  				"%s の %s が環境別ポリシー（1 つ上の段から %s を除外）と一致しない。"+
    84  					"意図的な変更なら targetStatusCodesPolicy を更新すること",
    85  				file, targetStatusCodesKey, strings.Join(tier.excluded, ","))
    86  		}
    87  		upper = readStatusCodes(t, root, tier.files[0])
    88  	}
    89  }

   100  // readEnvFileValues は、env ファイル群をキーごとの「ファイル → 値」へ読み替えて返します。
   101  // env/.env だけは readLocalEnv を通します。CI は埋め込みのため作業ツリーの env/.env を対象環境の
   102  // ファイルで上書きしており、そのまま読むと local と当該環境の差が消えて検証が空振りするためです。
   103  func readEnvFileValues(t *testing.T, root string) map[string]map[string]string {
   104  	t.Helper()
   105  
   106  	requireEnvValueFilesCoverDir(t, root)
   107  
   108  	values := map[string]map[string]string{}
   109  	for _, file := range envValueFiles {
   110  		kv := parseEnvFile(t, root, file)
   111  		if file == envLocalFile {
   112  			kv = readLocalEnv(t, root)
   113  		}
   114  
   115  		for key, value := range kv {
   116  			if values[key] == nil {
   117  				values[key] = map[string]string{}
   118  			}
   119  			values[key][file] = value
   120  		}
   121  	}
   122  
   123  	return values
   124  }

   126  // requireEnvValueFilesCoverDir は、envValueFiles が env/ 配下の env ファイル実体を網羅していることを
   127  // 確かめます。env ファイルは増える方向にしか壊れません（減れば読み込みが失敗する）。新しい環境の
   128  // ファイルを足して一覧への追記を忘れると、そのファイルだけ検証対象から静かに外れます。
   129  func requireEnvValueFilesCoverDir(t *testing.T, root string) {
   130  	t.Helper()
   131  
   132  	found, err := pkgfs.OS{}.Glob(filepath.Join(root, "env", ".env*"))
   133  	require.NoError(t, err, "env ファイルの一覧を取得できない")
   134  
   135  	onDisk := make([]string, 0, len(found))
   136  	for _, path := range found {
   137  		onDisk = append(onDisk, filepath.ToSlash(filepath.Join("env", filepath.Base(path))))
   138  	}
   139  
   140  	assert.ElementsMatch(t, envValueFiles, onDisk,
   141  		"env/ 配下の env ファイルと envValueFiles が一致しない。環境を追加したなら envValueFiles にも追記すること")
   142  }

   153  // excludeStatusCodes は、codes から excluded を除いた並びを返します。excluded に codes へ含まれない
   154  // コードがあれば、ポリシー宣言が陳腐化して除外が空振りしているため失敗させます。
   155  func excludeStatusCodes(t *testing.T, codes, excluded []string) []string {
   156  	t.Helper()
   157  
   158  	for _, code := range excluded {
   159  		require.Containsf(t, codes, code,
   160  			"ポリシーが除外する %s が 1 つ上の段に無く、除外指定が空振りしている", code)
   161  	}
   162  
   163  	out := make([]string, 0, len(codes))
   164  	for _, code := range codes {
   165  		if !slices.Contains(excluded, code) {
   166  			out = append(out, code)
   167  		}
   168  	}
   169  	return out
   170  }

   172  // readLocalEnv は、ローカル既定の env 値を返します。CI は埋め込みのため env/.env を
   173  // env/.env.<APP_ENV> で上書きする（make materialize-env）ので、作業ツリーの APP_ENV が
   174  // ローカルでなければコミット済みの内容を読み直します。
   175  func readLocalEnv(t *testing.T, root string) map[string]string {
   176  	t.Helper()
   177  
   178  	values := parseEnvContent(t, envLocalFile, readRepoFile(t, root, envLocalFile))
   179  	if values[appEnvKey] == envLocalProvenance {
   180  		return values
   181  	}
   182  	return parseEnvContent(t, envLocalFile, readCommittedFile(t, root, envLocalFile))
   183  }

   201  // parseEnvContent は、env ファイルの内容をキーと値の対応へ分解して返します。
   202  // アプリ本体のローダー（internal/config）と同じ godotenv で解釈します。独自パーサだと
   203  // クォートやコメントの扱いが分かれ、テストだけが実行時と違う値を見ることになります。
   204  func parseEnvContent(t *testing.T, file, content string) map[string]string {
   205  	t.Helper()
   206  
   207  	values, err := godotenv.Parse(strings.NewReader(content))
   208  	require.NoErrorf(t, err, "%s を dotenv として解釈できない", file)
   209  	require.NotEmptyf(t, values, "%s から代入行を 1 件も抽出できず、検証が空振りする", file)
   210  	return values
   211  }
