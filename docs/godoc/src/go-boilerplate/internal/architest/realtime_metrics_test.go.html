     3  // Realtime Delivery の計装規約（docs/design/realtime-delivery.md §3.4）を機械的に検査する。
     4  //
     5  //  1. instrument 名はすべて `realtime.` で始まり、リテラルで書かれている（feature 中立）。
     6  //  2. label のキーは宣言した 4 つだけで、主体の識別子（subject / stream / event / …）を label にしない。
     7  //  3. `realtime` meter を取るのは検査対象の 1 ファイルだけ（走査の外へ計装が逃げない）。
     8  //
     9  // 規約 2 が本体。ID を label にすると時系列の本数が使う側の都合で際限なく増え、いったん出した
    10  // 系列は取り消せない。lint も coverage もこれを見ないので、ここで止める。
    11  //
    12  // 突き合わせる 2 つの集合は「ソースが宣言している名前 / キー」と「規約が許す名前 / キー」。
    13  // ソース → 規約の向きが規約違反を捕まえ、逆向き（NotEmpty の canary）は走査器が壊れて
    14  // 何も拾わなくなったときに気づくためにある。規約 3 は、その走査対象そのものが逃げるのを止める。
    15  //
    16  // 残る限界: 走査はテキストなので（depguard が `go/ast` を禁じている）、属性の集合を別の場所で
    17  // 組んでから渡す形（`metric.WithAttributeSet(externallyBuiltSet)`）には到達できない。
    18  // キーを直接書く形はすべて拾うが、この 1 形だけは人手のレビューが受け持つ。
    19

    42  // realtimeAllowedLabelKeys は、label に使ってよいキー。いずれも取り得る値が有界。
    43  // ここを増やす変更は設計正本 §3.4 の書き換えを伴うべきなので、人手の確認を通すために宣言で持つ。
    44  var realtimeAllowedLabelKeys = []string{"outcome", "reason", "result", "trigger"}

    46  // instrumentCallRe は、meterBuilder の生成呼び出しから第 1 引数をそのまま捕捉する。
    47  // レシーバ名を固定しないのは、`b` 以外の名前に変えるだけで検査を外せてしまうため。
    48  var instrumentCallRe = regexp.MustCompile(
    49  	`\.(?:counter|gauge|histogram|countHistogram|upDownCounter)\(\s*([^,]+),`,
    50  )
    51  
    52  // rawInstrumentCallRe は、meterBuilder を迂回して OTel の生成子を直接呼ぶ形を捕捉する。
    53  // 迂回されると instrumentCallRe が何も拾わず、名前の検査が丸ごと空振りする。
    54  var rawInstrumentCallRe = regexp.MustCompile(`\.(?:Int64|Float64)(?:Counter|Histogram|Gauge|UpDownCounter)\(`)
    55  
    56  // attributeCallRe は、label のキーを与える呼び出しの第 1 引数を捕捉する。
    57  // `String` だけに絞ると、同じ package の既存作法である `attribute.Int` / リテラル引数が素通りする。
    58  var attributeCallRe = regexp.MustCompile(
    59  	`attribute\.(?:String|Int|Int64|Float64|Bool|StringSlice|IntSlice|Stringer|Key)\(\s*([^,)]+)[,)]`,
    60  )
    61  
    62  // attributeStructKeyRe は、`attribute.KeyValue{Key: …}` の形でキーを与える書き方を捕捉する。
    63  var attributeStructKeyRe = regexp.MustCompile(`attribute\.KeyValue\{\s*Key:\s*([^,}]+)`)
    64  
    65  // labelConstRe は、`attrXxx = "key"` 形式の定数宣言を捕捉する。
    66  var labelConstRe = regexp.MustCompile(`(?m)^\s*(?:const\s+)?([A-Za-z_][A-Za-z0-9_]*)\s*=\s*"([^"]+)"`)
    67  
    68  // stringLiteralRe は、引数がそのまま文字列リテラルである形を捕捉する。
    69  var stringLiteralRe = regexp.MustCompile(`^"([^"]*)"$`)
    70  
    71  // idLikeLabelRe は、主体の識別子とみなす label のキー。
    72  var idLikeLabelRe = regexp.MustCompile(
    73  	`(?i)(^id$|_id$|^id_|stream|subject|destination|ticket|trace|span|event|message|user|instance|connection)`,
    74  )

    77  	t.Parallel()
    78  
    79  	t.Run("正常系", func(t *testing.T) {
    80  		t.Parallel()
    81  
    82  		t.Run("instrument 名はすべてリテラルで realtime. から始まる", func(t *testing.T) {
    83  			t.Parallel()
    84  
    85  			src := readRealtimeMetricsSource(t)
    86  			names, violations := realtimeInstrumentNames(src)
    87  			// 計装は sample ではないので、sample 削除後もこの集合は空にならない。
    88  			require.NotEmpty(t, names, "instrument を 1 つも拾えていない。走査の対象かパターンを疑う")

    98  				"%s の instrument 名が規約（%q 始まりのリテラル）に反している（設計正本 §3.4）",

   101

   106  			assert.NotRegexp(t, rawInstrumentCallRe, readRealtimeMetricsSource(t),
   107  				"OTel の生成子を直接呼ぶと名前の検査を通らない。meterBuilder 経由で宣言すること")
   108  		})
   109  	})

   112  		t.Parallel()

   196  	t.Parallel()

   204  			// meter 名は文字列なので、別ファイルから同じ meter を取れば走査の外で

   206  			root := moduleRoot(t)

   208  			require.NoError(t, err)

   210  
   211  			var holders []string
   212  			for _, path := range matches {
   213  				if strings.HasSuffix(path, "_test.go") {
   214  					continue
   215  				}
   216  
   217  				b, err := pkgfs.OS{}.ReadFile(path)
   218  				require.NoError(t, err)
   219  
   220  				if strings.Contains(string(b), realtimeMeterNameLiteral) {
   221  					rel, err := filepath.Rel(root, path)
   222  					require.NoError(t, err)
   223  					holders = append(holders, rel)
   224  				}
   225  			}
   226  			slices.Sort(holders)
   227  
   228  			assert.Equal(t, []string{realtimeMetricsFile}, holders,
   229  				"realtime meter の宣言が増えると、そのファイルの label は検査されない")
   230  		})

   244  // realtimeInstrumentNames は、ソースから instrument 名を宣言順に拾い、
   245  // 名前を静的に読み取れなかった呼び出しを違反として返します。
   246  func realtimeInstrumentNames(src string) ([]string, []string) {
   247  	var names, violations []string
   248  
   249  	for _, m := range instrumentCallRe.FindAllStringSubmatch(src, -1) {
   250  		arg := strings.TrimSpace(m[1])
   251  
   252  		lit := stringLiteralRe.FindStringSubmatch(arg)
   253  		if lit == nil {
   254  			violations = append(violations, "instrument 名がリテラルでない: "+arg)
   255  
   256  			continue
   257  		}
   258  
   259  		names = append(names, lit[1])
   260  	}
   261  
   262  	return names, violations
   263  }

   265  // realtimeLabelKeys は、ソースから label のキーの実値を重複なく拾い、
   266  // 実値を静的に読み取れなかった呼び出しを違反として返します。
   267  // 引数はリテラルか、このソース内で宣言された定数のどちらかでなければなりません。
   268  func realtimeLabelKeys(src string) ([]string, []string) {
   269  	var keys, violations []string
   270  
   271  	values := map[string]string{}
   272  	for _, m := range labelConstRe.FindAllStringSubmatch(src, -1) {
   273  		values[m[1]] = m[2]
   274  	}
   275  
   276  	var args []string
   277  	for _, re := range []*regexp.Regexp{attributeCallRe, attributeStructKeyRe} {
   278  		for _, m := range re.FindAllStringSubmatch(src, -1) {
   279  			args = append(args, strings.TrimSpace(m[1]))
   280  		}
   281  	}
   282  
   283  	for _, arg := range args {
   284  		key, ok := resolveLabelKey(arg, values)
   285  		if !ok {
   286  			violations = append(violations, "label キーの実値を読み取れない: "+arg)
   287  
   288  			continue
   289  		}
   290  
   291  		if !slices.Contains(keys, key) {
   292  			keys = append(keys, key)
   293  		}
   294  	}
   295  
   296  	return keys, violations
   297  }

   299  // resolveLabelKey は、引数の字面を label キーの実値に解決します。
   300  func resolveLabelKey(arg string, consts map[string]string) (string, bool) {
   301  	if lit := stringLiteralRe.FindStringSubmatch(arg); lit != nil {
   302  		return lit[1], true
   303  	}
   304  
   305  	value, ok := consts[arg]
   306  
   307  	return value, ok
   308  }
