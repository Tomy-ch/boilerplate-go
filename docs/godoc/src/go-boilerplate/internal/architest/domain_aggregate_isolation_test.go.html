   157  // collectDomainAggregateViolations は、domain 配下の production code が持つ規約違反の import を
   158  // `file -> import` の形で返す。
   159  func collectDomainAggregateViolations(root string) ([]string, error) {
   160  	var violations []string
   161  
   162  	err := walkProductionGoFiles(filepath.Join(root, domainRoot), func(path string, content []byte) {
   163  		rel, _ := filepath.Rel(root, path)
   164  		rel = filepath.ToSlash(rel)
   165  
   166  		aggregate := aggregateOfDomainPath(rel)
   167  		if aggregate == "" || aggregate == serviceAggregate {
   168  			return
   169  		}
   170  
   171  		for _, m := range internalImportRe.FindAllStringSubmatch(string(content), -1) {
   172  			if domainImportViolation(aggregate, m[1]) {
   173  				violations = append(violations, rel+" -> "+m[1])
   174  			}
   175  		}
   176  	})
   177  	if err != nil {
   178  		return nil, err
   179  	}
   180  
   181  	return violations, nil
   182  }

   184  // domainImportViolation は、aggregate に属するファイルからの imp が規約違反かを返す。
   185  func domainImportViolation(aggregate, imp string) bool {
   186  	rest, ok := strings.CutPrefix(imp, domainImportPrefix)
   187  	if !ok {
   188  		return false
   189  	}
   190  
   191  	target, sub, _ := strings.Cut(rest, "/")
   192  	switch {
   193  	case target == lexiconAggregate:
   194  		return false
   195  	case target != aggregate:
   196  		return true
   197  	default:
   198  		// 自分の集約の中でも、sub-entity の目印を経由しないネストは別集約とみなす。
   199  		return sub != subEntityMarker && !strings.HasPrefix(sub, subEntityMarker+"/")
   200  	}
   201  }

   203  // aggregateOfDomainPath は、repo root 相対のパスから集約名を返す。domain 配下でなければ空を返す。
   204  func aggregateOfDomainPath(rel string) string {
   205  	rest, ok := strings.CutPrefix(rel, domainRoot+"/")
   206  	if !ok {
   207  		return ""
   208  	}
   209  
   210  	aggregate, _, found := strings.Cut(rest, "/")
   211  	if !found {
   212  		return ""
   213  	}
   214  
   215  	return aggregate
   216  }
