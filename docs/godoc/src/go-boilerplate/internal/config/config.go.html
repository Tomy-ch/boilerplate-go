   348  // validateSecurityConfig は、セキュリティ設定を検証します。
   349  // CIDR は parseCIDR に委ねる。
   350  func validateSecurityConfig(secCfg Security) error {
   351  	if len(secCfg.AllowedOrigins) == 0 {
   352  		return ErrEmptyAllowedOrigins
   353  	}
   354  
   355  	for _, origin := range secCfg.AllowedOrigins {
   356  		parsedURL, err := url.Parse(origin)
   357  		if err != nil {
   358  			return ErrHTTPOnlyAllowedForLocalhost
   359  		}
   360  		// スキームは url.Parse で小文字正規化されるが、念のため EqualFold で大小無視判定する。
   361  		if strings.EqualFold(parsedURL.Scheme, "http") &&
   362  			parsedURL.Hostname() != "localhost" && parsedURL.Hostname() != "127.0.0.1" {
   363  			return ErrHTTPOnlyAllowedForLocalhost
   364  		}
   365  	}
   366  
   367  	return nil
   368  }

   370  // parseCIDR は、CIDR 文字列を *net.IPNet へ解析します。解析が検証を兼ねる（parse, don't validate）ので
   371  // validateConfig には CIDR の検査を置かず、値が要る New が一度だけ呼んで結果を Config へ格納します。
   372  func parseCIDR(s string) (*net.IPNet, error) {
   373  	_, cidr, err := net.ParseCIDR(s)
   374  	if err != nil {
   375  		return nil, xerrors.Join(ErrFailedToParseCIDR, err)
   376  	}
   377  	return cidr, nil
   378  }
