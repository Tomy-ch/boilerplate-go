   570  // testDBName は、ホストから見たテスト用データベース名を返します。環境変数 DB_NAME_TEST があればそれを、
   571  // 無ければ既定の "test" を返します。DB スロットプール利用時は make が DB_NAME_TEST を各 worktree の
   572  // テスト用データベース（wt<N>_test）へ設定するため、共有 DB 内の自 worktree DB へ繋ぎます。ただし
   573  // deploy 系 env では本番 DB を誤指しないよう DB_NAME_TEST を無視します（IsLocalClassEnv、未設定は許可）。
   574  func testDBName() string {
   575  	v := os.Getenv("DB_NAME_TEST")
   576  	if v == "" {
   577  		return defaultTestDBName
   578  	}
   579  	if env := os.Getenv("APP_ENV"); env != "" && !IsLocalClassEnv(env) {
   580  		fmt.Fprintf(os.Stderr,
   581  			"[config] 警告: APP_ENV=%q は local/test 系でないため、DB スロットプールの DB_NAME_TEST=%q を無視します\n",
   582  			env, v)
   583  		return defaultTestDBName
   584  	}
   585  	return v
