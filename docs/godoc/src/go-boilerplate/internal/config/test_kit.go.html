    95  // setWorktreeDBName は、DB スロットプール利用時に対象 env の DB_NAME を自 worktree のデータベースへ
    96  // 上書きする。local は DB_NAME_LOCAL（wt<N>_local）、ci/test は DB_NAME_TEST（wt<N>_test）。deploy 系
    97  // env では本番 DB を誤指しないよう無視する（IsLocalClassEnv）。未使用時は .env.<env> の値のまま。
    98  func setWorktreeDBName(t *testing.T, env string) {
    99  	t.Helper()
   100  	switch {
   101  	case env == EnvLocal && IsLocalClassEnv(env):
   102  		if v := os.Getenv("DB_NAME_LOCAL"); v != "" {
   103  			t.Setenv("DB_NAME", v)
   104  		}
   105  	case IsLocalClassEnv(env):
   106  		t.Setenv("DB_NAME", testDBName())
   107  	case os.Getenv("DB_NAME_TEST") != "" || os.Getenv("DB_NAME_LOCAL") != "":
   108  		fmt.Fprintf(os.Stderr,
   109  			"[config] 警告: env=%q は local/test 系でないため、DB スロットプールの向き先変更を無視します\n", env)
   110  	}
   111  }

   113  // repoRoot は、go.mod を上方向に探索してリポジトリルートを返します。
   114  func repoRoot(t *testing.T) string {
   115  	t.Helper()
   116  
   117  	p, err := os.Getwd()
   118  	require.NoError(t, err)
   119  	for {
   120  		if _, err := os.Stat(filepath.Join(p, "go.mod")); err == nil {
   121  			return p
   122  		}
   123  		parent := filepath.Dir(p)
   124  		require.NotEqual(t, parent, p, "リポジトリルート（go.mod）が見つかりません")
   125  		p = parent
   126  	}
   127  }

   129  // NewRealtimeTestConnection は、contract test の接続先を返します。既定は DynamoDB Local
   130  // （`http://localhost:8000`）と GoAWS（`http://localhost:4100`）で、静的資格情報はダミーです。
   131  // endpoint の 2 変数と資格情報の 2 変数を空文字で設定し REALTIME_TEST_REGION を与えると、

   134  func NewRealtimeTestConnection(t *testing.T) RealtimeTestConnection {
   135  	t.Helper()
   136  
   137  	lookup := func(key, fallback string) string {
   138  		if v, ok := os.LookupEnv(key); ok {
   139  			return v
   140  		}
   141  
   142  		return fallback
   143  	}
   144  
   145  	return RealtimeTestConnection{
   146  		Endpoint:        lookup(realtimeTestEndpointEnvKey, defaultRealtimeTestEndpoint),
   147  		PubSubEndpoint:  lookup(realtimeTestPubSubEndpointKey, defaultRealtimeTestPubSub),
   148  		Region:          lookup(realtimeTestRegionEnvKey, defaultRealtimeTestRegion),
   149  		AccessKeyID:     lookup(realtimeTestAccessKeyEnvKey, defaultRealtimeTestCredential),
