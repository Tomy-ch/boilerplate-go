    25  // Endpoint は、このアプリが接続する外部サービスの所在をまとめて保持する。
    26  // この軸を分けている理由は env/README.md の "Endpoint" 節にある。
    27  // 全項目 required だが空文字を許容する。空の意味は接続先ごとに異なるため各フィールドに記す。
    28  type Endpoint struct {
    29  	// OTLP は OpenTelemetry Collector の送出先です。空なら送出しません。
    30  	OTLP string `env:"OTLP,required"`
    31  	// JWKS は公開鍵の取得先です。空なら OIDC discovery で issuer から解決します。
    32  	JWKS string `env:"JWKS,required"`
    33  	// ObjectStorage は S3 互換エンドポイントです。空なら SDK 既定の解決に委ねます（本番 AWS S3 等）。
    34  	ObjectStorage string `env:"OBJECT_STORAGE,required"`
    35  	// Realtime は Realtime Delivery の store（DynamoDB 互換）のエンドポイントです。空なら SDK 既定の解決に委ねます（本番 DynamoDB）。
    36  	Realtime string `env:"REALTIME,required"`
    37  	// RealtimePubSub は Realtime Delivery の fan-out（SNS / SQS 互換）のエンドポイントです。空なら SDK 既定の解決に委ねます（本番 SNS / SQS）。
    38  	RealtimePubSub string `env:"REALTIME_PUBSUB,required"`
    39  	// Outbox は PUBLISHER=http のときの送出先です。空のまま http を選ぶと DI で起動エラーにします。
    40  	Outbox string `env:"OUTBOX,required"`
    41  	// OutboxQueue は publish 端の SQS 互換エンドポイントです。空なら SDK 既定の解決に委ねます。
    42  	OutboxQueue string `env:"OUTBOX_QUEUE,required"`
    43  	// ConsumerQueue は consume 端の SQS 互換エンドポイントです。空なら SDK 既定の解決に委ねます。
    44  	ConsumerQueue string `env:"CONSUMER_QUEUE,required"`
    45  	// sample-api:begin
    46  	// ExchangeRate は外部為替レートサービスのベース URL です。空ならこの機能を使いません。
    47  	ExchangeRate string `env:"EXCHANGE_RATE,required"`
    48  	// sample-api:end

    64  // Realtime は、Realtime Delivery の EventLog / StreamTicket / InstanceLease を置く DynamoDB 互換 store と、
    65  // serve instance への fan-out（SNS / SQS 互換）の接続設定を保持する。中立境界の実装は DynamoDB / SNS / SQS の
    66  // adapter だが、env 名は vendor 非依存にする。
    67  // table 名は固定名（realtime_event_log 等）に TableSuffix を付けた形で、queue 名は QueuePrefix に instance の
    68  // 識別子を付けた形で、どちらも環境ごとに分かれる。
    69  type Realtime struct {
    70  	Region string `env:"REGION,required,notEmpty"`
    71  	// TableSuffix は table 名の末尾に付く環境識別子（例 local / ci / prd）。小文字・数字・アンダースコアで書く。
    72  	TableSuffix string `env:"TABLE_SUFFIX,required,notEmpty"`
    73  	// Topic は wakeup と失効通知を載せる topic の識別子（AWS では ARN）。deployment が用意する resource で、
    74  	// 空のまま fan-out を配線すると DI で起動エラーにする（ENDPOINT_OUTBOX と同じ扱い）。
    75  	Topic string `env:"TOPIC" envDefault:""`
    76  	// QueuePrefix は instance queue 名の先頭（例 realtime-local）。英数字・`-`・`_` で書く。
    77  	QueuePrefix string `env:"QUEUE_PREFIX,required,notEmpty"`
    78  	// DLQ は instance queue の redrive 先の識別子（AWS では ARN）。空なら RedrivePolicy を付けない。
    79  	DLQ string `env:"DLQ" envDefault:""`
    80  	// AccessKeyID / SecretAccessKey は ObjectStorage と同じ既定（両方空なら未設定を許す）。
    81  	AccessKeyID     string `env:"ACCESS_KEY_ID"     envDefault:""`
    82  	SecretAccessKey string `env:"SECRET_ACCESS_KEY" envDefault:""`
    83  	// MaxConnections は 1 instance が同時に保持する SSE 接続数の上限。負荷と機材で決まるため配備ごとに変わる。
    84  	MaxConnections int `env:"MAX_CONNECTIONS" envDefault:"1000"`
    85  	// ReplayConcurrency は replay と catch-up が 1 instance で同時に走る本数の上限。
    86  	ReplayConcurrency int `env:"REPLAY_CONCURRENCY" envDefault:"16"`
    87  }
