    42  	ctx, endSpan := h.tracer.Start(ctx)

    52  			if ctx.Err() != nil {
    53  				return nil
    54  			}
    55

    57  			h.metrics.LeaseHeartbeatFailed(ctx)
    58  		}
    59

    61  			return nil
    62  		}
    63  	}
