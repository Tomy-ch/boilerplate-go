    43  func newEngine(t *testing.T, set Settings) (*Engine, engineMocks) {
    44  	t.Helper()
    45  
    46  	ctrl := gomock.NewController(t)
    47  	m := engineMocks{
    48  		sub:           mock_realtime.NewMockInstanceSubscription(ctrl),
    49  		reprovisioner: mock_ctrlrealtime.NewMockReprovisioner(ctrl),
    50  		fanout:        mock_ctrlrealtime.NewMockFanoutObserver(ctrl),
    51  		sleeper:       mock_clock.NewMockSleeper(ctrl),
    52  		waker:         mock_ctrlrealtime.NewMockWaker(ctrl),
    53  		revoker:       mock_ctrlrealtime.NewMockRevoker(ctrl),
    54  		sinks:         &sinkRecord{wakeups: map[rt.StreamID][]rt.Sequence{}},
    55  	}
    56  	// 既定では作り直しを呼ばせない。AnyTimes で吸収すると、Run から repairIfGone の呼び出しを
    57  	// 削除する変更が全テスト緑のまま通ってしまう。呼ばれる想定のテストが個別に EXPECT を置く。
    58  	m.waker.EXPECT().Wake(gomock.Any(), gomock.Any(), gomock.Any()).
    59  		Do(func(_ context.Context, streamID rt.StreamID, upTo rt.Sequence) {
    60  			m.sinks.mu.Lock()
    61  			defer m.sinks.mu.Unlock()
    62  
    63  			m.sinks.wakeups[streamID] = append(m.sinks.wakeups[streamID], upTo)
    64  		}).AnyTimes()
    65  	m.revoker.EXPECT().Revoke(gomock.Any(), gomock.Any(), gomock.Any()).
    66  		Do(func(_ context.Context, subject string, destination rt.StreamID) {
    67  			m.sinks.mu.Lock()
    68  			defer m.sinks.mu.Unlock()
    69  
    70  			m.sinks.revocations = append(m.sinks.revocations, rt.Revocation{Subject: subject, Destination: destination})
    71  		}).AnyTimes()
    72  
    73  	m.fanout.EXPECT().ObserveFanout(gomock.Any()).AnyTimes()
    74  
    75  	log, logs := logging.NewObservedTestLogger(t)
    76  	m.logCount = func(msg string) int {
    77  		if msg == "" {
    78  			return logs.Len()
    79  		}
    80  
    81  		return logs.FilterMessage(msg).Len()
    82  	}
    83  
    84  	return NewEngine(m.sub, m.reprovisioner, m.fanout, m.waker, m.revoker, m.sleeper, log,
    85  		observability.NewNoopTracerFactory(t), observability.NewNoopRealtimeMetrics(t), set), m
    86  }
