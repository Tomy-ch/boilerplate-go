    53  // Run は、poll ループ本体です。ctx 完了まで常駐し、完了時に nil を返します。
    54  // 待機判断（連続消化 / PollInterval / ErrorBackoff）の表と根拠は README の Loop semantics を参照。
    55  // 待機は clock.Sleeper 経由で行い、ctx 完了で即座に抜けます。
    56  func (e *Engine) Run(ctx context.Context) error {
    57  	ctx, endSpan := e.tracer.Start(ctx)
    58  	defer endSpan()
    59  
    60  	log := e.logging.Named(relayLoggerName)
    61  	for {
    62  		if ctx.Err() != nil {
    63  			return nil
    64  		}
    65  
    66  		res, err := e.uc.RelayBatch(ctx, e.set.BatchSize)
    67  		if err != nil {
    68  			if ctx.Err() != nil {
    69  				return nil
    70  			}
    71  			log.Error(ctx, "outbox relay batch failed", logging.Error(logging.JobErrorKey, err))
    72  			if e.waitDone(ctx, e.set.ErrorBackoff) {
    73  				return nil
    74  			}
    75  			continue
    76  		}
    77  
    78  		e.observeLag(ctx, log)
    79  
    80  		if res.Claimed >= int(e.set.BatchSize) && res.Published > 0 {
    81  			continue
    82  		}
    83  		if e.waitDone(ctx, e.set.PollInterval) {
    84  			return nil
    85  		}
    86  	}
    87  }

    89  // waitDone は、d 待機します。ctx 完了で待機が中断された場合に true を返します。
    90  func (e *Engine) waitDone(ctx context.Context, d time.Duration) bool {
    91  	return e.sleeper.Sleep(ctx, d) != nil
    92  }

    94  // observeLag は、outbox lag(SLI) と停止中ストリーム数をベストエフォートで記録します。
    95  // 記録の詳細は README の Observability を参照。
    96  func (e *Engine) observeLag(ctx context.Context, log logging.Logger) {
    97  	if ctx.Err() != nil {
    98  		return
    99  	}
   100  	if err := e.uc.RecordLag(ctx); err != nil {
   101  		log.Error(ctx, "failed to record outbox lag", logging.Error(logging.JobErrorKey, err))
   102  	}
   103  	if err := e.uc.RecordBlockedStreams(ctx); err != nil {
   104  		log.Error(ctx, "failed to record blocked outbox streams", logging.Error(logging.JobErrorKey, err))
   105  	}
   106  }
