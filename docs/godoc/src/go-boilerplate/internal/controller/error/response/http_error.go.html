   173  // lookupErrorMetaByAppError は、アプリケーションエラーに対応するエラーメタデータを取得します。
   174  // 既知のエラー型に一致しない場合は 500 Internal Server Error のメタを返します。
   175  func lookupErrorMetaByAppError(err error) httpErrorMeta {
   176  	switch {
   177  	case xerrors.Is(err, apperror.ErrInvalidArgument):
   178  		return lookupErrorMetaByHTTPStatus(http.StatusBadRequest)
   179  	case xerrors.Is(err, apperror.ErrValidation):
   180  		return lookupErrorMetaByHTTPStatus(http.StatusUnprocessableEntity)
   181  	case xerrors.Is(err, apperror.ErrUnsupportedMediaType):
   182  		return lookupErrorMetaByHTTPStatus(http.StatusUnsupportedMediaType)
   183  	case xerrors.Is(err, apperror.ErrPayloadTooLarge):
   184  		return lookupErrorMetaByHTTPStatus(http.StatusRequestEntityTooLarge)
   185  	case xerrors.Is(err, apperror.ErrUnauthenticated):
   186  		return lookupErrorMetaByHTTPStatus(http.StatusUnauthorized)
   187  	case xerrors.Is(err, apperror.ErrPermissionDenied):
   188  		return lookupErrorMetaByHTTPStatus(http.StatusForbidden)
   189  	case xerrors.Is(err, apperror.ErrNotFound):
   190  		return lookupErrorMetaByHTTPStatus(http.StatusNotFound)
   191  	case xerrors.Is(err, apperror.ErrConflict):
   192  		return lookupErrorMetaByHTTPStatus(http.StatusConflict)
   193  	case xerrors.Is(err, apperror.ErrGone):
   194  		return lookupErrorMetaByHTTPStatus(http.StatusGone)
   195  	case xerrors.Is(err, apperror.ErrCanceled):
   196  		return lookupErrorMetaByHTTPStatus(statusClientClosedRequest)
   197  	case xerrors.Is(err, apperror.ErrUnavailable):
   198  		return lookupErrorMetaByHTTPStatus(http.StatusServiceUnavailable)
   199  	case xerrors.Is(err, apperror.ErrUnimplemented):
   200  		return lookupErrorMetaByHTTPStatus(http.StatusNotImplemented)
   201  	case xerrors.Is(err, apperror.ErrTooManyRequests):
   202  		return lookupErrorMetaByHTTPStatus(http.StatusTooManyRequests)
   203  	case xerrors.Is(err, apperror.ErrInternal):
   204  		return lookupErrorMetaByHTTPStatus(http.StatusInternalServerError)
