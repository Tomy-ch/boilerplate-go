    14  const (
    15  	frameEnd        = "\n\n"
    16  	idField         = "id: "
    17  	dataField       = "data: "
    18  	controlPreamble = "event: control\n"
    19  	heartbeatFrame  = ": heartbeat" + frameEnd

    22  // sseWriter は、確定済みのレスポンスへ SSE のフレームを 1 つずつ書きます。
    23  // 書き込みのたびに write deadline を張り直すので、接続そのものは maximum lifetime まで生き続け、
    24  // 止まった peer だけが deadline で切れます（http.Server の WriteTimeout はレスポンス単位なので、
    25  // 張り直さなければ server の WriteTimeout で stream 全体が切れます）。
    26  type sseWriter struct {
    27  	res      http.ResponseWriter
    28  	ctrl     *http.ResponseController
    29  	deadline time.Duration
    30  }
    31  
    32  // newSSEWriter は、res へ書く sseWriter を生成します。deadline は 1 回の書き込みに与える猶予です。
    33  func newSSEWriter(res http.ResponseWriter, deadline time.Duration) *sseWriter {
    34  	return &sseWriter{res: res, ctrl: http.NewResponseController(res), deadline: deadline}
    35  }

    37  // commit は、SSE のレスポンスヘッダを送ります。これ以降 HTTP ステータスは変えられず、
    38  // client へ伝えられるのは in-band の control event と切断だけになります。
    39  func (w *sseWriter) commit() error {
    40  	h := w.res.Header()
    41  	h.Set("Content-Type", "text/event-stream")
    42  	h.Set("Cache-Control", "no-store")
    43  	h.Set("X-Accel-Buffering", "no")
    44  	w.res.WriteHeader(http.StatusOK)
    45  
    46  	return w.flush()
    47  }

    49  // writeEvent は、business event を 1 フレーム書きます。SSE の id は sequence です（id を持つフレームは
    50  // これだけ — writeControl 参照）。
    51  func (w *sseWriter) writeEvent(e rt.DeliveryEvent) error {
    52  	body, err := e.MarshalJSON()
    53  	if err != nil {
    54  		return err
    55  	}
    56  
    57  	return w.write(idField + e.Sequence.String() + "\n" + dataField + string(body) + frameEnd)
    58  }
    59  
    60  // writeControl は、control event を 1 フレーム書きます。id を持ちません — id を持つのは business event
    61  // （writeEvent）だけで、持たせると client の Last-Event-ID が制御指示で上書きされ、再接続の位置が
    62  // business event の列から外れます（設計 §4.3）。heartbeat も同じ理由で id を持ちません。
    63  func (w *sseWriter) writeControl(ev gen.ControlEvent) error {
    64  	body, err := json.Marshal(ev)
    65  	if err != nil {
    66  		return xerrors.Wrap(err, "marshal control event")
    67  	}
    68  
    69  	return w.write(controlPreamble + dataField + string(body) + frameEnd)
    70  }
    71  
    72  // writeHeartbeat は、comment を 1 フレーム書きます（id を持たない理由は writeControl 参照）。到達しない peer をここで検出します。
    73  func (w *sseWriter) writeHeartbeat() error {
    74  	return w.write(heartbeatFrame)
    75  }

    77  // write は、deadline を張り直してから frame を書き、flush します。
    78  func (w *sseWriter) write(frame string) error {
    79  	if err := w.ctrl.SetWriteDeadline(time.Now().Add(w.deadline)); err != nil {
    80  		return xerrors.Wrap(err, "set write deadline")
    81  	}
    82  
    83  	if _, err := w.res.Write([]byte(frame)); err != nil {
    84  		return xerrors.Wrap(err, "write sse frame")
    85  	}
    86  
    87  	return w.flush()
    88  }
    89  
    90  // flush は、書いた分を client へ押し出します。proxy と Go の buffer を跨いで即時性を保ちます。
    91  // echo.Response 経由ではここがエラーを返すことはありません（echo は flush を支えない writer に対して
    92  // panic し、それ以外の失敗は握り潰すため）。返り値は素の ResponseWriter を渡したときだけ意味を持ちます。
    93  func (w *sseWriter) flush() error {
    94  	if err := w.ctrl.Flush(); err != nil {
    95  		return xerrors.Wrap(err, "flush sse frame")
    96  	}
    97  
    98  	return nil
    99  }
