    40  // GetStream は、接続を検証してから Streamer へ渡します。拒否はすべてレスポンスを確定する前に返します。
    41  // 拒否の内訳と HTTP ステータスは README.md「What happens on connect」を参照。
    42  func (s *server) GetStream(c *echo.Context, destination gen.StreamDestinationParam, params gen.GetStreamParams) error {
    43  	ctx, endSpan := s.tracer.Start(c.Request().Context())
    44  	defer endSpan()
    45  
    46  	grant, err := ctxhelper.RequireStreamGrant(ctx)
    47  	if err != nil {
    48  		return err
    49  	}
    50  
    51  	if grant.Destination != rt.StreamID(destination) {
    52  		return xerrors.Wrap(ucrealtime.ErrTicketInvalid, "ticket is bound to another destination")
    53  	}
    54  
    55  	cursor, err := resolveCursor(params.LastEventID, params.After, grant.InitialCursor)
    56  	if err != nil {
    57  		return apperror.WithMeta(err, apperror.NewMeta(codeInvalidStreamCursor))
    58  	}
    59  
    60  	if err := s.cursors.Validate(ctx, grant.Destination, cursor); err != nil {
    61  		if xerrors.Is(err, ucrealtime.ErrCursorExpired) {
    62  			return apperror.WithMeta(xerrors.Join(apperror.ErrGone, err), apperror.NewMeta(codeStreamCursorExpired))
    63  		}
    64  
    65  		// EventLog に届かないための拒否も、容量の拒否と同じく再試行できる 503 として返す。
    66  		if xerrors.Is(err, apperror.ErrUnavailable) {
    67  			hintRetryAfter(c)
    68  		}
    69  
    70  		return err
    71  	}
    72  
    73  	c.SetRequest(c.Request().WithContext(ctx))
    74  
    75  	revalidate, _ := ctxhelper.GetStreamRevalidator(ctx)
    76  
    77  	return s.streamer.Stream(c, StreamRequest{
    78  		Subject: grant.Subject, Destination: grant.Destination, Scope: grant.Scope, Cursor: cursor,
    79  		Resumed:    params.LastEventID != nil || params.After != nil,
    80  		Revalidate: revalidate,
    81  	})
    82  }
