    55  // captureSSE は、実 network 上で body を走らせ、client が受け取った生のバイト列を返します。
    56  // httptest.ResponseRecorder では write deadline も flush も検証できないため、実サーバーを立てます。
    57  func captureSSE(t *testing.T, body func(w *sseWriter) error) (string, error) {
    58  	t.Helper()
    59  
    60  	var writeErr error
    61  	done := make(chan struct{})
    62  	srv := httptest.NewServer(http.HandlerFunc(func(res http.ResponseWriter, _ *http.Request) {
    63  		defer close(done)
    64  		writeErr = body(newSSEWriter(res, sseTestDeadline))
    65  	}))
    66  	t.Cleanup(srv.Close)
    67  
    68  	req, err := http.NewRequestWithContext(context.Background(), http.MethodGet, srv.URL, nil)
    69  	require.NoError(t, err)
    70  	res, err := srv.Client().Do(req)
    71  	require.NoError(t, err)
    72  	t.Cleanup(func() { _ = res.Body.Close() })
    73  
    74  	raw, err := io.ReadAll(res.Body)
    75  	require.NoError(t, err)
    76  	<-done
    77  
    78  	return string(raw), writeErr
    79  }
