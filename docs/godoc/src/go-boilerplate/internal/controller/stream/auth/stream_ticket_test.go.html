   108

   114  		t.Run("検証にはバリデータの引数ではなくリクエストのcontextを渡す", func(t *testing.T) {
   115  			t.Parallel()
   116  			ctrl := gomock.NewController(t)
   117  			v := mock_realtime.NewMockTicketVerifier(ctrl)
   118  			type ctxKey struct{}
   119  			in := newInput(t, "/v1/streams/stream-1?ticket=raw-ticket", apiKeyScheme, true)
   120  			req := in.RequestValidationInput.Request
   121  			in.RequestValidationInput.Request = req.WithContext(context.WithValue(req.Context(), ctxKey{}, "request"))
   122  			v.EXPECT().Verify(gomock.Any(), "raw-ticket", rt.StreamID("stream-1")).
   123  				DoAndReturn(func(ctx context.Context, _ string, _ rt.StreamID) (rt.StreamGrant, error) {
   124  					assert.Equal(t, "request", ctx.Value(ctxKey{}))
   125  					return rt.StreamGrant{}, nil
   126  				})
   127  
   128  			require.NoError(t, New(v).Authenticate(context.WithValue(context.Background(), ctxKey{}, "validator"), in))
   129  		})
