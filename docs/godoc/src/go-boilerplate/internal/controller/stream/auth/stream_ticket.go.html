    47  // Authenticate は、scheme が宣言した query パラメータの ticket を path の destination に対して検証し、
    48  // 通れば束縛を StreamGrant スロットへ書き込みます。ticket の生値はエラーにも context にも残しません
    49  // — 後段が受け取るのは、同じ ticket をもう一度検証する関数値だけです。
    50  func (s *streamTicket) Authenticate(_ context.Context, input *openapi3filter.AuthenticationInput) error {
    51  	if input.SecurityScheme == nil || input.SecurityScheme.Name == "" {
    52  		return ErrSchemeDeclarationMissing
    53  	}
    54  
    55  	in := input.RequestValidationInput
    56  	// OpenAPI バリデータが渡す context は context.Background() から組み立てられており、
    57  	// スパン・deadline・キャンセルのいずれも持たない。検証は request の予算の内側で行う。
    58  	ctx := in.Request.Context()
    59  
    60  	value := in.GetQueryParams().Get(input.SecurityScheme.Name)
    61  	destination := rt.StreamID(in.PathParams[destinationParam])
    62  
    63  	//nolint:contextcheck // 引数の context ではなく input が内包する request の context を用いるため
    64  	grant, err := s.verifier.Verify(ctx, value, destination)
    65  	if err != nil {
    66  		return err
    67  	}
    68  
    69  	//nolint:contextcheck // input が内包する request の context のスロットへ書き戻すため
    70  	if !ctxhelper.SetStreamGrant(ctx, grant) {
    71  		return ErrStreamGrantSlotNotFound
    72  	}
    73  
    74  	//nolint:contextcheck // 同上
    75  	if !ctxhelper.SetStreamRevalidator(ctx, func(ctx context.Context) error {
    76  		_, err := s.verifier.Verify(ctx, value, destination)
    77  
    78  		return err
    79  	}) {
    80  		return ErrStreamGrantSlotNotFound
    81  	}
    82  
    83  	return nil
    84  }
