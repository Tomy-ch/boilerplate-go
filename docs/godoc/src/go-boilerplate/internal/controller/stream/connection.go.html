    67  // connection は、確定済みの 1 本の SSE 接続です。読む側（fetcher）と書く側（pump）の 2 つの goroutine が
    68  // channel だけで繋がります（README「Runtime」）。
    69  type connection struct {
    70  	// id は、registry の索引に使う instance 内で一意な番号です。
    71  	id uint64
    72  	// subject は、ticket が束縛された subject です。失効通知はこれで接続を引きます。
    73  	subject string
    74  	// stream は、この接続が配信する stream です。wakeup はこれで接続を引きます。
    75  	stream rt.StreamID
    76  	// openedAt は、索引に載った時刻です。閉じるときに滞在時間へ変えます。
    77  	openedAt time.Time
    78  
    79  	// events は、読んだ event を送出まで待たせる buffer です。溢れたら接続を閉じます。
    80  	events chan rt.DeliveryEvent
    81  	// control は、client へ渡す最後の指示を 1 つだけ積みます。到達は保証されません（設計 §4.3）。
    82  	control chan gen.ControlEvent
    83  	// signal は、wakeup が届いたことを fetcher へ伝えます。
    84  	signal chan struct{}
    85  	// pending は、wakeup が伝えてきた最大の位置です。fetcher が追いついていれば読み直しません。
    86  	pending atomic.Int64
    87  
    88  	// quit は、接続を閉じると決まったときに閉じられます。
    89  	quit chan struct{}
    90  	// done は、pump が戻って接続が実際に終わったときに閉じられます。drain はこれを待ちます。
    91  	done chan struct{}
    92  
    93  	quitOnce sync.Once
    94  	// reason は、最初に close を呼んだ側が記録した理由です。quitOnce の完了後にだけ読みます。
    95  	reason string

   164  // wake は、stream の位置が upTo まで進んだことを伝えます。重複と後戻りは冪等に吸収し、
   165  // batch をまたぐ通知は pending の最大値へ畳まれます（ADR-0073）。
   166  func (c *connection) wake(upTo rt.Sequence) {
   167  	for {
   168  		cur := c.pending.Load()
   169  		if int64(upTo) <= cur {
   170  			break
   171  		}
   172  
   173  		if c.pending.CompareAndSwap(cur, int64(upTo)) {
   174  			break
   175  		}
   176  	}
   177  
   178  	select {
   179  	case c.signal <- struct{}{}:
   180  	default:
   181  	}
   182  }

   184  // startTicker は、sleeper で interval() ごとに tick を送る goroutine を起こします。
   185  // 受け手が遅れた分の tick は畳まれ、goroutine は詰まりません。ctx が終わると goroutine も終わります。
   186  func startTicker(ctx context.Context, sleeper clock.Sleeper, interval func() time.Duration) <-chan struct{} {
   187  	ch := make(chan struct{}, 1)
   188  
   189  	go func() {
   190  		for {
   191  			if err := sleeper.Sleep(ctx, interval()); err != nil {
   192  				return
   193  			}
   194  
   195  			select {
   196  			case ch <- struct{}{}:
   197  			default:
   198  			}
   199  		}
   200  	}()
   201  
   202  	return ch
   203  }

   222  // run は、初回 replay を Stream が確保した枠のまま走らせ、以降は wakeup と周期 catch-up で読み直します。
   223  func (f *fetcher) run(ctx context.Context) {
   224  	defer f.releaseSlot()
   225  
   226  	f.drainPages(ctx, triggerInitial)
   227  	f.releaseSlot()
   228  
   229  	catchUp := startTicker(ctx, f.sleeper, jitteredCatchUp)
   230  	for {
   231  		trigger, ok := f.awaitTrigger(ctx, catchUp)
   232  		if !ok {
   233  			return
   234  		}
   235  
   236  		// CatchUpLag は通知を受けてから追いつき終わるまで — client から見た遅れ — なので、計測は枠を取る前から
   237  		// 始めます。枠待ちも設計 §2.6 が縮退の一形態として挙げる遅れです。
   238  		startedAt := time.Now()
   239  
   240  		if err := f.sem.Acquire(ctx, 1); err != nil {
   241  			return
   242  		}
   243  
   244  		f.drainPages(ctx, trigger)
   245  		f.sem.Release(1)
   246  
   247  		if trigger == triggerWakeup {
   248  			f.metrics.CatchUpLag(ctx, float64(time.Since(startedAt).Milliseconds()))
   249  		}
   250  	}
   251  }

   253  // awaitTrigger は、次に読み直す契機を待ち、その契機を返します。接続が終わるなら false を返します。
   254  // wakeup は既に追いついていれば読み直さず、周期 catch-up は通知の欠落を埋めるため常に読みます。
   255  func (f *fetcher) awaitTrigger(ctx context.Context, catchUp <-chan struct{}) (string, bool) {
   256  	for {
   257  		select {
   258  		case <-ctx.Done():
   259  			return "", false
   260  		case <-f.conn.quit:
   261  			return "", false
   262  		case <-catchUp:
   263  			return triggerCatchUp, true
   264  		case <-f.conn.signal:
   265  			if f.conn.pending.Load() > int64(f.fetched) {
   266  				return triggerWakeup, true
   267  			}
   268  		}
   269  	}
   270  }

   272  // drainPages は、追いつくまで EventLog を読み進めます。semaphore の枠を持った状態で呼びます。
   273  // trigger は、この読み進めを起こした契機です。
   274  func (f *fetcher) drainPages(ctx context.Context, trigger string) {
   275  	f.metrics.ReplayStarted(ctx)
   276  
   277  	var replayed int64
   278  	defer func() {
   279  		f.metrics.ReplayFinished(ctx)
   280  		f.metrics.ReplayExecuted(ctx, trigger, replayed)
   281  	}()
   282  
   283  	for {
   284  		events, hasMore, err := f.replayer.ReadPage(ctx, f.conn.stream, f.fetched)
   285  		if err != nil {
   286  			// 依存の不調でこの接続は閉じず、次の catch-up に委ねます（docs/design/realtime-delivery.md §2.6）。
   287  			f.log.Warn(ctx, "failed to read the event log for a stream connection",
   288  				logging.String(logging.StreamIDKey, string(f.conn.stream)),
   289  				logging.Error(logging.ErrorKey, err))
   290  			f.metrics.ReplayFailed(ctx)
   291  
   292  			return
   293  		}
   294  
   295  		if len(events) == 0 {
   296  			return
   297  		}
   298  
   299  		if events[0].Sequence != f.fetched+1 {
   300  			f.conn.closeWith(resyncControl(), closeReasonResync)
   301  
   302  			return
   303  		}
   304  
   305  		pushed := f.push(events)
   306  		replayed += int64(pushed)
   307  
   308  		if pushed < len(events) || !hasMore {
   309  			return
   310  		}
   311  	}
   312  }

   314  // push は、読んだ event を送出 buffer へ入れ、入れられた件数を返します。
   315  // 満杯なら event を捨てずに RETRY_LATER を試みて接続を閉じます（README「Runtime」）。
   316  func (f *fetcher) push(events []rt.DeliveryEvent) int {
   317  	for i, e := range events {
   318  		select {
   319  		case f.conn.events <- e:
   320  			f.fetched = e.Sequence
   321  		case <-f.conn.quit:
   322  			return i
   323  		default:
   324  			f.conn.closeWith(retryLaterControl(), closeReasonSlowClient)
   325  
   326  			return i
   327  		}
   328  	}
   329  
   330  	return len(events)
   331  }
