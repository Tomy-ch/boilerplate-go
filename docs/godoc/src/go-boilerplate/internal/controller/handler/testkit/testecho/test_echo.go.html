   176  // Build はテスト用のHTTPリクエストとレスポンスレコーダー、*echo.Contextを構築します。
   177  //
   178  // requestURL モードではルータ解決により *echo.Context のパスが設定されます。
   179  func (c *EchoTestClient) Build() (*http.Request, *httptest.ResponseRecorder, *echo.Context) {
   180  	c.t.Helper()
   181  
   182  	req, rec := c.buildRequest()
   183  	ec := c.e.NewContext(req, rec)
   184  
   185  	if c.requestURL != "" {
   186  		// 戻り値のハンドラは使わず、ルータがコンテキストへ設定する経路情報だけを利用する。
   187  		c.e.Router().Route(ec)
   188  		return req, rec, ec
   189  	}
   190  
   191  	ec.SetPath(c.routePattern)
   192  	if len(c.pathParams) > 0 {
   193  		values := make(echo.PathValues, len(c.pathParams))
   194  		for i, p := range c.pathParams {
   195  			values[i] = echo.PathValue{Name: p.Name, Value: p.Value}
   196  		}
   197  		ec.SetPathValues(values)
   198  	}
   199  
   200  	return req, rec, ec
   201  }

   211  // resolveTarget はリクエスト先URLを決定し、モードの排他違反を検出します。
   212  //
   213  // requestURL モード(ルータ登録済みの Echo 前提)と routePattern/pathParams モードは排他です。
   214  func (c *EchoTestClient) resolveTarget() (string, error) {
   215  	switch {
   216  	case c.requestURL != "":
   217  		if c.routePattern != "" || len(c.pathParams) > 0 {
   218  			return "", errModeConflict
   219  		}
   220  		return c.requestURL, nil
   221  	case c.routePattern != "":
   222  		return c.routePattern, nil
   223  	default:
   224  		return "", errTargetUnset
   225  	}
   226  }

   228  // buildRequest はリクエストとレスポンスレコーダーを構築します。
   229  func (c *EchoTestClient) buildRequest() (*http.Request, *httptest.ResponseRecorder) {
   230  	c.t.Helper()
   231  
   232  	target, err := c.resolveTarget()
   233  	if err != nil {
   234  		c.t.Fatal(err)
   235  	}
   236  
   237  	ctx := context.Background()
   238  	req := httptest.NewRequestWithContext(ctx, c.method, target, c.body)
   239  	for k, vv := range c.headers {
   240  		for _, v := range vv {
   241  			req.Header.Add(k, v)
   242  		}
   243  	}
   244  
   245  	q := req.URL.Query()
   246  	for _, p := range c.queryParams {
   247  		q.Add(p.Name, p.Value)
   248  	}
   249  	req.URL.RawQuery = q.Encode()
   250  
   251  	return req, httptest.NewRecorder()
   252  }
