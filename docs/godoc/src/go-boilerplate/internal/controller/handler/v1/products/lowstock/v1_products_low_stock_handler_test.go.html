    72  // wantProductResponse は、本番 toProductResponse とは独立な検証用オラクル（フィールド取り違え検出）。
    73  func wantProductResponse(dto productuc.ProductView) gen.ProductResponse {
    74  	//nolint:gosec // G115: テストデータは int32 範囲内の固定値です
    75  	quantity := int32(dto.Quantity)
    76  	var threshold *int32
    77  	if dto.StockWarningThreshold != nil {
    78  		//nolint:gosec // G115: テストデータは int32 範囲内の固定値です
    79  		v := int32(*dto.StockWarningThreshold)
    80  		threshold = &v
    81  	}
    82  	return gen.ProductResponse{
    83  		Id:                    dto.ID.ToPrimitive(),
    84  		Name:                  dto.Name,
    85  		Description:           dto.Description,
    86  		Price:                 dto.Price.String(),
    87  		Quantity:              quantity,
    88  		StockWarningThreshold: threshold,
    89  		Status: gen.ProductStatusRef{
    90  			Id:   dto.StatusID.ToPrimitive(),
    91  			Name: dto.StatusName,
    92  		},
    93  		Category: gen.ProductCategoryRef{
    94  			Id:   dto.CategoryID.ToPrimitive(),
    95  			Name: dto.CategoryName,
    96  		},
    97  		PublishedAt: dto.PublishedAt,
    98  		Images:      expectedImageItems(dto.Images),
    99  		//nolint:gosec // G115: テストデータは int32 範囲内の固定値です
   100  		Version: int32(dto.Version),
   101  	}
   102  }

   104  // expectedImageItems は、期待レスポンスの商品画像を組み立てます。
   105  // production の写像と同じ順序・同じ値になることを固定するため、要素は素直に書き下します。
   106  func expectedImageItems(dtos []productuc.ProductImageItemView) []gen.ProductImageItem {
   107  	items := make([]gen.ProductImageItem, len(dtos))
   108  	for i, dto := range dtos {
   109  		//nolint:gosec // G115: テストデータは int32 範囲内の固定値です
   110  		items[i] = gen.ProductImageItem{ImagePath: dto.Path, DisplaySort: int32(dto.DisplaySort)}
   111  	}
   112  	return items
   113  }
