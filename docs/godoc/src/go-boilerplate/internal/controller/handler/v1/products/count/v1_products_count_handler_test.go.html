    53  				func(_ context.Context, _ *auth.Authn, params productuc.CountProductsParams) (productuc.ProductCountView, error) {
    54  					require.NotNil(t, params.CategoryID)
    55  					assert.Equal(t, categoryID, *params.CategoryID)
    56  					require.NotNil(t, params.StatusID)

    76  					Keyword:  ptr.To("phone"),
    77  					MinPrice: ptr.To("10.50"), MaxPrice: ptr.To("99.99"),
    78  					MinQuantity: ptr.To[int32](2), MaxQuantity: ptr.To[int32](20),
    79  				},
    80  			})
    81  
    82  			require.NoError(t, err)
    83  			actual, ok := resp.(gen.GetProductsCount200JSONResponse)
    84  			require.True(t, ok)
