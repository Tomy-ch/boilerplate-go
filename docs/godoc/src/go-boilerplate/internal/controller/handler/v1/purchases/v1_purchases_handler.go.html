    54  func (s *server) GetPurchases(ctx context.Context, request gen.GetPurchasesRequestObject) (gen.GetPurchasesResponseObject, error) {
    55  	ctx, endSpan := s.tracer.Start(ctx)
    56  	defer endSpan()
    57  
    58  	authn, err := ctxhelper.RequireAuthn(ctx)
    59  	if err != nil {
    60  		return nil, err
    61  	}
    62  
    63  	cursor, err := paging.NewCursor(request.Params.After, request.Params.First)
    64  	if err != nil {
    65  		return nil, err
    66  	}
    67  
    68  	window, err := timewindow.New(timewindow.Bounds{
    69  		After:  request.Params.OrderedAfter,
    70  		Before: request.Params.OrderedBefore,
    71  	})
    72  	if err != nil {
    73  		return nil, err
    74  	}
    75  
    76  	statusCodes, err := conv.Int16sPtr(request.Params.StatusCodes)
    77  	if err != nil {
    78  		return nil, err
    79  	}
    80  
    81  	list, err := s.uc.GetPurchases(ctx, &authn, purchaseuc.ListPurchasesParams{
    82  		Cursor:            cursor,
    83  		Window:            window,
    84  		StatusCodes:       statusCodes,
    85  		ProductID:         conv.UUIDPtr(request.Params.ProductId),
    86  		IncludeOtherUsers: ptr.Deref(request.Params.IncludeOtherUsers, false),
    87  	})
    88  	if err != nil {
    89  		return nil, err
    90  	}
    91  
    92  	items := make([]gen.PurchaseSummaryResponse, len(list.Items))
    93  	for i, v := range list.Items {
    94  		items[i] = toPurchaseSummaryResponse(v)
    95  	}
    96  
    97  	return gen.GetPurchases200JSONResponse(gen.PurchaseListResponse{
    98  		Items:      items,
    99  		NextCursor: list.NextCursor,
   100  		HasNext:    list.NextCursor != nil,
   101  	}), nil
   102  }

   121  // 最初の結果を再生して返します。
   122  func (s *server) PostPurchases(ctx context.Context, request gen.PostPurchasesRequestObject) (gen.PostPurchasesResponseObject, error) {
   123  	ctx, endSpan := s.tracer.Start(ctx)
   124  	defer endSpan()
   125  
   126  	userID, err := ctxhelper.RequireUserID(ctx)
   127  	if err != nil {
   128  		return nil, err
   129  	}
   130  
   131  	details := make([]purchaseuc.DetailParam, len(request.Body.Details))
   132  	for i, d := range request.Body.Details {
   133  		details[i] = purchaseuc.DetailParam{
   134  			ProductID: conv.UUID(d.ProductId),
   135  			Quantity:  int(d.Quantity),
   136  		}
   137  	}
   138  
   139  	var displayCurrency *string
   140  	if request.Params.DisplayCurrency != nil {
   141  		s := string(*request.Params.DisplayCurrency)
   142  		displayCurrency = &s
   143  	}
   144  
   145  	view, _, err := idempotency.Run(ctx, s.idem, http.StatusCreated, func(ctx context.Context) (checkoutuc.PurchaseView, error) {
   146  		return s.checkout.CreatePurchase(ctx, checkoutuc.CreatePurchaseParams{
   147  			UserID:          userID,
   148  			Details:         details,
   149  			DisplayCurrency: displayCurrency,
   150  			CouponID:        conv.UUIDPtr(request.Body.CouponId),
   151  		})
   152  	})
   153  	if err != nil {
   154  		return nil, err
   155  	}
   156  
   157  	res, err := toPurchaseResponse(view.Purchase, view.ReferenceAmount)
   158  	if err != nil {
   159  		return nil, err
   160  	}
   161  	return gen.PostPurchases201JSONResponse(res), nil
   162  }

   189  // toPurchaseResponse は、ユースケースの DTO を HTTP レスポンスへ変換します。
   190  // 数量が int32 に収まらない場合はエラーを返します。
   191  func toPurchaseResponse(v purchaseuc.PurchaseView, ref *checkoutuc.ReferenceAmountView) (gen.PurchaseResponse, error) {
   192  	details := make([]gen.PurchaseDetailResponse, len(v.Details))
   193  	for i, d := range v.Details {
   194  		quantity, err := safecast.IntToInt32(d.Quantity)
   195  		if err != nil {
   196  			return gen.PurchaseResponse{}, xerrors.Wrap(err, "invalid purchase detail quantity")
   197  		}
   198  		details[i] = gen.PurchaseDetailResponse{
   199  			ProductId: d.ProductID.ToPrimitive(),
   200  			Quantity:  quantity,
   201  			UnitPrice: d.UnitPrice.String(),
   202  		}
   203  	}
   204  
   205  	return gen.PurchaseResponse{
   206  		Code:            v.Code,
   207  		UserId:          v.UserID.ToPrimitive(),
   208  		StatusId:        v.StatusID.ToPrimitive(),
   209  		SubtotalAmount:  int64(v.SubtotalAmount),
   210  		DiscountAmount:  int64(v.DiscountAmount),
   211  		AppliedCoupon:   toAppliedCouponResponse(v.AppliedCoupon),
   212  		TaxAmount:       int64(v.TaxAmount),
   213  		ShippingFee:     int64(v.ShippingFee),
   214  		TotalAmount:     int64(v.TotalAmount),
   215  		Details:         details,
   216  		OrderedAt:       v.OrderedAt,
   217  		ReferenceAmount: toReferenceAmount(ref),
   218  	}, nil
   219  }
