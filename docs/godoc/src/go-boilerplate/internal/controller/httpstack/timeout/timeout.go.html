    16  // Middleware は、リクエスト context に timeout の deadline を設定するミドルウェアを返します。
    17  // deadline 超過は apperror.ErrUnavailable(503) へ、それ以外のエラーはそのまま伝播します。
    18  // SSE stream endpoint だけは deadline を持ちません — 接続は分単位で開き続けるのが正常な状態で、
    19  // 単一 budget を当てると必ず途中で切れます。接続側の境界（write deadline・heartbeat・
    20  // connection maximum lifetime）がこの budget の代わりを務めます。
    21  // 単一 deadline budget の共有と ContextTimeout を基底に選んだ理由は README を参照してください。
    22  func Middleware(timeout time.Duration) echo.MiddlewareFunc {
    23  	return middleware.ContextTimeoutWithConfig(middleware.ContextTimeoutConfig{
    24  		Timeout: timeout,
    25  		Skipper: func(c *echo.Context) bool { return streampath.Is(c.Request().URL.Path) },
    26  		ErrorHandler: func(_ *echo.Context, err error) error {
    27  			if xerrors.Is(err, context.DeadlineExceeded) {
    28  				return xerrors.Join(apperror.ErrUnavailable, err)
    29  			}
    30  			return err
    31  		},
    32  	})
