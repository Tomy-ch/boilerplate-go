    41  func SecretQueryParamNames(spec *openapi3.T) []string {
    42  	if spec == nil || spec.Components == nil {
    43  		return nil
    44  	}
    45  
    46  	var names []string
    47  	for _, ref := range spec.Components.SecuritySchemes {
    48  		if ref == nil || ref.Value == nil {
    49  			continue
    50  		}
    51  		if ref.Value.Type == securitySchemeAPIKey && ref.Value.In == securitySchemeInQuery && ref.Value.Name != "" {
    52  			names = append(names, ref.Value.Name)
    53  		}
    54  	}
    55  	sort.Strings(names)
    56  
    57  	return names
    58  }

    60  // URI は、raw（path と query を含むリクエスト URI）から秘匿対象パラメータの値を取り除いた文字列を返します。
    61  // query の並びと符号化はそのまま保ちます。標準の構文解析が受け付けない query（`;` 区切りや壊れた符号化）は
    62  // 組ごとに判定できないため、query 全体を置き換えます（fail-closed）。
    63  func (r Redactor) URI(raw string) string {
    64  	if len(r.names) == 0 {
    65  		return raw
    66  	}
    67  
    68  	path, query, found := strings.Cut(raw, "?")
    69  	if !found {
    70  		return raw
    71  	}
    72  
    73  	if _, err := url.ParseQuery(query); err != nil {
    74  		return path + "?" + RedactedValue
    75  	}
    76  
    77  	pairs := strings.Split(query, "&")
    78  	for i, pair := range pairs {
    79  		key, _, hasValue := strings.Cut(pair, "=")
    80  		if !hasValue {
    81  			continue
    82  		}
    83  		if r.secret(key) {
    84  			pairs[i] = key + "=" + RedactedValue
    85  		}
    86  	}
    87  
    88  	return path + "?" + strings.Join(pairs, "&")
    89  }

    93  func (r Redactor) QueryParams(params map[string][]string) map[string][]string {
    94  	if len(r.names) == 0 || params == nil {
    95  		return params
    96  	}
    97  
    98  	out := make(map[string][]string, len(params))
    99  	for name, values := range params {
   100  		if _, ok := r.names[name]; !ok {
   101  			out[name] = values
   102  			continue
   103  		}

   105  		for i := range values {
   106  			redacted[i] = RedactedValue
   107  		}
   108  		out[name] = redacted
   109  	}
   110  
   111  	return out
   112  }
