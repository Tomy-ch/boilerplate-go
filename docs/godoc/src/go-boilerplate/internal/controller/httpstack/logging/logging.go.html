    24  // Middleware は、Echoフレームワークのミドルウェアで、リクエストのログを出力します。
    25  // ただし /health, /metrics 等の運用系エンドポイントはログ出力をスキップし、SSE として確定した接続は
    26  // 応答ログだけをスキップします（その 1 行は接続が閉じるまで出ず、duration もリクエストではなく接続の
    27  // 長さになるため）。確定前の拒否は普通のレスポンスなので、要求ログも応答ログもそのまま出ます。
    28  // レスポンスを取り出せない場合はログを出さず素通しします。
    29  // URI と query は red を通してから出すため、query で運ばれる資格情報はログに残りません。
    30  func Middleware(logger logging.Logger, lf logging.LogFieldBuilder, red redaction.Redactor) echo.MiddlewareFunc {
    31  	return func(next echo.HandlerFunc) echo.HandlerFunc {
    32  		return func(c *echo.Context) error {
    33  			if ops.IsOpsPath(c.Request().URL.Path) {
    34  				return next(c)
    35  			}
    36  
    37  			res := server.ResponseOf(c)
    38  			if res == nil {
    39  				return next(c)
    40  			}
    41  
    42  			start := time.Now()
    43  			ctx := c.Request().Context()
    44  
    45  			l := requestLog{
    46  				c:   c,
    47  				res: res,
    48  				lf:  lf,
    49  				red: red,
    50  			}
    51  
    52  			reqFields := l.buildRequestLogFields(start)
    53  			logger.Named("http.request").Info(ctx, "request received", reqFields...)
    54  
    55  			res.After(func() {
    56  				if streampath.IsCommittedStream(res.Header()) {
    57  					return
    58  				}
    59  
    60  				end := time.Now()
    61  				latency := end.Sub(start)
    62  
    63  				fields := l.buildResponseLogFields(end, latency)
    64  				resLogger := logger.Named("http.response")
    65  				if res.Status >= MinStatusError {
    66  					resLogger.Error(ctx, "request handled", fields...)
    67  				} else {
    68  					resLogger.Info(ctx, "request handled", fields...)
    69  				}
    70  			})
    71  
    72  			return next(c)
    73  		}
    74  	}
    75  }

    86  		RemoteIP:      req.RemoteAddr,
    87  		Host:          req.Host,
    88  		Scheme:        l.c.Scheme(),
    89  		Proto:         req.Proto,
    90  		UserAgent:     req.UserAgent(),
    91  		ContentType:   req.Header.Get(echo.HeaderContentType),
    92  		ContentLength: req.ContentLength,
    93  		PathParams:    server.ExtractPathParams(l.c),
    94  		QueryParams:   l.red.QueryParams(server.ExtractQueryParams(l.c)),
    95  	}
    96
