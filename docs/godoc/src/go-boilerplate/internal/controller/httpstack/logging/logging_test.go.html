   261  func Test_requestLog_buildResponseLogFields(t *testing.T) {
   262  	t.Parallel()
   263  
   264  	lf := logging.NewTestLogFieldBuilder(t)
   265  
   266  	// 各サブテストは並列実行されるため、共有の *echo.Context を使うと Response() の
   267  	// ステータス/ヘッダ書き込みが競合する。サブテストごとに専用の Context を生成する。
   268  	newContext := func() *echo.Context {
   269  		e := echo.New()
   270  		ctx := context.Background()
   271  		req := httptest.NewRequestWithContext(ctx, http.MethodGet, "/resp", nil)
   272  		req.RemoteAddr = "5.6.7.8:9000"
   273  		rec := httptest.NewRecorder()
   274  		return e.NewContext(req, rec)
   275  	}
   276  
   277  	t.Run("正常系", func(t *testing.T) {
   278  		t.Parallel()
   279  
   280  		t.Run("基本フィールドを返す（trace_id/span_idはLoggerがctxから注入する）", func(t *testing.T) {
   281  			t.Parallel()
   282  
   283  			c := newContext()
   284  			expectedStatus := http.StatusCreated
   285  			expectedRequestID := "req-123"
   286  			expectedEventAt := time.Date(2026, time.July, 27, 12, 0, 0, 0, time.UTC)
   287  
   288  			res := server.ResponseOf(c)
   289  			require.NotNil(t, res)
   290  			res.Status = expectedStatus
   291  			c.Response().Header().Set("X-Request-Id", expectedRequestID)
   292  
   293  			l := requestLog{c: c, res: res, lf: lf}
   294  			fields := l.buildResponseLogFields(expectedEventAt, 150*time.Millisecond)
   295  
   296  			assert.Contains(t, fields, logging.Int(logging.StatusKey, expectedStatus))
   297  			assert.Contains(t, fields, logging.String(logging.RequestIDKey, expectedRequestID))
   298  			assert.Contains(t, fields, logging.Time(logging.EventAtKey, expectedEventAt))
   299  		})
   300  
   301  		t.Run("秘匿対象のqueryはレスポンス側のURIでも値が置き換わる", func(t *testing.T) {
   302  			t.Parallel()
   303  
   304  			e := echo.New()
   305  			req := httptest.NewRequestWithContext(context.Background(), http.MethodGet, "/v1/streams/s?ticket=raw-secret", nil)
   306  			c := e.NewContext(req, httptest.NewRecorder())

   315  			assert.NotContains(t, text, "raw-secret")

   317  		})
   318  	})

   322  // どのキーに載ったかを問わず生値の有無を見るために使います。
   323  func loggedJSON(ctx context.Context, t *testing.T, fields []*logging.Field) string {
   324  	t.Helper()
   325  	logger, observed := logging.NewObservedTestLogger(t)
   326  	logger.Info(ctx, "probe", fields...)
   327  	entries := observed.All()
   328  	require.Len(t, entries, 1)
   329  	raw, err := json.Marshal(entries[0].ContextMap())
   330  	require.NoError(t, err)
   331  	return string(raw)
