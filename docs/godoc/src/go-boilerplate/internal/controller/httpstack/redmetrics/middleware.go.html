    14  // Middleware は、HTTP リクエストの count / duration / status を計測する Echo ミドルウェアを返します。
    15  // Observe は 1 リクエストにつき厳密に 1 回だけ呼び出されます。/metrics などの運用系パスと、SSE として
    16  // 確定した接続は計測対象から除外し（接続の長さを request latency として数えると分布が歪むため）、
    17  // レスポンスを取り出せない場合は計測せず素通しします。stream endpoint の確定前の拒否は普通の
    18  // レスポンスなので計測します — ticket 総当たりや容量枯渇はここにしか現れません。After フック方式に起因する既知の限界
    19  // （ボディ無し応答が計測されない等）は README を参照してください。
    20  func Middleware(rec Recorder) echo.MiddlewareFunc {
    21  	return func(next echo.HandlerFunc) echo.HandlerFunc {
    22  		return func(c *echo.Context) error {
    23  			if ops.IsOpsPath(c.Request().URL.Path) {
    24  				return next(c)
    25  			}
    26  
    27  			res := server.ResponseOf(c)
    28  			if res == nil {
    29  				return next(c)
    30  			}
    31  
    32  			start := time.Now()
    33  			method := c.Request().Method
    34  
    35  			var once sync.Once
    36  			res.After(func() {
    37  				once.Do(func() {
    38  					if streampath.IsCommittedStream(res.Header()) {
    39  						return
    40  					}
    41  
    42  					status := normalizeStatus(res.Status)
    43  					rec.Observe(method, routeOf(c), status, statusClass(status), time.Since(start))
    44  				})
    45  			})
    46  
    47  			return next(c)
    48  		}
    49  	}
    50  }
