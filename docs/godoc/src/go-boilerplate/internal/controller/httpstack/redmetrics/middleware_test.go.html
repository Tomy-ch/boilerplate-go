    17  // serveCfg は、ミドルウェア経由のリクエスト実行設定です。

    24  // serve は、redmetrics ミドルウェアを適用した Echo に rec を計測先として 1 リクエスト流します。
    25  // 計測の検証は rec（生成 mock）に設定した EXPECT で行います。
    26  func serve(t *testing.T, rec Recorder, cfg serveCfg) {
    27  	t.Helper()
    28  
    29  	e := echo.New()
    30  	e.Use(Middleware(rec))
    31  	if cfg.registerPath != "" {
    32  		e.GET(cfg.registerPath, cfg.handler)
    33  	}
    34  
    35  	req := httptest.NewRequestWithContext(context.Background(), http.MethodGet, cfg.requestPath, nil)
    36  	res := httptest.NewRecorder()
    37  	e.ServeHTTP(res, req)
    38  }

    40  // okHandler は、After フックを発火させるためボディ付きで応答するハンドラを返します。
    41  func okHandler() echo.HandlerFunc {
    42  	return func(c *echo.Context) error {
    43  		return c.String(http.StatusOK, "ok")
    44  	}
    45  }
