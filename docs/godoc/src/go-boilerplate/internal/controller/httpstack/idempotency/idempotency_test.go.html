    51  func newEcho(t *testing.T, key string, withAuthn bool, subject string) *echo.Context {
    52  	t.Helper()
    53  
    54  	ctx := context.Background()
    55  	if withAuthn {
    56  		ctx = ctxhelper.WithAuthn(ctx)
    57  		a, err := auth.New(subject, "test", nil, nil)
    58  		require.NoError(t, err)
    59  		if id, perr := uuid.Parse(subject); perr == nil {
    60  			a, err = a.WithUserID(id)
    61  			require.NoError(t, err)
    62  		}
    63  		ctxhelper.SetAuthn(ctx, *a)
    64  	}
    65  	req := httptest.NewRequestWithContext(ctx, http.MethodPost, testPath, nil)
    66  	if key != "" {
    67  		req.Header.Set(headerName, key)
    68  	}
    69  	return echo.New().NewContext(req, httptest.NewRecorder())
