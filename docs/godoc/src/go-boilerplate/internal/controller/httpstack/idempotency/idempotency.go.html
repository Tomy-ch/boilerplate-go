    44  func handle(ec *echo.Context, request any, operationID string, next NextFunc) (any, error) {
    45  	r := ec.Request()
    46  	key := strings.TrimSpace(r.Header.Get(headerName))
    47  	if key == "" {
    48  		// ヘッダ無し = 非冪等（既存挙動のまま素通し）。
    49  		return next(ec, request)
    50  	}
    51  	if err := validateKey(key); err != nil {
    52  		return nil, err
    53  	}
    54  
    55  	// 未認証なら安全側で素通しする。
    56  	authn, ok := ctxhelper.GetAuthn(r.Context())
    57  	if !ok {
    58  		return next(ec, request)
    59  	}
    60  
    61  	fp, err := fingerprint(r.Method, r.URL.Path, request)
    62  	if err != nil {
    63  		return nil, xerrors.Join(apperror.ErrInternal, xerrors.Wrap(err, "failed to fingerprint idempotent request"))
    64  	}
    65  
    66  	reqCtx := idempotencyuc.WithRequest(r.Context(), idempotencyuc.Request{
    67  		// スコープは認証主体そのもの（ADR-0063 (idempotency-scope-required)）。内部ユーザー ID ではない。
    68  		Scope:       authn.Subject(),
    69  		Key:         key,
    70  		Fingerprint: fp,
    71  		Method:      r.Method,
    72  		Path:        r.URL.Path,
    73  		OperationID: operationID,
    74  	})
    75  	ec.SetRequest(r.WithContext(reqCtx))
    76  
    77  	return next(ec, request)
    78  }

    80  // validateKey は、Idempotency-Key の健全性を検証します（≤255 / 印字可能 ASCII）。違反は 400。
    81  func validateKey(key string) error {
    82  	if len(key) > maxKeyLength {
    83  		return xerrors.Wrap(apperror.ErrInvalidArgument, "Idempotency-Key must be 255 characters or fewer")
    84  	}
    85  	for _, c := range []byte(key) {
    86  		if c < 0x21 || c > 0x7E {
    87  			return xerrors.Wrap(apperror.ErrInvalidArgument, "Idempotency-Key must contain only printable ASCII characters")
    88  		}
    89  	}
    90  	return nil
    91  }

    93  // fingerprint は、method + path + typed request の決定的 marshal を SHA-256 した指紋を返します。
    94  // request の marshal に失敗した場合は弱い指紋を作らずエラーを返します（fail-closed）。
    95  func fingerprint(method, path string, request any) ([]byte, error) {
    96  	b, err := json.Marshal(request)
    97  	if err != nil {
    98  		return nil, err
    99  	}
   100  	h := sha256.New()
   101  	h.Write([]byte(method))
   102  	h.Write([]byte{'\n'})
   103  	h.Write([]byte(path))
   104  	h.Write([]byte{'\n'})
   105  	h.Write(b)
   106  	return h.Sum(nil), nil
   107  }
