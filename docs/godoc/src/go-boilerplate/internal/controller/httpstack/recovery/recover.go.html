    25  // Middleware は、Echoフレームワークのミドルウェアで、パニックからのリカバリを行います。
    26  //
    27  // [middleware.RecoverWithConfig] は復帰した panic を [middleware.PanicStackError] へ包んで
    28  // 戻り値のエラーとして返すため、その外側でスタック付きのログを出し、
    29  // エラーハンドラへは包む前のエラーを渡します。
    30  func Middleware(
    31  	z logging.Logger, lf logging.LogFieldBuilder, appCfg *config.ApplicationConfig, red redaction.Redactor,
    32  ) echo.MiddlewareFunc {
    33  	recoverMW := middleware.RecoverWithConfig(newRecoverConfig(z, appCfg))
    34  	logPanic := newPanicLogFunc(z, lf, red)
    35  
    36  	return func(next echo.HandlerFunc) echo.HandlerFunc {
    37  		recovered := recoverMW(next)
    38  		return func(c *echo.Context) error {
    39  			err := recovered(c)
    40  
    41  			var pse *middleware.PanicStackError
    42  			if !xerrors.As(err, &pse) {
    43  				return err
    44  			}
    45  			logPanic(c, pse.Err, pse.Stack)
    46  			return pse.Err
    47  		}
    48  	}
    49  }

    51  // newRecoverConfig は、環境設定に基づいてリカバリミドルウェアの設定を生成します。
    52  func newRecoverConfig(logger logging.Logger, appCfg *config.ApplicationConfig) middleware.RecoverConfig {
    53  	switch {
    54  	case appCfg.IsDevelopmentMode():
    55  		return developmentConfig()
    56  	case appCfg.IsProductionMode():
    57  		return productionConfig()
    58  	default:
    59  		logger.Named("middleware.recover").Warn(
    60  			context.Background(),
    61  			"Unknown environment, using production config for recover middleware",
    62  			logging.String("env", appCfg.Mode()),
    63  		)
    64  		return productionConfig()
    65  	}
    66  }

    68  // newPanicLogFunc は、復帰したパニックのログ出力関数を生成します。
    69  func newPanicLogFunc(
    70  	logger logging.Logger, lf logging.LogFieldBuilder, red redaction.Redactor,
    71  ) func(c *echo.Context, err error, stack []byte) {
    72  	return func(c *echo.Context, err error, stack []byte) {
    73  		reqIn := server.BuildHTTPRequestLogInput(c, logging.EventTypePanic, red)
    74  		recoverFields := []*logging.Field{
    75  			logging.String(logging.InternalErrorKey, err.Error()),
    76  			logging.Strings(logging.InternalStackTraceKey, logging.SplitStackLines(string(stack))),
    77  		}
    78  		fields := append(lf.BuildHTTPRequestFields(reqIn), recoverFields...)
    79  		logger.Named("middleware.recover").Error(c.Request().Context(), "panic recovered", fields...)
    80  		// ログ済みを記録する（二重ログは ctxhelper.GetRecoveredFromEcho で抑止）。
    81  		ctxhelper.SetRecoveredToEcho(c, true)
    82  	}
    83  }
