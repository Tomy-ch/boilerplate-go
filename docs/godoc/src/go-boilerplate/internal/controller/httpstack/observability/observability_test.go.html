    63  			t.Parallel()

    65  			e := echo.New()
    66  			e.Use(Middleware())
    67  
    68  			ctx := context.Background()
    69  			req := httptest.NewRequestWithContext(ctx, http.MethodGet, "/", nil)
    70  			rec := httptest.NewRecorder()
    71  			e.ServeHTTP(rec, req)
    72  
    73  			assert.Equal(t, http.StatusNotFound, rec.Code)

    76  		t.Run("span属性にqueryの値を載せない", func(t *testing.T) { //nolint:paralleltest // otel の global provider を差し替えるため
    77  			// 秘匿すべき資格情報（stream ticket）は query で運ばれる（ADR-0074）。ミドルウェアが記録する span 属性は
    78  			// url.path までで url.query を持たないことを、属性名の許可リストではなく値の全走査で固定する
    79  			// （upstream が属性を増やしても、生値が現れた時点で落ちる）。
    80  			recorded := obs.InstallRecordingTracerProvider(t)
    81  
    82  			e := echo.New()
    83  			e.Use(Middleware())
    84  			e.GET("/v1/streams/:destination", func(c *echo.Context) error { return c.NoContent(http.StatusOK) })
    85  
    86  			req := httptest.NewRequestWithContext(context.Background(), http.MethodGet, "/v1/streams/s?ticket=raw-secret-value", nil)
    87  			rec := httptest.NewRecorder()
    88  			e.ServeHTTP(rec, req)
    89  
    90  			require.Equal(t, http.StatusOK, rec.Code)
    91  			// 並行する別ケースの span が混ざりうるので、被験リクエストの span（url.path が一致するもの）に絞ってから走査する。
    92  			values := obs.SpanAttributeValues(recorded(), "url.path", "/v1/streams/s")
    93  			require.NotEmpty(t, values, "被験リクエストの span が記録されていない")
    94  			for _, v := range values {
    95  				assert.NotContains(t, v, "raw-secret-value")
    96  			}
    97  		})
    98  	})
