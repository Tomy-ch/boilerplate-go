    34  // NewAuthenticator が選び、検証結果は request の context スロットへ書き込みます。

    42  // NewAuthenticator は、認証用のOpenAPIオプションを返します。
    43  // operation が宣言した securityScheme の名前が schemes のいずれかと一致すればその認証器へ委譲し、
    44  // 一致しなければ Bearer として扱います。Bearer でもない scheme に認証器が無い場合は検証できないため拒否します。
    45  // 同じ scheme を担当する認証器が 2 つあれば ErrDuplicateScheme を返します（起動時に結線の不具合として落とす）。
    46  func NewAuthenticator(
    47  	authenticator authbd.Authenticator,
    48  	resolver authbd.IdentityResolver,
    49  	schemes []SchemeAuthenticator,
    50  ) (openapi3filter.AuthenticationFunc, error) {
    51  	byName := make(map[string]SchemeAuthenticator, len(schemes))
    52  	for _, s := range schemes {
    53  		if _, dup := byName[s.Scheme()]; dup {
    54  			return nil, xerrors.Wrap(ErrDuplicateScheme, s.Scheme())
    55  		}
    56  		byName[s.Scheme()] = s
    57  	}
    58  
    59  	return func(ctx context.Context, input *openapi3filter.AuthenticationInput) error {
    60  		req := input.RequestValidationInput.Request
    61  
    62  		if s, ok := byName[input.SecuritySchemeName]; ok {
    63  			if err := s.Authenticate(ctx, input); err != nil {
    64  				failure := withHTTPStatus(err)
    65  				//nolint:contextcheck // input が内包する request の context のスロットへ書き戻すため
    66  				ctxhelper.SetAuthnFailure(req.Context(), failure)
    67  				return failure
    68  			}
    69  			return nil
    70  		}
    71  
    72  		if !isBearerScheme(input.SecurityScheme) {
    73  			failure := withHTTPStatus(xerrors.Wrap(ErrUnauthorizedSchemeUnsupported, input.SecuritySchemeName))
    74  			//nolint:contextcheck // 同上
    75  			ctxhelper.SetAuthnFailure(req.Context(), failure)
    76  			return failure
    77  		}
    78  
    79  		// OpenAPI バリデータが渡す context は context.Background() から組み立てられており、
    80  		// スパン・deadline・キャンセルのいずれも持たない。認証は request の予算の内側で行う。
    81  		//nolint:contextcheck // 引数の context ではなく input が内包する request の context を用いるため
    82  		authn, err := authExtractor(req.Context(), req, authenticator, resolver)
    83  		if err != nil {
    84  			failure := withHTTPStatus(err)
    85  			// 記録するのは資格情報が提示されたうえでの失敗だけ。未提示は失敗ではない。
    86  			//nolint:contextcheck // input が内包する request の context のスロットへ書き戻すため
    87  			ctxhelper.SetAuthnFailure(req.Context(), failure)
    88  			return failure
    89  		}
    90  		if authn == nil {
    91  			return withHTTPStatus(ErrUnauthorizedTokenNotProvided)
    92  		}
    93  
    94  		//nolint:contextcheck // input が内包する request の context のスロットへ書き戻すため
    95  		if !ctxhelper.SetAuthn(req.Context(), *authn) {
    96  			failure := withHTTPStatus(ErrAuthnSlotNotFound)
    97  			// 認証は成立したが結果を運べていない。記録しなければ、認証済みの主体が匿名として通る。
    98  			//nolint:contextcheck // 同上
    99  			ctxhelper.SetAuthnFailure(req.Context(), failure)
   100  			return failure
   101  		}
   102  		return nil
   103  	}, nil
   104  }

   106  // isBearerScheme は、scheme を Bearer 経路で扱ってよいかを返します。
   107  // 宣言が渡されない場合（nil）は Bearer とみなします — 既定の経路であり、spec 外からの呼び出しがこれに当たります。
   108  func isBearerScheme(scheme *openapi3.SecurityScheme) bool {
   109  	if scheme == nil {
   110  		return true
   111  	}
   112  	return scheme.Type == securitySchemeTypeHTTP && strings.EqualFold(scheme.Scheme, securitySchemeBearer)
   113  }

   115  // withHTTPStatus は、認証フェーズのエラーへ apperror の分類に対応するステータスを持たせます（元のエラーは保持）。
   116  // OpenAPI バリデータはステータスを解決できないエラーを 403 へ丸めるため、この段で持たせないと、
   117  // 認可の判定を行っていないものが認可の結論として外へ出ます。
   118  // メッセージは空のまま返します（本文は errorhandler が組み立てる）。
   119  func withHTTPStatus(err error) error {
   120  	var he *echo.HTTPError
   121  	if xerrors.As(err, &he) {
   122  		return err
   123  	}
   124  	return echo.NewHTTPError(response.NewHTTPErrorFromAppError(err).HTTPStatus, "").Wrap(err)
   125  }

   127  // authExtractor は、Bearer トークンを検証して内部ユーザーを解決した Authn を返します。
   128  // トークンが無い場合は nil, nil を返します。
   129  func authExtractor(
   130  	ctx context.Context,
   131  	req *http.Request,
   132  	authenticator authbd.Authenticator,
   133  	resolver authbd.IdentityResolver,
   134  ) (*authbd.Authn, error) {
   135  	scheme, token := extractBearerToken(req)
   136  	if token == "" {
   137  		//nolint:nilnil // トークン未提供を表す。呼び出し側で authn==nil を判定し未提供エラーへ変換するため意図的にnil,nilを返す
   138  		return nil, nil
   139  	}
   140  
   141  	cred, err := authbd.NewCredential(scheme, token)
   142  	if err != nil {
   143  		return nil, err
   144  	}
   145  
   146  	// 資格情報についての結論だけを認証エラーへ寄せ、原因は外へ出さない。
   147  	// 検証を遂行できなかった場合（鍵が引けない / 予算切れ）は結論ではないため、分類を保ったまま返す。
   148  	authn, err := authenticator.Authenticate(ctx, cred)
   149  	if err != nil {
   150  		if xerrors.Is(err, apperror.ErrUnauthenticated) {
   151  			return nil, ErrUnauthorizedInvalidToken
   152  		}
   153  		return nil, err
   154  	}
   155  	if authn == nil {
   156  		//nolint:nilnil // Authenticate が nil,nil を返した場合は未提供として扱い、呼び出し側で ErrUnauthorizedTokenNotProvided へ変換する
   157  		return nil, nil
   158  	}
   159  
   160  	// 認証済みの外部アイデンティティ（issuer + subject）を内部ユーザーへ解決する。
   161  	resolved, err := resolver.Resolve(ctx, authn)
   162  	if err != nil {
   163  		return nil, err
   164  	}
   165  	if resolved == nil {
   166  		// resolver は成功時に非 nil の Authn を返す契約。防御的に未認証として扱う。
   167  		return nil, authbd.ErrIdentityNotFound
   168  	}
   169  
   170  	return resolved, nil
   171  }

   173  // extractBearerToken は、Authorization ヘッダから Bearer トークンを抽出します（ヘッダ名・スキームは RFC 6750 により固定）。
   174  // Authorization: Bearer <token> 形式のときだけ scheme=SchemeBearer と token を返し、
   175  // それ以外は scheme/token とも空を返します。
   176  func extractBearerToken(r *http.Request) (string, string) {
   177  	raw := strings.TrimSpace(r.Header.Get(echo.HeaderAuthorization))
   178  	if raw == "" {
   179  		return "", ""
   180  	}
   181  	after, ok := strings.CutPrefix(raw, prefixBearer)
   182  	if !ok {
   183  		return "", ""
   184  	}
   185  	return authbd.SchemeBearer, strings.TrimSpace(after)
   186  }
