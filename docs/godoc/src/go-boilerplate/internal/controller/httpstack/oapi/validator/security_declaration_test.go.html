    99  func effectiveSecurity(spec *openapi3.T, op *openapi3.Operation) openapi3.SecurityRequirements {
   100  	if op.Security != nil {
   101  		return *op.Security
   102  	}
   103  	return spec.Security

   106  // classifySecurity は security 要件を 3 つの姿勢へ分類する。
   107  // 要件リストは OR 結合で、空の要件オブジェクト {} は必ず満たされるため匿名アクセスを許す。
   108  // 空の要件しか無い宣言は資格情報を受け取る余地が無く、完全公開と区別できない。
   109  func classifySecurity(reqs openapi3.SecurityRequirements) securityKind {
   110  	if len(reqs) == 0 {
   111  		return fullyPublic
   112  	}
   113  	if emptyRequirementIndex(reqs) < 0 {
   114  		return authRequired
   115  	}
   116  	if len(reqs) == 1 {
   117  		return fullyPublic
   118  	}
   119  	return optionalAuth
   120  }

   153  // staleEntries は、spec に存在しない operation を指したままの許可リストのキーを返す。
   154  func staleEntries(allowList map[string]string, seen map[string]bool) []string {
   155  	stale := make([]string, 0, len(allowList))
   156  	for key := range allowList {
   157  		if !seen[key] {
   158  			stale = append(stale, key)
   159  		}
   160  	}
   161  	sort.Strings(stale)
   162  	return stale
   163  }

   166  func assertAllowListMembership(t *testing.T, key string, kind securityKind, reqs openapi3.SecurityRequirements) {
   167  	t.Helper()
   168  
   169  	switch kind {
   170  	case authRequired:
   171  		assert.NotContains(t, publicOperations, key,
   172  			"%s は認証必須のため許可リストのエントリが不要です。publicOperations から削除してください", key)
   173  		assert.NotContains(t, optionalAuthOperations, key,
   174  			"%s は認証必須のため許可リストのエントリが不要です。optionalAuthOperations から削除してください", key)
   175  		if _, ok := streamTicketOperations[key]; ok {
   176  			assert.False(t, declaresBearer(reqs),
   177  				"%s は stream ticket 認証として登録されていますが BearerAuth も宣言しています。経路はどちらか一方です", key)
   178  		}
   179  	case fullyPublic:
   180  		assert.Contains(t, publicOperations, key,
   181  			"%s に認証必須の security 宣言がありません。認証が必要なら OpenAPI 定義に security: [BearerAuth] を追加し、"+
   182  				"意図的な公開 API なら publicOperations に理由コメント付きで追加してください", key)
   183  		assert.NotContains(t, optionalAuthOperations, key,
   184  			"%s は資格情報を受け取らないため任意認証ではありません。publicOperations へ移してください", key)
   185  		assert.NotContains(t, streamTicketOperations, key,
   186  			"%s は資格情報を受け取らないため stream ticket 認証ではありません。publicOperations へ移してください", key)
   187  	case optionalAuth:
   188  		assert.Contains(t, optionalAuthOperations, key,
   189  			"%s は任意認証（BearerAuth と空要件の両方を宣言）です。意図どおりなら optionalAuthOperations へ "+
   190  				"理由コメント付きで追加してください（ADR-0021 (optional-authentication-fail-closed)）", key)
   191  		assert.NotContains(t, publicOperations, key,
   192  			"%s は無効な資格情報を 401 で拒否するため完全公開ではありません。optionalAuthOperations へ移してください", key)
   193  		assert.NotContains(t, streamTicketOperations, key,
   194  			"%s は任意認証であり stream ticket 認証ではありません。optionalAuthOperations へ移してください", key)
   195  	}
   196  }

   313  	t.Run("正常系", func(t *testing.T) {

   326  			for path, item := range spec.Paths.Map() {
   327  				for method, op := range item.Operations() {
   328  					reqs := effectiveSecurity(spec, op)
   329  					idx := emptyRequirementIndex(reqs)
   330  					if idx < 0 || len(reqs) == 1 {
   331  						continue
   332  					}
   333  					assert.Equal(t, len(reqs)-1, idx,
   334  						"%s %s: 空の security 要件は末尾に置いてください。先に置くと認証関数が呼ばれず、"+
