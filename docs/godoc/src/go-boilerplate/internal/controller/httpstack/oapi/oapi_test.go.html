    71  	t.Helper()
    72  
    73  	ctrl := gomock.NewController(t)
    74  	authenticator := mock_auth.NewMockAuthenticator(ctrl)
    75  	resolver := mock_auth.NewMockIdentityResolver(ctrl)
    76  
    77  	authn, err := authbd.New("user-john-doe", authbd.IssuerMock, nil, nil)
    78  	require.NoError(t, err)
    79  
    80  	switch {
    81  	case stub.authenticateErr != nil:
    82  		authenticator.EXPECT().Authenticate(gomock.Any(), gomock.Any()).
    83  			Return(nil, stub.authenticateErr).AnyTimes()
    84  	case stub.resolveErr != nil:
    85  		authenticator.EXPECT().Authenticate(gomock.Any(), gomock.Any()).Return(authn, nil).AnyTimes()
    86  		resolver.EXPECT().Resolve(gomock.Any(), gomock.Any()).Return(nil, stub.resolveErr).AnyTimes()
    87  	default:
    88  		authenticator.EXPECT().Authenticate(gomock.Any(), gomock.Any()).Return(authn, nil).AnyTimes()
    89  		resolver.EXPECT().Resolve(gomock.Any(), gomock.Any()).Return(authn, nil).AnyTimes()
    90  	}
    91  
    92  	fn, err := auth.NewAuthenticator(authenticator, resolver, nil)
    93  	require.NoError(t, err)
    94  	return fn

    97  // serveTestSpec は、合成 spec のミドルウェアへ 1 リクエスト通し、ステータスとハンドラ到達の有無を返す。
    98  func serveTestSpec(t *testing.T, stub authnStub, path, authorization string) (int, bool) {
    99  	t.Helper()
   100  
   101  	loader := openapi3.NewLoader()
   102  	spec, err := loader.LoadFromData([]byte(testSpec))
   103  	require.NoError(t, err)
   104  	require.NoError(t, spec.Validate(context.Background()))
   105  
   106  	e := echo.New()
   107  	mw := Middleware(spec, nil, newTestAuthFunc(t, stub))
   108  
   109  	reached := false
   110  	handler := mw(func(c *echo.Context) error {
   111  		reached = true
   112  		return c.NoContent(http.StatusOK)
   113  	})
   114  
   115  	req := httptest.NewRequestWithContext(context.Background(), http.MethodGet, path, nil)
   116  	if authorization != "" {
   117  		req.Header.Set(echo.HeaderAuthorization, authorization)
   118  	}
   119  	rec := httptest.NewRecorder()
   120  	c := e.NewContext(req, rec)
   121  	c.SetPath(path)
   122  
   123  	if herr := handler(c); herr != nil {
   124  		var he *echo.HTTPError
   125  		if xerrors.As(herr, &he) {
   126  			return he.Code, reached
   127  		}
   128  		return http.StatusInternalServerError, reached
   129  	}
   130  	return rec.Code, reached
   131  }
