    15  // Middleware は、OpenAPI スキーマに基づくリクエストバリデーション（認証は authFunc 経由）を行うミドルウェアを返します。
    16  // バリデーション実行前にリクエストコンテキストへ authn スロット（ctxhelper.WithAuthn）と stream grant スロット
    17  // （ctxhelper.WithStreamGrant）を注入するため、authFunc はそのスロットへ認証結果を書き込めます。
    18  // 認証が失敗したリクエストは、spec がそれを任意と宣言していてもハンドラへ到達しません（failClosed）。
    19  func Middleware(
    20  	spec *openapi3.T,
    21  	skipper echomw.Skipper,
    22  	authFunc openapi3filter.AuthenticationFunc,
    23  ) echo.MiddlewareFunc {
    24  	oapiValidator := oapimw.OapiRequestValidatorWithOptions(spec, &oapimw.Options{
    25  		SilenceServersWarning: true,
    26  		Skipper:               skipper,
    27  		Options: openapi3filter.Options{
    28  			AuthenticationFunc: authFunc,
    29  		},
    30  	})
    31  
    32  	return func(next echo.HandlerFunc) echo.HandlerFunc {
    33  		return func(c *echo.Context) error {
    34  			req := c.Request()
    35  			req = req.WithContext(ctxhelper.WithStreamGrant(ctxhelper.WithAuthn(req.Context())))
    36  			c.SetRequest(req)
    37  
    38  			return oapiValidator(failClosed(next))(c)
    39  		}
    40  	}
    41  }
    42  
    43  // failClosed は、認証に失敗したリクエストをハンドラへ到達させず、その失敗を返します。
    44  // 認証を任意と宣言した operation でも、提示されたうえで検証に失敗した資格情報は通さない。
    45  // 根拠は ADR-0021 (optional-authentication-fail-closed) と README.md「Fail-closed authentication」。
    46  func failClosed(next echo.HandlerFunc) echo.HandlerFunc {
    47  	return func(c *echo.Context) error {
    48  		if err := ctxhelper.AuthnFailure(c.Request().Context()); err != nil {
    49  			return err
    50  		}
    51  		return next(c)
    52  	}
    53  }
