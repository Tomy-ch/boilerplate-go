    68  // Execute は、回収できる instance をすべて処理します。引数は取りません。
    69  // 一部が失敗した場合も、確定した内訳を出力してからエラーを返します（内訳に載せる項目は README「Result counts」）。
    70  func (j *jobImpl) Execute(ctx context.Context, args []string) error {
    71  	ctx, endSpan := j.tracer.Start(ctx)
    72  	defer endSpan()
    73  
    74  	if len(args) > 0 {
    75  		return xerrors.Wrap(errUnknownFlag, args[0])
    76  	}
    77  
    78  	sweeper, err := j.newSweeper(ctx)
    79  	if err != nil {
    80  		j.metrics.CleanupExecuted(ctx, observability.RealtimeResultError)
    81  
    82  		return err
    83  	}
    84  
    85  	result, err := sweeper.Sweep(ctx)
    86  	// 内訳は成否によらず確定しているので、先に計上します。
    87  	j.recordOutcomes(ctx, result)
    88  
    89  	if err != nil {
    90  		j.logging.Named(jobName).Warn(ctx, abortedMessage, resultFields(result)...)
    91  		j.metrics.CleanupExecuted(ctx, observability.RealtimeResultError)
    92  
    93  		return err
    94  	}
    95  
    96  	j.logging.Named(jobName).Info(ctx, resultMessage, resultFields(result)...)
    97  	j.metrics.CleanupExecuted(ctx, observability.RealtimeResultOK)
    98  
    99  	return nil
   100  }

   104  func (j *jobImpl) recordOutcomes(ctx context.Context, r ucrealtime.SweepResult) {
   105  	for outcome, n := range map[string]int{
   106  		outcomeDetected:  r.Detected,
   107  		outcomeClaimed:   r.Claimed,
   108  		outcomeReclaimed: r.Reclaimed,
   109  		outcomeSkipped:   r.Skipped,
   110  		outcomeFailed:    r.Failed,
   111  	} {
   112  		if n > 0 {
   113  			j.metrics.CleanupInstances(ctx, outcome, int64(n))
   114  		}
   115  	}
   116  }
