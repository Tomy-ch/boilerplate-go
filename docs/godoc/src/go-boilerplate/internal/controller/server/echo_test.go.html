    99  		t.Parallel()
   100  
   101  		t.Run("リクエストの主要フィールドが入力構造体に詰め直される", func(t *testing.T) {
   102  			t.Parallel()
   103  
   104  			e := echo.New()
   105  			ctx := context.Background()
   106  			req := httptest.NewRequestWithContext(ctx, http.MethodPost, "/users/123?q=v", nil)
   107  			req.Header.Set(echo.HeaderContentType, echo.MIMEApplicationJSON)
   108  			req.Header.Set("User-Agent", "test-agent")
   109  			req.Host = "example.com"
   110  			rec := httptest.NewRecorder()
   111  			c := e.NewContext(req, rec)
   112  			c.SetPath("/users/:id")
   113  			c.SetPathValues(echo.PathValues{{Name: "id", Value: "123"}})
   114  
   115  			got := BuildHTTPRequestLogInput(c, logging.EventTypeError, redaction.Redactor{})
   116  
   117  			assert.Equal(t, logging.EventTypeError, got.EventType)
   118  			assert.Equal(t, http.MethodPost, got.Method)
   119  			assert.Equal(t, "/users/:id", got.Path)
   120  			assert.Equal(t, "/users/123?q=v", got.URI)
   121  			assert.Equal(t, "example.com", got.Host)
   122  			assert.Equal(t, "http", got.Scheme)
   123  			assert.Equal(t, "test-agent", got.UserAgent)
   124  			assert.Equal(t, echo.MIMEApplicationJSON, got.ContentType)
   125  			assert.Equal(t, map[string]string{"id": "123"}, got.PathParams)
   126  			assert.Equal(t, []string{"v"}, got.QueryParams["q"])
   127  			assert.False(t, got.EventAt.IsZero())
   128  		})
   129
