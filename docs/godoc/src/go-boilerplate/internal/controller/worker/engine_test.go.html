    54  // startEngine は、worker を別 goroutine で起動し、cancel と完了 channel を返します。
    55  func startEngine(t *testing.T, set Settings, log logging.Logger, w bw.Worker) (context.CancelFunc, <-chan error) {
    56  	t.Helper()
    57  	eng, err := New([]bw.Worker{w}, set, observability.NewNoopTracerFactory(t), observability.NewNoopWorkerMetrics(t), log)
    58  	require.NoError(t, err)
    59  
    60  	ctx, cancel := context.WithCancel(context.Background())
    61  	done := make(chan error, 1)
    62  	go func() { done <- eng.Run(ctx, w.Name()) }()
    63  	return cancel, done
    64  }

   411  func engineRunCircuitOpenAndRecover(t *testing.T) {
   412  	t.Parallel()
   413  
   414  	f := testkit.NewFake()
   415  	f.Enqueue(bw.Message{ID: "a"})
   416  	var failing atomic.Bool
   417  	failing.Store(true)
   418  	w := testWorker{name: "w", cons: f, handler: handlerFunc(func(context.Context, bw.Message) error {
   419  		if failing.Load() {
   420  			return xerrors.Wrap(apperror.ErrRetryable, "downstream down")
   421  		}
   422  		return nil
   423  	})}
   424  	set := baseSettings()
   425  	set.Concurrency = 1
   426  	set.MaxInFlight = 1
   427  	set.BatchSize = 1
   428  	set.CircuitFailureThreshold = 2
   429  	// cooldown を 1s 級へ引き上げ、-race のスローダウン下でも観測窓に Half-open probe が割り込まないようにする。
   430  	set.CircuitOpenBackoffInitial = time.Second
   431  	set.CircuitOpenBackoffMax = time.Second
   432  
   433  	cancel, done := startEngine(t, set, logging.NewTestLogger(t), w)
   434  	defer func() { cancel(); <-done }()
   435  
   436  	// B4-1: 連続 2 失敗で Open へ遷移する。MaxInFlight=1 のため poll loop は失敗確定
   437  	// （onFailure で trip）まで次の Receive に進めず、Open 遷移後は cooldown 待機に入る。
   438  	require.Eventually(t, func() bool { return len(f.NackedIDs()) >= 2 }, eventually, tick)
   439  	// Open 中は intake が止まり Receive されないため、cooldown（1s）より十分短い窓で
   440  	// Nack も Ack も増えないことを確認する（実時間 sleep には依存しない）。
   441  	require.Never(t, func() bool {
   442  		return len(f.NackedIDs()) > 2 || len(f.AckedIDs()) > 0
   443  	}, 200*time.Millisecond, tick)
   444  	// 再配送された "a" が Receive されず queue に滞留していることが intake 停止の証跡。
   445  	assert.Equal(t, 1, f.QueueLen())
   446  
   447  	// B4-2: 回復させると cooldown 経過後の Half-open 試行で成功し Ack される。
   448  	failing.Store(false)
   449  	require.Eventually(t, func() bool { return len(f.AckedIDs()) >= 1 }, 5*time.Second, tick)
   450  }

   452  func engineRunFailureLogStructuredFields(t *testing.T) {
   453  	t.Parallel()
   454  
   455  	logger, observed := logging.NewObservedTestLogger(t)
   456  	f := testkit.NewFake()
   457  	f.Enqueue(bw.Message{ID: "a"})
   458  	w := testWorker{name: "w", cons: f, handler: handlerFunc(func(context.Context, bw.Message) error {
   459  		return xerrors.Wrap(apperror.ErrRetryable, "downstream")
   460  	})}
   461  
   462  	cancel, done := startEngine(t, baseSettings(), logger, w)
   463  	defer func() { cancel(); <-done }()
   464  
   465  	require.Eventually(t, func() bool {
   466  		for _, e := range observed.FilterMessage("retryable failure, nacked").All() {
   467  			cm := e.ContextMap()
   468  			if cm[logging.MessageIDKey] == "a" && cm[logging.ReceiveCountKey] == int64(1) {
   469  				return true
   470  			}
   471  		}
   472  		return false
   473  	}, eventually, tick)
   474  
   475  	// 再配送(retryable Nack)で ReceiveCount=2 の失敗ログも生じ、並行処理下ではログ順序が
   476  	// 保証されないため index を固定せず、初回配送(ReceiveCount=1)のログを ID で特定して検証する。
   477  	var target map[string]any
   478  	for _, e := range observed.FilterMessage("retryable failure, nacked").All() {
   479  		cm := e.ContextMap()
   480  		if cm[logging.MessageIDKey] == "a" && cm[logging.ReceiveCountKey] == int64(1) {
   481  			target = cm
   482  			break
   483  		}
   484  	}
   485  	require.NotNil(t, target, "初回配送(ReceiveCount=1)の失敗ログが存在すること")

   663  func TestEngine_drain(t *testing.T) {
   664  	t.Parallel()
   665  
   666  	t.Run("正常系", func(t *testing.T) {
   667  		t.Parallel()
   668  
   669  		t.Run("ctx cancel 後も in-flight は drain 期限内に完了して Ack される", func(t *testing.T) {
   670  			t.Parallel()
   671  
   672  			f := testkit.NewFake()
   673  			f.Enqueue(bw.Message{ID: "a"})
   674  			started := make(chan struct{}, 1)
   675  			release := make(chan struct{})
   676  			w := testWorker{name: "w", cons: f, handler: handlerFunc(func(context.Context, bw.Message) error {
   677  				started <- struct{}{}
   678  				<-release
   679  				return nil
   680  			})}
   681  
   682  			cancel, done := startEngine(t, baseSettings(), logging.NewTestLogger(t), w)
   683  
   684  			<-started // in-flight に入った
   685  			cancel()  // Receive 停止
   686  			close(release)
   687  
   688  			select {
   689  			case <-done:
   690  			case <-time.After(eventually):
   691  				t.Fatal("drain が完了しなかった")
   692  			}
   693  			assert.Equal(t, []string{"a"}, f.AckedIDs())
   694  		})
   695  	})
   696  
   697  	t.Run("異常系", func(t *testing.T) {
   698  		t.Parallel()
   699  
   700  		t.Run("DrainTimeout を超えて完了しない in-flight は待たずに抜け Ack されない", func(t *testing.T) {
   701  			t.Parallel()
   702  
   703  			f := testkit.NewFake()
   704  			w := testWorker{name: "w", cons: f, handler: handlerFunc(func(context.Context, bw.Message) error { return nil })}
   705  			set := baseSettings()
   706  			set.DrainTimeout = 20 * time.Millisecond
   707  
   708  			r := newRun(newTestEngine(t, set, w), w)
   709  			r.wg.Add(1) // 完了しない in-flight を模す
   710  
   711  			start := time.Now()
   712  			r.drain()
   713  			elapsed := time.Since(start)
   714  			r.wg.Done() // drain 用 goroutine をリークさせないため後始末
   715  
   716  			assert.GreaterOrEqual(t, elapsed, set.DrainTimeout)
   717  			assert.Less(t, elapsed, eventually)
   718  			assert.Empty(t, f.AckedIDs())
   719  		})
