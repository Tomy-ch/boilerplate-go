    10  // maxTraceStateBytes は、持ち回る tracestate の上限です（W3C の推奨上限）。
    11  // tracestate は client が任意に注入でき、32 member × 512 バイトまで文法上は通ります。
    12  // link を張るのに必要なのは traceparent だけなので、超えた分は運ばずに捨てます。
    13  const maxTraceStateBytes = 512

    15  // traceContextPropagator は、traceparent/tracestate のみを伝搬する propagator です（Baggage を含まない）。
    16  var traceContextPropagator propagation.TextMapPropagator = propagation.TraceContext{}

    18  // mapCarrier は、map[string]string を propagation のキャリアとして扱うアダプタです。
    19  type mapCarrier map[string]string
    20  
    21  func (c mapCarrier) Get(key string) string { return c[key] }
    22  func (c mapCarrier) Set(key, value string) { c[key] = value }
    23  func (c mapCarrier) Keys() []string {
    24  	keys := make([]string, 0, len(c))
    25  	for k := range c {
    26  		keys = append(keys, k)
    27  	}
    28  	return keys
    29  }

    37  // extractFromCarrier は、prop から trace context を抽出して ctx に付与します。
    38  func extractFromCarrier(ctx context.Context, attrs map[string]string, prop propagation.TextMapPropagator) context.Context {
    39  	if len(attrs) == 0 {
    40  		return ctx
    41  	}
    42  	return prop.Extract(ctx, mapCarrier(attrs))
    43  }

    45  // TraceContextFromCarrier は、attrs から trace context の項目だけを取り出した carrier を返します。
    46  // 非同期のホップで起点 trace を持ち回るとき、運ぶ側が項目名を知らずに済むようにするためのものです。
    47  // 該当が無ければ nil を返します。
    48  func TraceContextFromCarrier(attrs map[string]string) map[string]string {
    49  	if len(attrs) == 0 {
    50  		return nil
    51  	}
    52  
    53  	out := map[string]string{}
    54  	for _, key := range traceContextPropagator.Fields() {
    55  		value, ok := attrs[key]
    56  		if !ok || value == "" || len(value) > maxTraceStateBytes {
    57  			continue
    58  		}
    59  
    60  		out[key] = value
    61  	}
    62  
    63  	if len(out) == 0 {
    64  		return nil
    65  	}
    66  
    67  	return out
    68  }
