   259  // lagSecondsOfChannel は、channel 属性で該当するデータ点の値を返します。
   260  // 単一系列前提の DataPoints[0] 読みでは、複数チャネルが同時に記録されたときに
   261  // 系列が混ざる退行（属性を落として 1 系列へ潰す等）を検出できません。
   262  func lagSecondsOfChannel(t *testing.T, rm metricdata.ResourceMetrics, channel string) int64 {
   263  	t.Helper()
   264  
   265  	g, ok := metricByName(t, rm, "outbox.lag_seconds").Data.(metricdata.Gauge[int64])
   266  	require.True(t, ok)
   267  	for _, dp := range g.DataPoints {
   268  		v, found := dp.Attributes.Value(attribute.Key("channel"))
   269  		if found && v.AsString() == channel {
   270  			return dp.Value
   271  		}
   272  	}
   273  	t.Fatalf("channel %s のデータ点が見つかりません", channel)
   274  	return 0
   275  }
