   119  // CounterValue は、metricName の counter のうち labelKey=labelValue の点の値を返します。
   120  // 該当する点が無ければ -1 を返します（0 は「0 が計上された」であり、両者は区別すべき別の事実です）。
   121  func (o *ObservedRealtimeMetrics) CounterValue(t *testing.T, metricName, labelKey, labelValue string) int64 {
   122  	t.Helper()
   123  
   124  	var rm metricdata.ResourceMetrics
   125  	if err := o.reader.Collect(context.Background(), &rm); err != nil {
   126  		t.Fatalf("failed to collect realtime metrics: %v", err)
   127  	}
   128  
   129  	m, found := findMetric(rm, metricName)
   130  	if !found {
   131  		return -1
   132  	}
   133  
   134  	sum, ok := m.Data.(metricdata.Sum[int64])
   135  	if !ok {
   136  		t.Fatalf("metric %s is not an int64 counter", metricName)
   137  	}
   138  
   139  	for _, dp := range sum.DataPoints {
   140  		if matchesLabel(dp.Attributes, labelKey, labelValue) {
   141  			return dp.Value
   142  		}
   143  	}
   144  
   145  	return -1
   146  }

   169  func (o *ObservedRealtimeMetrics) HistogramCount(t *testing.T, metricName string) uint64 {
   170  	t.Helper()
   171  
   172  	var rm metricdata.ResourceMetrics
   173  	if err := o.reader.Collect(context.Background(), &rm); err != nil {
   174  		t.Fatalf("failed to collect realtime metrics: %v", err)

   190  		return 0
   191  	}
   192  }

   218  // LabelValues は、metricName の counter に付いた labelKey の値を返します（未計上なら空）。
   219  // 返るのは counter のデータポイント単位、すなわち計上された label 組であり、記録回数ではありません
   220  // （同一 label 組へ何回記録しても cumulative counter の 1 データポイントに集約されます）。
   221  // metricName が int64 counter でない場合は、テストキットの誤用としてテストを失敗させます。
   222  func (o *ObservedHTTPClientMetrics) LabelValues(t *testing.T, metricName, labelKey string) []string {
   223  	t.Helper()
   224  
   225  	var rm metricdata.ResourceMetrics
   226  	if err := o.reader.Collect(context.Background(), &rm); err != nil {
   227  		t.Fatalf("failed to collect http client metrics: %v", err)
   228  	}
   229  
   230  	values, ok := counterLabelValues(rm, metricName, labelKey)
   231  	if !ok {
   232  		t.Fatalf("metric %s is not an int64 counter", metricName)
   233  	}
   234  	return values
   235  }

   237  // counterLabelValues は、rm から metricName の int64 counter を探し、各データポイントの labelKey の値を返します。
   238  // metricName の指標が int64 counter でない場合は ok=false を返します（未計上の場合は空・ok=true）。
   239  func counterLabelValues(rm metricdata.ResourceMetrics, metricName, labelKey string) ([]string, bool) {
   240  	var values []string
   241  	for _, sm := range rm.ScopeMetrics {
   242  		for _, m := range sm.Metrics {
   243  			if m.Name != metricName {
   244  				continue
   245  			}
   246  			sum, ok := m.Data.(metricdata.Sum[int64])
   247  			if !ok {
   248  				return nil, false
   249  			}
   250  			for _, dp := range sum.DataPoints {
   251  				if v, found := dp.Attributes.Value(attribute.Key(labelKey)); found {
   252  					values = append(values, v.String())
   253  				}
   254  			}
   255  		}
   256  	}
   257  	return values, true
   258  }

   319  // NewRecordingTracerProvider は、終了した span を保持する TracerProvider と、保持した span を返す関数を返します。
   320  // 計装が span の属性に何を載せたかをテストで検証するためのもので、テスト終了時に provider を停止します。
   321  func NewRecordingTracerProvider(t *testing.T) (trace.TracerProvider, func() []sdktrace.ReadOnlySpan) {
   322  	t.Helper()
   323  
   324  	rec := &spanRecorder{}
   325  	tp := sdktrace.NewTracerProvider(sdktrace.WithSyncer(rec))
   326  	t.Cleanup(func() { require.NoError(t, tp.Shutdown(context.Background())) })
   327  
   328  	return tp, func() []sdktrace.ReadOnlySpan {
   329  		rec.mu.Lock()

   331  		return append([]sdktrace.ReadOnlySpan(nil), rec.spans...)
   332  	}
   333  }

   358  // SpanAttributeValues は、属性 filterKey が filterValue に一致する span について、その全属性の値を文字列で返します。
   359  // 計装が span に載せた値の中に特定の文字列（資格情報など）が無いことを表明するために使います。
   360  func SpanAttributeValues(spans []sdktrace.ReadOnlySpan, filterKey, filterValue string) []string {
   361  	var values []string
   362  	for _, span := range spans {
   363  		if !hasAttribute(span, filterKey, filterValue) {
   364  			continue
   365  		}
   366  		for _, attr := range span.Attributes() {
   367  			values = append(values, attr.Value.AsString())
   368  		}
   369  	}
   370  	return values
   371  }

   373  // hasAttribute は、span が key に value を持つかを返します。
   374  func hasAttribute(span sdktrace.ReadOnlySpan, key, value string) bool {
   375  	for _, attr := range span.Attributes() {
   376  		if string(attr.Key) == key && attr.Value.AsString() == value {
   377  			return true
   378  		}
   379  	}
   380  	return false
   381  }
