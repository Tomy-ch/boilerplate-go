    10  // outboxMeterName は、outbox relay 計装の meter 名です。
    11  const outboxMeterName = "go-boilerplate/outbox"
    12  
    13  // OutboxMetrics は、outbox relay 所有の outbox 固有計装です。
    14  // publish 自体の count/latency/error は httpclient(downstream=outbox) の計装で賄うため、
    15  // ここでは outbox 固有の SLI（lag / blocked streams）と dead 数のみを持ちます。
    16  // relay は配送チャネルごとに独立して動くため、いずれもチャネル属性を伴います。
    17  type OutboxMetrics struct {
    18  	lagSeconds     metric.Int64Gauge
    19  	dead           metric.Int64Counter
    20  	blockedStreams metric.Int64Gauge
    21  }
    22  
    23  // NewOutboxMetrics は、注入された MeterProvider から outbox relay の計装一式を生成します。
    24  func NewOutboxMetrics(mp metric.MeterProvider) (*OutboxMetrics, error) {
    25  	b := &meterBuilder{m: mp.Meter(outboxMeterName)}
    26  	om := &OutboxMetrics{
    27  		lagSeconds:     b.gauge("outbox.lag_seconds", "最古 pending 行の経過秒数（SLI=outbox lag）"),
    28  		dead:           b.counter("outbox.dead", "恒久エラーと判定して dead 化したメッセージ数"),
    29  		blockedStreams: b.gauge("outbox.blocked_streams", "先頭行が dead で進行が止まっているストリーム数"),
    30  	}
    31  	if b.err != nil {
    32  		return nil, b.err
    33  	}
    34  	return om, nil
    35  }

    37  // SetLagSeconds は、最古 pending 行の経過秒数（outbox lag）をチャネル別に記録します（pending 無しは 0）。
    38  func (m *OutboxMetrics) SetLagSeconds(ctx context.Context, channel string, seconds int64) {
