    56  // sumInt64 は、指定名の Sum[int64] から属性が一致する 1 点の値を取り出します。
    57  func sumInt64(t *testing.T, rm metricdata.ResourceMetrics, name string, attrs ...attribute.KeyValue) int64 {
    58  	t.Helper()
    59  
    60  	s, ok := metricByName(t, rm, name).Data.(metricdata.Sum[int64])
    61  	require.True(t, ok, "%s が Sum[int64] であること", name)
    62  
    63  	want := attribute.NewSet(attrs...)
    64  	for _, dp := range s.DataPoints {
    65  		if dp.Attributes.Equals(&want) {
    66  			return dp.Value
    67  		}
    68  	}
    69  	t.Fatalf("metric %s に期待した属性の点が無い: %v", name, attrs)
    70  
    71  	return 0
    72  }
