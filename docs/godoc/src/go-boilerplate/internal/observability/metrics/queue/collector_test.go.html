   352  // drainDescs は、Describe が channel へ送出した Desc を全件読み出します。
   353  // Describe は consumer が読むまでブロックし得るため、送出側を別 goroutine に置きます。
   354  func drainDescs(t *testing.T, describe func(chan<- *prometheus.Desc)) []*prometheus.Desc {
   355  	t.Helper()
   356  	ch := make(chan *prometheus.Desc)
   357  	go func() {
   358  		describe(ch)
   359  		close(ch)
   360  	}()
   361  	var got []*prometheus.Desc
   362  	for d := range ch {
   363  		got = append(got, d)
   364  	}
   365  	return got
   366  }
