    25  // collectTimeout は、1 回の scrape（Collect）で全 target の滞留量取得に許す上限時間です。
    26  // registry は Gather のたびに Collect を goroutine で並行起動するため、deadline なしで
    27  // broker API（SQS GetQueueAttributes 等）を呼ぶと、応答不能時に goroutine が蓄積して
    28  // リソースを枯渇させ得ます。これを防ぐため scrape 単位で timeout を設けます。

   107  // Collect は、各 target から滞留量を取得して Prometheus に提供します。
   108  // provider がエラーを返した場合は depth を出さず、収集失敗 counter を増やします。
   109  func (c *StatsCollector) Collect(ch chan<- prometheus.Metric) {
   110  	ctx, cancel := context.WithTimeout(context.Background(), collectTimeout)
   111  	defer cancel()
   112  
   113  	// 遅い 1 provider が共有 deadline を食い潰して健全な target まで収集失敗させるのを防ぐため、
   114  	// 全 target を並行収集する。channel 送信は Wait 後に単一 goroutine から直列で行う（並行送信回避）。
   115  	results := make([]collectResult, len(c.targets))
   116  	var wg sync.WaitGroup
   117  	for i, t := range c.targets {
   118  		wg.Add(1)
   119  		go func(i int, t Target) {
   120  			defer wg.Done()
   121  			stats, err := t.Provider.QueueStats(ctx)
   122  			results[i] = collectResult{stats: stats, err: err}
   123  		}(i, t)
   124  	}
   125  	wg.Wait()
   126  
   127  	for i, t := range c.targets {
   128  		if results[i].err != nil {
   129  			// QueueStats は source / DLQ いずれの失敗かを区別しないため queue="unknown" とします。
   130  			c.recordFailure(t.WorkerName, t.Adapter, queueUnknown)
   131  			continue
   132  		}
   133  		c.collectDepth(ch, t, queueSource, results[i].stats.Source)
   134  		if results[i].stats.DLQ != nil {
   135  			c.collectDepth(ch, t, queueDLQ, *results[i].stats.DLQ)
   136  		}
   137  	}
   138  	c.emitFailures(ch)
   139  }

   141  // collectDepth は、1 queue の滞留量を state 別の gauge として出力します。
   142  func (c *StatsCollector) collectDepth(ch chan<- prometheus.Metric, t Target, queue string, d worker.QueueDepth) {
   143  	ch <- prometheus.MustNewConstMetric(
   144  		c.depthDesc, prometheus.GaugeValue, float64(d.Visible), t.WorkerName, t.Adapter, queue, stateVisible,

   147  		c.depthDesc, prometheus.GaugeValue, float64(d.InFlight), t.WorkerName, t.Adapter, queue, stateNotVisible,
   148  	)
   149  	ch <- prometheus.MustNewConstMetric(
   150  		c.depthDesc, prometheus.GaugeValue, float64(d.Delayed), t.WorkerName, t.Adapter, queue, stateDelayed,
   151  	)
   152  }

   154  // recordFailure は、収集失敗の累積回数を増やします。
   155  func (c *StatsCollector) recordFailure(workerName, adapter, queue string) {
   156  	c.mu.Lock()
   157  	defer c.mu.Unlock()
   158  	c.failures[failureKey{worker: workerName, adapter: adapter, queue: queue}]++
   159  }

   161  // emitFailures は、累積した収集失敗回数を counter として出力します。
   162  // channel 送信は consumer が読むまでブロックし得るため、ロック下では failures の
   163  // スナップショットを取るに留め、送信はロック解放後に行います。
   164  func (c *StatsCollector) emitFailures(ch chan<- prometheus.Metric) {
   165  	c.mu.Lock()
   166  	snapshot := make([]struct {
   167  		key failureKey
   168  		v   int64
   169  	}, 0, len(c.failures))
   170  	for k, v := range c.failures {
   171  		snapshot = append(snapshot, struct {
   172  			key failureKey
   173  			v   int64
   174  		}{key: k, v: v})
   175  	}
   176  	c.mu.Unlock()
   177  
   178  	for _, s := range snapshot {
   179  		ch <- prometheus.MustNewConstMetric(
   180  			c.failureDesc, prometheus.CounterValue, float64(s.v), s.key.worker, s.key.adapter, s.key.queue,
   181  		)
   182  	}
   183  }
