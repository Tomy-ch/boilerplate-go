    65  // NewRealtimeMetrics は、注入された MeterProvider から Realtime Delivery の計装一式を生成します。
    66  // いずれかの計装生成失敗で error を返します。
    67  func NewRealtimeMetrics(mp metric.MeterProvider) (*RealtimeMetrics, error) {
    68  	b := &meterBuilder{m: mp.Meter(realtimeMeterName)}
    69  	rm := &RealtimeMetrics{
    70  		connectionsActive:     b.upDownCounter("realtime.connections.active", "この instance が索引に持っている SSE 接続数"),
    71  		connectionsAccepted:   b.counter("realtime.connections.accepted", "レスポンスを確定して配信を始めた接続数"),
    72  		connectionsReconnects: b.counter("realtime.connections.reconnects", "cursor を伴って張り直された接続数"),
    73  		connectionsRejected:   b.counter("realtime.connections.rejected", "確定せず 503 で断った接続数"),
    74  		connectionsClosed:     b.counter("realtime.connections.closed", "索引から外れた接続数"),
    75  		connectionDurationMs:  b.histogram("realtime.connections.duration_ms", "接続が索引に居た時間(ミリ秒)"),
    76  
    77  		replayExecutions:        b.counter("realtime.replay.executions", "EventLog を読み進めた回数"),
    78  		replayEvents:            b.counter("realtime.replay.events", "読み進めて送出した event 数"),
    79  		replayDepth:             b.countHistogram("realtime.replay.depth", "1 回の読み進めで追いついた event 数"),
    80  		replayFailures:          b.counter("realtime.replay.failures", "EventLog の読み取りに失敗した回数"),
    81  		replayInFlight:          b.upDownCounter("realtime.replay.in_flight", "同時に走っている読み進めの本数"),
    82  		replayAdmissionTimeouts: b.counter("realtime.replay.admission_timeouts", "初回 replay の枠を待ち切れず諦めた回数"),
    83  		catchUpLagMs:            b.histogram("realtime.catchup.lag_ms", "wakeup 受信から読み終わりまでの時間(ミリ秒)"),
    84  
    85  		deliveryLatencyMs:     b.histogram("realtime.delivery.latency_ms", "event の発生から SSE へ書き終わるまでの時間(ミリ秒)"),
    86  		eventLogAppends:       b.counter("realtime.eventlog.appends", "EventLog への追記回数"),
    87  		eventLogLagMs:         b.histogram("realtime.eventlog.lag_ms", "outbox 行の作成から EventLog 追記までの時間(ミリ秒)"),
    88  		wakeupPublishFailures: b.counter("realtime.wakeup.publish_failures", "wakeup の publish に失敗した回数"),
    89  		recoveryExecutions:    b.counter("realtime.recovery.executions", "消えた受信先を作り直した回数"),
    90  
    91  		leaseHeartbeatFailures: b.counter("realtime.lease.heartbeat_failures", "instance lease の heartbeat に失敗した回数"),
    92  		cleanupExecutions:      b.counter("realtime.cleanup.executions", "orphan cleanup ジョブの実行回数"),
    93  		cleanupInstances:       b.counter("realtime.cleanup.instances", "orphan cleanup が扱った instance 数"),
    94  	}
    95  	if b.err != nil {

   109  func (m *RealtimeMetrics) ConnectionEstablished(ctx context.Context, resumed bool) {
