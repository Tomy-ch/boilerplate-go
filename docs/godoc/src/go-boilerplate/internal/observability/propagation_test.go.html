   113  func TestExtractFromCarrier(t *testing.T) {
   114  	t.Run("正常系", func(t *testing.T) {
   115  		t.Run("グローバル伝播器で attrs の traceparent から trace を復元する", func(t *testing.T) {
   116  			// 公開関数は otel.GetTextMapPropagator() を参照するため、TraceContext を一時登録する。
   117  			prev := otel.GetTextMapPropagator()
   118  			otel.SetTextMapPropagator(propagation.TraceContext{})
   119  			t.Cleanup(func() { otel.SetTextMapPropagator(prev) })
   120  
   121  			sc := testSampledSpanContext(true)
   122  			traceID := sc.TraceID()
   123  			spanID := sc.SpanID()
   124  			carrier := propagation.MapCarrier{}
   125  			propagation.TraceContext{}.Inject(trace.ContextWithSpanContext(context.Background(), sc), carrier)
   126  
   127  			got := ExtractFromCarrier(context.Background(), map[string]string(carrier))
   128  
   129  			gsc := trace.SpanContextFromContext(got)
   130  			assert.True(t, gsc.HasTraceID())
   131  			assert.Equal(t, traceID, gsc.TraceID())
   132  			assert.Equal(t, spanID, gsc.SpanID())
   133  		})
   134  	})
   135  
   136  	t.Run("異常系", func(t *testing.T) {
   137  		t.Run("attrs が空なら ctx をそのまま返す", func(t *testing.T) {
   138  			ctx := context.Background()
   139  
   140  			// グローバル伝播器を変更せず公開経路（otel.GetTextMapPropagator 利用）を通す。
   141  			assert.Equal(t, ctx, ExtractFromCarrier(ctx, nil))
   142  		})
   143  	})

   147  func TestInjectTraceContextToCarrier(t *testing.T) {
   148  	t.Run("正常系", func(t *testing.T) {
   149  		t.Run("trace context を inject しつつ baggage は転送しない", func(t *testing.T) {
   150  			// span が無いと traceparent も書かれず NotContains が疑似陽性になるため、
   151  			// span 有りで traceparent が書かれた上で baggage だけが落ちることを検証する。
   152  			sc := testSampledSpanContext(false)
   153  			ctx := trace.ContextWithSpanContext(context.Background(), sc)
   154  
   155  			member, err := baggage.NewMember("tenant", "acme")
   156  			require.NoError(t, err)
   157  			bag, err := baggage.New(member)
   158  			require.NoError(t, err)
   159  			ctx = baggage.ContextWithBaggage(ctx, bag)
   160  
   161  			attrs := map[string]string{}
   162  			InjectTraceContextToCarrier(ctx, attrs)
   163  
   164  			assert.Contains(t, attrs, "traceparent")
   165  			assert.NotContains(t, attrs, "baggage")
   166  		})
   167  
   168  		t.Run("グローバル伝播器に依らず TraceContext 限定で inject する", func(t *testing.T) {
   169  			// グローバル伝播器を Baggage 単独に差し替えても traceparent が書かれる。
   170  			// すなわち traceContextPropagator 固定で otel.GetTextMapPropagator を見ない。
   171  			prev := otel.GetTextMapPropagator()
   172  			otel.SetTextMapPropagator(propagation.Baggage{})
   173  			t.Cleanup(func() { otel.SetTextMapPropagator(prev) })
   174  
   175  			sc := testSampledSpanContext(false)
   176  
   177  			attrs := map[string]string{}
   178  			InjectTraceContextToCarrier(trace.ContextWithSpanContext(context.Background(), sc), attrs)
   179  
   180  			assert.Contains(t, attrs, "traceparent")
   181  		})
   182  	})
   183  
   184  	t.Run("異常系", func(t *testing.T) {
   185  		t.Run("attrs が nil なら何もせずパニックしない", func(t *testing.T) {
   186  			assert.NotPanics(t, func() { InjectTraceContextToCarrier(newSampledContext(), nil) })
   187  		})
   188  	})

   252  func TestTraceContextFromCarrier(t *testing.T) {
   253  	t.Parallel()
   254  
   255  	const traceParent = "00-4bf92f3577b34da6a3ce929d0e0e4736-00f067aa0ba902b7-01"
   256  
   257  	t.Run("正常系", func(t *testing.T) {
   258  		t.Parallel()
   259  
   260  		t.Run("trace context の項目だけを取り出す", func(t *testing.T) {
   261  			t.Parallel()
   262  
   263  			got := TraceContextFromCarrier(map[string]string{
   264  				"traceparent":     traceParent,
   265  				"tracestate":      "vendor=1",
   266  				"Idempotency-Key": "must-not-be-carried",
   267  			})
   268  
   269  			// 運ぶのは trace の再構成に要る分だけ。carrier を丸ごと持ち回ると、
   270  			// 無関係なヘッダまで EventLog の item に残ってしまう。
   271  			assert.Equal(t, map[string]string{"traceparent": traceParent, "tracestate": "vendor=1"}, got)
   272  		})
   273  
   274  		t.Run("取り出した carrier から SpanContext を復元できる", func(t *testing.T) {
   275  			t.Parallel()
   276  
   277  			got := TraceContextFromCarrier(map[string]string{"traceparent": traceParent})
   278  			sc := trace.SpanContextFromContext(
   279  				extractFromCarrier(context.Background(), got, traceContextPropagator),
   280  			)
   281  
   282  			require.True(t, sc.IsValid())
   283  			assert.Equal(t, "4bf92f3577b34da6a3ce929d0e0e4736", sc.TraceID().String())
   284  		})
   285  
   286  		t.Run("長すぎる tracestate は運ばない", func(t *testing.T) {
   287  			t.Parallel()
   288  
   289  			// tracestate は client が任意に注入できる。上限を置かないと 1 event あたり
   290  			// 16 KiB 近くまで膨らみ、封筒の直列化上限の外側で item を太らせる。
   291  			long := strings.Repeat("a", maxTraceStateBytes+1)
   292  
   293  			got := TraceContextFromCarrier(map[string]string{"traceparent": traceParent, "tracestate": long})
   294  
   295  			assert.Equal(t, map[string]string{"traceparent": traceParent}, got)
   296  		})
   297  	})
   298  
   299  	t.Run("異常系", func(t *testing.T) {
   300  		t.Parallel()
   301  
   302  		t.Run("trace context が無ければ nil を返す", func(t *testing.T) {
   303  			t.Parallel()
   304  
   305  			assert.Nil(t, TraceContextFromCarrier(nil))
   306  			assert.Nil(t, TraceContextFromCarrier(map[string]string{"Idempotency-Key": "k"}))
   307  			assert.Nil(t, TraceContextFromCarrier(map[string]string{"traceparent": ""}))
   308  		})
   309  	})
