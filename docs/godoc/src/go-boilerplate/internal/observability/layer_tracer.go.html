    63  // StartWithLink は、新しい span を開始し、carrier が指す trace への link を張ります。

    66  func (lt LayerTracer) StartWithLink(
    67  	ctx context.Context,
    68  	carrier map[string]string,
    69  	opts ...trace.SpanStartOption,
    70  ) (context.Context, func()) {
    71  	if lt.funcName == "" {
    72  		full := getCallerFullName()
    73  		lt.funcName = fnmeta.ExtractFunctionName(full)

   102  func RunWithSpan[T any](
   103  	parentCtx context.Context,
   104  	lt LayerTracer,
   105  	layer layerName,
   106  	pkg, funcName string,
   107  	fn func(ctx context.Context) (T, error),
   108  ) (context.Context, T, error) {
   109  	spanName := BuildSpanName(string(layer), pkg, funcName)
   110  
   111  	_, childCtx, end := StartSpanWithParent(parentCtx, lt, spanName)
   112  	defer end()
   113  
   114  	v, err := fn(childCtx)
   115  	if err != nil {
   116  		var zero T
   117  		return childCtx, zero, err
   118  	}
   119  
   120  	return childCtx, v, nil
   121  }
