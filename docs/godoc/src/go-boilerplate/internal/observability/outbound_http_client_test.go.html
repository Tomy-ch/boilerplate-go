   143  func Test_limitedRedirectForOutbound(t *testing.T) {
   144  	t.Parallel()
   145  
   146  	// redirectTo は、resp を受けて next へ向かう CheckRedirect の入力を組み立てます。
   147  	redirectTo := func(t *testing.T, status, hops int) (*http.Request, []*http.Request) {
   148  		t.Helper()
   149  		req, err := http.NewRequestWithContext(context.Background(), http.MethodPost, "http://example.com/next", nil)
   150  		require.NoError(t, err)
   151  		req.Response = &http.Response{StatusCode: status}
   152  
   153  		return req, make([]*http.Request, hops)
   154  	}
   155  
   156  	t.Run("正常系", func(t *testing.T) {
   157  		t.Parallel()
   158  
   159  		t.Run("メソッドと本文を保つ 307 / 308 は追従する", func(t *testing.T) {
   160  			t.Parallel()
   161  
   162  			for _, status := range []int{http.StatusTemporaryRedirect, http.StatusPermanentRedirect} {
   163  				req, via := redirectTo(t, status, 1)
   164  				assert.NoErrorf(t, limitedRedirectForOutbound(req, via), "status %d で追従していない", status)
   165  			}
   166  		})
   167  
   168  		t.Run("メソッドが変わりうる 301 / 302 / 303 は追従しない", func(t *testing.T) {
   169  			t.Parallel()
   170  
   171  			for _, status := range []int{http.StatusMovedPermanently, http.StatusFound, http.StatusSeeOther} {
   172  				req, via := redirectTo(t, status, 1)
   173  				assert.Equalf(t, http.ErrUseLastResponse, limitedRedirectForOutbound(req, via),
   174  					"status %d で追従している", status)
   175  			}
   176  		})
   177  
   178  		t.Run("追従回数の上限を超えたら 307 でも止める", func(t *testing.T) {
   179  			t.Parallel()
   180  
   181  			req, via := redirectTo(t, http.StatusTemporaryRedirect, maxOutboundRedirects)
   182  
   183  			assert.Equal(t, http.ErrUseLastResponse, limitedRedirectForOutbound(req, via))
   184  		})
   185  
   186  		t.Run("元レスポンスが無ければ追従しない", func(t *testing.T) {
   187  			t.Parallel()

   189  			req, err := http.NewRequestWithContext(context.Background(), http.MethodGet, "http://example.com", nil)
   190  			require.NoError(t, err)
