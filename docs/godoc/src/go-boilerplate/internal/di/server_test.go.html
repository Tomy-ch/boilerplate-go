    75  // realtimeSubstrate は、Realtime Delivery の外部依存を IF レベルで差し替えます。
    76  // serve lifecycle（起動時の probe → 受信先の用意 → 常駐処理 → drain → 片付け）は本物のまま、
    77  // DynamoDB と SNS / SQS へ出ていく往復だけが止まります。
    78  //
    79  // 差し替えた型の構築子とその上流（DynamoDB クライアント・fan-out の組み立て）は走りません。
    80  // fx の decorator は元の構築子を呼ばないためで、REALTIME_TOPIC 未設定のような
    81  // 「app.Start より前に落ちる」分岐はここでは露見せず、app-di-startup-check が受け持ちます。
    82  //
    83  // 期待は回数で縛らず AnyTimes にします。sample の realtime adapter を消すと runtime ごと
    84  // graph から外れて 1 度も呼ばれなくなるため、回数を固定すると撤去後の世界で落ちます。
    85  // 参加者が実際に走る順序と回数は internal/di/module/realtime_test.go が固定しています。
    86  func realtimeSubstrate(t *testing.T, ctrl *gomock.Controller) fx.Option {
    87  	t.Helper()
    88  
    89  	eventLog := mock_realtime.NewMockEventLogStore(ctrl)
    90  	eventLog.EXPECT().Latest(gomock.Any(), gomock.Any()).Return(rt.DeliveryEvent{}, false, nil).AnyTimes()
    91  
    92  	tickets := mock_realtime.NewMockStreamTicketStore(ctrl)
    93  
    94  	leases := mock_realtime.NewMockInstanceLeaseStore(ctrl)
    95  	leases.EXPECT().Heartbeat(gomock.Any(), gomock.Any()).Return(nil).AnyTimes()
    96  	leases.EXPECT().Delete(gomock.Any(), gomock.Any()).Return(nil).AnyTimes()
    97  
    98  	sub := mock_realtime.NewMockInstanceSubscription(ctrl)
    99  	sub.EXPECT().Provision(gomock.Any(), gomock.Any()).Return(nil).AnyTimes()
   100  	sub.EXPECT().Teardown(gomock.Any()).Return(nil).AnyTimes()
   101  	// 空を即座に返すと consumer の loop が待たずに回り続けるので、停止まで待たせる。
   102  	sub.EXPECT().Receive(gomock.Any(), gomock.Any()).
   103  		DoAndReturn(func(ctx context.Context, _ int) ([]rt.Notification, error) {
   104  			<-ctx.Done()
   105  
   106  			return nil, ctx.Err()
   107  		}).AnyTimes()
   108  
   109  	notifier := mock_realtime.NewMockRevocationNotifier(ctrl)
   110  
   111  	return fx.Replace(
   112  		fx.Annotate(eventLog, fx.As(new(rt.EventLogStore))),
   113  		fx.Annotate(tickets, fx.As(new(rt.StreamTicketStore))),
   114  		fx.Annotate(leases, fx.As(new(rt.InstanceLeaseStore))),
   115  		fx.Annotate(sub, fx.As(new(rt.InstanceSubscription))),
   116  		fx.Annotate(notifier, fx.As(new(rt.RevocationNotifier))),
   117  	)
   118  }

   154  func TestNewApplicationServer(t *testing.T) {
   155  	t.Parallel()
   156  
   157  	// ライフサイクルフックの発火有無で、ラッパーが実際に app.Start / app.Stop を駆動したことを検証する。
   158  	// 空アプリだと start/stop が常に nil を返し、駆動していなくても通ってしまうため。
   159  	newHookedApp := func(started, stopped *bool) *fx.App {
   160  		return fx.New(
   161  			fx.Invoke(func(lc fx.Lifecycle) {
   162  				lc.Append(fx.Hook{
   163  					OnStart: func(context.Context) error { *started = true; return nil },
   164  					OnStop:  func(context.Context) error { *stopped = true; return nil },
   165  				})
   166  			}),
   167  			fx.NopLogger,
   168  		)
   169  	}
   170  
   171  	t.Run("正常系", func(t *testing.T) {
   172  		t.Parallel()
   173  
   174  		t.Run("返した start/stop が app のライフサイクルを起動・停止する", func(t *testing.T) {
   175  			t.Parallel()
   176  
   177  			var started, stopped bool
   178  			start, stop := NewApplicationServer(newHookedApp(&started, &stopped))
   179  			require.NotNil(t, start)
   180  			require.NotNil(t, stop)
   181  
   182  			ctx, cancel := context.WithTimeout(context.Background(), 2*time.Second)
   183  			defer cancel()
   184  			require.NoError(t, start(ctx))
   185  			assert.True(t, started, "start ラッパーが app.Start を呼びライフサイクルが起動すること")
   186  
   187  			stopCtx, stopCancel := context.WithTimeout(context.Background(), 2*time.Second)
   188  			defer stopCancel()
   189  			require.NoError(t, stop(stopCtx))
   190  			assert.True(t, stopped, "stop ラッパーが app.Stop を呼びライフサイクルが停止すること")
   191  		})
   192  	})
   193  
   194  	t.Run("異常系", func(t *testing.T) {
   195  		t.Parallel()
   196  
   197  		t.Run("キャンセル済みコンテキストの start は app.Start のエラーをそのまま返す", func(t *testing.T) {
   198  			t.Parallel()
   199  
   200  			var started, stopped bool
   201  			start, _ := NewApplicationServer(newHookedApp(&started, &stopped))
   202  
   203  			ctx, cancel := context.WithCancel(context.Background())
   204  			cancel()
   205  
   206  			require.ErrorIs(t, start(ctx), context.Canceled)
   207  			assert.False(t, started, "起動前に中断されフックへ到達しないこと")
   208  		})
   209  	})
   210  }
