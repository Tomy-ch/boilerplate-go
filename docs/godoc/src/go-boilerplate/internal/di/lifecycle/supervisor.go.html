     7  // SupervisedRunner は、detached goroutine で常駐する background runner を fx ライフサイクルへ
     8  // 結線する共通プリミティブです。job / worker / outbox relay の各 hook が共有します。
     9  //
    10  //   - OnStart: [SupervisedRunner.OnStartAux]（任意）を同期実行した後、[SupervisedRunner.Body] を
    11  //     goroutine で起動する（OnStart はブロックしない）。
    12  //   - OnStop:  実行 context をキャンセルし、Body の完了を stopCtx（grace）の範囲で待ち、
    13  //     その後 [SupervisedRunner.OnStopAux]（任意）を実行する。
    14  //
    15  // Body に渡される実行 context は startCtx とは独立しており、OnStop でのみキャンセルされます（detached）。
    16  type SupervisedRunner struct {
    17  	// Body は、background ループ本体です。渡される context は OnStop でキャンセルされます。
    18  	// nil の場合は何も起動しません。
    19  	Body func(ctx context.Context)
    20  	// OnStartAux は、Body 起動前に OnStart 内で同期実行される任意の処理です（例: health listener 起動）。
    21  	OnStartAux func()
    22  	// OnStopAux は、drain 完了後に実行される任意の処理です（例: health listener 停止）。
    23  	// 引数の stopCtx は OnStop の grace context です。
    24  	OnStopAux func(stopCtx context.Context)
    25  }
    26  
    27  // Register は、SupervisedRunner を reg のライフサイクルへ登録します。
    28  func (s SupervisedRunner) Register(reg Registrar) {
    29  	start, stop := s.Bind()
    30  	reg.RegisterStart(start)
    31  	reg.RegisterStop(stop)
    32  }
    33  
    34  // Bind は、SupervisedRunner の start / stop 関数を返します。順序を呼び出し側で決めたいときに使い、
    35  // Register はこれを 1 組ずつ登録します。1 度の Bind で返る組は 1 つの実行 context を共有するので、
    36  // start は 1 回だけ呼びます。
    37  func (s SupervisedRunner) Bind() (func(ctx context.Context) error, func(ctx context.Context) error) {
    38  	// 実行 context は startCtx に紐づけない（[SupervisedRunner] の detached 契約）。
    39  	runCtx, cancel := context.WithCancel(context.Background())
    40  	done := make(chan struct{})
    41  
    42  	start := func(_ context.Context) error {
    43  		if s.OnStartAux != nil {
    44  			s.OnStartAux()
    45  		}
    46  		go func() {
    47  			defer close(done)
    48  			if s.Body != nil {
    49  				s.Body(runCtx)
    50  			}
    51  		}()
    52  		return nil
    53  	}
    54  
    55  	stop := func(stopCtx context.Context) error {
    56  		cancel()
    57  		select {
    58  		case <-done: // Body 完了（drain 完了）
    59  		case <-stopCtx.Done(): // 猶予切れ
    60  		}
    61  		if s.OnStopAux != nil {
    62  			s.OnStopAux(stopCtx)
    63  		}
    64  		return nil
    65  	}
    66  
    67  	return start, stop
    68  }
