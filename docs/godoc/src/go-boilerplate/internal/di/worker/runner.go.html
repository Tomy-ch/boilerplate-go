    32  // ValidateShutdownGrace は、drain 完了前に停止猶予が尽きないことを起動時に検証します。
    33  //
    34  // WORKER_DRAIN_TIMEOUT >= APP_SHUTDOWN_TIMEOUT だと OnStop の drain が停止猶予で打ち切られ、
    35  // 未完了の in-flight handler と後続の停止処理（DB pool close 等）が競合します。
    36  // 設定不整合は起動時に失敗させ、運用時の競合を未然に防ぎます。
    37  func ValidateShutdownGrace(appCfg *config.ApplicationConfig, workerCfg *config.WorkerConfig) error {
    38  	return validateShutdownGrace(workerCfg.DrainTimeout(), appCfg.ShutdownTimeout())
    39  }

    41  // validateShutdownGrace は、drain < grace を検証します。
    42  func validateShutdownGrace(drain, grace time.Duration) error {
    43  	if drain >= grace {
    44  		return xerrors.Wrap(ErrInvalidShutdownGrace, fmt.Sprintf(
    45  			"WORKER_DRAIN_TIMEOUT (%s) must be shorter than APP_SHUTDOWN_TIMEOUT (%s)", drain, grace,
    46  		))
    47  	}
    48  	return nil
    49  }
