    26  // TestRealtimeProvisionerContract は、serve lifecycle の realtime 参加者（lease + instance の受信先）を実 DynamoDB Local と
    27  // 実 GoAWS に対して往復させ、Provision で lease と queue / subscription が現れ、Teardown で消えることを固定します。
    28  // 起動・停止の順序そのものは internal/di/server/hook の unit test が固定し、ここは substrate 上の結果を見ます。
    29  func TestRealtimeProvisionerContract(t *testing.T) {
    30  	t.Parallel()
    31  
    32  	dynamo := dynamotestkit.NewTestClient(t)
    33  	table := dynamotestkit.TableName(t, "lifecycle_lease")
    34  	require.NoError(t, dynamodbclient.EnsureTable(t.Context(), dynamo, instanceleasedynamo.TableSpec(table)))
    35  	dynamotestkit.DeleteOnCleanup(t, dynamo, table)
    36  
    37  	tf := observability.NewNoopTracerFactory(t)
    38  	leases := instancelease.New(dynamo, table, tf)
    39  	keeper := provideLeaseKeeper(leases, system.NewClock(), tf)
    40  
    41  	clients := testkit.NewTestClients(t)
    42  	topicARN := testkit.CreateTopic(t, clients, testkit.Name(t, "lifecycle"))
    43  	cfg := config.NewRealtimeConfig(config.MockConfigForTest(t))
    44  	sub := provideInstanceSubscription(
    45  		realtimeFanout{clients: clients, topicARN: topicARN},
    46  		cfg,
    47  		realtimeinfra.NewEmulatorQueueAttributes(),
    48  		tf,
    49  	)
    50  
    51  	id, err := provideInstanceID()
    52  	require.NoError(t, err)
    53  
    54  	// queue 名は <prefix>-<instance id> なので、この instance の queue だけを prefix で引ける。
    55  	prefix := cfg.QueuePrefix() + "-" + string(id)
    56  
    57  	p := provideRealtimeProvisioner(sub, keeper, id)
    58  	require.NoError(t, p.Provision(t.Context()))
    59  	t.Cleanup(func() {
    60  		// t.Context は Cleanup の直前に cancel されるので、片付けは独立の ctx で行う。
    61  		ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
    62  		defer cancel()
    63  		_ = p.Teardown(ctx)
    64  	})
    65  
    66  	assert.True(t, leaseExists(t, leases, id), "lease が書かれている")
    67  	assert.Len(t, queueURLs(t, clients, prefix), 1, "instance queue が 1 つある")
    68  
    69  	require.NoError(t, p.Teardown(t.Context()))
    70  
    71  	assert.False(t, leaseExists(t, leases, id), "lease が消えている")
    72  	assert.Empty(t, queueURLs(t, clients, prefix), "instance queue が消えている")
    73  }
