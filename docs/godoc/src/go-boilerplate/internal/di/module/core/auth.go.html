    68  // provideAuthenticator は、環境に対応した Authenticator を選んで返します。
    69  // 対応する case が無い環境は誤った Authenticator を配線しないよう起動エラーにします（fail-closed）。
    70  func provideAuthenticator(p authenticatorParams) (authbd.Authenticator, error) {
    71  	logger := p.Logger.Named("core.authn").CallerSkip(callerSkipCount)
    72  
    73  	switch p.AppCfg.Env() {
    74  	case config.EnvCI, config.EnvTest:
    75  		logger.Warn(
    76  			context.Background(),
    77  			"Local authenticator wired: authentication is stubbed (non-production only)",
    78  			logging.String("env", p.AppCfg.Env()),
    79  		)
    80  
    81  		return local.New(), nil
    82  	case config.EnvLocal, config.EnvDast, config.EnvDevelopment:
    83  		return provideJWKSAuthenticator(p, logger)
    84  	default:
    85  		logger.Error(
    86  			context.Background(),
    87  			"No authenticator configured for the current environment",
    88  			logging.String("env", p.AppCfg.Env()),
    89  		)
    90  
    91  		return nil, xerrors.Wrap(errNoAuthenticatorForEnv, p.AppCfg.Env())
    92  	}
    93  }

   102  		return false

   104  }

   106  // provideJWKSAuthenticator は、AUTH_* 設定から JWKS backed の JWT authenticator を構築します。
   107  // 取得はリクエストから切り離して走るため、停止時に取り残さないよう完了待ちを lifecycle へ登録します（遅延取得）。
   108  func provideJWKSAuthenticator(p authenticatorParams, logger logging.Logger) (authbd.Authenticator, error) {
   109  	authenticator, err := jwt.NewJWKS(jwt.JWKSParams{
   110  		Issuer:             p.AuthCfg.Issuer(),
   111  		Audience:           p.AuthCfg.Audience(),
   112  		AllowedAlgs:        p.AuthCfg.AllowedAlgorithms(),
   113  		Leeway:             p.AuthCfg.ClockSkew(),
   114  		ExpectedType:       accessTokenType,
   115  		Clock:              p.Clock,
   116  		JWKSURL:            p.EndpointCfg.JWKS(),
   117  		CacheTTL:           p.AuthCfg.JWKSCacheTTL(),
   118  		DiscoveryTTL:       p.AuthCfg.DiscoveryTTL(),
   119  		UnknownKidCooldown: p.AuthCfg.UnknownKidCooldown(),
   120  		AllowInsecureURL:   allowInsecureJWKSURL(p.AppCfg.Env()),
   121  		TracerFactory:      p.TracerFtry,
   122  	}, p.HTTPClient)
   123  	if err != nil {
   124  		logger.Error(
   125  			context.Background(),
   126  			"Failed to wire JWKS JWT authenticator",
   127  			logging.String("env", p.AppCfg.Env()),
   128  			logging.Error("error", err),
   129  		)
   130  
   131  		return nil, err
   132  	}
   133  
   134  	if d, ok := authenticator.(jwt.Drainer); ok {
   135  		p.Lifecycle.Append(fx.Hook{OnStop: d.Drain})
   136  	}
   137  
   138  	logger.Info(
   139  		context.Background(),
   140  		"JWKS JWT authenticator wired",
   141  		logging.String("env", p.AppCfg.Env()),
   142  		logging.String("issuer", p.AuthCfg.Issuer()),
   143  	)
   144  
   145  	return authenticator, nil
   146  }
