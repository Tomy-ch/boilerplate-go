   117  func newRealtimeFanout(ctx context.Context, topicARN string, cc realtimeinfra.ClientConfig) (realtimeFanout, error) {

   252  // provideRealtimeProvisioner は、lease の記録と instance の受信先を 1 つの参加者に合成し、
   253  // 起動は lease → 受信先、片付けは 受信先 → lease の順に固定します
   254  // （fx の value group は順序を保証しない。順序そのものの根拠は docs/design/realtime-delivery.md §2.5）。
   255  func provideRealtimeProvisioner(
   256  	sub rt.InstanceSubscription,
   257  	keeper ucrealtime.LeaseKeeper,
   258  	id rt.InstanceID,
   259  ) hook.Provisioner {
   260  	return hook.Provisioner{
   261  		Name: realtimeParticipantName,
   262  		Provision: func(ctx context.Context) error {
   263  			if err := keeper.Beat(ctx, id); err != nil {
   264  				return err
   265  			}
   266  
   267  			if err := sub.Provision(ctx, id); err != nil {
   268  				if terr := sub.Teardown(ctx); terr != nil {
   269  					return xerrors.Join(err, terr) // 受信先が残ったので lease は残す
   270  				}
   271  
   272  				return xerrors.Join(err, keeper.Release(ctx, id))
   273  			}
   274  
   275  			return nil
   276  		},
   277  		Teardown: func(ctx context.Context) error {
   278  			if err := sub.Teardown(ctx); err != nil {
   279  				return err // 受信先が残ったので lease は残す
   280  			}
   281  
   282  			return keeper.Release(ctx, id)
   283  		},
   284  	}
   285  }
