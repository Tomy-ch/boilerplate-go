    48  // realtimeRunDeps は、constructor を実際に走らせて value group を集めるための依存です。ConfigModule /
    49  // ObservabilityModule / DatabaseModule は global（設定の読み込みと prometheus の既定 registry）へ触るため、
    50  // 並列テストと競合する。realtime が要るのは設定 3 つと Logger / TracerFactory / outbound クライアントだけ
    51  // なので、いずれも test 用の実装を直接差し込む。
    52  func realtimeRunDeps(t *testing.T) fx.Option {
    53  	t.Helper()
    54  
    55  	cfg := config.MockConfigForTest(t)
    56  
    57  	return fx.Options(
    58  		lifecycle.Module(), clockModule(), realtimeModule(),
    59  		fx.Provide(func() *config.ApplicationConfig { return config.NewApplicationConfig(cfg) }),
    60  		fx.Provide(func() *config.EndpointConfig { return config.NewEndpointConfig(cfg) }),
    61  		fx.Provide(func() *config.RealtimeConfig { return config.NewRealtimeConfig(cfg) }),
    62  		fx.Provide(func() logging.Logger { return logging.NewTestLogger(t) }),
    63  		fx.Provide(func() observability.TracerFactory { return observability.NewNoopTracerFactory(t) }),
    64  		fx.Provide(func() *observability.OutboundHTTPClient { return observability.NewDisabledOutboundHTTPClient(true) }),
    65  		fx.Provide(echo.New),
    66  		fx.Replace(testFanout(t)),
    67  		// 計装は MeterProvider を要るが、ObservabilityModule は global の registry を触るため
    68  		// 並列テストと競合する。noop の実体で差し替えて構築子ごと迂回する。
    69  		fx.Replace(observability.NewNoopRealtimeMetrics(t)),
    70  	)
    71  }
