    45  // OutboxRelayModule は、1 つの配送チャネルを担う outbox relay engine とそのライフサイクルフックを
    46  // 提供するfx.Moduleです。relay 専用プロセス（cmd outbox-relay）でのみ使用します。
    47  // チャネル隔離と publisher profile の閉じ込めは internal/di/README.md の Optional を参照。
    48  func OutboxRelayModule(channel outboxbndry.Channel) fx.Option {
    49  	return fx.Module("outbox-relay",
    50  		publisherModuleFor(channel),
    51  		fx.Supply(channel),
    52  		fx.Provide(
    53  			fx.Annotate(
    54  				observability.NewOutboxMetrics,
    55  				fx.As(new(outboxuc.Metrics)),
    56  			),
    57  			provideRelayUsecase,
    58  			provideRelaySettings,
    59  			outboxengine.NewEngine,
    60  		),
    61  		fx.Invoke(relayhook.RegisterRelayHooks),
    62  	)
    63  }

    65  // publisherModuleFor は、channel を配送できる publisher module を返します。対応する module の無い channel は、
    66  // 担当者の居ない relay を黙って起動させないため構築エラーにします（fail-closed）。
    67  func publisherModuleFor(channel outboxbndry.Channel) fx.Option {
    68  	switch channel {
    69  	case outboxbndry.ChannelHTTP:
    70  		return fx.Options(outboxPublisherModule(), fx.Invoke(publisher.VerifyChannel))
    71  	case outboxbndry.ChannelRealtime:
    72  		return realtimePublisherModule()
    73  	default:
    74  		return fx.Error(
    75  			xerrors.Wrap(
    76  				publisher.ErrChannelUnsupported,
    77  				"no publisher module serves delivery channel "+channel.String(),
    78  			),
    79  		)
    80  	}
    81  }

    96  // provideRelaySettings は、OutboxConfig から relay engine の設定を生成します
    97  // （clamp の理由は internal/controller/outbox/README.md の Public API を参照）。
    98  // BatchSize が engine の int32 に収まらないときは、既定値へ丸めて誤設定を隠すのではなく起動を失敗させます。
    99  func provideRelaySettings(cfg *config.OutboxConfig) (outboxengine.Settings, error) {
   100  	batchSize := outboxuc.DefaultBatchSize
   101  	if cfg.BatchSize() > 0 {
   102  		v, err := safecast.IntToInt32(cfg.BatchSize())
   103  		if err != nil {
   104  			return outboxengine.Settings{}, xerrors.Wrap(err, "outbox relay batch size is out of range")
   105  		}
   106  		batchSize = v
   107  	}
   108  
   109  	pollInterval := defaultRelayPollInterval
   110  	if cfg.PollInterval() > 0 {
   111  		pollInterval = cfg.PollInterval()
   112  	}
   113  
   114  	errorBackoff := defaultRelayErrorBackoff
   115  	if cfg.ErrorBackoff() > 0 {
   116  		errorBackoff = cfg.ErrorBackoff()
   117  	}
   118  
   119  	return outboxengine.Settings{
   120  		BatchSize:    batchSize,
   121  		PollInterval: pollInterval,
   122  		ErrorBackoff: errorBackoff,
   123  	}, nil
   124  }
