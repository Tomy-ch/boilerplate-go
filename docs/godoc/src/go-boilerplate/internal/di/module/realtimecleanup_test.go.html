    53  func TestRealtimeCleanupModule(t *testing.T) {
    54  	t.Parallel()
    55  
    56  	t.Run("正常系", func(t *testing.T) {
    57  		t.Parallel()
    58  
    59  		t.Run("Realtime が未設定でも jobs グループを組める", func(t *testing.T) {
    60  			t.Parallel()
    61  
    62  			// これが崩れると、REALTIME_TOPIC が空の環境（dast / dev / stg / prd）で outbox-gc のような
    63  			// 無関係なジョブまで起動できなくなる。fx は Runner を組むために登録済みジョブの constructor を
    64  			// すべて実行するので、Realtime の構築が graph に載っていないことをここで締める。
    65  			require.Empty(t, config.NewRealtimeConfig(config.MockConfigForTest(t)).Topic())
    66  
    67  			got := collectGroup[job.Job](t, `group:"jobs"`, realtimeCleanupRunDeps(t))
    68  
    69  			names := make([]string, 0, len(got))
    70  			for _, j := range got {
    71  				names = append(names, j.Name())
    72  			}
    73  			assert.Equal(t, []string{"orphan-cleanup"}, names)
    74  		})
    75  	})
