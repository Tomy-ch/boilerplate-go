    43  // RunOutboxReplay は、dead 状態の outbox 行を pending へ戻すワンショット実行を行い、
    44  // 書き換えた行数と発生したエラーを返します。messageID が nil の場合は全 dead 行を対象とします。
    45  // 停止処理のエラーは runErr と xerrors.Join で結合して返します。
    46  // DI グラフの構築に失敗した場合、replay を実行せずその構築エラーを返します。
    47  func RunOutboxReplay(ctx context.Context, messageID *uuid.UUID) (int64, error) {
    48  	var (
    49  		replay outboxuc.ReplayUsecase
    50  		appCfg *config.ApplicationConfig
    51  	)
    52  
    53  	opts := append(outboxRelayCommonOptions(),
    54  		fx.Populate(&replay, &appCfg), fx.WithLogger(NewFxEventLogger))
    55  	app := fx.New(opts...)
    56  
    57  	// fx.Populate は構築失敗時に nil を残すため、replay / appCfg の参照より先にエラーを返す。
    58  	if err := app.Err(); err != nil {
    59  		return 0, err
    60  	}
    61  
    62  	if err := app.Start(ctx); err != nil {
    63  		return 0, err
    64  	}
    65  
    66  	result, runErr := replay.ReplayDead(ctx, messageID)
    67  
    68  	// 停止猶予は APP_SHUTDOWN_TIMEOUT に一本化（独立した magic number を持たない）。
    69  	stopCtx, cancel := context.WithTimeout(context.Background(), appCfg.ShutdownTimeout())
    70  	defer cancel()
    71  	stopErr := app.Stop(stopCtx) //nolint:contextcheck // 停止用 context は意図的に ctx を継承しない
    72  
    73  	// replay 失敗と OnStop 失敗（DB プール Close 等）の双方をオペレータへ可視化する。
    74  	return result, xerrors.Join(runErr, stopErr)
    75  }
