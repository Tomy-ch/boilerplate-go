    20  // newStartServerFunc は、HTTPサーバーを起動する関数を生成します。
    21  func newStartServerFunc(
    22  	srv *http.Server,
    23  	log logging.Logger,
    24  	appCfg *config.ApplicationConfig,
    25  	secCfg *config.SecurityConfig,
    26  	srvCfg *config.ServerConfig,
    27  	osCfg *config.OperatingSystemConfig,
    28  ) func(context.Context) error {
    29  	return func(ctx context.Context) error {
    30  		port := srvCfg.Port()
    31  
    32  		lc := &net.ListenConfig{}
    33  		ln, err := lc.Listen(ctx, "tcp", ":"+strconv.Itoa(port))
    34  		if err != nil {

    38  
    39  		fields := append(lifecycleEventFields(logging.EventTypeStart, osCfg.TimeZone()),

    41  			logging.Strings("allowed_origins", secCfg.AllowedOrigins()),
    42  			logging.String("cidr", secCfg.CIDR().IP.String()),
    43  			logging.String("mode", appCfg.Mode()),
    44  		)
    45  		log.Named("server.Start").CallerSkip(serverCallerSkip).Info(ctx, "http started", fields...)
    46  		return nil
    47  	}

    50  // serveHTTP は、ln 上で HTTP サーバーを稼働させます。
    51  // 正常停止（[http.ErrServerClosed]）以外で終了した場合はエラーログを残します。
    52  func serveHTTP(ctx context.Context, srv *http.Server, ln net.Listener, log logging.Logger) {
    53  	if err := srv.Serve(ln); err != nil && !xerrors.Is(err, http.ErrServerClosed) {
    54  		log.Named("server.Start").Error(ctx, "failed to start http server", logging.Error(logging.ErrorKey, err))
    55  	}
    56  }

    59  func newStopServerFunc(
    60  	srv *http.Server, log logging.Logger, osCfg *config.OperatingSystemConfig,
    61  ) func(context.Context) error {
    62  	return func(ctx context.Context) error {
    63  		l := log.Named("server.Stop").CallerSkip(serverCallerSkip)
    64  		l.Info(ctx, "http stopping", lifecycleEventFields(logging.EventTypeEnd, osCfg.TimeZone())...)
    65  		if err := srv.Shutdown(ctx); err != nil {
    66  			l.Error(ctx, "failed to shutdown http server", logging.Error(logging.ErrorKey, err))
    67  			return err
    68  		}
    69  		return nil
    70  	}
    71  }
