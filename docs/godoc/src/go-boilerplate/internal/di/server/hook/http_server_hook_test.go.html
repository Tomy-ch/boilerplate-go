    42  		fn, ok := args[0].(func(context.Context) error)
    43  		require.True(t, ok)
    44  		shutdownFn = fn
    45  	}).Times(1)
    46  
    47  	cfg := config.MockConfigForTest(t)
    48  	appCfg := config.NewApplicationConfig(cfg)
    49  	secCfg := config.NewSecurityConfig(cfg)
    50  	srvCfg := config.NewServerConfig(cfg)
    51  	osCfg := config.NewOperatingSystemConfig(cfg)
    52  
    53  	_, srv := newTestHTTPServer(t, srvCfg)
    54  
    55  	mockLogger.EXPECT().Named(serveLifecycleLoggerName).Return(mockLogger)
    56  	mockLogger.EXPECT().CallerSkip(serverCallerSkip).Return(mockLogger)
    57  	// 登録された stop を駆動すると HTTP shutdown まで進む。その経路のログは別テストが見る。
    58  	mockLogger.EXPECT().Named("server.Stop").Return(mockLogger).AnyTimes()
    59  	mockLogger.EXPECT().CallerSkip(serverCallerSkip).Return(mockLogger).AnyTimes()
    60  	mockLogger.EXPECT().Info(gomock.Any(), gomock.Any(), gomock.Any(), gomock.Any(), gomock.Any()).AnyTimes()
    61  
    62  	drained := false
    63  	RegisterHTTPServerHooks(HTTPServerHooksIn{
    64  		Srv: srv, Reg: mockReg, Log: mockLogger, AppCfg: appCfg, SecCfg: secCfg, SrvCfg: srvCfg, OSCfg: osCfg,
    65  		Applied: &extension.AppliedServerExtends{},
    66  		Startup: []StartupProbe{{Name: "store", Probe: func(context.Context) error { return errParticipant }}},
    67  		Drainers: []Drainer{{Name: "sse", Drain: func(context.Context) error {
    68  			drained = true
    69  
    70  			return nil
    71  		}}},
    72  	})
    73  	require.NotNil(t, startFn)
    74  	require.NotNil(t, shutdownFn)
    75  
    76  	// 登録されたのが参加者を含む serveLifecycle の start / stop であること: start は到達確認の失敗で止まり、
    77  	// stop は drain を通ってから HTTP shutdown へ進む。
    78  	require.ErrorIs(t, startFn(t.Context()), errParticipant)
    79  	require.NoError(t, shutdownFn(t.Context()))
    80  	assert.True(t, drained)
    81  }

   268  			ctrl := gomock.NewController(t)

   281  			// ハンドラを処理中にして接続を active に保ち、Shutdown を idle 完了させない
   282  			entered := make(chan struct{})
   283  			release := make(chan struct{})
   284  			e.GET("/block", func(c *echo.Context) error {
   285  				close(entered)

   289  
   290  			lc := &net.ListenConfig{}
   291  			ln, err := lc.Listen(context.Background(), "tcp", "127.0.0.1:0")

   297  				req, rerr := http.NewRequestWithContext(
   298  					context.Background(),
   299  					http.MethodGet,
   300  					"http://"+ln.Addr().String()+"/block",
   301  					nil,
   302  				)
   303  				if rerr != nil {
   304  					return
   305  				}
   306  				resp, gerr := http.DefaultClient.Do(req)
   307  				if gerr == nil {
   308  					_ = resp.Body.Close()
   309  				}
   310  			}()
   311  			<-entered // リクエストが処理中＝接続が active になったことを保証
   312  
   313  			// 既に期限の切れた context で Shutdown → 処理中接続が残り context error を返す
   314  			ctx, cancel := context.WithTimeout(context.Background(), 20*time.Millisecond)
   315  			defer cancel()
   316  			fn := newStopServerFunc(srv, mockLogger, osCfg)
   317  			require.Error(t, fn(ctx))
   318  		})
   319  	})
   320  }
