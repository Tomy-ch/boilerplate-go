    56  // RegisterHTTPServerHooks は、serve instance の起動・停止の順序を 1 箇所で決めて登録します。
    57  //
    58  //	Start: 依存の到達確認 → 受信 resource の作成 → 常駐処理の開始 → HTTP listen
    59  //	Stop:  drain（新規接続の拒否を含む）→ 常駐処理の停止 → 受信 resource の片付け → HTTP shutdown
    60  //
    61  // fx の OnStop は登録の逆順で走るため、順序を hook の登録順に委ねず、start / stop 各 1 本の中で明示的に
    62  // 呼びます（参加者が増減しても順序が変わらず、drain が HTTP shutdown より前に完了することを固定する）。
    63  // Start は途中で失敗したら、そこまでに起動・作成したものを逆順に止めて片付けてから失敗を返します。
    64  // Stop は参加者が失敗しても記録して次へ進み（途中で止めると resource が残る）、HTTP shutdown まで
    65  // 終えてからすべての失敗をまとめて返します。
    66  func RegisterHTTPServerHooks(in HTTPServerHooksIn) {
    67  	lc := newServeLifecycle(in)
    68  	in.Reg.RegisterStart(lc.start)
    69  	in.Reg.RegisterStop(lc.stop)
    70  }

    72  func newServeLifecycle(in HTTPServerHooksIn) *serveLifecycle {
    73  	runners := make([]boundRunner, 0, len(in.Runners))
    74  	for _, r := range in.Runners {
    75  		start, stop := r.Runner.Bind()
    76  		runners = append(runners, boundRunner{name: r.Name, start: start, stop: stop})
    77  	}
    78  
    79  	return &serveLifecycle{
    80  		log:          in.Log.Named(serveLifecycleLoggerName).CallerSkip(serverCallerSkip),
    81  		startup:      in.Startup,
    82  		provisioners: in.Provisioners,
    83  		runners:      runners,
    84  		drainers:     in.Drainers,
    85  		httpStart:    newStartServerFunc(in.Srv, in.Log, in.AppCfg, in.SecCfg, in.SrvCfg, in.OSCfg),
    86  		httpStop:     newStopServerFunc(in.Srv, in.Log, in.OSCfg),
    87  	}

    90  // start は、[RegisterHTTPServerHooks] が定める Start 順で起動し、失敗したら逆順に巻き戻します。
    91  func (l *serveLifecycle) start(ctx context.Context) error {
    92  	for _, p := range l.startup {
    93  		if err := p.Probe(ctx); err != nil {
    94  			return xerrors.Wrap(err, "serve startup probe "+p.Name+" failed")
    95  		}
    96  	}
    97  
    98  	provisioned := 0
    99  	for _, p := range l.provisioners {
   100  		if err := p.Provision(ctx); err != nil {
   101  			l.teardown(ctx, provisioned)
   102  
   103  			return xerrors.Wrap(err, "serve provisioner "+p.Name+" failed")
   104  		}
   105  
   106  		provisioned++
   107  	}
   108  
   109  	started := 0
   110  	for _, r := range l.runners {
   111  		if err := r.start(ctx); err != nil {
   112  			l.stopRunners(ctx, started)
   113  			l.teardown(ctx, provisioned)
   114  
   115  			return xerrors.Wrap(err, "serve runner "+r.name+" failed to start")
   116  		}
   117  
   118  		started++
   119  	}
   120  
   121  	if err := l.httpStart(ctx); err != nil {
   122  		l.stopRunners(ctx, started)
   123  		l.teardown(ctx, provisioned)
   124  
   125  		return err
   126  	}
   127  
   128  	return nil
   129  }

   131  // stop は、[RegisterHTTPServerHooks] が定める Stop 順で停止し、失敗をまとめて返します。
   132  func (l *serveLifecycle) stop(ctx context.Context) error {
   133  	var errs []error
   134  
   135  	for _, d := range l.drainers {
   136  		if err := d.Drain(ctx); err != nil {
   137  			l.log.Error(
   138  				ctx,
   139  				"serve drainer failed",
   140  				logging.String("participant", d.Name),
   141  				logging.Error(logging.ErrorKey, err),
   142  			)
   143  			errs = append(errs, err)
   144  		}
   145  	}
   146  
   147  	errs = append(errs, l.stopRunners(ctx, len(l.runners))...)
   148  	errs = append(errs, l.teardown(ctx, len(l.provisioners))...)
   149  
   150  	if err := l.httpStop(ctx); err != nil {
   151  		errs = append(errs, err)
   152  	}
   153  
   154  	return xerrors.Join(errs...)
   155  }

   157  // stopRunners は、先頭から n 個の常駐処理を逆順に止め、失敗を記録して返します。
   158  func (l *serveLifecycle) stopRunners(ctx context.Context, n int) []error {
   159  	var errs []error
   160  
   161  	for i := n - 1; i >= 0; i-- {
   162  		if err := l.runners[i].stop(ctx); err != nil {
   163  			l.log.Error(
   164  				ctx,
   165  				"serve runner failed to stop",
   166  				logging.String("participant", l.runners[i].name),
   167  				logging.Error(logging.ErrorKey, err),
   168  			)
   169  			errs = append(errs, err)
   170  		}
   171  	}
   172  
   173  	return errs
   174  }
