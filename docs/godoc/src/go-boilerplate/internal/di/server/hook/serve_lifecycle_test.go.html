   242  	t.Parallel()
   243  
   244  	t.Run("正常系", func(t *testing.T) {
   245  		t.Parallel()
   246  
   247  		t.Run("drain → 常駐処理の停止 → resource の片付け → HTTP shutdown の順で、drain が終わるまで shutdown は始まらない", func(t *testing.T) {
   248  			t.Parallel()
   249  
   250  			f := newFixture(t)
   251  			drained := false
   252  			f.lc.drainers = []Drainer{{Name: "sse", Drain: func(context.Context) error {
   253  				f.log.add("drain.sse")
   254  				drained = true
   255  
   256  				return nil
   257  			}}}
   258  			f.lc.httpStop = func(context.Context) error {
   259  				assert.True(t, drained, "HTTP shutdown は drain の完了後")
   260  				f.log.add("http.stop")
   261  
   262  				return nil
   263  			}
   264  			f.lc.provisioners = []Provisioner{f.provisioner("lease", nil, nil), f.provisioner("inbox", nil, nil)}
   265  			f.lc.runners = []boundRunner{f.runner("heartbeat"), f.runner("consumer")}
   266  			require.NoError(t, f.lc.start(t.Context()))
   267  			f.log.reset()
   268  
   269  			require.NoError(t, f.lc.stop(t.Context()))
   270  			assert.Equal(t, []string{
   271  				"drain.sse", "runner.stop.consumer", "runner.stop.heartbeat", "teardown.inbox", "teardown.lease", "http.stop",
   272  			}, f.log.get())
   273  		})
   274  
   275  		t.Run("参加者がゼロなら HTTP shutdown だけ", func(t *testing.T) {
   276  			t.Parallel()
   277  
   278  			f := newFixture(t)
   279  			require.NoError(t, f.lc.stop(t.Context()))
   280  			assert.Equal(t, []string{"http.stop"}, f.log.get())
   281  		})
   282  	})
   283  
   284  	t.Run("異常系", func(t *testing.T) {
   285  		t.Parallel()
   286  
   287  		t.Run("drain と片付けの失敗は記録して続け、まとめて返す。HTTP shutdown は必ず呼ぶ", func(t *testing.T) {
   288  			t.Parallel()
   289  
   290  			f := newFixture(t)
   291  			f.lc.drainers = []Drainer{f.drainer("sse", errParticipant)}
   292  			f.lc.provisioners = []Provisioner{f.provisioner("inbox", nil, errTeardown)}
   293  			f.lc.runners = []boundRunner{{name: "bad", start: func(context.Context) error { return nil }, stop: func(context.Context) error {
   294  				f.log.add("runner.stop.bad")
   295  
   296  				return errRunnerStop
   297  			}}}
   298  
   299  			err := f.lc.stop(t.Context())
   300  			require.ErrorIs(t, err, errParticipant)
   301  			require.ErrorIs(t, err, errRunnerStop, "常駐処理の停止失敗も返す")
   302  			require.ErrorIs(t, err, errTeardown, "片付けの失敗も返す")
   303  			assert.Equal(t, []string{"drain.sse", "runner.stop.bad", "teardown.inbox", "http.stop"}, f.log.get())
   304  			assert.Equal(t, 1, f.logCount("serve drainer failed"))
   305  			assert.Equal(t, 1, f.logCount("serve provisioner failed to tear down"))
   306  		})
   307  
   308  		t.Run("HTTP shutdown の失敗は返す", func(t *testing.T) {
   309  			t.Parallel()
   310  
   311  			f := newFixture(t)
   312  			f.lc.httpStop = func(context.Context) error { return errParticipant }
   313  
   314  			require.ErrorIs(t, f.lc.stop(t.Context()), errParticipant)
   315  		})
   316  	})
