    62  // RunJob は、ジョブ実行用の開始関数・停止関数を生成して返します。context／タイムアウトの制御は返した関数の呼出側が担います。
    63  // grace（APP_SHUTDOWN_TIMEOUT）を fx.StopTimeout に設定し、停止時に fx 既定（15s）が
    64  // 停止猶予より先に teardown を打ち切らないようにします（停止軸を grace に一本化）。
    65  // DI グラフの構築に失敗した場合、返す start／stop はいずれもその構築エラーを返します。
    66  func RunJob(grace time.Duration) (StartFunc, StopFunc) {
    67  	var (
    68  		state  job.State
    69  		logger logging.Logger
    70  		osCfg  *config.OperatingSystemConfig
    71  	)
    72  
    73  	app := fx.New(
    74  		NewJobCore(),
    75  		fx.StopTimeout(grace),
    76  		fx.Populate(&state, &logger, &osCfg),
    77  		fx.WithLogger(NewFxEventLogger),
    78  	)
    79  
    80  	start := func(ctx context.Context, name string, args []string) <-chan error {
    81  		// fx.Populate は fx.New 時点の invoke なので、グラフ構築が失敗すると対象は nil のまま残る。
    82  		// logger / osCfg を触る前に構築エラーを返し、nil 参照ではなく DI エラーをオペレータへ届ける。
    83  		if err := app.Err(); err != nil {
    84  			return failClosedChan(err)
    85  		}
    86  
    87  		l := logger.Named("job.RunJob").CallerSkip(callSkip)
    88  		done := make(chan error, 1)
    89  		state.Set(name, args, done)
    90  
    91  		if err := app.Start(ctx); err != nil {
    92  			fields := append(jobEventFields(logging.EventTypeStart, osCfg.TimeZone()),
    93  				logging.String(logging.JobNameKey, name),
    94  				logging.Strings(logging.JobArgsKey, args),
    95  				logging.Error(logging.JobErrorKey, err),
    96  			)
    97  			l.Error(ctx, "failed to start job application", fields...)
    98  
    99  			return failClosedChan(err)
   100  		}
   101  
   102  		fields := append(jobEventFields(logging.EventTypeStart, osCfg.TimeZone()),
   103  			logging.String(logging.JobNameKey, name),
   104  			logging.Strings(logging.JobArgsKey, args),
   105  		)
   106  		l.Info(ctx, "job started", fields...)
   107  
   108  		return done
   109  	}
   110  
   111  	stop := func(ctx context.Context) error {
   112  		// start と同じく、構築失敗時は nil の logger / osCfg を参照せずに構築エラーを返す。
   113  		if err := app.Err(); err != nil {
   114  			return err
   115  		}
   116  
   117  		l := logger.Named("job.RunJob").CallerSkip(callSkip)
   118  		if err := app.Stop(ctx); err != nil {
   119  			fields := append(jobEventFields(logging.EventTypeEnd, osCfg.TimeZone()),
   120  				logging.Error(logging.JobErrorKey, err),
   121  			)
   122  			l.Error(ctx, "failed to stop job application", fields...)
   123  			return err
   124  		}
   125  
   126  		l.Info(ctx, "job application finished", jobEventFields(logging.EventTypeEnd, osCfg.TimeZone())...)
   127  		return nil
   128  	}
   129  
   130  	return start, stop
   131  }
