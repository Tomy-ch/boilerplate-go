    81  // CreatePurchase は、購入の作成を購入ユースケースへ委ね、その後に参考換算額を添えます。
    82  // 参考換算の取得に失敗しても購入は成立したまま nil で degrade します。
    83  func (u *usecase) CreatePurchase(ctx context.Context, params CreatePurchaseParams) (PurchaseView, error) {
    84  	ctx, endSpan := u.tracer.Start(ctx)
    85  	defer endSpan()
    86  
    87  	created, err := u.purchase.CreatePurchase(ctx, purchaseuc.CreatePurchaseParams{
    88  		UserID:   params.UserID,
    89  		Details:  params.Details,
    90  		CouponID: params.CouponID,
    91  	})
    92  	if err != nil {
    93  		return PurchaseView{}, err
    94  	}
    95  
    96  	view := PurchaseView{Purchase: created}
    97  	if params.DisplayCurrency != nil {
    98  		view.ReferenceAmount = u.referenceAmount(ctx, created.TotalAmount, *params.DisplayCurrency)
    99  	}
   100  	return view, nil
   101  }

   103  // referenceAmount は、合計金額（基軸通貨のセント）の表示通貨での参考換算額を算出します。
   104  // 為替 gateway の障害時は nil を返します。
   105  func (u *usecase) referenceAmount(ctx context.Context, totalCents int, displayCurrency string) *ReferenceAmountView {
   106  	// 決済スケール（整数セント）の合計を価格スケール（基軸通貨の decimal）へ戻して換算入力にする。
   107  	amount := decimal.FromInt(int64(totalCents)).DivRound(decimal.FromInt(centsPerBaseUnit), minorUnitDigits)
   108  	res, err := u.xr.Convert(ctx, exchangerate.ConvertInput{
   109  		Base:            baseCurrency,
   110  		Quote:           displayCurrency,
   111  		Amount:          amount,
   112  		DisplayCurrency: &displayCurrency,
   113  	})
   114  	if err != nil || res.Reference == nil {
   115  		return nil
   116  	}
   117  	return &ReferenceAmountView{
   118  		Currency: res.Reference.Currency,
   119  		Amount:   res.Reference.Amount,
   120  		Rate:     res.Reference.Rate,
   121  		RateDate: res.Reference.RateDate,
   122  	}
   123  }
