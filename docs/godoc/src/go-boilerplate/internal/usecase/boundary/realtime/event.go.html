     1  // Package realtime は、Realtime Delivery（docs/design/realtime-delivery.md）の seam を定義します。
     2  // feature 中立な封筒 DeliveryEvent と、それを保存・配送する側が実装する境界
     3  // （EventLogStore / StreamTicketStore / InstanceLeaseStore / SecretGenerator / SequenceAllocator）を置きます。
     4  // DynamoDB / AWS の語彙（table / partition / TTL 等）はこの
     5  // package に現れません。feature の語彙（会話・メッセージ等）も現れません。

    16  const (
    17  	// MaxSerializedBytes は、DeliveryEvent を JSON に直列化したときの上限（64 KiB）です。
    18  	// payload 単体ではなく封筒全体で判定します（SSE の 1 event として送る大きさがこれだからです）。
    19  	MaxSerializedBytes = 64 * 1024
    20  
    21  	// EventLogRetention は、event を replay できる期間です。store はこの期間を過ぎた item を掃除してよく、
    22  	// usecase は OccurredAt がこの期間より古い位置への cursor を失効と判定します（ADR-0072）。
    23  	EventLogRetention = 7 * 24 * time.Hour

    33  // DeliveryEvent は、feature 中立な配送封筒です。EventID は outbox の message_id と同じ値で、
    34  // 同じ (StreamID, Sequence) への再 append が冪等かどうかを判定する基準になります。
    35  type DeliveryEvent struct {
    36  	// EventID は、event の一意な識別子（= outbox message_id）です。
    37  	EventID string
    38  	// StreamID は、event が属する stream です。
    39  	StreamID StreamID
    40  	// Sequence は、stream 内の位置です。
    41  	Sequence Sequence
    42  	// Type は、event 種別（`<feature>.<noun>.<verb>.vN`）です。機構は解釈しません。
    43  	Type string
    44  	// OccurredAt は、event が起きた時刻（UTC）です。保持期間の判定に使います。
    45  	OccurredAt time.Time
    46  	// SchemaVersion は、Payload の schema 版です。
    47  	SchemaVersion int
    48  	// Payload は、feature が定義する JSON です。機構は中身を解釈しません。
    49  	Payload json.RawMessage
    50  	// Origin は、この event を生んだ command の trace を運ぶ不透明な carrier です。項目名は observability
    51  	// だけが知り、機構は解釈せず運ぶだけです（outbox の Message.Headers と同じ扱い）。client には届きません
    52  	// （直列化形に含めない）。nil でも構いません — link の無い span になるだけです
    53  	// （配送・replay の span がここへ link を張る契約は docs/design/realtime-delivery.md §3.4）。
    54  	Origin map[string]string
    55  }

    73  // MarshalJSON は、client に届く形（sequence は 10 進文字列、時刻は RFC 3339）で直列化します。
    74  func (e DeliveryEvent) MarshalJSON() ([]byte, error) {
    75  	payload := e.Payload
    76  	if len(payload) == 0 {
    77  		payload = json.RawMessage("null")
    78  	}
    79  
    80  	b, err := json.Marshal(wireEvent{
    81  		EventID:       e.EventID,
    82  		StreamID:      e.StreamID,
    83  		Sequence:      e.Sequence.String(),
    84  		Type:          e.Type,
    85  		OccurredAt:    e.OccurredAt.UTC(),
    86  		SchemaVersion: e.SchemaVersion,
    87  		Payload:       payload,
    88  	})
    89  	if err != nil {
    90  		return nil, xerrors.Wrap(ErrInvalidEvent, err.Error())
    91  	}
    92  
    93  	return b, nil
    94  }

    96  // ParseDeliveryEvent は、MarshalJSON の形（sequence は 10 進文字列）から封筒を復元します。JSON でないか
    97  // sequence が 10 進整数でなければ ErrInvalidEvent を返します。復元した封筒が保存・配送できる形かは Validate が別に判定します。
    98  func ParseDeliveryEvent(b []byte) (DeliveryEvent, error) {
    99  	var w wireEvent
   100  	if err := json.Unmarshal(b, &w); err != nil {
   101  		return DeliveryEvent{}, xerrors.Wrap(ErrInvalidEvent, err.Error())
   102  	}
   103  
   104  	seq, err := strconv.ParseInt(w.Sequence, 10, 64)
   105  	if err != nil {
   106  		return DeliveryEvent{}, xerrors.Wrap(ErrInvalidEvent, "sequence must be a decimal integer: "+w.Sequence)
   107  	}
   108  
   109  	return DeliveryEvent{
   110  		EventID:       w.EventID,
   111  		StreamID:      w.StreamID,
   112  		Sequence:      Sequence(seq),
   113  		Type:          w.Type,
   114  		OccurredAt:    w.OccurredAt,
   115  		SchemaVersion: w.SchemaVersion,
   116  		Payload:       w.Payload,
   117  	}, nil
   118  }
   119  
   120  // Validate は、封筒が保存・配送できる形かを判定します。必須項目の欠落は ErrInvalidEvent、
   121  // 直列化後の大きさが MaxSerializedBytes を超えるものは ErrPayloadTooLarge を返します。
   122  func (e DeliveryEvent) Validate() error {
   123  	switch {
   124  	case e.EventID == "":
   125  		return xerrors.Wrap(ErrInvalidEvent, "event id is empty")
   126  	case e.StreamID == "":
   127  		return xerrors.Wrap(ErrInvalidEvent, "stream id is empty")
   128  	case e.Sequence <= 0:
   129  		return xerrors.Wrap(ErrInvalidEvent, "sequence must be positive")
   130  	case e.Type == "":
   131  		return xerrors.Wrap(ErrInvalidEvent, "type is empty")
   132  	case e.OccurredAt.IsZero():
   133  		return xerrors.Wrap(ErrInvalidEvent, "occurred at is zero")
   134  	case len(e.Payload) > 0 && !json.Valid(e.Payload):
   135  		return xerrors.Wrap(ErrInvalidEvent, "payload is not valid json")
   136  	}
   137  
   138  	b, err := e.MarshalJSON()
   139  	if err != nil {
   140  		return err
   141  	}
   142  
   143  	if len(b) > MaxSerializedBytes {
   144  		return xerrors.Wrap(ErrPayloadTooLarge, "serialized event is "+strconv.Itoa(len(b))+" bytes")
   145  	}
   146  
   147  	return nil
   148  }
