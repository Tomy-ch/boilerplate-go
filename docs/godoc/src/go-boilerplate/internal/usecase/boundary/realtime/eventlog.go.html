     7  // EventLogStore は、stream ごとに sequence 順で event を保持する有界の replay store です
     8  // （ADR-0072: 現在状態の正本ではなく、保持期間内の replay のためだけに置く）。
     9  // 失敗は apperror sentinel（読めない・書けないは ErrUnavailable）で返します。
    10  type EventLogStore interface {
    11  	// Append は、event を (StreamID, Sequence) の位置へ 1 度だけ書きます。
    12  	// 同じ位置に同じ EventID の event が既にあれば成功として返します。
    13  	// 同じ位置に異なる EventID の event があれば ErrSequenceConflict を返します。
    14  	// 実装は書く前に event.Validate() を呼び、不正な封筒を保存しません。
    15  	Append(ctx context.Context, event DeliveryEvent) error
    16  	// ReadAfter は、q.After より後ろの event を sequence 昇順に、最大 q.Limit 件返します。
    17  	// 書き込み直後の読み取りでも取りこぼさない（強い一貫性の）読み取りです。
    18  	// 続きがあれば HasMore が true になり、呼び出し側は最後の event の Sequence を次の After に渡します。
    19  	ReadAfter(ctx context.Context, q ReadAfterQuery) (ReadAfterResult, error)
    20  	// Latest は、stream の最後の event を返します。1 件も無ければ ok=false を返します。
    21  	Latest(ctx context.Context, streamID StreamID) (DeliveryEvent, bool, error)
    22  	// Find は、stream の指定 sequence の event を返します。無ければ ok=false を返します。
    23  	Find(ctx context.Context, streamID StreamID, seq Sequence) (DeliveryEvent, bool, error)
    24  	// AppendedThrough は、この stream へ追記した最大の位置を返します。1 度も追記していなければ 0 です
    25  	// （この値の使い方は docs/design/realtime-delivery.md の Glossary「append watermark」）。
    26  	AppendedThrough(ctx context.Context, streamID StreamID) (Sequence, error)
    27  }
