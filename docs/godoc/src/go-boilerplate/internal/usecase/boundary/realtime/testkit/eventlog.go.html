    25  // EventLog は、in-memory の rt.EventLogStore です。読み取りは常に最新の書き込みを反映します
    26  // （実装は強い一貫性の読み取りとして振る舞います）。
    27  //
    28  // 正規の Append では作れない状態を組み立てる専用の口（Seed / SetUnavailable / Hold /
    29  // SeedAppendedThrough）を持ちます。
    30  // 各口の使い分けは README を参照してください。
    31  type EventLog struct {
    32  	mu          sync.Mutex
    33  	streams     map[rt.StreamID][]rt.DeliveryEvent
    34  	appended    map[rt.StreamID]rt.Sequence
    35  	unavailable bool
    36  	// held は、閉じられるまで読み取りを待たせる関門です。nil のときは待たせません。
    37  	held chan struct{}
    38  }

    48  // Seed は、Validate も冪等判定も通さずに event を置きます。飛び番や保持期間外の位置を作れます
    49  // （EventLog の doc にある専用の口の一つ）。
    50  func (l *EventLog) Seed(events ...rt.DeliveryEvent) {
    51  	l.mu.Lock()
    52  	defer l.mu.Unlock()
    53  
    54  	for _, e := range events {
    55  		l.streams[e.StreamID] = insert(l.streams[e.StreamID], e)
    56  	}
    57  }
    58  
    59  // Hold は、以降の読み取りを、返った関数が呼ばれるまで待たせます
    60  // （EventLog の doc にある専用の口の一つ）。
    61  func (l *EventLog) Hold() func() {
    62  	l.mu.Lock()
    63  	gate := make(chan struct{})
    64  	l.held = gate
    65  	l.mu.Unlock()
    66  
    67  	var once sync.Once
    68  
    69  	return func() {
    70  		once.Do(func() {
    71  			l.mu.Lock()
    72  			if l.held == gate {
    73  				l.held = nil
    74  			}
    75  			l.mu.Unlock()
    76  			close(gate)
    77  		})
    78  	}
    79  }

    89  // Append は、event を (StreamID, Sequence) へ 1 度だけ書きます。同じ位置に同じ EventID があれば成功、
    90  // 異なる EventID があれば rt.ErrSequenceConflict を返します。
    91  func (l *EventLog) Append(_ context.Context, event rt.DeliveryEvent) error {
    92  	if err := event.Validate(); err != nil {
    93  		return err
    94  	}
    95  
    96  	l.mu.Lock()
    97  	defer l.mu.Unlock()
    98  
    99  	if l.unavailable {
   100  		return errEventLogUnavailable
   101  	}
   102  
   103  	for _, e := range l.streams[event.StreamID] {
   104  		if e.Sequence == event.Sequence {
   105  			if e.EventID == event.EventID {
   106  				return nil
   107  			}
   108  
   109  			return rt.ErrSequenceConflict
   110  		}
   111  	}
   112  
   113  	l.streams[event.StreamID] = insert(l.streams[event.StreamID], event)
   114  	if event.Sequence > l.appended[event.StreamID] {
   115  		l.appended[event.StreamID] = event.Sequence
   116  	}
   117  
   118  	return nil
   119  }

   142  // ReadAfter は、q.After より後ろの event を sequence 昇順に最大 q.Limit 件返します。
   143  func (l *EventLog) ReadAfter(ctx context.Context, q rt.ReadAfterQuery) (rt.ReadAfterResult, error) {
   144  	if err := l.awaitRelease(ctx); err != nil {
   145  		return rt.ReadAfterResult{}, err
   146  	}
   147  
   148  	l.mu.Lock()
   149  	defer l.mu.Unlock()
   150  
   151  	if l.unavailable {
   152  		return rt.ReadAfterResult{}, errEventLogUnavailable
   153  	}
   154  
   155  	limit := int(q.Limit)
   156  	if limit <= 0 {
   157  		limit = defaultReadLimit
   158  	}
   159  
   160  	var found []rt.DeliveryEvent
   161  	for _, e := range l.streams[q.StreamID] {
   162  		if e.Sequence > q.After {
   163  			found = append(found, e)
   164  		}
   165  	}
   166  
   167  	if len(found) > limit {
   168  		return rt.ReadAfterResult{Events: found[:limit], HasMore: true}, nil
   169  	}
   170  
   171  	return rt.ReadAfterResult{Events: found}, nil
   172  }

   174  // Latest は、sequence 昇順に保った slice の末尾を返します。stream が空なら ok=false です。
   175  func (l *EventLog) Latest(_ context.Context, streamID rt.StreamID) (rt.DeliveryEvent, bool, error) {
   176  	l.mu.Lock()
   177  	defer l.mu.Unlock()
   178  
   179  	if l.unavailable {
   180  		return rt.DeliveryEvent{}, false, errEventLogUnavailable
   181  	}
   182  
   183  	events := l.streams[streamID]
   184  	if len(events) == 0 {
   185  		return rt.DeliveryEvent{}, false, nil
   186  	}
   187  
   188  	return events[len(events)-1], true, nil
   189  }

   192  func (l *EventLog) Find(_ context.Context, streamID rt.StreamID, seq rt.Sequence) (rt.DeliveryEvent, bool, error) {
   193  	l.mu.Lock()
   194  	defer l.mu.Unlock()
   195  
   196  	if l.unavailable {
   197  		return rt.DeliveryEvent{}, false, errEventLogUnavailable
   198  	}
   199  
   200  	for _, e := range l.streams[streamID] {
   201  		if e.Sequence == seq {

   209  // awaitRelease は、Hold が掛かっていれば解けるまで待ちます。
   210  func (l *EventLog) awaitRelease(ctx context.Context) error {
   211  	l.mu.Lock()
   212  	gate := l.held
   213  	l.mu.Unlock()
   214  
   215  	if gate == nil {
   216  		return nil
   217  	}
   218  
   219  	select {
   220  	case <-gate:
   221  		return nil
   222  	case <-ctx.Done():
   223  		return ctx.Err()
   224  	}
   225  }

   227  // insert は、sequence 昇順を保ったまま event を差し込みます。同じ位置の event は置き換えます。
   228  func insert(events []rt.DeliveryEvent, e rt.DeliveryEvent) []rt.DeliveryEvent {
   229  	for i, cur := range events {
   230  		if cur.Sequence == e.Sequence {
   231  			events[i] = e
   232  
   233  			return events
   234  		}
   235  	}
   236  
   237  	events = append(events, e)
   238  	sort.Slice(events, func(i, j int) bool { return events[i].Sequence < events[j].Sequence })
   239  
   240  	return events
   241  }
