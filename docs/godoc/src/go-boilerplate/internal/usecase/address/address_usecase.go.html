    64  func (u *usecase) LookupByPostalCode(ctx context.Context, postalCode string) (*Result, error) {
    65  	ctx, endSpan := u.tracer.Start(ctx)
    66  	defer endSpan()
    67  
    68  	candidates, err := u.gateway.Lookup(ctx, postalCode)
    69  	if err != nil {
    70  		// 外部 lookup 障害時は degrade する（docs/spec/usecase/address.md）。
    71  		return &Result{Candidates: []*CandidateView{}, IsFallback: true}, nil
    72  	}
    73  
    74  	// 県名は同一郵便番号でほぼ単一のため memo 化して FindByName 呼び出しを最小化する
    75  	// （degrade の範囲は docs/spec/usecase/address.md）。
    76  	resolved := make(map[string]*uuid.UUID, 1)
    77  	views := make([]*CandidateView, 0, len(candidates))
    78  	for _, c := range candidates {
    79  		prefID, ok := resolved[c.PrefectureName]
    80  		if !ok {
    81  			p, ferr := u.prefectureQ.FindByName(ctx, c.PrefectureName)
    82  			switch {
    83  			case ferr == nil:
    84  				prefID = p.ID().ToPtr()
    85  			case xerrors.Is(ferr, apperror.ErrNotFound):
    86  				prefID = nil
    87  			default:
    88  				return nil, ferr
    89  			}
    90  			resolved[c.PrefectureName] = prefID
    91  		}
    92  		views = append(views, &CandidateView{
    93  			PrefectureID:   prefID,
    94  			PrefectureName: c.PrefectureName,
    95  			City:           c.City,
    96  			Town:           c.Town,
    97  		})
    98  	}
    99  
   100  	return &Result{Candidates: views, IsFallback: false}, nil
   101  }
