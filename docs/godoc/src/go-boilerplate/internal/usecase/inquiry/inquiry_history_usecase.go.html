    27  	ctx, endSpan := u.tracer.Start(ctx)
    28  	defer endSpan()
    29  
    30  	return tx.DoWithResult(ctx, u.txm, func(ctx context.Context) (*HistoryView, error) {
    31  		i, err := u.repo.FindActiveByUserID(ctx, params.UserID)
    32  		if err != nil {

    42  // historyOf は、現在位置を先に読み、その位置までのメッセージを 1 ページ読み出します。
    43  //
    44  // この順序は入れ替えてはなりません。現在位置を先に読み、それを上限にすることで
    45  // 「現在位置とメッセージを同じ snapshot で読んだ」のと等価になります
    46  // （docs/spec/usecase/inquiry.md の streamCursor と snapshot）。
    47  func (u *usecase) historyOf(
    48  	ctx context.Context,
    49  	i *inquiry.Inquiry,
    50  	afterSequence *int64,
    51  	first *int,
    52  ) (*HistoryView, error) {
    53  	limit := paging.NewLimit(first, historyLimitPolicy).Value()
    54  
    55  	cursor, ok, err := u.sequences.Current(ctx, conversationStreamID(i.ID().String()))
    56  	if err != nil {
    57  		return nil, err
    58  	}
    59  	if !ok {
    60  		return &HistoryView{InquiryID: i.ID(), Messages: []MessageView{}}, nil
    61  	}
    62  
    63  	messages, err := u.repo.ListMessages(ctx, i.ID(), inquiry.HistoryParams{
    64  		AfterSequence: afterSequence,
    65  		UpToSequence:  int64(cursor),
    66  		Limit:         limit + 1,
    67  	})
    68  	if err != nil {
    69  		return nil, err
    70  	}
    71  
    72  	views, next := pageOf(i.ID(), messages, limit)
    73  	return &HistoryView{
    74  		InquiryID:         i.ID(),
    75  		Messages:          views,
    76  		NextAfterSequence: next,
    77  		StreamCursor:      int64(cursor),
    78  	}, nil
    79  }

    81  // pageOf は、1 件多く読んだ結果を 1 ページ分の DTO と次ページの開始位置へ分けます。
    82  func pageOf(inquiryID uuid.UUID, messages []*inquiry.Message, limit int) ([]MessageView, *int64) {
    83  	var next *int64
    84  	if len(messages) > limit {
    85  		messages = messages[:limit]
    86  		last := messages[len(messages)-1].Sequence()
    87  		next = &last
    88  	}
    89  
    90  	views := make([]MessageView, 0, len(messages))
    91  	for _, m := range messages {
    92  		views = append(views, toMessageView(inquiryID, m))
    93  	}
    94  	return views, next
    95  }
