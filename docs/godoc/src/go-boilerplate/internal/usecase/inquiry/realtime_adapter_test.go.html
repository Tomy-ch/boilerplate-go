   104  				func(_ context.Context, in ucoutbox.EmitInput) (uuid.UUID, error) {

   110  			require.NoError(t, u.emitDelivery(context.Background(), i, m, 9))
   111  			require.Len(t, payloads, 2)
   112  
   113  			// feature の payload をそのまま載せると publish 側の ParseDeliveryEvent が読めず、
   114  			// relay が恒久エラーとして dead 化する。両者を繋ぐのはこの直列化形だけなので、
   115  			// 封筒として読めることと、機構が要る値が入っていることをここで固定する。
   116  			conversation, err := rt.ParseDeliveryEvent(payloads[0])
   117  			require.NoError(t, err)
   118  			assert.Equal(t, rt.StreamID(i.ID().String()), conversation.StreamID)
   119  			assert.Equal(t, rt.Sequence(m.Sequence()), conversation.Sequence)
   120  			assert.Equal(t, event.TypeMessageCreated, conversation.Type)
   121  			assert.Equal(t, event.SchemaVersionMessageCreated, conversation.SchemaVersion)
   122  			assert.False(t, conversation.OccurredAt.IsZero())
   123  			// EventID は outbox が採番する message_id と同じでなければならず、それが決まるのは
   124  			// Emit の後。publish 側が埋める前提で空のまま出す。
   125  			assert.Empty(t, conversation.EventID)
   126  
   127  			feed, err := rt.ParseDeliveryEvent(payloads[1])
   128  			require.NoError(t, err)
   129  			assert.Equal(t, feedStreamID, feed.StreamID)
   130  			// feed の位置は会話 stream とは別に採番される。取り違えると一覧の再開位置がずれる。
   131  			assert.Equal(t, rt.Sequence(9), feed.Sequence)

   174  			t.Parallel()
   175  
   176  			occurred := time.Date(2026, time.September, 3, 1, 2, 3, 0, time.UTC)
   177  			b, err := envelope(rt.DeliveryEvent{
   178  				StreamID: "stream-1", Sequence: 7, Type: "inquiry.message.created.v1",
   179  				OccurredAt: occurred, SchemaVersion: 1, Payload: []byte(`{"k":"v"}`),
   180  			})
   181  			require.NoError(t, err)
   182  
   183  			// publish 側が ParseDeliveryEvent で読む形であることが、両者を繋ぐ唯一の契約。
   184  			got, err := rt.ParseDeliveryEvent(b)
   185  			require.NoError(t, err)
   186  			assert.Equal(t, rt.StreamID("stream-1"), got.StreamID)
   187  			assert.Equal(t, rt.Sequence(7), got.Sequence)
   188  			assert.Equal(t, "inquiry.message.created.v1", got.Type)
   189  			assert.True(t, occurred.Equal(got.OccurredAt))
   190  			assert.Equal(t, 1, got.SchemaVersion)
   191  			assert.JSONEq(t, `{"k":"v"}`, string(got.Payload))
   192  			// EventID は outbox が採番する message_id と等しくなければならず、決まるのは Emit の後。
   193  			assert.Empty(t, got.EventID)
   194  		})
   195  	})
