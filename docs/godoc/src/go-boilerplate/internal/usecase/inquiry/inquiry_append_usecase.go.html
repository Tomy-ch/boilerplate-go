    11  // appendMessage は、投稿と回答が共有する 1 トランザクション分の追加処理です。
    12  //
    13  // 呼び出しの順序を入れ替えてはなりません。同一 stream への並行追加は先頭の採番で直列化されます

    15  func (u *usecase) appendMessage(
    16  	ctx context.Context,
    17  	i *inquiry.Inquiry,
    18  	author inquiry.Author,
    19  	body string,
    20  ) (MessageView, error) {
    21  	sequence, err := u.sequences.Allocate(ctx, conversationStreamID(i.ID().String()))
    22  	if err != nil {
    23  		return MessageView{}, err
    24  	}
    25  
    26  	messageID, err := uuid.New()
    27  	if err != nil {
    28  		return MessageView{}, xerrors.Wrap(err, "failed to generate inquiry message id")
    29  	}
    30  
    31  	m, err := i.AppendMessage(messageID, inquiry.MessageAttributes{
    32  		Author:   author,
    33  		Body:     body,
    34  		Sequence: int64(sequence),
    35  	}, u.clock.Now())
    36  	if err != nil {
    37  		return MessageView{}, err
    38  	}
    39  	if cerr := u.repo.CreateMessage(ctx, i.ID(), m); cerr != nil {
    40  		return MessageView{}, cerr
    41  	}
    42  	if uerr := u.repo.Update(ctx, i); uerr != nil {
    43  		return MessageView{}, uerr
    44  	}
    45  
    46  	feedSequence, err := u.sequences.Allocate(ctx, feedStreamID)
    47  	if err != nil {
    48  		return MessageView{}, err
    49  	}
    50  
    51  	// 作成日時は DB の既定値が刻むため、書き込んだ行を読み直してから event と応答を組み立てます。
    52  	// 読み直しは書き込みの検証も兼ねます（internal/usecase/README.md の
    53  	// Verifying infrastructure against the domain）。
    54  	stored, err := u.readBackMessage(ctx, i.ID(), int64(sequence))
    55  	if err != nil {
    56  		return MessageView{}, err
    57  	}
    58  
    59  	if eerr := u.emitDelivery(ctx, i, stored, int64(feedSequence)); eerr != nil {
    60  		return MessageView{}, eerr
    61  	}
    62  
    63  	return toMessageView(i.ID(), stored), nil
    64  }
