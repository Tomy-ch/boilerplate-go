   125  // decodeInquiryCursor は、不透明キー列を keyset 境界へ解釈します。
   126  // 先頭ページ（カーソル無し）では境界なし（nil）を正常値として返します。
   127  func decodeInquiryCursor(cursor *paging.Cursor) (*inquiry.Cursor, error) {
   128  	if !cursor.HasCursor() {
   129  		return nil, nil //nolint:nilnil // 先頭ページは境界なし（nil）を正常値として返す
   130  	}
   131  
   132  	keys := cursor.Keys()
   133  	if len(keys) != inquiryCursorKeyCount {
   134  		return nil, xerrors.Wrap(
   135  			apperror.ErrInvalidArgument,
   136  			fmt.Sprintf("invalid cursor: expected %d keys", inquiryCursorKeyCount),
   137  		)
   138  	}
   139  
   140  	updatedAt, err := time.Parse(time.RFC3339Nano, keys[0])
   141  	if err != nil {
   142  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, "invalid cursor: updated_at is not RFC3339Nano")
   143  	}
   144  	id, err := uuid.Parse(keys[1])
   145  	if err != nil {
   146  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, "invalid cursor: id is not a valid UUID")
   147  	}
   148  
   149  	return &inquiry.Cursor{UpdatedAt: updatedAt, ID: id}, nil
   150  }
