    48  // issueTicket は、認可済みの subject × stream に対して ticket を発行します。
    49  //
    50  // 開始位置には stream の現在位置を渡します。cursor を持たずに接続した client へ、履歴 API が担う
    51  // 過去分を購読側から再生し直さないためです（開始位置は認可の下限ではありません:
    52  // docs/design/realtime-delivery.md §2.3）。
    53  func (u *usecase) issueTicket(
    54  	ctx context.Context,
    55  	subject string,
    56  	destination rt.StreamID,
    57  	scope string,
    58  ) (TicketView, error) {
    59  	cursor, ok, err := u.sequences.Current(ctx, destination)
    60  	if err != nil {
    61  		return TicketView{}, err
    62  	}
    63  	if !ok {
    64  		cursor = 0
    65  	}
    66  
    67  	issued, err := u.tickets.Issue(ctx, ucrealtime.IssueTicketInput{
    68  		Subject:       subject,
    69  		Destination:   destination,
    70  		Scope:         scope,
    71  		InitialCursor: cursor,
    72  	})
    73  	if err != nil {
    74  		return TicketView{}, err
    75  	}
    76  
    77  	return TicketView{
    78  		Ticket:    issued.Value,
    79  		StreamID:  destination.String(),
    80  		ExpiresAt: issued.ExpiresAt,
    81  	}, nil
    82  }
