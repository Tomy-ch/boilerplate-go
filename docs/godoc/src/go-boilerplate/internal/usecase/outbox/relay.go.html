    62  // RelayUsecase は、pending のエントリを claim して publish するユースケースです。relay engine が周期的に呼びます。

   119  func (u *relayUsecase) RecordLag(ctx context.Context) error {
   120  	ctx, endSpan := u.tracer.Start(ctx)
   121  	defer endSpan()
   122  
   123  	createdAt, ok, err := u.store.OldestPendingCreatedAt(ctx, u.channel)
   124  	if err != nil {
   125  		return err
   126  	}
   127  
   128  	var lag time.Duration
   129  	if ok {
   130  		if lag = u.clock.Now().Sub(createdAt); lag < 0 {
   131  			lag = 0
   132  		}
   133  	}
   134  	u.metrics.SetLagSeconds(ctx, u.channel.String(), int64(lag.Seconds()))
   135  	return nil
   136  }

   150  func (u *relayUsecase) RelayBatch(ctx context.Context, batchSize int32) (RelayResult, error) {
   151  	ctx, endSpan := u.tracer.Start(ctx)
   152  	defer endSpan()
   153  
   154  	if batchSize <= 0 {
   155  		batchSize = DefaultBatchSize
   156  	}
   157  
   158  	return tx.DoWithResult(ctx, u.txm, func(ctx context.Context) (RelayResult, error) {
   159  		msgs, err := u.store.ClaimPending(ctx, u.channel, batchSize)
   160  		if err != nil {
   161  			return RelayResult{}, err
   162  		}
   163  		res := RelayResult{Claimed: len(msgs)}
   164  		for i := range msgs {
   165  			published, derr := u.deliver(ctx, msgs[i])
   166  			if derr != nil {
   167  				return RelayResult{}, derr
   168  			}
   169  			if published {
   170  				res.Published++
   171  			}
   172  		}
   173  		return res, nil
   174  	})
   175  }

   177  // deliver は、1 件を publish し、結果に応じて published / failed / dead をマークします（tx 方針は RelayBatch を参照）。
   178  // publish 成功なら published=true を、DB マーク自体の失敗時のみ error を返します。
   179  func (u *relayUsecase) deliver(ctx context.Context, m outboxbndry.PendingMessage) (bool, error) {
   180  	perr := u.publisher.Publish(ctx, publisher.Message{
   181  		MessageID: m.MessageID,
   182  		EventType: m.EventType,
   183  		Payload:   m.Payload,
   184  		Headers:   decodeHeaders(m.Headers),
   185  		CreatedAt: m.CreatedAt,
   186  	})
   187  	if perr == nil {
   188  		return true, u.store.MarkPublished(ctx, m.ID)
   189  	}
   190  
   191  	// 失敗理由は dead でも残すため、dead 化の前に必ず記録する。
   192  	permanent := isPermanent(perr)
   193  	nextAttemptAt := u.clock.Now()
   194  	if !permanent {
   195  		nextAttemptAt = nextAttemptAt.Add(retryDelay(m.Attempts))
   196  	}
   197  	if ferr := u.store.MarkFailed(ctx, m.ID, perr.Error(), nextAttemptAt); ferr != nil {
   198  		return false, ferr
   199  	}
   200  	if !permanent {
   201  		return false, nil
   202  	}
   203  
   204  	if derr := u.store.MarkDead(ctx, m.ID); derr != nil {
   205  		return false, derr
   206  	}
   207  	u.metrics.IncDead(ctx, u.channel.String())
   208  	u.logging.Named(relayLoggerName).Warn(
   209  		ctx,
   210  		"outbox message marked dead by permanent publish failure",
   211  		logging.String(logging.MessageIDKey, m.MessageID.String()),
   212  		logging.String(logging.EventTypeKey, m.EventType),
   213  		logging.Error(logging.JobErrorKey, perr),
   214  	)
   215  	return false, nil
   216  }

   218  // isPermanent は、publish の失敗が再試行で結果の変わらない永久失敗かを返します。
   219  // どちらの分類も持たないエラーは一時失敗として扱います（既定の理由は ADR-0058）。
   220  func isPermanent(err error) bool {
   221  	return xerrors.Is(err, apperror.ErrPermanent)
   222  }

   224  // retryDelay は、attempts 回失敗した行を次に claim してよくなるまでの間隔を、指数バックオフへ
   225  // full jitter を重ねて返します（同時に失敗した行が同じ poll へ殺到しないため）。
   226  func retryDelay(attempts int32) time.Duration {
   227  	return retry.Full(retryBackoff.Duration(int(attempts)))
   228  }
