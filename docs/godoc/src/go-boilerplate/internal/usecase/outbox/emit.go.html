    61  func (u *emitUsecase) Emit(ctx context.Context, in EmitInput) (uuid.UUID, error) {
    62  	ctx, endSpan := u.tracer.Start(ctx)
    63  	defer endSpan()
    64  
    65  	if err := validateEmit(in); err != nil {
    66  		return uuid.UUID{}, err
    67  	}
    68  
    69  	headers := make(map[string]string, len(in.Headers)+1)
    70  	for k, v := range in.Headers {
    71  		// defense-in-depth としての除去（契約は EmitInput.Headers を参照）。
    72  		if httpheader.IsSensitive(k) {
    73  			continue
    74  		}
    75  		headers[k] = v
    76  	}
    77  	// trace context を headers へ注入する（契約は Emit の doc を参照）。
    78  	observability.InjectTraceContextToCarrier(ctx, headers)
    79  
    80  	var headerBytes []byte
    81  	if len(headers) > 0 {
    82  		b, err := json.Marshal(headers)
    83  		if err != nil {
    84  			return uuid.UUID{}, xerrors.Join(apperror.ErrInternal, xerrors.Wrap(err, "failed to encode outbox headers"))
    85  		}
    86  		headerBytes = b
    87  	}
    88  
    89  	return u.store.Insert(ctx, outboxbndry.EmitParams{
    90  		AggregateType:    in.AggregateType,
    91  		AggregateID:      in.AggregateID,
    92  		EventType:        in.EventType,
    93  		Payload:          in.Payload,
    94  		Headers:          headerBytes,
    95  		Channel:          in.Channel,
    96  		OrderingKey:      in.OrderingKey,
    97  		OrderingSequence: in.OrderingSequence,
    98  	})
    99  }

   101  // validateEmit は、行を作る前に配送チャネルと順序指定を検査します。
   102  // 未知のチャネルの行はどの relay も claim せず、対になっていない順序指定は DB の制約で弾かれるため、
   103  // いずれも呼び出し元へ即座に返します。
   104  func validateEmit(in EmitInput) error {
   105  	if _, err := outboxbndry.ParseChannel(in.Channel.String()); err != nil {
   106  		return err
   107  	}
   108  
   109  	switch {
   110  	case in.OrderingKey == "" && in.OrderingSequence != 0:
   111  		return xerrors.Wrap(ErrInvalidOrdering, "ordering sequence requires an ordering key")
   112  	case in.OrderingKey != "" && in.OrderingSequence <= 0:
   113  		return xerrors.Wrap(ErrInvalidOrdering, "ordering key requires a positive ordering sequence")
   114  	default:
   115  		return nil
   116  	}
   117  }
