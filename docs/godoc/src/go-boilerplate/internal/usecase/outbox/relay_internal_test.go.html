   131  			ctrl := gomock.NewController(t)

   177  		t.Run("一時失敗の待機は毎回ゼロではない", func(t *testing.T) {
   178  			t.Parallel()
   179  			ctrl := gomock.NewController(t)
   180  			store := mock_outbox.NewMockStore(ctrl)
   181  			pub := mock_publisher.NewMockPublisher(ctrl)
   182  			msg := deliverMessage(t)
   183  
   184  			// full jitter は [0, d] の一様乱数なので、1 回の呼び出しでは「待機 0」と
   185  			// 「バックオフの加算そのものが無い」を区別できない。繰り返して 1 度でも
   186  			// 前進すれば、加算を落とす退行を捕まえられる。
   187  			const attempts = 32
   188  			var advanced bool
   189  
   190  			pub.EXPECT().Publish(gomock.Any(), gomock.Any()).
   191  				Return(xerrors.Join(apperror.ErrRetryable, xerrors.New("publish failed"))).Times(attempts)
   192  			store.EXPECT().MarkFailed(gomock.Any(), msg.ID, gomock.Any(), gomock.Any()).DoAndReturn(
   193  				func(_ context.Context, _ int64, _ string, nextAttemptAt time.Time) error {
   194  					if nextAttemptAt.After(deliverNow) {
   195  						advanced = true
   196  					}
   197  					return nil
   198  				},
   199  			).Times(attempts)
   200  
   201  			u := newDeliverUsecase(t, store, pub, observability.NewNoopOutboxMetrics(t))
   202  			for range attempts {
   203  				_, err := u.deliver(context.Background(), msg)
   204  				require.NoError(t, err)
   205  			}
   206  
   207  			assert.True(t, advanced, "一時失敗は次回試行時刻を前進させる")
   208  		})
