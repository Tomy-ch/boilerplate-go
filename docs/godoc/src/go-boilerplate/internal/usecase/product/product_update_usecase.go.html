    33  	StatusID *uuid.UUID

    45  // UpdateProduct は、読み込みから更新までを 1 つのトランザクションで行い、読み込み時点のバージョンを
    46  // 条件に更新することで並行編集による上書き（lost update）を防ぎます。
    47  // バージョン不一致の 409 は tx.Manager の透過リトライでは解消しないため、リトライ対象と混同されては
    48  // なりません（クライアントの再試行手順は docs/spec/usecase/product.md の UpdateProduct）。
    49  func (u *usecase) UpdateProduct(
    50  	ctx context.Context, authn *auth.Authn, id uuid.UUID, params UpdateProductParams,
    51  ) (ProductView, error) {
    52  	ctx, endSpan := u.tracer.Start(ctx)
    53  	defer endSpan()
    54  
    55  	if authn == nil {
    56  		return ProductView{}, apperror.ErrUnauthenticated
    57  	}
    58  	if err := u.authorizer.Authorize(ctx, authn, authz.ActionProductUpdate, authz.NewResource("product", nil)); err != nil {
    59  		return ProductView{}, err
    60  	}
    61  
    62  	price, err := parseOptionalPrice(params.Price)
    63  	if err != nil {
    64  		return ProductView{}, err
    65  	}
    66  
    67  	images, err := resolveUpdatedImages(params.Images)
    68  	if err != nil {
    69  		return ProductView{}, err
    70  	}
    71  
    72  	var entity *product.Product
    73  	var updatedVersion int
    74  	err = u.txm.Do(ctx, func(ctx context.Context) error {
    75  		loaded, err := u.repo.FindByID(ctx, id)
    76  		if err != nil {
    77  			return err
    78  		}
    79  		entity = loaded
    80  
    81  		if err = entity.EnsureVersion(params.Version); err != nil {
    82  			return err
    83  		}
    84  
    85  		statusRef, categoryRef, err := u.resolveUpdatedRefs(ctx, entity, params)
    86  		if err != nil {
    87  			return err
    88  		}
    89  
    90  		if err = entity.Update(product.Attributes{
    91  			Name:                  ptr.Deref(params.Name, entity.Name()),
    92  			Description:           params.Description.Resolve(entity.Description()),
    93  			Price:                 ptr.Deref(price, entity.Price()),
    94  			Quantity:              ptr.Deref(params.Quantity, entity.Quantity()),
    95  			StockWarningThreshold: params.StockWarningThreshold.Resolve(entity.StockWarningThreshold()),
    96  			Status:                statusRef,
    97  			Category:              categoryRef,
    98  			PublishedAt:           params.PublishedAt.Resolve(entity.PublishedAt()),
    99  			// DiscontinuedAt は UpdateProductParams が持たないため、現在値を据え置きます。
   100  			DiscontinuedAt: entity.DiscontinuedAt(),
   101  			Images:         ptr.Deref(images, entity.Images()),
   102  		}); err != nil {
   103  			return err
   104  		}
   105  
   106  		updatedVersion, err = u.repo.Update(ctx, entity)
   107  
   108  		return err
   109  	})
   110  	if err != nil {
   111  		return ProductView{}, err
   112  	}
   113  
   114  	view := toProductView(entity)
   115  	// バージョンの採番は永続化時に DB が行うため、エンティティが保持する読み込み時点の値を採番後の値で置き換えます。
   116  	view.Version = updatedVersion
   117  
   118  	return view, nil
   119  }

   121  // resolveUpdatedRefs は、部分更新後の商品ステータス参照・カテゴリ参照を解決します。
   122  // 参照はペアで扱い、両方とも未指定の場合のみ現在値を据え置いてマスタ問い合わせを行いません。
   123  // いずれかが指定された場合は、未指定側も現在の ID でマスタと突合し、参照整合を再確認します。
   124  func (u *usecase) resolveUpdatedRefs(
   125  	ctx context.Context, entity *product.Product, params UpdateProductParams,
   126  ) (product.StatusRef, product.CategoryRef, error) {
   127  	if params.StatusID == nil && params.CategoryID == nil {
   128  		return entity.Status(), entity.Category(), nil
   129  	}
   130  
   131  	statusRef, categoryRef, err := u.resolveRefs(
   132  		ctx,
   133  		ptr.Deref(params.StatusID, entity.Status().ID()),
   134  		ptr.Deref(params.CategoryID, entity.Category().ID()),
   135  	)
   136  	if err != nil {
   137  		return product.StatusRef{}, product.CategoryRef{}, err
   138  	}
   139  
   140  	return statusRef, categoryRef, nil
   141  }

   144  // 未指定の場合は nil を返し、呼び出し側が現在の画像を据え置きます。null 指定は空の集合として扱い、

   146  func resolveUpdatedImages(field patch.Field[[]ProductImageParams]) (*[]product.Image, error) {
   147  	if !field.IsSpecified() {
   148  		return nil, nil //nolint:nilnil // 未指定は「画像の指定なし」を表すため nil, nil が妥当です
   149  	}
   150  
   151  	specified := field.Resolve(nil)
   152  	if specified == nil {
   153  		return &[]product.Image{}, nil
   154  	}
   155  
   156  	images, err := buildImages(*specified)
   157  	if err != nil {
   158  		return nil, err
   159  	}

   164  // parseOptionalPrice は、指定された場合のみ十進文字列を価格へ解釈します。
   165  // 非数値は入力エラー（400）、負値は業務不変条件違反（422）として扱います。
   166  func parseOptionalPrice(v *string) (*money.Price, error) {
   167  	if v == nil {
   168  		return nil, nil //nolint:nilnil // 未指定は「価格の指定なし」を表すため nil, nil が妥当です
   169  	}
   170  
   171  	parsed, err := decimal.Parse(*v)
   172  	if err != nil {
   173  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, "price is not a valid decimal: "+err.Error())
   174  	}
   175  	price, err := money.NewPrice(parsed)
   176  	if err != nil {
   177  		return nil, err
   178  	}
   179  
   180  	return &price, nil
   181  }
