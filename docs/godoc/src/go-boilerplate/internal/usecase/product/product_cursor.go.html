    41  // decodeProductCursor は、cursor の不透明キー列を keyset 境界（productCursor）へ解釈します。
    42  // 先頭ページ（カーソル無し）の場合は nil を返します。キーの個数・型が不正な場合は ErrInvalidArgument を返します。
    43  func decodeProductCursor(cursor *paging.Cursor) (*productCursor, error) {
    44  	if !cursor.HasCursor() {
    45  		return nil, nil //nolint:nilnil // 先頭ページは境界なし（nil）を正常値として返す
    46  	}
    47  
    48  	keys := cursor.Keys()
    49  	if len(keys) != productCursorKeyCount {
    50  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument,
    51  			"invalid cursor: cursor was issued for a different includeUnpublished")
    52  	}
    53  
    54  	publishedAt, err := time.Parse(time.RFC3339Nano, keys[0])
    55  	if err != nil {
    56  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, "invalid cursor: published_at is not RFC3339Nano")
    57  	}
    58  
    59  	id, err := uuid.Parse(keys[1])
    60  	if err != nil {
    61  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, "invalid cursor: id is not a valid UUID")
    62  	}
    63  
    64  	return &productCursor{publishedAt: publishedAt, id: id}, nil
    65  }
