   233  func parseProductPriceFilter(name, value string) (*money.Price, error) {
   234  	if len(value) > maxProductPriceFilterLength {
   235  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, name+" is too long")
   236  	}
   237  	parsed, err := decimal.Parse(value)
   238  	if err != nil {
   239  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, name+" must be a non-negative decimal")
   240  	}
   241  	price, err := money.NewPrice(parsed)
   242  	if err != nil {
   243  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, name+" must be a non-negative decimal")
   244  	}
   245  	return &price, nil
   246  }

   248  // validateMasterFilter は、マスタ参照の絞り込みが非推奨の ID と後継のコードで二重に指定されていないことを
   249  // 検証します。両方が来た場合にどちらかを優先すると、送ったのに無視された側が静かに効かなくなるため拒否します。
   250  func validateMasterFilter(filter SearchFilter) error {
   251  	if filter.CategoryID != nil && len(filter.CategoryCodes) > 0 {
   252  		return xerrors.Wrap(apperror.ErrInvalidArgument, "categoryId and categoryCodes must not be specified together")
   253  	}
   254  	if filter.StatusID != nil && len(filter.StatusCodes) > 0 {
   255  		return xerrors.Wrap(apperror.ErrInvalidArgument, "statusId and statusCodes must not be specified together")
   256  	}
   257  	return nil
   258  }

   260  func parseProductListRange(filter SearchFilter) (productListRange, error) {
   261  	if err := validateMasterFilter(filter); err != nil {
   262  		return productListRange{}, err
   263  	}
   264  
   265  	result := productListRange{minQuantity: filter.MinQuantity, maxQuantity: filter.MaxQuantity}
   266  	if filter.MinPrice != nil {
   267  		minPrice, err := parseProductPriceFilter("minPrice", *filter.MinPrice)
   268  		if err != nil {
   269  			return productListRange{}, err
   270  		}
   271  		result.minPrice = minPrice
   272  	}
   273  	if filter.MaxPrice != nil {
   274  		maxPrice, err := parseProductPriceFilter("maxPrice", *filter.MaxPrice)
   275  		if err != nil {
   276  			return productListRange{}, err
   277  		}
   278  		result.maxPrice = maxPrice
   279  	}
   280  	if result.minPrice != nil && result.maxPrice != nil &&
   281  		result.minPrice.Decimal().Cmp(result.maxPrice.Decimal()) > 0 {
   282  		return productListRange{}, xerrors.Wrap(apperror.ErrInvalidArgument, "minPrice must not exceed maxPrice")
   283  	}
   284  	if result.minQuantity != nil && *result.minQuantity < 0 {
   285  		return productListRange{}, xerrors.Wrap(apperror.ErrInvalidArgument, "minQuantity must be non-negative")
   286  	}
   287  	if result.maxQuantity != nil && *result.maxQuantity < 0 {
   288  		return productListRange{}, xerrors.Wrap(apperror.ErrInvalidArgument, "maxQuantity must be non-negative")
   289  	}
   290  	if result.minQuantity != nil && result.maxQuantity != nil && *result.minQuantity > *result.maxQuantity {
   291  		return productListRange{}, xerrors.Wrap(apperror.ErrInvalidArgument, "minQuantity must not exceed maxQuantity")
   292  	}
   293  	return result, nil
   294  }

   296  func (u *usecase) ListProducts(
   297  	ctx context.Context, authn *auth.Authn, params ListProductsParams,
   298  ) (ProductListView, error) {
   299  	if params.Cursor == nil {
   300  		return ProductListView{}, xerrors.Wrap(apperror.ErrInvalidArgument, "cursor must not be nil")
   301  	}
   302  
   303  	rangeFilter, err := parseProductListRange(params.SearchFilter)
   304  	if err != nil {
   305  		return ProductListView{}, err
   306  	}
   307  
   308  	ctx, endSpan := u.tracer.Start(ctx)
   309  	defer endSpan()
   310  
   311  	if err := u.authorizeUnpublishedRead(ctx, authn, params.IncludeUnpublished); err != nil {
   312  		return ProductListView{}, err
   313  	}
   314  
   315  	// 母集団が変わると並び順の軸も変わるため、取得とカーソル生成は同じ枝で揃えます。
   316  	var (
   317  		products product.Products
   318  		encode   func(last *product.Product) string
   319  	)
   320  	if params.IncludeUnpublished {
   321  		products, err = u.findAllPage(ctx, params, rangeFilter)
   322  		encode = encodeAllProductCursor
   323  	} else {
   324  		products, err = u.findPublishedPage(ctx, params, rangeFilter)
   325  		encode = encodeProductCursor
   326  	}
   327  	if err != nil {
   328  		return ProductListView{}, err
   329  	}
   330  
   331  	limit := params.Cursor.Limit()
   332  	hasNext := len(products) > limit
   333  	if hasNext {
   334  		products = products[:limit]
   335  	}
   336  
   337  	items := make([]ProductView, len(products))
   338  	for i, p := range products {
   339  		items[i] = toProductView(p)
   340  	}
   341  
   342  	var nextCursor *string
   343  	if hasNext && len(products) > 0 {
   344  		// len 判定は防御的な安全弁。hasNext は len > limit なので limit >= 1 の下では冗長だが、
   345  		// limit の下限保証は paging.NewCursor 依存であり、ゼロ値 Cursor 混入時の products[-1] panic を防ぐ。
   346  		encoded := encode(products[len(products)-1])
   347  		nextCursor = &encoded
   348  	}
   349  
   350  	return ProductListView{Items: items, NextCursor: nextCursor}, nil
   351  }

   353  func (u *usecase) CountProducts(
   354  	ctx context.Context, authn *auth.Authn, params CountProductsParams,
   355  ) (ProductCountView, error) {
   356  	rangeFilter, err := parseProductListRange(params.SearchFilter)
   357  	if err != nil {
   358  		return ProductCountView{}, err
   359  	}
   360  
   361  	ctx, endSpan := u.tracer.Start(ctx)
   362  	defer endSpan()
   363  
   364  	if err := u.authorizeUnpublishedRead(ctx, authn, params.IncludeUnpublished); err != nil {
   365  		return ProductCountView{}, err
   366  	}
   367  
   368  	filter := toDomainSearchFilter(params.SearchFilter, rangeFilter)
   369  
   370  	var count int64
   371  	if params.IncludeUnpublished {
   372  		count, err = u.repo.CountAll(ctx, filter)
   373  	} else {
   374  		count, err = u.repo.CountPublished(ctx, filter)
   375  	}
   376  	if err != nil {
   377  		return ProductCountView{}, err
   378  	}
   379  
   380  	return ProductCountView{Count: count}, nil
   381  }
