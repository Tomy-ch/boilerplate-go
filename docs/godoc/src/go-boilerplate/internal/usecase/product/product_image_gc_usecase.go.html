    75  func (u *imageGCUsecase) SweepOrphans(
    76  	ctx context.Context, grace time.Duration, batchSize int32, dryRun bool,
    77  ) (ImageGCResult, error) {
    78  	ctx, endSpan := u.tracer.Start(ctx)
    79  	defer endSpan()
    80  
    81  	if grace <= 0 {
    82  		grace = DefaultImageGCGrace
    83  	}
    84  	if batchSize <= 0 {
    85  		batchSize = DefaultImageGCBatchSize
    86  	}
    87  	cutoff := u.clock.Now().Add(-grace)
    88  
    89  	var (
    90  		result ImageGCResult
    91  		cursor string
    92  	)
    93  	for {
    94  		page, err := u.sweepPage(ctx, cutoff, cursor, batchSize, dryRun)
    95  		if err != nil {
    96  			return result, err
    97  		}
    98  		result.Deleted += page.deleted
    99  		result.Scanned += page.scanned
   100  
   101  		if page.nextCursor == "" {
   102  			return result, nil
   103  		}
   104  		cursor = page.nextCursor
   105  	}
   106  }

   108  // sweepPage は、1 ページ分の列挙・参照照合・削除を実行します。
   109  func (u *imageGCUsecase) sweepPage(
   110  	ctx context.Context, cutoff time.Time, cursor string, batchSize int32, dryRun bool,
   111  ) (imageGCPageResult, error) {
   112  	listed, err := u.storage.List(ctx, objectstorage.ListQuery{
   113  		Prefix: imageKeyPrefix,
   114  		Cursor: cursor,
   115  		Limit:  batchSize,
   116  	})
   117  	if err != nil {
   118  		return imageGCPageResult{}, err
   119  	}
   120  
   121  	page := imageGCPageResult{nextCursor: listed.NextCursor}
   122  	candidates := agedImageKeys(listed.Objects, cutoff)
   123  	if len(candidates) == 0 {
   124  		return page, nil
   125  	}
   126  	page.scanned = int64(len(candidates))
   127  
   128  	// 参照照合の失敗はそのまま伝播し、このページの削除は行わない（fail-open 禁止。理由は
   129  	// docs/spec/usecase/product.md の SweepOrphans notes）。
   130  	referenced, err := u.images.FilterExistingImagePaths(ctx, candidates)
   131  	if err != nil {
   132  		return imageGCPageResult{}, err
   133  	}
   134  
   135  	orphans := excludeKeys(candidates, referenced)
   136  	if len(orphans) == 0 {
   137  		return page, nil
   138  	}
   139  	if dryRun {
   140  		page.deleted = int64(len(orphans))
   141  		return page, nil
   142  	}
   143  
   144  	if err := u.storage.Delete(ctx, orphans); err != nil {
   145  		return imageGCPageResult{}, err
   146  	}
   147  	page.deleted = int64(len(orphans))
   148  	return page, nil
   149  }

   151  // agedImageKeys は、猶予期間を過ぎた商品画像オブジェクトのキーを返します。接頭辞を再検査する理由は
   152  // docs/spec/usecase/product.md の SweepOrphans notes を参照。
   153  func agedImageKeys(objects []objectstorage.Object, cutoff time.Time) []string {
   154  	keys := make([]string, 0, len(objects))
   155  	for _, o := range objects {
   156  		if !strings.HasPrefix(o.Key, imageKeyPrefix) {
   157  			continue
   158  		}
   159  		if !o.ModifiedAt.Before(cutoff) {
   160  			continue
   161  		}
   162  		keys = append(keys, o.Key)
   163  	}
   164  	return keys
   165  }

   167  // excludeKeys は、keys から excluded に含まれるものを取り除いた並びを返します。
   168  func excludeKeys(keys, excluded []string) []string {
   169  	if len(excluded) == 0 {
   170  		return keys
   171  	}
   172  	excludedSet := make(map[string]struct{}, len(excluded))
   173  	for _, k := range excluded {
   174  		excludedSet[k] = struct{}{}
   175  	}
   176  	remained := make([]string, 0, len(keys))
   177  	for _, k := range keys {
   178  		if _, ok := excludedSet[k]; !ok {
   179  			remained = append(remained, k)
   180  		}
   181  	}
   182  	return remained
   183  }
