    44  func (u *usecase) CreateProduct(ctx context.Context, authn *auth.Authn, params CreateProductParams) (ProductView, error) {
    45  	ctx, endSpan := u.tracer.Start(ctx)
    46  	defer endSpan()
    47  
    48  	if authn == nil {
    49  		return ProductView{}, apperror.ErrUnauthenticated
    50  	}
    51  	if err := u.authorizer.Authorize(ctx, authn, authz.ActionProductCreate, authz.NewResource("product", nil)); err != nil {
    52  		return ProductView{}, err
    53  	}
    54  
    55  	price, err := decimal.Parse(params.Price)
    56  	if err != nil {
    57  		return ProductView{}, xerrors.Wrap(apperror.ErrInvalidArgument, "price is not a valid decimal: "+err.Error())
    58  	}
    59  	productPrice, err := money.NewPrice(price)
    60  	if err != nil {
    61  		return ProductView{}, apperror.WithDetails(err, product.FieldPrice)
    62  	}
    63  
    64  	id, err := uuid.New()
    65  	if err != nil {
    66  		return ProductView{}, xerrors.Wrap(err, "failed to generate product id")
    67  	}
    68  	images, err := buildImages(params.Images)
    69  	if err != nil {
    70  		return ProductView{}, err
    71  	}
    72  
    73  	var entity *product.Product
    74  	err = u.txm.Do(ctx, func(ctx context.Context) error {
    75  		statusRef, categoryRef, err := u.resolveRefs(ctx, params.StatusID, params.CategoryID)
    76  		if err != nil {
    77  			return err
    78  		}
    79  
    80  		entity, err = product.New(id, product.Attributes{
    81  			Name:                  params.Name,
    82  			Description:           params.Description,
    83  			Price:                 productPrice,
    84  			Quantity:              params.Quantity,
    85  			StockWarningThreshold: params.StockWarningThreshold,
    86  			Status:                statusRef,
    87  			Category:              categoryRef,
    88  			PublishedAt:           params.PublishedAt,
    89  			Images:                images,
    90  		}, u.clock.Now())
    91  		if err != nil {
    92  			return err
    93  		}
    94  
    95  		return u.repo.Create(ctx, entity)
    96  	})
    97  	if err != nil {
    98  		return ProductView{}, err
    99  	}
   100  
   101  	return toProductView(entity), nil

   119  // resolveRefs は、status / category の ID から名称を解決し、商品ステータス参照・カテゴリ参照を構築します。
   120  // マスタ不在に加え、取得できたマスタの名称が参照不変条件（NewStatusRef / NewCategoryRef の検証）に
   121  // 反する場合も、いずれもサーバ側データ不整合による参照整合性破れ（errMissingStatus / errMissingCategory）
   122  // としてサーバ内部エラーへ正規化します。クライアント起因の入力エラーではないため 4xx へは露出させません。
   123  func (u *usecase) resolveRefs(
   124  	ctx context.Context, statusID, categoryID uuid.UUID,
   125  ) (product.StatusRef, product.CategoryRef, error) {
   126  	statusEntity, err := u.statusRepo.FindByID(ctx, statusID)
   127  	if err != nil {
   128  		if xerrors.Is(err, apperror.ErrNotFound) {
   129  			return product.StatusRef{}, product.CategoryRef{}, errMissingStatus
   130  		}
   131  		return product.StatusRef{}, product.CategoryRef{}, err
   132  	}
   133  	statusRef, err := product.NewStatusRef(statusEntity.ID(), statusEntity.Name())
   134  	if err != nil {
   135  		return product.StatusRef{}, product.CategoryRef{}, errMissingStatus
   136  	}
   137  
   138  	categoryEntity, err := u.categoryRepo.FindByID(ctx, categoryID)
   139  	if err != nil {
   140  		if xerrors.Is(err, apperror.ErrNotFound) {
   141  			return product.StatusRef{}, product.CategoryRef{}, errMissingCategory
   142  		}
   143  		return product.StatusRef{}, product.CategoryRef{}, err
   144  	}
   145  	categoryRef, err := product.NewCategoryRef(categoryEntity.ID(), categoryEntity.Name())
   146  	if err != nil {
   147  		return product.StatusRef{}, product.CategoryRef{}, errMissingCategory
   148  	}
   149  
   150  	return statusRef, categoryRef, nil
   151  }
