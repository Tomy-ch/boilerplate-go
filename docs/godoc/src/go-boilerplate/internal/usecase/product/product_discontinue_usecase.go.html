    47  // DiscontinueProduct は、admin が商品を廃番にし、何が起きたかを件数で返します。
    48  // 非公開化・進行中購入の判定・代替クーポンの一括発行を 1 トランザクションで行います。
    49  // 分岐の判別根拠は ADR-0034 (commandservice-atomicity-criterion) の Worked instances、
    50  // ロック順序と再実行時の規律は docs/spec/usecase/product.md の Workflow — DiscontinueProduct を参照。
    51  func (u *usecase) DiscontinueProduct(
    52  	ctx context.Context,
    53  	authn *auth.Authn,
    54  	id uuid.UUID,
    55  	params DiscontinueProductParams,
    56  ) (DiscontinueProductView, error) {
    57  	ctx, endSpan := u.tracer.Start(ctx)
    58  	defer endSpan()
    59  
    60  	if err := u.authorizer.Authorize(ctx, authn, authz.ActionProductDiscontinue, authz.NewResource("product", nil)); err != nil {
    61  		return DiscontinueProductView{}, err
    62  	}
    63  
    64  	discount, err := coupon.NewRateDiscount(params.CouponDiscountRate)
    65  	if err != nil {
    66  		return DiscontinueProductView{}, err
    67  	}
    68  
    69  	now := u.clock.Now()
    70  
    71  	var view DiscontinueProductView
    72  	err = u.txm.Do(ctx, func(ctx context.Context) error {
    73  		entity, lerr := u.repo.LockByID(ctx, id)
    74  		if lerr != nil {
    75  			return lerr
    76  		}
    77  
    78  		if entity.IsDiscontinued() {
    79  			// 件数はいずれもこの実行が起こしたことを表すため、何も起こさない再実行では 0 のままにします。
    80  			view = DiscontinueProductView{DiscontinuedAt: *entity.DiscontinuedAt()}
    81  
    82  			return nil
    83  		}
    84  
    85  		statuses, perr := u.purchaseRepo.FindStatusesByProductID(ctx, id)
    86  		if perr != nil {
    87  			return perr
    88  		}
    89  		if derr := discontinuation.EnsureDiscontinuable(statuses); derr != nil {
    90  			return derr
    91  		}
    92  
    93  		if derr := entity.Discontinue(now); derr != nil {
    94  			return derr
    95  		}
    96  		if _, uerr := u.repo.Update(ctx, entity); uerr != nil {
    97  			return uerr
    98  		}
    99  
   100  		// 適用範囲は廃番商品のカテゴリで固定です。廃番商品自身を範囲にすると買えない商品にしか
   101  		// 使えないため、この journey が配るクーポンの範囲は 1 つに決まっています。
   102  		scope, serr := coupon.NewCategoryScope(entity.Category().ID())
   103  		if serr != nil {
   104  			return serr
   105  		}
   106  
   107  		result, ierr := u.discontinueCmd.IssueDiscontinuationCoupons(ctx, command.IssueDiscontinuationCouponsParams{
   108  			ProductID: id,
   109  			Scope:     scope,
   110  			Discount:  discount,
   111  			ExpiresAt: now.Add(params.CouponValidity),
   112  			IssuedAt:  now,
   113  		})
   114  		if ierr != nil {
   115  			return ierr
   116  		}
   117  
   118  		view = DiscontinueProductView{
   119  			DiscontinuedAt:    now,
   120  			AffectedCartCount: result.AffectedCartCount,
   121  			AffectedUserCount: result.AffectedUserCount,
   122  			IssuedCouponCount: result.IssuedCouponCount,
   123  		}
   124  
   125  		return nil
   126  	})
   127  	if err != nil {
   128  		return DiscontinueProductView{}, err
   129  	}
   130  
   131  	return view, nil
   132  }
