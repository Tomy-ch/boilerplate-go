   174  // toSearchResults は、ユーザーエンティティ列を UserSearchResult の DTO 列へ変換します。
   175  // いずれかのユーザーが参照する都道府県を解決できない場合は参照整合性破れ（errOrphanPrefecture）を返します。
   176  func (u *usecase) toSearchResults(ctx context.Context, us user.Users) (UserSearchResults, error) {
   177  	_, prefectureMap, err := observability.RunWithSpan(
   178  		ctx, u.tracer, "usecase", "user", "prefectureMap", func(ctx context.Context) (map[uuid.UUID]*prefecture.Prefecture, error) {
   179  			pids := make([]uuid.UUID, len(us))
   180  			for i, ue := range us {
   181  				pids[i] = ue.PrefectureID()
   182  			}
   183  
   184  			ps, pftErr := u.pftRepo.FindByIDs(ctx, pids)
   185  			if pftErr != nil {
   186  				return nil, pftErr
   187  			}
   188  
   189  			prefectureMap := make(map[uuid.UUID]*prefecture.Prefecture, len(ps))
   190  			for _, p := range ps {
   191  				prefectureMap[p.ID()] = p
   192  			}
   193  
   194  			return prefectureMap, nil
   195  		},
   196  	)
   197  	if err != nil {
   198  		return nil, err
   199  	}
   200  
   201  	results := make(UserSearchResults, len(us))
   202  	for i, ue := range us {
   203  		p, ok := prefectureMap[ue.PrefectureID()]
   204  		if !ok {
   205  			return nil, errOrphanPrefecture
   206  		}
   207  		results[i] = toSearchResult(ue, p.Name())
   208  	}
   209  
   210  	return results, nil
   211  }
