   109  // Usecase は、ユーザーに関するユースケースを定義します。
   110  type Usecase interface {
   111  	// ListUsers は、認可を確認したうえでユーザー一覧を取得します。他ユーザーを列挙する操作のため
   112  	// admin ロールを持つ主体のみ許可し、拒否された場合は authz.ErrForbidden を返します。
   113  	ListUsers(ctx context.Context, authn *authbd.Authn, active *bool, page *paging.Page) ([]UserView, error)
   114  	// ListUsersWithTotal は、認可を確認したうえでユーザー一覧と総件数をまとめて取得します。
   115  	// 認可条件は ListUsers と同じく admin 限定です。
   116  	ListUsersWithTotal(ctx context.Context, authn *authbd.Authn, active *bool, page *paging.Page) (*UserListView, error)
   117  	// ListUsersFeed は、認可を確認したうえで未削除ユーザーを作成日時の降順（cursor ページネーション）で
   118  	// 取得します。認可条件は ListUsers と同じく admin 限定です。
   119  	ListUsersFeed(ctx context.Context, authn *authbd.Authn, cursor *paging.Cursor) (*UserFeedView, error)
   120  	// CreateUser は、ユーザーを作成します。
   121  	CreateUser(ctx context.Context, dto *CreateParamsDTO) (UserView, error)
   122  	// CountUsers は、ユーザーの総件数を返します。件数のみを返し個々のユーザーを開示しないため、
   123  	// 認可を要求しません。
   124  	CountUsers(ctx context.Context, active *bool) (int64, error)
   125  	// GetUser は、認可を確認したうえで ID から単一ユーザーを取得します。
   126  	// 認可が拒否された場合は authz.ErrForbidden（apperror.ErrPermissionDenied をラップ）を返します。
   127  	GetUser(ctx context.Context, authn *authbd.Authn, id uuid.UUID) (UserView, error)
   128  	// UpdateUser は、認可を確認したうえでユーザーのプロフィールを全更新します。
   129  	// 認可が拒否された場合は authz.ErrForbidden（apperror.ErrPermissionDenied をラップ）を返します。
   130  	UpdateUser(ctx context.Context, authn *authbd.Authn, id uuid.UUID, dto *UpdateProfileParams) (UserView, error)
   131  	// UpdateUserPartially は、認可を確認したうえでユーザーを部分更新します。指定されたフィールドのみを
   132  	// 反映し、未指定 / null は据え置きます（値のクリアは非対応で、クリアには全更新の UpdateUser を使います）。
   133  	// 認可が拒否された場合は authz.ErrForbidden（apperror.ErrPermissionDenied をラップ）を返します。
   134  	UpdateUserPartially(ctx context.Context, authn *authbd.Authn, id uuid.UUID, dto *PatchParamsDTO) (UserView, error)
   135  	// DeleteUser は、認可を確認したうえでユーザーを退会させます。ユーザーを論理削除し、

   138  	// 認可が拒否された場合は authz.ErrForbidden（apperror.ErrPermissionDenied をラップ）を返します。
   139  	DeleteUser(ctx context.Context, authn *authbd.Authn, id uuid.UUID) error

   246  func (u *usecase) ListUsersFeed(ctx context.Context, authn *authbd.Authn, cursor *paging.Cursor) (*UserFeedView, error) {
   247  	ctx, endSpan := u.tracer.Start(ctx)
   248  	defer endSpan()
   249  
   250  	if err := u.authorizeUserCollection(ctx, authn); err != nil {
   251  		return nil, err
   252  	}
   253  
   254  	if cursor == nil {
   255  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, "cursor must not be nil")
   256  	}
   257  
   258  	after, err := decodeFeedCursor(cursor)
   259  	if err != nil {
   260  		return nil, err
   261  	}
   262  
   263  	us, err := u.userRepo.FindFeed(ctx, after, cursor.Limit32()+1)
   264  	if err != nil {
   265  		return nil, err
   266  	}
   267  
   268  	limit := cursor.Limit()
   269  	hasNext := len(us) > limit
   270  	if hasNext {
   271  		us = us[:limit]
   272  	}
   273  
   274  	items, err := u.toUserViews(ctx, us)
   275  	if err != nil {
   276  		return nil, err
   277  	}
   278  
   279  	var nextCursor *string
   280  	if hasNext && len(us) > 0 {
   281  		encoded := encodeFeedCursor(us[len(us)-1])
   282  		nextCursor = &encoded
   283  	}
   284  
   285  	return &UserFeedView{Items: items, NextCursor: nextCursor}, nil
   286  }

   418  func (u *usecase) DeleteUser(ctx context.Context, authn *authbd.Authn, id uuid.UUID) error {
   419  	ctx, endSpan := u.tracer.Start(ctx)
   420  	defer endSpan()
   421  
   422  	if err := u.authorizeUserAccess(ctx, authn, authz.ActionUserDelete, id); err != nil {
   423  		return err
   424  	}
   425  
   426  	now := u.clock.Now()
   427  	// 論理削除と退会イベントの発行を単一 tx にまとめ、退会だけが成立してイベントが失われることを防ぐ。
   428  	// 退会を拒む条件（進行中の購入）も同じ tx で判定し、拒否時は論理削除もイベントも残さない。
   429  	return u.txm.Do(ctx, func(ctx context.Context) error {
   430  		// 進行中購入の判定より前に排他ロックを取ることが、購入作成（共有ロック）との直列化の成立条件。
   431  		// 判定より後だと「判定通過 → 購入の成立 → 退会の確定」の順序を止められない。
   432  		userEntity, err := u.userLock.LockByID(ctx, id)
   433  		if err != nil {
   434  			return err
   435  		}
   436  
   437  		statuses, err := u.purchaseRepo.FindStatusesByUserID(ctx, id)
   438  		if err != nil {
   439  			return err
   440  		}
   441  		if err = membership.EnsureWithdrawable(userEntity, statuses); err != nil {
   442  			return err
   443  		}
   444  
   445  		if err = userEntity.MarkAsDeleted(now); err != nil {
   446  			return err
   447  		}
   448  		if err = u.userRepo.Update(ctx, userEntity); err != nil {
   449  			return err
   450  		}
   451  
   452  		payload, err := event.BuildWithdrawn(userEntity)
   453  		if err != nil {
   454  			return err
   455  		}
   456  		_, err = u.emit.Emit(ctx, outbox.EmitInput{
   457  			AggregateType: aggregateType,
   458  			AggregateID:   id.String(),
   459  			EventType:     event.TypeWithdrawn,
   460  			Payload:       payload,
   461  			Channel:       outboxbndry.ChannelHTTP,
   462  		})
   463  		return err
   464  	})
   465  }

   504  func (u *usecase) toUserViews(ctx context.Context, us user.Users) ([]UserView, error) {
   505  	_, prefectureMap, err := observability.RunWithSpan(
   506  		ctx, u.tracer, "usecase", "user", "prefectureMap", func(ctx context.Context) (map[uuid.UUID]*prefecture.Prefecture, error) {
   507  			pids := make([]uuid.UUID, len(us))
   508  			for i, ue := range us {
   509  				pids[i] = ue.PrefectureID()
   510  			}
   511

   513  			if pftErr != nil {

   515  			}
   516  
   517  			prefectureMap := make(map[uuid.UUID]*prefecture.Prefecture, len(ps))
   518  			for _, p := range ps {
   519  				prefectureMap[p.ID()] = p
   520  			}
   521  
   522  			return prefectureMap, nil
   523  		},
   524  	)
   525  	if err != nil {
   526  		return nil, err
   527  	}
   528  
   529  	dtos := make([]UserView, len(us))
   530  	for i, ue := range us {
   531  		p, ok := prefectureMap[ue.PrefectureID()]
   532  		if !ok {
   533  			return nil, errOrphanPrefecture
   534  		}
   535  		dtos[i] = toUserView(ue, p.Name())
