   117  // Usecase は、ユーザーに関するユースケースを定義します。
   118  type Usecase interface {
   119  	// ListUsers は、認可を確認したうえでユーザー一覧を取得します。他ユーザーを列挙する操作のため
   120  	// admin ロールを持つ主体のみ許可し、拒否された場合は authz.ErrForbidden を返します。
   121  	ListUsers(ctx context.Context, authn *authbd.Authn, active *bool, page *paging.Page) ([]UserView, error)
   122  	// ListUsersWithTotal は、認可を確認したうえでユーザー一覧と総件数をまとめて取得します。
   123  	// 認可条件は ListUsers と同じく admin 限定です。
   124  	ListUsersWithTotal(ctx context.Context, authn *authbd.Authn, active *bool, page *paging.Page) (*UserListView, error)
   125  	// ListUsersFeed は、認可を確認したうえで未削除ユーザーを作成日時の降順（cursor ページネーション）で
   126  	// 取得します。認可条件は ListUsers と同じく admin 限定です。
   127  	ListUsersFeed(ctx context.Context, authn *authbd.Authn, cursor *paging.Cursor) (*UserFeedView, error)
   128  	// CreateUser は、認証済みの主体を内部ユーザーとして登録します。ユーザーの作成・外部アイデンティティの
   129  	// 結び付け・ウェルカムクーポンの発行を単一トランザクションで行います。
   130  	// 既に登録済みの主体からの呼び出しは apperror.ErrConflict を返し、何も残しません。
   131  	CreateUser(ctx context.Context, dto *CreateParamsDTO) (UserView, error)
   132  	// CountUsers は、ユーザーの総件数を返します。件数のみを返し個々のユーザーを開示しないため、
   133  	// 認可を要求しません。
   134  	CountUsers(ctx context.Context, active *bool) (int64, error)
   135  	// GetUser は、認可を確認したうえで ID から単一ユーザーを取得します。
   136  	// 認可が拒否された場合は authz.ErrForbidden（apperror.ErrPermissionDenied をラップ）を返します。
   137  	GetUser(ctx context.Context, authn *authbd.Authn, id uuid.UUID) (UserView, error)
   138  	// UpdateUser は、認可を確認したうえでユーザーのプロフィールを全更新します。
   139  	// 認可が拒否された場合は authz.ErrForbidden（apperror.ErrPermissionDenied をラップ）を返します。
   140  	UpdateUser(ctx context.Context, authn *authbd.Authn, id uuid.UUID, dto *UpdateProfileParams) (UserView, error)
   141  	// UpdateUserPartially は、認可を確認したうえでユーザーを部分更新します。指定されたフィールドのみを
   142  	// 反映し、未指定 / null は据え置きます（値のクリアは非対応で、クリアには全更新の UpdateUser を使います）。
   143  	// 認可が拒否された場合は authz.ErrForbidden（apperror.ErrPermissionDenied をラップ）を返します。
   144  	UpdateUserPartially(ctx context.Context, authn *authbd.Authn, id uuid.UUID, dto *PatchParamsDTO) (UserView, error)
   145  	// DeleteUser は、認可を確認したうえでユーザーを退会させます。ユーザーを論理削除し、
   146  	// 同一トランザクションで退会イベントを発行します（配信は非同期・結果整合。ADR-0054 (transactional-outbox)）。
   147  	// 進行中の購入が残っている場合は apperror.ErrConflict を返し、退会させません。
   148  	// 認可が拒否された場合は authz.ErrForbidden（apperror.ErrPermissionDenied をラップ）を返します。
   149  	DeleteUser(ctx context.Context, authn *authbd.Authn, id uuid.UUID) error
   150  }

   192  func (u *usecase) CreateUser(ctx context.Context, dto *CreateParamsDTO) (UserView, error) {
   193  	ctx, endSpan := u.tracer.Start(ctx)
   194  	defer endSpan()
   195  
   196  	now := u.clock.Now()
   197  
   198  	userID, err := uuid.New()
   199  	if err != nil {
   200  		return UserView{}, xerrors.Wrap(err, "failed to generate user id")
   201  	}
   202  
   203  	var (
   204  		userEntity *user.User
   205  		pftName    string
   206  	)
   207  	err = u.txm.Do(ctx, func(ctx context.Context) error {
   208  		pftDomain, err := u.pftRepo.FindByName(ctx, dto.PrefectureName)
   209  		if err != nil {
   210  			return err
   211  		}
   212  		pftName = pftDomain.Name()
   213  
   214  		userEntity, err = user.New(userID, user.Attributes{
   215  			FirstName:    dto.FirstName,
   216  			LastName:     dto.LastName,
   217  			Email:        dto.Email,
   218  			Phone:        dto.Phone,
   219  			PrefectureID: pftDomain.ID(),
   220  			City:         dto.City,
   221  			Street:       dto.Street,
   222  			Building:     dto.Building,
   223  			PostalCode:   dto.PostalCode,
   224  			CreatedAt:    now,
   225  			UpdatedAt:    now,
   226  		})
   227  		if err != nil {
   228  			return err
   229  		}
   230  
   231  		if err = u.userRepo.Create(ctx, userEntity); err != nil {
   232  			return err
   233  		}
   234  
   235  		if err = u.identities.Register(ctx, userID, dto.Issuer, dto.Subject); err != nil {
   236  			return err
   237  		}
   238  
   239  		welcome, err := newWelcomeCoupon(userID, now)
   240  		if err != nil {
   241  			return err
   242  		}
   243  
   244  		return u.couponRepo.Create(ctx, welcome)
   245  	})
   246  	if err != nil {
   247  		return UserView{}, err
   248  	}
   249  
   250  	return toUserView(userEntity, pftName), nil
   251  }

   278  func (u *usecase) ListUsersFeed(ctx context.Context, authn *authbd.Authn, cursor *paging.Cursor) (*UserFeedView, error) {
   279  	ctx, endSpan := u.tracer.Start(ctx)
   280  	defer endSpan()
   281  
   282  	if err := u.authorizeUserCollection(ctx, authn); err != nil {
   283  		return nil, err
   284  	}
   285  
   286  	if cursor == nil {
   287  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, "cursor must not be nil")
   288  	}
   289  
   290  	after, err := decodeFeedCursor(cursor)
   291  	if err != nil {
   292  		return nil, err
   293  	}
   294  
   295  	us, err := u.userRepo.FindFeed(ctx, after, cursor.Limit32()+1)
   296  	if err != nil {
   297  		return nil, err
   298  	}
   299  
   300  	limit := cursor.Limit()
   301  	hasNext := len(us) > limit
   302  	if hasNext {
   303  		us = us[:limit]
   304  	}
   305  
   306  	items, err := u.toUserViews(ctx, us)
   307  	if err != nil {
   308  		return nil, err
   309  	}
   310  
   311  	var nextCursor *string
   312  	if hasNext && len(us) > 0 {
   313  		encoded := encodeFeedCursor(us[len(us)-1])
   314  		nextCursor = &encoded
   315  	}
   316  
   317  	return &UserFeedView{Items: items, NextCursor: nextCursor}, nil
   318  }

   450  func (u *usecase) DeleteUser(ctx context.Context, authn *authbd.Authn, id uuid.UUID) error {
   451  	ctx, endSpan := u.tracer.Start(ctx)
   452  	defer endSpan()
   453  
   454  	if err := u.authorizeUserAccess(ctx, authn, authz.ActionUserDelete, id); err != nil {
   455  		return err
   456  	}
   457  
   458  	now := u.clock.Now()
   459  	// 論理削除と退会イベントの発行を単一 tx にまとめ、退会だけが成立してイベントが失われることを防ぐ。
   460  	// 退会を拒む条件（進行中の購入）も同じ tx で判定し、拒否時は論理削除もイベントも残さない。
   461  	return u.txm.Do(ctx, func(ctx context.Context) error {
   462  		// 進行中購入の判定より前に排他ロックを取ることが、購入作成（共有ロック）との直列化の成立条件。
   463  		// 判定より後だと「判定通過 → 購入の成立 → 退会の確定」の順序を止められない。
   464  		userEntity, err := u.userLock.LockByID(ctx, id)
   465  		if err != nil {
   466  			return err
   467  		}
   468  
   469  		statuses, err := u.purchaseRepo.FindStatusesByUserID(ctx, id)
   470  		if err != nil {
   471  			return err
   472  		}
   473  		if err = membership.EnsureWithdrawable(userEntity, statuses); err != nil {
   474  			return err
   475  		}
   476  
   477  		if err = userEntity.MarkAsDeleted(now); err != nil {
   478  			return err
   479  		}
   480  		if err = u.userRepo.Update(ctx, userEntity); err != nil {
   481  			return err
   482  		}
   483  
   484  		payload, err := event.BuildWithdrawn(userEntity)
   485  		if err != nil {
   486  			return err
   487  		}
   488  		_, err = u.emit.Emit(ctx, outbox.EmitInput{
   489  			AggregateType: aggregateType,
   490  			AggregateID:   id.String(),
   491  			EventType:     event.TypeWithdrawn,
   492  			Payload:       payload,

   534  // toUserViews は、ユーザーエンティティ列を UserView の DTO 列へ変換します。
   535  // いずれかのユーザーが参照する都道府県を解決できない場合は参照整合性破れ（errOrphanPrefecture）を返します。
   536  func (u *usecase) toUserViews(ctx context.Context, us user.Users) ([]UserView, error) {
   537  	_, prefectureMap, err := observability.RunWithSpan(
   538  		ctx, u.tracer, "usecase", "user", "prefectureMap", func(ctx context.Context) (map[uuid.UUID]*prefecture.Prefecture, error) {
   539  			pids := make([]uuid.UUID, len(us))
   540  			for i, ue := range us {
   541  				pids[i] = ue.PrefectureID()
   542  			}
   543  
   544  			ps, pftErr := u.pftRepo.FindByIDs(ctx, pids)
   545  			if pftErr != nil {
   546  				return nil, pftErr
   547  			}
   548  
   549  			prefectureMap := make(map[uuid.UUID]*prefecture.Prefecture, len(ps))
   550  			for _, p := range ps {
   551  				prefectureMap[p.ID()] = p
   552  			}
   553  
   554  			return prefectureMap, nil
   555  		},
   556  	)
   557  	if err != nil {
   558  		return nil, err
   559  	}
   560  
   561  	dtos := make([]UserView, len(us))
   562  	for i, ue := range us {
   563  		p, ok := prefectureMap[ue.PrefectureID()]
   564  		if !ok {
   565  			return nil, errOrphanPrefecture
   566  		}
   567  		dtos[i] = toUserView(ue, p.Name())
   568  	}
   569  
   570  	return dtos, nil

   575  func (u *usecase) resolvePatchPrefecture(ctx context.Context, name *string, currentID uuid.UUID) (*prefecture.Prefecture, error) {
