    31  // IssueCoupon は、admin が受給者を名指ししてクーポンを 1 枚発行します。
    32  //
    33  // Repository への書き込みで足りる理由は docs/spec/usecase/coupon.md の Workflow — IssueCoupon を参照。
    34  func (u *usecase) IssueCoupon(
    35  	ctx context.Context,
    36  	authn *auth.Authn,
    37  	params IssueCouponParams,
    38  ) (CouponView, error) {
    39  	ctx, endSpan := u.tracer.Start(ctx)
    40  	defer endSpan()
    41  
    42  	if err := u.authorizer.Authorize(
    43  		ctx, authn, authz.ActionCouponIssue, authz.NewResource("coupon", nil),
    44  	); err != nil {
    45  		return CouponView{}, err
    46  	}
    47  
    48  	discount, err := newDiscount(params.DiscountKind, params.DiscountValue)
    49  	if err != nil {
    50  		return CouponView{}, err
    51  	}
    52  
    53  	scope, err := newScope(params.ScopeKind, params.ScopeTargetID)
    54  	if err != nil {
    55  		return CouponView{}, err
    56  	}
    57  
    58  	id, err := uuid.New()
    59  	if err != nil {
    60  		return CouponView{}, xerrors.Wrap(err, "failed to generate coupon id")
    61  	}
    62  
    63  	now := u.clock.Now()
    64  
    65  	var issued *coupon.Coupon
    66  	err = u.txm.Do(ctx, func(ctx context.Context) error {
    67  		// 存在確認をトランザクションの外へ出さないこと。理由は docs/spec/usecase/coupon.md の
    68  		// Workflow — IssueCoupon の invariants を参照。
    69  		if serr := u.ensureScopeTargetExists(ctx, scope); serr != nil {
    70  			return serr
    71  		}
    72  
    73  		// 退会済みの利用者へは発行しません。FindByID は在籍者だけを返すため、
    74  		// 不存在と退会済みは同じ NotFound になります。
    75  		if _, uerr := u.userRepo.FindByID(ctx, params.UserID); uerr != nil {
    76  			return uerr
    77  		}
    78  
    79  		c, cerr := coupon.New(id, coupon.Attributes{
    80  			UserID:    params.UserID,
    81  			Discount:  discount,
    82  			Scope:     scope,
    83  			ExpiresAt: params.ExpiresAt,
    84  			IssuedAt:  now,
    85  		})
    86  		if cerr != nil {
    87  			return cerr
    88  		}
    89  
    90  		if cerr = u.couponRepo.Create(ctx, c); cerr != nil {
    91  			return cerr
    92  		}
    93  		issued = c
    94  
    95  		return nil
    96  	})
    97  	if err != nil {
    98  		return CouponView{}, err
    99  	}
   100  
   101  	return toCouponView(issued), nil
   102  }
