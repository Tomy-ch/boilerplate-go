    66  // IssuePromotionalCoupons は、admin が退会していないすべての利用者へクーポンを一括発行し、
    67  // 何が起きたかを件数で返します。
    68  //
    69  // 受給者は述語でしか決まらないため、書き込みは [command.CommandService.IssuePromotionalCoupons]
    70  // が担います。
    71  func (u *usecase) IssuePromotionalCoupons(
    72  	ctx context.Context,
    73  	authn *auth.Authn,
    74  	params IssuePromotionalCouponsParams,
    75  ) (IssuePromotionalCouponsView, error) {
    76  	ctx, endSpan := u.tracer.Start(ctx)
    77  	defer endSpan()
    78  
    79  	if err := u.authorizer.Authorize(
    80  		ctx, authn, authz.ActionCouponBulkIssue, authz.NewResource("coupon", nil),
    81  	); err != nil {
    82  		return IssuePromotionalCouponsView{}, err
    83  	}
    84  
    85  	discount, err := newDiscount(params.DiscountKind, params.DiscountValue)
    86  	if err != nil {
    87  		return IssuePromotionalCouponsView{}, err
    88  	}
    89  	if err = ensureFlatDiscountWithinCap(discount); err != nil {
    90  		return IssuePromotionalCouponsView{}, err
    91  	}
    92  
    93  	scope, err := newScope(params.ScopeKind, params.ScopeTargetID)
    94  	if err != nil {
    95  		return IssuePromotionalCouponsView{}, err
    96  	}
    97  
    98  	now := u.clock.Now()
    99  
   100  	var view IssuePromotionalCouponsView
   101  	err = u.txm.Do(ctx, func(ctx context.Context) error {
   102  		// 対象の存在確認は書き込みと同じトランザクションで行います。Idempotency-Key の有無で
   103  		// idempotency.Run がトランザクションを開くかどうかが変わるため、外に置くと境界が要求次第になります。
   104  		if serr := u.ensureScopeTargetExists(ctx, scope); serr != nil {
   105  			return serr
   106  		}
   107  
   108  		// 上限は書き込みの前に判定します（分類は docs/spec/usecase/coupon.md の
   109  		// Workflow — IssuePromotionalCoupons の invariants を参照）。
   110  		recipients, cerr := u.userRepo.CountByActive(ctx, ptr.To(true))
   111  		if cerr != nil {
   112  			return cerr
   113  		}
   114  		if recipients > maxPromotionRecipients {
   115  			return ErrTooManyRecipients
   116  		}
   117  
   118  		result, ierr := u.bulkIssueCmd.IssuePromotionalCoupons(ctx, command.IssuePromotionalCouponsParams{
   119  			Scope:     scope,
   120  			Discount:  discount,
   121  			ExpiresAt: params.ExpiresAt,
   122  			IssuedAt:  now,
   123  		})
   124  		if ierr != nil {
   125  			return ierr
   126  		}
   127  
   128  		// 事前判定だけでは上限が破れるため、実際に書いた枚数で締めます（母集団の性質は
   129  		// docs/spec/usecase/coupon.md の Workflow — IssuePromotionalCoupons の invariants を参照）。
   130  		if result.IssuedCouponCount > maxPromotionRecipients {
   131  			return ErrTooManyRecipients
   132  		}
   133  
   134  		view = IssuePromotionalCouponsView{
   135  			IssuedAt:          now,
   136  			ExpiresAt:         params.ExpiresAt,
   137  			RecipientCount:    result.RecipientCount,
   138  			IssuedCouponCount: result.IssuedCouponCount,
   139  		}
   140  
   141  		return nil
   142  	})
   143  	if err != nil {
   144  		return IssuePromotionalCouponsView{}, err
   145  	}
   146  
   147  	return view, nil
   148  }

   161  // ensureFlatDiscountWithinCap は、定額値引きが一括発行の上限に収まることを確かめます。
   162  //
   163  // 定率はドメインが 1 を上限に持つため、際限なく大きくなり得るのは定額だけです。
   164  func ensureFlatDiscountWithinCap(discount coupon.Discount) error {
   165  	if discount.Kind() != coupon.DiscountKindFlat {
   166  		return nil
   167  	}
   168  	if discount.Value().Cmp(maxPromotionFlatDiscount) > 0 {
   169  		return ErrFlatDiscountTooLarge
   170  	}
   171  
   172  	return nil
   173  }

   186  // ensureScopeTargetExists は、適用範囲が指す対象の存在を確かめます。
   187  //
   188  // 存在しない対象を範囲にしたクーポンは誰にも使えません。発行は取り消せないため、配ってから
   189  // 気づくことを避けます。全体の適用範囲は対象を持たないので何も確認しません。
   190  func (u *usecase) ensureScopeTargetExists(ctx context.Context, scope coupon.Scope) error {
   191  	targetID := scope.TargetID()
   192  	if targetID == nil {
   193  		return nil
   194  	}
   195  
   196  	switch scope.Kind() {
   197  	case coupon.ScopeKindCategory:
   198  		_, err := u.categoryRepo.FindByID(ctx, *targetID)
   199  
   200  		return err
   201  	case coupon.ScopeKindProduct:
   202  		_, err := u.productRepo.FindByID(ctx, *targetID)
   203  
   204  		return err
   205  	default:
   206  		return nil
   207  	}
   208  }
