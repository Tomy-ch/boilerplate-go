   113  // ListApplicableToMyCart は、カートの明細を対象にクーポンごとの値引き額を求め、0 になるものを落とします。
   114  // 対象明細の絞り込みは docs/spec/usecase/coupon.md の Workflow を参照。
   115  func (u *usecase) ListApplicableToMyCart(ctx context.Context, authn *auth.Authn) ([]CartCouponView, error) {
   116  	ctx, endSpan := u.tracer.Start(ctx)
   117  	defer endSpan()
   118  
   119  	userID, err := requireUserID(authn)
   120  	if err != nil {
   121  		return nil, err
   122  	}
   123  
   124  	coupons, err := u.couponRepo.FindByUserID(ctx, userID)
   125  	if err != nil {
   126  		return nil, err
   127  	}
   128  	if len(coupons) == 0 {
   129  		return []CartCouponView{}, nil
   130  	}
   131  
   132  	now := u.clock.Now()
   133  
   134  	lines, err := u.buildLines(ctx, userID, now)
   135  	if err != nil {
   136  		return nil, err
   137  	}
   138  	if len(lines) == 0 {
   139  		return []CartCouponView{}, nil
   140  	}
   141  
   142  	views := make([]CartCouponView, 0, len(coupons))
   143  	for _, c := range coupons {
   144  		if c.IsUsed() || c.IsExpired(now) {
   145  			continue
   146  		}
   147  
   148  		amount, derr := c.DiscountFor(lines)
   149  		if derr != nil {
   150  			return nil, derr
   151  		}
   152  		if amount <= 0 {
   153  			continue
   154  		}
   155  
   156  		views = append(views, CartCouponView{Coupon: toCouponView(c), DiscountAmount: amount})
   157  	}
   158  
   159  	return views, nil
   160  }

   162  // buildLines は、カートの明細のうち購入できるものだけをクーポンの対象明細へ写します。
   163  // カートを持たない場合と期限切れの場合は、いずれも空を返します。
   164  //
   165  // Repository は期限切れのカートも返し、期限の判定は Cart.IsExpired が持ちます
   166  // （internal/domain/cart/cart_repository.go）。期限切れのカートは購入へ進めないため、
   167  // カート表示（internal/usecase/cart）と同じく「表示するカートが無い」に畳みます。
   168  func (u *usecase) buildLines(
   169  	ctx context.Context, userID uuid.UUID, now time.Time,
   170  ) ([]coupon.Line, error) {
   171  	c, err := u.cartRepo.FindByOwnerID(ctx, userID)
   172  	if err != nil {
   173  		if xerrors.Is(err, apperror.ErrNotFound) {
   174  			return nil, nil
   175  		}
   176  		return nil, err
   177  	}
   178  	if c.IsExpired(now) {
   179  		return nil, nil
   180  	}
   181  
   182  	items := c.Items()
   183  	if len(items) == 0 {
   184  		return nil, nil
   185  	}
   186  
   187  	ids := make([]uuid.UUID, len(items))
   188  	for i, item := range items {
   189  		ids[i] = item.ProductID()
   190  	}
   191  	products, err := u.productRepo.FindByIDs(ctx, ids)
   192  	if err != nil {
   193  		return nil, err
   194  	}
   195  
   196  	byID := make(map[uuid.UUID]*product.Product, len(products))
   197  	for _, p := range products {
   198  		byID[p.ID()] = p
   199  	}
   200  
   201  	lines := make([]coupon.Line, 0, len(items))
   202  	for _, item := range items {
   203  		p, ok := byID[item.ProductID()]
   204  		if !ok {
   205  			continue
   206  		}
   207  
   208  		snapshot := cart.NewProductSnapshot(cart.ProductSnapshotAttributes{
   209  			Quantity:     p.Quantity(),
   210  			Price:        p.Price(),
   211  			Published:    p.IsPublished(),
   212  			Discontinued: p.IsDiscontinued(),
   213  		})
   214  		if len(item.Evaluate(&snapshot).Issues()) > 0 {
   215  			continue
   216  		}
   217  
   218  		lines = append(lines, coupon.NewLine(coupon.LineAttributes{
   219  			ProductID:  p.ID(),
   220  			CategoryID: p.Category().ID(),
   221  			Subtotal:   p.Price().Decimal().Mul(decimal.FromInt(int64(item.Quantity()))),
   222  		}))
   223  	}
   224  
   225  	return lines, nil
   226  }

   228  // toCouponView は、クーポン集約を出力 DTO の語彙へ写します。種別は code ではなく名前で出します。
   229  func toCouponView(c *coupon.Coupon) CouponView {
   230  	return CouponView{
   231  		ID:            c.ID(),
   232  		DiscountKind:  c.Discount().Kind().Name(),
   233  		DiscountValue: c.Discount().Value(),
   234  		ScopeKind:     c.Scope().Kind().Name(),
   235  		ScopeTargetID: c.Scope().TargetID(),
   236  		ExpiresAt:     c.ExpiresAt(),
   237  		UsedAt:        c.UsedAt(),
   238  		IssuedAt:      c.IssuedAt(),
   239  	}
   240  }
