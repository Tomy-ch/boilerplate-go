   142  // ListApplicableToMyCart は、カートの明細を対象にクーポンごとの値引き額を求め、0 になるものを落とします。
   143  // 対象明細の絞り込みは docs/spec/usecase/coupon.md の Workflow を参照。
   144  func (u *usecase) ListApplicableToMyCart(ctx context.Context, authn *auth.Authn) ([]CartCouponView, error) {
   145  	ctx, endSpan := u.tracer.Start(ctx)
   146  	defer endSpan()
   147  
   148  	userID, err := requireUserID(authn)
   149  	if err != nil {
   150  		return nil, err
   151  	}
   152  
   153  	coupons, err := u.couponRepo.FindByUserID(ctx, userID)
   154  	if err != nil {
   155  		return nil, err
   156  	}
   157  	if len(coupons) == 0 {
   158  		return []CartCouponView{}, nil
   159  	}
   160  
   161  	now := u.clock.Now()
   162  
   163  	lines, err := u.buildLines(ctx, userID, now)
   164  	if err != nil {
   165  		return nil, err
   166  	}
   167  	if len(lines) == 0 {
   168  		return []CartCouponView{}, nil
   169  	}
   170  
   171  	views := make([]CartCouponView, 0, len(coupons))
   172  	for _, c := range coupons {
   173  		if c.IsUsed() || c.IsExpired(now) {
   174  			continue
   175  		}
   176  
   177  		amount, derr := c.DiscountFor(lines)
   178  		if derr != nil {
   179  			return nil, derr
   180  		}
   181  		if amount <= 0 {
   182  			continue
   183  		}
   184  
   185  		views = append(views, CartCouponView{Coupon: toCouponView(c), DiscountAmount: amount})
   186  	}
   187  
   188  	return views, nil
   189  }

   191  // buildLines は、カートの明細のうち購入できるものだけをクーポンの対象明細へ写します。
   192  // カートを持たない場合と期限切れの場合は、いずれも空を返します。
   193  //
   194  // Repository は期限切れのカートも返し、期限の判定は Cart.IsExpired が持ちます
   195  // （internal/domain/cart/cart_repository.go）。期限切れのカートは購入へ進めないため、
   196  // カート表示（internal/usecase/cart）と同じく「表示するカートが無い」に畳みます。
   197  func (u *usecase) buildLines(
   198  	ctx context.Context, userID uuid.UUID, now time.Time,
   199  ) ([]coupon.Line, error) {
   200  	c, err := u.cartRepo.FindByOwnerID(ctx, userID)
   201  	if err != nil {
   202  		if xerrors.Is(err, apperror.ErrNotFound) {
   203  			return nil, nil
   204  		}
   205  		return nil, err
   206  	}
   207  	if c.IsExpired(now) {
   208  		return nil, nil
   209  	}
   210  
   211  	items := c.Items()
   212  	if len(items) == 0 {
   213  		return nil, nil
   214  	}
   215  
   216  	ids := make([]uuid.UUID, len(items))
   217  	for i, item := range items {
   218  		ids[i] = item.ProductID()
   219  	}
   220  	products, err := u.productRepo.FindByIDs(ctx, ids)
   221  	if err != nil {
   222  		return nil, err
   223  	}
   224  
   225  	byID := make(map[uuid.UUID]*product.Product, len(products))
   226  	for _, p := range products {
   227  		byID[p.ID()] = p
   228  	}
   229  
   230  	lines := make([]coupon.Line, 0, len(items))
   231  	for _, item := range items {
   232  		p, ok := byID[item.ProductID()]
   233  		if !ok {
   234  			continue
   235  		}
   236  
   237  		snapshot := cart.NewProductSnapshot(cart.ProductSnapshotAttributes{
   238  			Quantity:     p.Quantity(),
   239  			Price:        p.Price(),
   240  			Published:    p.IsPublished(),
   241  			Discontinued: p.IsDiscontinued(),
   242  		})
   243  		if len(item.Evaluate(&snapshot).Issues()) > 0 {
   244  			continue
   245  		}
   246  
   247  		lines = append(lines, coupon.NewLine(coupon.LineAttributes{
   248  			ProductID:  p.ID(),
   249  			CategoryID: p.Category().ID(),
   250  			Subtotal:   p.Price().Decimal().Mul(decimal.FromInt(int64(item.Quantity()))),
   251  		}))
   252  	}
   253  
   254  	return lines, nil
   255  }
