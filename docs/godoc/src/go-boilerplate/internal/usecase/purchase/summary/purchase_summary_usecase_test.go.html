   351  func Test_toGroups(t *testing.T) {
   352  	t.Parallel()
   353  
   354  	laptopID := uuidtestkit.NewTestFromSalt(t, "tg_laptop")
   355  	tabletID := uuidtestkit.NewTestFromSalt(t, "tg_tablet")
   356  	bookID := uuidtestkit.NewTestFromSalt(t, "tg_book")
   357  
   358  	rows := func(t *testing.T) []query.PurchaseItemSummaryReadModel {
   359  		t.Helper()
   360  		return []query.PurchaseItemSummaryReadModel{
   361  			{CategoryName: "電子機器", ProductID: laptopID, ProductName: "ノートPC", ItemsTotal: dec(t, "980.00")},
   362  			{CategoryName: "電子機器", ProductID: tabletID, ProductName: "タブレット", ItemsTotal: dec(t, "300.25")},
   363  			{CategoryName: "書籍", ProductID: bookID, ProductName: "技術書", ItemsTotal: dec(t, "20.50")},
   364  		}
   365  	}
   366  
   367  	t.Run("正常系", func(t *testing.T) {
   368  		t.Parallel()
   369  
   370  		t.Run("カテゴリ単位では名称をキーに同カテゴリの商品を合算する", func(t *testing.T) {
   371  			t.Parallel()
   372  
   373  			total, groups := toGroups(rows(t), []GroupKind{GroupByCategory})
   374  
   375  			assert.Equal(t, "1300.75", total.String())
   376  			require.Len(t, groups, 2)
   377  			assert.Equal(t, "電子機器", groups["電子機器"].Name)
   378  			assert.Equal(t, "1280.25", groups["電子機器"].ItemsTotal.String())
   379  			assert.Equal(t, "20.5", groups["書籍"].ItemsTotal.String())
   380  			// 最下位の階層は下位グループを持たない。
   381  			assert.Nil(t, groups["電子機器"].Groups)
   382  		})
   383  
   384  		t.Run("指定順がネストの階層順になり親は子の総和を持つ", func(t *testing.T) {
   385  			t.Parallel()
   386  
   387  			total, groups := toGroups(rows(t), []GroupKind{GroupByCategory, GroupByProduct})
   388  
   389  			electronics := groups["電子機器"]
   390  			assert.Equal(t, "1280.25", electronics.ItemsTotal.String())
   391  			require.Len(t, electronics.Groups, 2)
   392  			assert.Equal(t, "980", electronics.Groups[laptopID.String()].ItemsTotal.String())
   393  			assert.Equal(t, "300.25", electronics.Groups[tabletID.String()].ItemsTotal.String())
   394  
   395  			// どの階層でも内訳の総和が親と一致する（丸めを挟まないため誤差は生じない）。
   396  			sum := electronics.Groups[laptopID.String()].ItemsTotal.Add(electronics.Groups[tabletID.String()].ItemsTotal)
   397  			assert.True(t, sum.Equal(electronics.ItemsTotal))
   398  			assert.True(t, electronics.ItemsTotal.Add(groups["書籍"].ItemsTotal).Equal(total))
   399  		})
