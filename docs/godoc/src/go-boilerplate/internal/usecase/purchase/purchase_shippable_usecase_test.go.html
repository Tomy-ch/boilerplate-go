    82  				},
    83  			),
    84  		},

    91  	}

    94  		attrs.ShippedAt = &shippedAt
    95  	}
    96  
    97  	p, err := domainpurchase.Reconstruct(uuidtestkit.NewTestFromSalt(t, salt), attrs)
