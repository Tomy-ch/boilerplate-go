    43  // BuildCreated は、購入集約から purchase.created.v1 の自己完結 snapshot payload を marshal します。

    45  	src := p.Details()

    47  	for i, d := range src {

    51  			UnitPrice: d.UnitPrice().String(),
    52  		}
    53  	}
    54

    56  	if id := p.CouponID(); id != nil {
    57  		couponID = ptr.To(id.String())

    59  
    60  	payload, err := json.Marshal(created{
    61  		PurchaseID:     p.ID().String(),
    62  		Code:           p.Code(),
    63  		UserID:         p.UserID().String(),
    64  		StatusCode:     p.StatusCode(),
    65  		SubtotalAmount: p.SubtotalAmount(),
    66  		DiscountAmount: p.DiscountAmount(),
    67  		TaxAmount:      p.TaxAmount(),
    68  		ShippingFee:    p.ShippingFee(),
    69  		TotalAmount:    p.TotalAmount(),
    70  		CouponID:       couponID,
    71  		Details:        details,
    72  	})
    73  	if err != nil {
    74  		return nil, xerrors.Wrap(err, "failed to encode purchase.created payload")
    75  	}
