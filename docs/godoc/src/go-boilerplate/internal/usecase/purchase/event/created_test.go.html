   128  			var decoded struct {
   129  				SubtotalAmount int     `json:"subtotalAmount"`
   130  				DiscountAmount int     `json:"discountAmount"`
   131  				TaxAmount      int     `json:"taxAmount"`
   132  				ShippingFee    int     `json:"shippingFee"`
   133  				TotalAmount    int     `json:"totalAmount"`
   134  				CouponID       *string `json:"couponId"`
   135  			}
   136  			require.NoError(t, json.Unmarshal(payload, &decoded))
   137  			require.NotNil(t, decoded.CouponID)
   138  			assert.Equal(t, couponID.String(), *decoded.CouponID)
   139  			// subtotal=160000 / discount=16000 / 課税基礎=144000 → tax=14400 / shipping=500 / total=158900
   140  			assert.Equal(t, 160000, decoded.SubtotalAmount)
   141  			assert.Equal(t, 16000, decoded.DiscountAmount)
   142  			assert.Equal(t, 14400, decoded.TaxAmount)

   144  			assert.Equal(t, 158900, decoded.TotalAmount)
   145  			// 購読側が snapshot だけで合計を組み立て直せることを固定する。

   148  			// しているため、規約を満たしている証拠としてここに置く。
   149  			assert.Equal(t,
   150  				decoded.TotalAmount,
   151  				decoded.SubtotalAmount-decoded.DiscountAmount+decoded.TaxAmount+decoded.ShippingFee,
