    48  	ctx, endSpan := u.tracer.Start(ctx)
    49  	defer endSpan()
    50  
    51  	if authn == nil {
    52  		return PurchaseGetDetailView{}, xerrors.Wrap(apperror.ErrUnauthenticated, "requires authenticated user")
    53  	}
    54  	userID, err := authn.UserID()
    55  	if err != nil {
    56  		return PurchaseGetDetailView{}, xerrors.Wrap(err, "failed to get user ID from authenticator")

    58  
    59  	rm, err := u.detailQS.FindDetailByUserAndCode(ctx, userID, purchaseCode)

    63  
    64  	return toPurchaseGetDetailView(rm), nil
