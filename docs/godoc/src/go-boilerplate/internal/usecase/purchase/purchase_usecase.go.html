   256  // detailProductIDs は、明細が参照する商品 ID を重複なく返します。
   257  func detailProductIDs(details []purchase.PurchaseDetail) []uuid.UUID {
   258  	seen := make(map[uuid.UUID]struct{}, len(details))
   259  	ids := make([]uuid.UUID, 0, len(details))
   260  	for _, d := range details {
   261  		if _, ok := seen[d.ProductID()]; ok {
   262  			continue
   263  		}
   264  		seen[d.ProductID()] = struct{}{}
   265  		ids = append(ids, d.ProductID())
   266  	}
   267  
   268  	return ids
   269  }

   271  // newPurchaseDraft は、購入・購入コード・各明細の ID を採番し、ドメイン入力と商品 ID 列を組み立てます。
   272  // 採番はトランザクションの外で行うため、リトライされても同じ ID が再利用されることはありません。
   273  func newPurchaseDraft(details []DetailParam) (*purchaseDraft, error) {
   274  	purchaseID, err := uuid.New()
   275  	if err != nil {
   276  		return nil, xerrors.Wrap(err, "failed to generate purchase id")
   277  	}
   278  	codeUUID, err := uuid.New()
   279  	if err != nil {
   280  		return nil, xerrors.Wrap(err, "failed to generate purchase code")
   281  	}
   282  
   283  	draft := &purchaseDraft{
   284  		purchaseID: purchaseID,
   285  		code:       codeUUID.String(),
   286  		inputs:     make([]purchase.DetailInput, len(details)),
   287  		productIDs: make([]uuid.UUID, len(details)),
   288  	}
   289  	for i, d := range details {
   290  		detailID, derr := uuid.New()
   291  		if derr != nil {
   292  			return nil, xerrors.Wrap(derr, "failed to generate purchase detail id")
   293  		}
   294  		draft.inputs[i] = purchase.DetailInput{ID: detailID, ProductID: d.ProductID, Quantity: d.Quantity}
   295  		draft.productIDs[i] = d.ProductID
   296  	}
   297  	return draft, nil
   298  }

   300  func (u *usecase) CreatePurchase(ctx context.Context, params CreatePurchaseParams) (PurchaseView, error) {
   301  	ctx, endSpan := u.tracer.Start(ctx)
   302  	defer endSpan()
   303  
   304  	draft, err := newPurchaseDraft(params.Details)
   305  	if err != nil {
   306  		return PurchaseView{}, err
   307  	}
   308  
   309  	now := u.clock.Now()
   310  
   311  	var (
   312  		created       *purchase.Purchase
   313  		appliedCoupon *coupon.Coupon
   314  	)
   315  	// nested で最外 tx に乗る（tx 所有については docs/spec/usecase/purchase.md 冒頭を参照）。
   316  	if txErr := u.txm.Do(ctx, func(ctx context.Context) error {
   317  		entity, redeemed, cerr := u.createPurchaseInTx(ctx, params, draft, now)
   318  		if cerr != nil {
   319  			return cerr
   320  		}
   321  		created, appliedCoupon = entity, redeemed
   322  
   323  		return nil
   324  	}); txErr != nil {
   325  		return PurchaseView{}, txErr
   326  	}
   327  
   328  	return toPurchaseView(created, appliedCoupon), nil
   329  }

   331  // applyRedeemedCoupon は、引き換えたクーポンの値引きを購入へ適用します。
   332  // 対象明細の組み立てと拒否の条件は docs/spec/usecase/purchase.md の § クーポンの適用（CreatePurchase）を参照。
   333  func applyRedeemedCoupon(entity *purchase.Purchase, redeemed *coupon.Coupon, products product.Products) error {
   334  	if redeemed == nil {
   335  		return nil
   336  	}
   337  
   338  	categories := make(map[uuid.UUID]uuid.UUID, len(products))
   339  	for _, p := range products {
   340  		categories[p.ID()] = p.Category().ID()
   341  	}
   342  
   343  	details := entity.Details()
   344  	lines := make([]coupon.Line, len(details))
   345  	for i, d := range details {
   346  		lines[i] = coupon.NewLine(coupon.LineAttributes{
   347  			ProductID:  d.ProductID(),
   348  			CategoryID: categories[d.ProductID()],
   349  			Subtotal:   d.LineTotal(),
   350  		})
   351  	}
   352  
   353  	amount, err := redeemed.DiscountFor(lines)
   354  	if err != nil {
   355  		return err
   356  	}
   357  
   358  	return entity.ApplyCoupon(redeemed.ID(), amount)
   359  }

   361  func (u *usecase) CancelPurchase(ctx context.Context, params CancelPurchaseParams) (CancelPurchaseView, error) {
   362  	ctx, endSpan := u.tracer.Start(ctx)
   363  	defer endSpan()
   364  
   365  	now := u.clock.Now()
   366  
   367  	var detail *purchase.Detail
   368  	// tx 境界は PayPurchase のコメントを参照。二重キャンセル対策は
   369  	// docs/spec/usecase/purchase.md § PATCH キャンセル を参照。
   370  	if txErr := u.txm.Do(ctx, func(ctx context.Context) error {
   371  		locked, lerr := u.repo.LockByCode(ctx, params.PurchaseCode)
   372  		if lerr != nil {
   373  			return lerr
   374  		}
   375  
   376  		// 存在秘匿のため NotFound へ畳む理由は CancelPurchase の doc コメントを参照。
   377  		if locked.UserID() != params.UserID {
   378  			return xerrors.Wrap(apperror.ErrNotFound, "purchase not found")
   379  		}
   380  
   381  		domainEvent, cerr := locked.Cancel(now)
   382  		if cerr != nil {
   383  			return cerr
   384  		}
   385  
   386  		if serr := u.restoreStock(ctx, locked.Details()); serr != nil {
   387  			return serr
   388  		}
   389  
   390  		if perr := u.repo.UpdateCancelled(ctx, locked); perr != nil {
   391  			return perr
   392  		}
   393  
   394  		payload, berr := event.BuildCanceled(locked)
   395  		if berr != nil {
   396  			return berr
   397  		}
   398  		eventType, terr := event.WireType(domainEvent.Type())
   399  		if terr != nil {
   400  			return terr
   401  		}
   402  		if _, eerr := u.emit.Emit(ctx, outbox.EmitInput{
   403  			AggregateType: aggregateType,
   404  			AggregateID:   locked.ID().String(),
   405  			EventType:     eventType,
   406  			Payload:       payload,
   407  			Channel:       outboxbndry.ChannelHTTP,
   408  		}); eerr != nil {
   409  			return eerr
   410  		}
   411  
   412  		reread, rerr := u.repo.FindDetailByID(ctx, locked.ID())
   413  		if rerr != nil {
   414  			return rerr
   415  		}
   416  		detail = reread
   417  		return nil
   418  	}); txErr != nil {
   419  		return CancelPurchaseView{}, txErr
   420  	}
   421  
   422  	return toCancelPurchaseView(detail), nil
   423  }

   425  func (u *usecase) PayPurchase(ctx context.Context, params PayPurchaseParams) (PayPurchaseView, error) {
   426  	ctx, endSpan := u.tracer.Start(ctx)
   427  	defer endSpan()
   428  
   429  	now := u.clock.Now()
   430  
   431  	var detail *purchase.Detail
   432  	// この Do が最外 tx（本経路は Idempotency-Key を配線しない）。
   433  	// 二重支払い対策は docs/spec/usecase/purchase.md § PATCH 支払い を参照。
   434  	if txErr := u.txm.Do(ctx, func(ctx context.Context) error {
   435  		locked, lerr := u.repo.LockByCode(ctx, params.PurchaseCode)
   436  		if lerr != nil {
   437  			return lerr
   438  		}
   439  
   440  		// 存在秘匿のため NotFound へ畳む理由は CancelPurchase の doc コメントを参照。
   441  		if locked.UserID() != params.UserID {
   442  			return xerrors.Wrap(apperror.ErrNotFound, "purchase not found")
   443  		}
   444  
   445  		domainEvent, perr := locked.Pay(now)
   446  		if perr != nil {
   447  			return perr
   448  		}
   449  
   450  		if perr := u.repo.UpdatePaid(ctx, locked); perr != nil {
   451  			return perr
   452  		}
   453  
   454  		payload, berr := event.BuildPaid(locked)
   455  		if berr != nil {
   456  			return berr
   457  		}
   458  		eventType, terr := event.WireType(domainEvent.Type())
   459  		if terr != nil {
   460  			return terr
   461  		}

   786  //
   787  // ロック順序（ユーザー行 → クーポン行 → 商品行、id 昇順）は docs/spec/usecase/purchase.md の
   788  // § クーポンの適用（CreatePurchase）を参照。書き込み後は Repository 経由で再検証します
   789  // （internal/usecase/README.md の Verifying infrastructure against the domain）。
   790  func (u *usecase) createPurchaseInTx(
   791  	ctx context.Context, params CreatePurchaseParams, draft *purchaseDraft, now time.Time,
   792  ) (*purchase.Purchase, *coupon.Coupon, error) {
   793  	if uerr := u.ensurePurchaserActive(ctx, params.UserID); uerr != nil {
   794  		return nil, nil, uerr
   795  	}
   796  
   797  	redeemed, rerr := u.redeemRequestedCoupon(ctx, params, now)
   798  	if rerr != nil {
   799  		return nil, nil, rerr
   800  	}
   801  
   802  	products, lerr := u.productRepo.LockByIDs(ctx, draft.productIDs)
   803  	if lerr != nil {
   804  		return nil, nil, lerr
   805  	}
   806  	locked := make([]purchase.LockedProduct, len(products))
   807  	for i, p := range products {
   808  		locked[i] = purchase.NewLockedProduct(p.ID(), p.Price(), p.Quantity())
   809  	}
   810  
   811  	entity, nerr := purchase.New(draft.purchaseID, draft.code, params.UserID, draft.inputs, locked)
   812  	if nerr != nil {
   813  		return nil, nil, nerr
   814  	}
   815  	if aerr := applyRedeemedCoupon(entity, redeemed, products); aerr != nil {
   816  		return nil, nil, aerr

   818  
   819  	// 充足（在庫超過が無いこと）は直前の purchase.New が locked と突き合わせて検証済み。
   820  	if serr := u.applyStockDelta(ctx, products, entity.Details(), -1); serr != nil {

   833  
   834  	reread, frerr := u.repo.FindByID(ctx, draft.purchaseID)

   911  // applyStockDelta は、明細の数量 × sign を対象商品の在庫へ適用して永続化します。
   912  // sign は購入時に -1、キャンセルによる復元時に 1 を取ります。products はロック済みである必要があり、
   913  // 在庫の範囲判定は商品集約の AdjustStock が行います。
   914  func (u *usecase) applyStockDelta(
   915  	ctx context.Context, products product.Products, details []purchase.PurchaseDetail, sign int,
   916  ) error {
   917  	byID := make(map[uuid.UUID]*product.Product, len(products))

   919  		byID[p.ID()] = p
   920  	}
   921

   924  		if !ok {
   925  			return xerrors.Wrap(apperror.ErrNotFound, "product not found for purchase detail")
   926  		}
   927  		if aerr := p.AdjustStock(sign * d.Quantity()); aerr != nil {
   928  			return aerr
   929  		}
   930  		if _, uerr := u.productRepo.UpdateStock(ctx, p); uerr != nil {
   931  			return uerr
   932  		}
   933  	}
