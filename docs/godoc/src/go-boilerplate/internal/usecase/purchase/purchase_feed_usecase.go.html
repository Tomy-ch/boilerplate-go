    66  // 対象が無い場合は QueryService が空一覧を返すため、他ユーザーの購入が混ざる経路は存在しません
    67  // （母集団の切り替え条件は Usecase.GetPurchases のインターフェース doc を参照）。

    69  // 注文日時の対象期間は params.Window をそのまま QueryService へ渡し、境界を持たない側には条件を付けません。

    71  	ctx context.Context, authn *auth.Authn, params ListPurchasesParams,
    72  ) (*PurchaseListView, error) {
    73  	if params.Cursor == nil {
    74  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, "cursor must not be nil")
    75  	}
    76  	if authn == nil {
    77  		return nil, apperror.ErrUnauthenticated
    78  	}
    79  
    80  	ctx, endSpan := u.tracer.Start(ctx)
    81  	defer endSpan()
    82  
    83  	cursor := params.Cursor
    84  	after, err := decodePurchaseCursor(cursor)
    85  	if err != nil {
    86  		return nil, err
    87  	}
    88  
    89  	feedParams := query.ListFeedParams{
    90  		Limit:       cursor.Limit32() + 1,
    91  		Window:      params.Window,
    92  		StatusCodes: params.StatusCodes,
    93  		ProductID:   params.ProductID,
    94  	}
    95  	if after != nil {
    96  		feedParams.AfterOrderedAt = &after.orderedAt
    97  		feedParams.AfterID = &after.id
    98  	}
    99  
   100  	feed, err := u.findFeedPage(ctx, authn, params.IncludeOtherUsers, feedParams)
   101  	if err != nil {
   102  		return nil, err
   103  	}
   104  
   105  	limit := cursor.Limit()
   106  	hasNext := len(feed) > limit
   107  	if hasNext {
   108  		feed = feed[:limit]
   109  	}
   110  
   111  	items := make([]PurchaseSummaryView, len(feed))
   112  	for i, f := range feed {
   113  		items[i] = PurchaseSummaryView{
   114  			Code:          f.Code,
   115  			TotalAmount:   f.TotalAmount,
   116  			StatusID:      f.StatusID,
   117  			StatusCode:    f.StatusCode,
   118  			StatusName:    f.StatusName,
   119  			FirstItemName: f.FirstItemName,
   120  			ItemCount:     f.ItemCount,
   121  			OrderedAt:     f.OrderedAt,
   122  		}
   123  	}
   124  
   125  	var nextCursor *string
   126  	if hasNext && len(feed) > 0 {
   127  		// len 判定は防御的な安全弁。hasNext は len > limit なので limit >= 1 の下では冗長だが、
   128  		// limit の下限保証は paging.NewCursor 依存であり、ゼロ値 Cursor 混入時の feed[-1] panic を防ぐ。

   130  		nextCursor = &encoded
   131  	}
   132  
   133  	return &PurchaseListView{Items: items, NextCursor: nextCursor}, nil
   134  }

   159  // decodePurchaseCursor は、cursor の不透明キー列を keyset 境界（purchaseCursor）へ解釈します。
   160  // 先頭ページ（カーソル無し）の場合は nil を返します。キーの個数・型が不正な場合は ErrInvalidArgument を返します。
   161  func decodePurchaseCursor(cursor *paging.Cursor) (*purchaseCursor, error) {
   162  	if !cursor.HasCursor() {
   163  		return nil, nil //nolint:nilnil // 先頭ページは境界なし（nil）を正常値として返す
   164  	}
   165  
   166  	keys := cursor.Keys()
   167  	if len(keys) != purchaseCursorKeyCount {
   168  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, fmt.Sprintf("invalid cursor: expected %d keys", purchaseCursorKeyCount))
   169  	}
   170  
   171  	orderedAt, err := time.Parse(time.RFC3339Nano, keys[0])
   172  	if err != nil {
   173  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, "invalid cursor: ordered_at is not RFC3339Nano")
   174  	}
   175  
   176  	id, err := uuid.Parse(keys[1])
   177  	if err != nil {
   178  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, "invalid cursor: id is not a valid UUID")
   179  	}
   180  
   181  	return &purchaseCursor{orderedAt: orderedAt, id: id}, nil
   182  }

   185  func encodePurchaseCursor(last query.PurchaseFeedReadModel) string {
   186  	return paging.EncodeCursor(last.OrderedAt.Format(time.RFC3339Nano), last.ID.String())
