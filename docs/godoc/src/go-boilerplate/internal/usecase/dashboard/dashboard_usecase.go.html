    75  func (u *usecase) GetDashboardSummary(
    76  	ctx context.Context, authn *auth.Authn, window timewindow.Window,
    77  ) (SummaryView, error) {
    78  	ctx, endSpan := u.tracer.Start(ctx)
    79  	defer endSpan()
    80  
    81  	if authn == nil {
    82  		return SummaryView{}, xerrors.Wrap(apperror.ErrUnauthenticated, "requires authenticated user")
    83  	}
    84  	// 所有者なしリソースとして認可する（docs/spec/usecase/dashboard.md 参照）。
    85  	if err := u.authorizer.Authorize(
    86  		ctx, authn, authz.ActionDashboardRead, authz.NewResource(resourceKindDashboard, nil),
    87  	); err != nil {
    88  		return SummaryView{}, err
    89  	}
    90  
    91  	sales, err := u.qs.SummarizeSales(ctx, window)
    92  	if err != nil {
    93  		return SummaryView{}, err
    94  	}
    95  	statusCounts, err := u.qs.CountPurchasesByStatus(ctx, window)
    96  	if err != nil {
    97  		return SummaryView{}, err
    98  	}
    99  	// 商品数は Repository から取得する（配置根拠は docs/spec/usecase/dashboard.md 参照）。
   100  	productCounts, err := u.productRepo.Count(ctx)
   101  	if err != nil {
   102  		return SummaryView{}, err
   103  	}
   104  
   105  	return SummaryView{
   106  		SalesAmount:           sales.Amount,
   107  		SalesCount:            sales.Count,
   108  		PurchaseStatusCounts:  toStatusCountViews(statusCounts),
   109  		TotalProductCount:     productCounts.Total,
   110  		PublishedProductCount: productCounts.Published,
   111  	}, nil
   112  }
