    85  func (u *usecase) CheckHealth(ctx context.Context) (*DTO, error) {
    86  	ctx, endSpan := u.tracer.Start(ctx)
    87  	defer endSpan()
    88  
    89  	applTime := u.clock.Now()
    90  	dbHealth, err := u.dbSystemCqrs.CheckDBHealth(ctx)
    91  	if err != nil {
    92  		return nil, err
    93  	}
    94  
    95  	deps, status := u.checkDependencies(ctx)
    96  
    97  	return &DTO{
    98  		Status:          status,
    99  		ApplicationTime: applTime,
   100  		DBHealthCheck:   dbHealth,
   101  		Dependencies:    deps,
   102  	}, nil
   103  }

   105  // checkDependencies は、各 probe を検査して状態の一覧と総合ステータスを返します。
   106  // 1 つでも落ちていれば degraded で、エラーにはしません（理由は Probe を参照）。
   107  func (u *usecase) checkDependencies(ctx context.Context) ([]DependencyStatus, string) {
   108  	if len(u.probes) == 0 {
   109  		return nil, Ok
   110  	}
   111  
   112  	deps := make([]DependencyStatus, 0, len(u.probes))
   113  	status := Ok
   114  	for _, p := range u.probes {
   115  		depStatus := Ok
   116  		if u.checkProbe(ctx, p) != nil {
   117  			depStatus = Degraded
   118  			status = Degraded
   119  		}
   120  
   121  		deps = append(deps, DependencyStatus{Name: p.Name, Status: depStatus})
   122  	}
   123  
   124  	return deps, status
   125  }

   127  // checkProbe は、1 つの probe を probeTimeout で有界に検査します。超過は degraded として扱います
   128  // （上限の根拠は probeTimeout を参照）。
   129  func (u *usecase) checkProbe(ctx context.Context, p Probe) error {
   130  	ctx, cancel := context.WithTimeout(ctx, probeTimeout)
   131  	defer cancel()
   132  
   133  	return p.Check(ctx)
   134  }
