    22  		c, found, err := u.resolveCart(ctx, subject, now)
    23  		if err != nil {
    24  			return err
    25  		}

    27  			return nil

    30  		locked, lerr := u.cartRepo.LockByID(ctx, c.ID())

    32  			// 解決とロックの間にカートが消えていても、カートを持たない主体と同じく成功で返します。
    33  			// 集約の不在は 1 つの事実であり、どちらの読み取りで見つけたかで応答を変えません
    34  			// （この op は 404 を宣言していません）。
    35  			if xerrors.Is(lerr, apperror.ErrNotFound) {
    36  				return nil
    37  			}

    39  		}
    40  
    41  		locked.Clear()
    42  		locked.Touch(now, cartTTL)
    43  		return u.cartRepo.Update(ctx, locked)
    44  	})
