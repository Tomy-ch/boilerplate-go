    27  func (u *usecase) setItem(ctx context.Context, params SetItemParams) (CartView, error) {
    28  	now := u.clock.Now()
    29  
    30  	c, issuedToken, err := u.resolveOrCreateCart(ctx, params.Subject, now)
    31  	if err != nil {
    32  		return CartView{}, err
    33  	}
    34  
    35  	if aerr := u.ensureProductAvailable(ctx, params.ProductID); aerr != nil {
    36  		return CartView{}, aerr
    37  	}
    38  
    39  	itemID, err := uuid.New()
    40  	if err != nil {
    41  		return CartView{}, xerrors.Wrap(err, "failed to generate cart item id")
    42  	}
    43  	if serr := c.SetItem(cart.SetItemAttributes{
    44  		ItemID:    itemID,
    45  		ProductID: params.ProductID,
    46  		Quantity:  params.Quantity,
    47  	}, now); serr != nil {
    48  		return CartView{}, serr
    49  	}
    50  
    51  	view, verr := u.buildView(ctx, c, now)
    52  	if verr != nil {
    53  		return CartView{}, verr
    54  	}
    55  	if uerr := u.cartRepo.Update(ctx, c); uerr != nil {
    56  		return CartView{}, uerr
    57  	}
    58  
    59  	view.SessionToken = issuedToken
    60  	return view, nil

    86  // resolveOwnerCart は、所有者が確定したカートを解決します。
    87  // ユーザー 1 人につきカートは高々 1 件のため、期限切れでも作り直せません
    88  // （docs/spec/usecase/cart.md の SetItem）。
    89  //
    90  // 解決とロックの間にカートが消えていた場合は、引けなかった場合と同じく確保し直します
    91  // （docs/spec/usecase/cart.md の SetItem）。
    92  func (u *usecase) resolveOwnerCart(
    93  	ctx context.Context, userID uuid.UUID, now time.Time,
    94  ) (*cart.Cart, *string, error) {
    95  	found, err := u.cartRepo.FindByOwnerID(ctx, userID)
    96  	if err != nil {
    97  		if !xerrors.Is(err, apperror.ErrNotFound) {
    98  			return nil, nil, err
    99  		}
   100  		return u.createOwnerCart(ctx, userID, now)
   101  	}
   102  
   103  	c, lerr := u.cartRepo.LockByID(ctx, found.ID())
   104  	if lerr != nil {
   105  		if !xerrors.Is(lerr, apperror.ErrNotFound) {
   106  			return nil, nil, lerr
   107  		}
   108  		return u.createOwnerCart(ctx, userID, now)
   109  	}
   110  	if c.IsExpired(now) {
   111  		c.Clear()
   112  	}
   113  	return c, nil, nil
   114  }

   116  // resolveGuestCart は、ゲストのカートを解決します。
   117  // 提示されたトークンで引けなかった場合と、引けたが期限切れだった場合は、どちらも採番し直します
   118  // （docs/spec/usecase/cart.md の SetItem）。
   119  //
   120  // 解決とロックの間にカートが消えていた場合も同じく採番し直します
   121  // （docs/spec/usecase/cart.md の SetItem）。
   122  func (u *usecase) resolveGuestCart(
   123  	ctx context.Context, presented *string, now time.Time,
   124  ) (*cart.Cart, *string, error) {
   125  	if presented == nil {
   126  		return u.createGuestCart(ctx, now)
   127  	}
   128  
   129  	token, err := cart.NewSessionToken(*presented)
   130  	if err != nil {
   131  		return nil, nil, err
   132  	}
   133  
   134  	found, ferr := u.cartRepo.FindBySessionToken(ctx, token)
   135  	if ferr != nil {
   136  		if !xerrors.Is(ferr, apperror.ErrNotFound) {
   137  			return nil, nil, ferr
   138  		}
   139  		return u.createGuestCart(ctx, now)
   140  	}
   141  
   142  	c, lerr := u.cartRepo.LockByID(ctx, found.ID())
   143  	if lerr != nil {
   144  		if !xerrors.Is(lerr, apperror.ErrNotFound) {
   145  			return nil, nil, lerr
   146  		}
   147  		return u.createGuestCart(ctx, now)
   148  	}
   149  	if c.IsExpired(now) {
   150  		return u.createGuestCart(ctx, now)
   151  	}
   152  	return c, nil, nil
   153  }

   158  func (u *usecase) createOwnerCart(
   159  	ctx context.Context, userID uuid.UUID, now time.Time,
   160  ) (*cart.Cart, *string, error) {
   161  	id, err := uuid.New()

   165  
   166  	candidate, nerr := cart.NewForOwner(id, cart.Attributes{OwnerID: &userID, ExpiresAt: now.Add(cartTTL)})

   168  		return nil, nil, nerr

   172  	if cerr != nil {

   182  func (u *usecase) createGuestCart(ctx context.Context, now time.Time) (*cart.Cart, *string, error) {
   183  	raw, err := u.tokenGen.Generate()
   184  	if err != nil {
   185  		return nil, nil, xerrors.Wrap(err, "failed to generate session token")
   186  	}
   187  
   188  	token, terr := cart.NewSessionToken(raw)
   189  	if terr != nil {
   190  		return nil, nil, terr
   191  	}
   192  
   193  	id, ierr := uuid.New()
   194  	if ierr != nil {
   195  		return nil, nil, xerrors.Wrap(ierr, "failed to generate cart id")
   196  	}
   197  
   198  	c, nerr := cart.NewForGuest(id, token, now.Add(cartTTL))
   199  	if nerr != nil {
   200  		return nil, nil, nerr
   201  	}
   202  	if cerr := u.cartRepo.Create(ctx, c); cerr != nil {
   203  		return nil, nil, cerr
   204  	}
   205  	return c, &raw, nil
   206  }
