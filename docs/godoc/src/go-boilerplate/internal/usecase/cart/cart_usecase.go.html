   191  // resolveCart は、主体から表示すべきカートを解決します。
   192  // 主体を持たない・カートが無い・有効期限を過ぎているのいずれも「表示するカートが無い」であり、
   193  // found=false を返します。いずれも呼び出し側の失敗ではありません。
   194  func (u *usecase) resolveCart(ctx context.Context, subject Subject, now time.Time) (*cart.Cart, bool, error) {
   195  	c, found, err := u.findCart(ctx, subject)
   196  	if err != nil {
   197  		if xerrors.Is(err, apperror.ErrNotFound) {
   198  			return nil, false, nil
   199  		}
   200  		return nil, false, err
   201  	}
   202  	if !found || c.IsExpired(now) {
   203  		return nil, false, nil
   204  	}
   205  	return c, true, nil
   206  }

   208  // findCart は、主体の種類に応じてカートを引きます。主体を持たない場合は found=false を返します。
   209  // 認証済みユーザーはゲストセッションより優先されます。
   210  func (u *usecase) findCart(ctx context.Context, subject Subject) (*cart.Cart, bool, error) {
   211  	switch {
   212  	case subject.UserID != nil:
   213  		c, err := u.cartRepo.FindByOwnerID(ctx, *subject.UserID)
   214  		return c, err == nil, err
   215  	case subject.SessionToken != nil:
   216  		token, err := cart.NewSessionToken(*subject.SessionToken)
   217  		if err != nil {
   218  			return nil, false, err
   219  		}
   220  		c, err := u.cartRepo.FindBySessionToken(ctx, token)
   221  		return c, err == nil, err
   222  	default:
   223  		return nil, false, nil
   224  	}
   225  }

   227  // findProducts は、明細が指す商品を ID で引き当てた表を返します。
   228  // 引けなかった商品は表に現れません。明細が無い場合は問い合わせません。
   229  func (u *usecase) findProducts(
   230  	ctx context.Context, items []cart.CartItem,
   231  ) (map[uuid.UUID]*product.Product, error) {
   232  	if len(items) == 0 {
   233  		return map[uuid.UUID]*product.Product{}, nil
   234  	}
   235  
   236  	ids := make([]uuid.UUID, 0, len(items))
   237  	for _, item := range items {
   238  		ids = append(ids, item.ProductID())
   239  	}
   240  
   241  	products, err := u.productRepo.FindByIDs(ctx, ids)
   242  	if err != nil {
   243  		return nil, err
   244  	}
   245  
   246  	found := make(map[uuid.UUID]*product.Product, len(products))
   247  	for _, p := range products {
   248  		found[p.ID()] = p
   249  	}
   250  	return found, nil
   251  }

   253  // buildView は、明細を再評価して出力 DTO を組み立て、あわせてカートの状態を進めます。
   254  // 提示した価格の記録と有効期限の延長までを含むため、呼び出し側は戻り値を返す前にカートを保存します。
   255  //
   256  // 判定も合算も提示価格を書き換える前に済ませます（docs/spec/domain/cart.md の Cart.Subtotal）。
   257  //
   258  // SessionToken は埋めません。トークンを新しく発行したかどうかを知るのはカートを解決した側だけです。
   259  func (u *usecase) buildView(ctx context.Context, c *cart.Cart, now time.Time) (CartView, error) {
   260  	items := c.Items()
   261  	products, err := u.findProducts(ctx, items)
   262  	if err != nil {
   263  		return CartView{}, err
   264  	}
   265  
   266  	views, seen := evaluateItems(items, products)
   267  	subtotal, serr := c.Subtotal(toSnapshots(products))
   268  	if serr != nil {
   269  		return CartView{}, serr
   270  	}
   271  
   272  	c.MarkSeen(seen)
   273  	c.Touch(now, cartTTL)
   274  
   275  	expiresAt := c.ExpiresAt()
   276  	return CartView{Items: views, SubtotalAmount: subtotal, ExpiresAt: &expiresAt}, nil
   277  }

   279  // evaluateItems は、明細を商品の現在値と突き合わせ、明細ごとの View と、提示した価格の表を返します。
   280  // 価格の表には引けた商品のぶんだけが入ります（引けなかった明細の提示価格は据え置かれます）。
   281  func evaluateItems(
   282  	items []cart.CartItem, products map[uuid.UUID]*product.Product,
   283  ) ([]CartItemView, map[uuid.UUID]money.Price) {
   284  	views := make([]CartItemView, 0, len(items))
   285  	seen := make(map[uuid.UUID]money.Price, len(items))
   286  
   287  	for _, item := range items {
   288  		p := products[item.ProductID()]
   289  		views = append(views, evaluateItem(item, p))
   290  		if p != nil {
   291  			seen[p.ID()] = p.Price()
   292  		}
   293  	}
   294  
   295  	return views, seen
   296  }

   298  // toSnapshots は、引けた商品を再評価用の観測値へ切り出します。
   299  // 引けなかった商品は含めません（この表に無い明細の扱いは docs/spec/domain/cart.md の Cart.Subtotal）。

   301  	snapshots := make(map[uuid.UUID]cart.ProductSnapshot, len(products))
   302  	for id, p := range products {
   303  		snapshots[id] = cart.NewProductSnapshot(cart.ProductSnapshotAttributes{
   304  			Quantity:     p.Quantity(),
   305  			Price:        p.Price(),
   306  			Published:    p.IsPublished(),
   307  			Discontinued: p.IsDiscontinued(),
   308  		})
   309  	}
   310  
   311  	return snapshots

   314  // evaluateItem は、明細 1 件の再評価を cart.CartItem.Evaluate へ委ね、結果を出力 DTO の語彙へ写します。
   315  func evaluateItem(item cart.CartItem, p *product.Product) CartItemView {
   316  	view := CartItemView{ProductID: item.ProductID(), Quantity: item.Quantity()}
   317  
   318  	var snapshot *cart.ProductSnapshot
   319  	if p != nil {
   320  		name, price := p.Name(), p.Price().Decimal()
   321  		view.ProductName, view.UnitPrice = &name, &price
   322  		if image, ok := p.PrimaryImage(); ok {
   323  			imagePath := image.ImagePath()
   324  			view.ImagePath = &imagePath
   325  		}
   326  		s := cart.NewProductSnapshot(cart.ProductSnapshotAttributes{
   327  			Quantity:     p.Quantity(),
   328  			Price:        p.Price(),
   329  			Published:    p.IsPublished(),
   330  			Discontinued: p.IsDiscontinued(),
   331  		})
   332  		snapshot = &s
   333  	}
   334  
   335  	evaluation := item.Evaluate(snapshot)
   336  	view.Issues = toItemIssues(evaluation.Issues())
   337  	view.AvailableQuantity = evaluation.AvailableQuantity()
   338  
   339  	return view
   340  }
