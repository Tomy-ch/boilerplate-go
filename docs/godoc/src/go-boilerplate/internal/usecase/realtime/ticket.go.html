    85  func (s *ticketService) Issue(ctx context.Context, in IssueTicketInput) (TicketView, error) {
    86  	ctx, endSpan := s.tracer.Start(ctx)
    87  	defer endSpan()
    88  
    89  	value, err := s.secrets.Generate()
    90  	if err != nil {
    91  		return TicketView{}, xerrors.Wrap(err, "issue ticket")
    92  	}
    93  
    94  	now := s.clock.Now()
    95  	ticket := rt.StreamTicket{
    96  		Hash:          hashTicket(value),
    97  		Subject:       in.Subject,
    98  		Destination:   in.Destination,
    99  		Scope:         in.Scope,
   100  		InitialCursor: in.InitialCursor,
   101  		IssuedAt:      now,
   102  		ExpiresAt:     now.Add(TicketTTL),
   103  	}
   104  	if err := s.store.Save(ctx, ticket); err != nil {
   105  		return TicketView{}, err
   106  	}
   107  
   108  	return TicketView{Value: value, ExpiresAt: ticket.ExpiresAt}, nil
   109  }

   111  func (s *ticketService) Verify(ctx context.Context, value string, destination rt.StreamID) (rt.StreamGrant, error) {
   112  	ctx, endSpan := s.tracer.Start(ctx)
   113  	defer endSpan()
   114  
   115  	if value == "" {
   116  		return rt.StreamGrant{}, xerrors.Wrap(ErrTicketInvalid, "ticket is empty")
   117  	}
   118  
   119  	ticket, ok, err := s.store.Find(ctx, hashTicket(value), s.clock.Now())
   120  	if err != nil {
   121  		return rt.StreamGrant{}, err
   122  	}
   123  
   124  	if !ok {
   125  		return rt.StreamGrant{}, xerrors.Wrap(ErrTicketInvalid, "ticket is unknown or expired")
   126  	}
   127  
   128  	if ticket.Destination != destination {
   129  		return rt.StreamGrant{}, xerrors.Wrap(ErrTicketInvalid, "ticket is bound to another destination")
   130  	}
   131  
   132  	return rt.StreamGrant{
   133  		Subject: ticket.Subject, Destination: ticket.Destination, Scope: ticket.Scope, InitialCursor: ticket.InitialCursor,
   134  	}, nil
   135  }

   137  // hashTicket は、生値の保存形（SHA-256 の 16 進）を返します。生値は 256 bit の乱数なので salt は要りません。
   138  func hashTicket(value string) rt.TicketHash {
   139  	sum := sha256.Sum256([]byte(value))
   140  
   141  	return rt.TicketHash(hex.EncodeToString(sum[:]))
   142  }
