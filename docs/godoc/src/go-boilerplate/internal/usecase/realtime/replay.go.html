    21  	// EventLog が読めなければ apperror.ErrUnavailable（retry 可能）を返します。
    22  	ReadPage(ctx context.Context, streamID rt.StreamID, after rt.Sequence) (events []rt.DeliveryEvent, hasMore bool, err error)
    23  }

    36  // 自分の位置と突き合わせて決めることで、ここは「その位置より後ろにある物」をそのまま返します。
