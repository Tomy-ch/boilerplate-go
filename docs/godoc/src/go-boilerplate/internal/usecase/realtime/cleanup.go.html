    15  // LeaseCleanupOwnershipTTL は、回収の引き受けが有効な期間です。回収は数回の API 呼び出しで終わるため
    16  // これで足り、引き受けたまま倒れた掃除役が居ても、この期間を過ぎれば次の掃除役が引き受け直せます。
    17  // 死んだと見なすまでの余裕（LeaseCleanupMargin）とは別の量なので、値を共有しません。
    18  const LeaseCleanupOwnershipTTL = 10 * time.Minute

    34  // OrphanSweeper は、死んだ instance が残した受信先を回収する seam です。回収は引き受けを条件付きで
    35  // 取った主体だけが行い、同時に走った掃除役のうち 1 つだけが 1 つの instance を回収します（ADR-0073）。

    61  // Sweep は、期限切れが安全余裕を過ぎた lease を引き受けてから、受信先を片付け、lease を閉じます。
    62  // 順序が固定なのは、lease が受信先を辿る唯一の索引だからです。先に lease を閉じると、片付けに失敗した
    63  // 受信先を誰も見つけられなくなります（docs/design/realtime-delivery.md §2.5 / ADR-0073）。
    64  func (s *orphanSweeper) Sweep(ctx context.Context) (SweepResult, error) {
    65  	ctx, endSpan := s.tracer.Start(ctx)
    66  	defer endSpan()
    67  
    68  	now := s.clock.Now()
    69  	// 期限切れからさらに安全余裕を過ぎたものだけを対象にする。引き受けと閉じるときの条件式にも
    70  	// 同じ時刻を渡し、列挙と条件判定が別の基準を使わないようにする。
    71  	cutoff := now.Add(-LeaseCleanupMargin)
    72  
    73  	leases, err := s.leases.ListExpired(ctx, cutoff)
    74  	if err != nil {
    75  		return SweepResult{}, err
    76  	}
    77  
    78  	result := SweepResult{Detected: len(leases)}
    79  
    80  	var errs []error
    81  	for _, lease := range leases {
    82  		if err := s.sweepOne(ctx, lease.InstanceID, now, cutoff, &result); err != nil {
    83  			errs = append(errs, err)
    84  		}
    85  	}
    86  
    87  	return result, xerrors.Join(errs...)
    88  }

    90  // sweepOne は、1 つの instance を引き受けから lease の削除まで処理し、内訳を result へ加算します。
    91  func (s *orphanSweeper) sweepOne(
    92  	ctx context.Context, id rt.InstanceID, now, cutoff time.Time, result *SweepResult,
    93  ) error {
    94  	claimed, err := s.leases.AcquireCleanup(ctx, rt.CleanupClaim{
    95  		InstanceID:    id,
    96  		Owner:         s.owner,
    97  		ExpiredBefore: cutoff,
    98  		Now:           now,
    99  		OwnerUntil:    now.Add(LeaseCleanupOwnershipTTL),
   100  	})
   101  	if err != nil {
   102  		result.Failed++
   103  
   104  		return err
   105  	}
   106  
   107  	if !claimed {
   108  		result.Skipped++
   109  
   110  		return nil
   111  	}
   112  
   113  	result.Claimed++
   114  
   115  	if err := s.reclaimer.Reclaim(ctx, id); err != nil {
   116  		// 受信先が残ったので lease は閉じない（順序の理由は Sweep の doc comment）。次の掃除が引き受け直す。
   117  		result.Failed++
   118  
   119  		return err
   120  	}
   121  
   122  	released, err := s.leases.ReleaseCleanup(ctx, rt.CleanupRelease{
   123  		InstanceID:    id,
   124  		Owner:         s.owner,
   125  		ExpiredBefore: cutoff,
   126  	})
   127  	if err != nil {
   128  		result.Failed++
   129  
   130  		return err
   131  	}
   132  
   133  	if !released {
   134  		// 引き受けている間に instance が復帰したか、引き受けが他へ移った。受信先は既に片付いており、
   135  		// 復帰した instance は次の受信で自分の受信先を張り直す。
   136  		result.Skipped++
   137  
   138  		return nil
   139  	}
   140  
   141  	result.Reclaimed++
   142  
   143  	return nil
   144  }
