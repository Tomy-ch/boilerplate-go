    18  // CursorValidator は、client が提示した cursor から replay を始められるかを判定します。
    19  type CursorValidator interface {
    20  	// Validate は、cursor より後ろを欠けなく replay できるなら nil を返します。replay floor（ADR-0072 —
    21  	// 保存せず導出する）より前なら ErrCursorExpired、EventLog が読めなければ apperror.ErrUnavailable
    22  	// （retry 可能）を返します。cursor 0 は初期位置（stream の先頭から）です。
    23  	Validate(ctx context.Context, streamID rt.StreamID, cursor rt.Sequence) error
    24  }

    37  // Validate は、cursor より後ろを 1 回読み、続きが無ければ追記済みの位置と比べて判定します。
    38  //
    39  // 前半を 1 回の読み取りにまとめるのは、「cursor+1 の有無」と「後ろの event の有無」を別々に読むと、
    40  // その間に relay が cursor+1 を append しただけで gap に見えるためです。
    41  //
    42  // 後半は「cursor より後ろに追記されたものが、消えているか」を問います。cursor 自身の item が
    43  // 残っているかは判定に要りません。最後の event を見た client は、その event が保持期間で消えても
    44  // 何も失っておらず、そこで断ると正本を読み直しても同じ cursor が返り回復できません。
    45  func (v *cursorValidator) Validate(ctx context.Context, streamID rt.StreamID, cursor rt.Sequence) error {
    46  	ctx, endSpan := v.tracer.Start(ctx)
    47  	defer endSpan()
    48  
    49  	res, err := v.log.ReadAfter(ctx, rt.ReadAfterQuery{StreamID: streamID, After: cursor, Limit: 1})
    50  	if err != nil {
    51  		return err
    52  	}
    53  
    54  	if len(res.Events) > 0 {
    55  		next := res.Events[0]
    56  		if next.Sequence != cursor+1 {
    57  			return xerrors.Wrap(ErrCursorExpired, "the next event is gone while a later one exists")
    58  		}
    59  
    60  		if v.clock.Now().Sub(next.OccurredAt) > rt.EventLogRetention {
    61  			return xerrors.Wrap(ErrCursorExpired, "the next event is older than the retention")
    62  		}
    63  
    64  		return nil
    65  	}
    66  
    67  	appended, err := v.log.AppendedThrough(ctx, streamID)
    68  	if err != nil {
    69  		return err
    70  	}
    71  
    72  	if cursor >= appended {
    73  		return nil
    74  	}
    75  
    76  	return xerrors.Wrap(ErrCursorExpired, "the events after the cursor are gone")
    77  }
