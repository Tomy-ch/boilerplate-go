    88  			expectListExpired(t, leases, clk, "inst-1", "inst-2", "inst-3")

    93  			leases.EXPECT().ReleaseCleanup(gomock.Any(), gomock.Any()).Return(true, nil).Times(2)
    94  
    95  			got, err := s.Sweep(t.Context())
    96  			require.NoError(t, err)

    98  		})

   135  func Test_orphanSweeper_sweepOne(t *testing.T) {

   149  					ExpiredBefore: sweepCutoff,

   151  					OwnerUntil:    sweepNow.Add(LeaseCleanupOwnershipTTL),
   152  				}).Return(true, nil),
   153  				reclaimer.EXPECT().Reclaim(gomock.Any(), rt.InstanceID("inst-1")).Return(nil),
   154  				leases.EXPECT().ReleaseCleanup(gomock.Any(), rt.CleanupRelease{
   155  					InstanceID:    "inst-1",
   156  					Owner:         "cleaner-1",
   157  					ExpiredBefore: sweepCutoff,
   158  				}).Return(true, nil),
   159  			)
   160  
   161  			var got SweepResult
   162  			require.NoError(t, s.sweepOne(t.Context(), "inst-1", sweepNow, sweepCutoff, &got))
   163  			assert.Equal(t, SweepResult{Claimed: 1, Reclaimed: 1}, got)
   164  		})
