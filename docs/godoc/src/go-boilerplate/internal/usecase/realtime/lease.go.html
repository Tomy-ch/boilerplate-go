    24  // LeaseKeeper は、serve instance の生存記録を書き、正常終了時に取り消す seam です。
    25  // 記録は crash した instance の resource を後から回収するためだけにあります（ADR-0073）。
    26  type LeaseKeeper interface {

    29  	// Release は、id の記録を取り消します。resource を自分で片付けた instance が呼び、無くてもエラーになりません。
    30  	Release(ctx context.Context, id rt.InstanceID) error
    31  }

    44  func (k *leaseKeeper) Beat(ctx context.Context, id rt.InstanceID) error {
