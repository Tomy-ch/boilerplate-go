    74  func (u *usecase) Convert(ctx context.Context, in ConvertInput) (*ConvertResult, error) {
    75  	ctx, endSpan := u.tracer.Start(ctx)
    76  	defer endSpan()
    77  
    78  	rate, err := u.gateway.GetRate(ctx, in.Base, in.Quote)
    79  	if err != nil {
    80  		return nil, err
    81  	}
    82  
    83  	result := &ConvertResult{Converted: in.Amount.Mul(rate.Value)}
    84  
    85  	// degrade 方針は docs/spec/usecase/exchangerate.md を参照。
    86  	if in.DisplayCurrency != nil {
    87  		refRate, refErr := u.gateway.GetRate(ctx, in.Base, *in.DisplayCurrency)
    88  		switch {
    89  		case refErr != nil:
    90  			u.logging.Warn(ctx, "reference amount degraded: display rate unavailable", logging.Error(logging.ErrorKey, refErr))
    91  		default:
    92  			ref, buildErr := BuildReferenceAmount(refRate, in.Amount)
    93  			if buildErr != nil {
    94  				u.logging.Warn(ctx, "reference amount degraded: amount exceeds settlement range", logging.Error(logging.ErrorKey, buildErr))
    95  			} else {
    96  				result.Reference = ref
    97  			}
    98  		}
    99  	}
   100  	return result, nil
   101  }

   103  // BuildReferenceAmount は、レートと base 通貨建て金額から参考換算額を組み立てます。
   104  // 金額が表示通貨の最小単位整数の範囲を超える場合はエラーを返します。
   105  func BuildReferenceAmount(rate *boundary.Rate, amount decimal.Decimal) (*ReferenceAmount, error) {
   106  	minor, err := money.ApplyRateHalfUp(amount, rate.Value, displayMinorUnitDigits)
   107  	if err != nil {
   108  		return nil, err
   109  	}
   110  	return &ReferenceAmount{
   111  		Currency: rate.Quote,
   112  		Amount:   minor,
   113  		Rate:     rate.Value,
   114  		RateDate: rate.Date,
   115  	}, nil
   116  }
