    53  				func(_ context.Context, req *httpclient.Request) (*httpclient.Response, error) {
    54  					assert.Equal(t, httpclient.Downstream("address"), req.Downstream())
    55  					assert.Equal(t, httpclient.MethodGet(), req.Method())
    56  					// endpoint / パス / クエリまで含めて組み立てを検証する（endpoint 欠落や別ホスト固定の退行を捕捉）。
    57  					assert.Equal(t, string(testEndpoint)+"/api/search?zipcode=1000001", req.URL())
    58  					return &httpclient.Response{StatusCode: 200, Body: []byte(
    59  						`{"status":200,"message":null,"results":[` +
    60  							`{"address1":"東京都","address2":"千代田区","address3":"千代田"},` +
    61  							`{"address1":"東京都","address2":"千代田区","address3":"大手町"}]}`,
    62  					)}, nil
    63  				},
    64  			)
    65  
    66  			gw := address.New(testEndpoint, client, observability.NewNoopTracerFactory(t))
    67  			candidates, err := gw.Lookup(context.Background(), "1000001")
    68  
    69  			require.NoError(t, err)
    70  			require.Len(t, candidates, 2)
    71  			assert.Equal(t, "東京都", candidates[0].PrefectureName)
