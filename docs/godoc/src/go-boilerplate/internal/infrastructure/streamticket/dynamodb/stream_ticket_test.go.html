    25  var (

    48  // waitIndexed は、GSI（結果整合）に item が載るのを短く待ちます。
    49  func waitIndexed(t *testing.T, s *store, subject string, destination realtime.StreamID, want int) {
    50  	t.Helper()
    51  
    52  	require.Eventually(t, func() bool {
    53  		out, err := s.c.Query(t.Context(), &dynamodb.QueryInput{
    54  			TableName:              aws.String(s.table),
    55  			IndexName:              aws.String(indexBySubjectDestination),
    56  			KeyConditionExpression: aws.String(attrSubject + " = :s AND " + attrDestination + " = :d"),
    57  			ExpressionAttributeValues: map[string]types.AttributeValue{
    58  				":s": &types.AttributeValueMemberS{Value: subject},
    59  				":d": &types.AttributeValueMemberS{Value: string(destination)},
    60  			},
    61  		})
    62  
    63  		// 1 件でも見えた時点で打ち切ると、同じ subject × destination の残りが索引に載る前に
    64  		// Invalidate が走り、取りこぼした ticket が生き残る。期待件数まで待つ。
    65  		return err == nil && len(out.Items) >= want
    66  	}, 5*time.Second, 50*time.Millisecond)
    67  }
