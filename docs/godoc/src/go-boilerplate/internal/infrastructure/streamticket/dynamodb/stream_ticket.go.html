    35  func (s *store) Save(ctx context.Context, ticket realtime.StreamTicket) error {
    36  	ctx, endSpan := s.tracer.Start(ctx)
    37  	defer endSpan()
    38  
    39  	if ticket.Hash == "" {
    40  		return xerrors.Wrap(apperror.ErrInvalidArgument, "save ticket: hash is empty")

    42  
    43  	_, err := s.c.PutItem(ctx, &dynamodb.PutItemInput{TableName: aws.String(s.table), Item: toItem(ticket)})

    51  // Find は、hash の ticket を強い一貫性で読みます。無い、または asOf 時点で期限切れなら ok=false です。
    52  func (s *store) Find(ctx context.Context, hash realtime.TicketHash, asOf time.Time) (realtime.StreamTicket, bool, error) {
    53  	ctx, endSpan := s.tracer.Start(ctx)
    54  	defer endSpan()
    55  
    56  	out, err := s.c.GetItem(ctx, &dynamodb.GetItemInput{
    57  		TableName:      aws.String(s.table),
    58  		Key:            key(hash),
    59  		ConsistentRead: aws.Bool(true),
    60  	})
    61  	if err != nil {
    62  		return realtime.StreamTicket{}, false, dynamodbclient.Normalize(err, "find ticket")
    63  	}
    64  
    65  	if len(out.Item) == 0 {
    66  		return realtime.StreamTicket{}, false, nil
    67  	}
    68  
    69  	ticket, err := fromItem(out.Item)
    70  	if err != nil {
    71  		return realtime.StreamTicket{}, false, err
    72  	}
    73  
    74  	if !asOf.Before(ticket.ExpiresAt) {
    75  		return realtime.StreamTicket{}, false, nil
    76  	}
    77  
    78  	return ticket, true, nil
    79  }

    81  // Invalidate は、subject × destination の ticket を複合キーの GSI で引いてすべて削除します。
    82  // GSI は結果整合なので、直前に Save した ticket が見えないことがあります。revocation の主機構は
    83  // fan-out による接続の close で、ここは STOP を無視する client への保険です（設計正本 §2.4）。
    84  func (s *store) Invalidate(ctx context.Context, subject string, destination realtime.StreamID) error {
    85  	ctx, endSpan := s.tracer.Start(ctx)
    86  	defer endSpan()
    87  
    88  	var start map[string]types.AttributeValue
    89  	for {
    90  		out, err := s.c.Query(ctx, &dynamodb.QueryInput{
    91  			TableName:              aws.String(s.table),
    92  			IndexName:              aws.String(indexBySubjectDestination),
    93  			KeyConditionExpression: aws.String(attrSubject + " = :s AND " + attrDestination + " = :d"),
    94  			ExpressionAttributeValues: map[string]types.AttributeValue{
    95  				":s": &types.AttributeValueMemberS{Value: subject},
    96  				":d": &types.AttributeValueMemberS{Value: string(destination)},
    97  			},
    98  			ExclusiveStartKey: start,
    99  		})
   100  		if err != nil {
   101  			return dynamodbclient.Normalize(err, "invalidate tickets")
   102  		}
   103  
   104  		for _, item := range out.Items {
   105  			hash, ok := item[attrTicketHash].(*types.AttributeValueMemberS)
   106  			if !ok {
   107  				return xerrors.Wrap(apperror.ErrInternal, "stream ticket index item: ticket_hash is not a string")
   108  			}
   109  
   110  			if _, err := s.c.DeleteItem(ctx, &dynamodb.DeleteItemInput{
   111  				TableName: aws.String(s.table),
   112  				Key:       key(realtime.TicketHash(hash.Value)),
   113  			}); err != nil {
   114  				return dynamodbclient.Normalize(err, "invalidate tickets")
   115  			}
   116  		}
   117  
   118  		if len(out.LastEvaluatedKey) == 0 {
   119  			return nil
   120  		}
   121  
   122  		start = out.LastEvaluatedKey
   123  	}
   124  }

   130  // toItem は、ticket を item に写します。expires_at は epoch 秒（TTL 属性でもある）なので秒精度です。
   131  func toItem(t realtime.StreamTicket) map[string]types.AttributeValue {
   132  	item := key(t.Hash)
   133  	item[attrSubject] = &types.AttributeValueMemberS{Value: t.Subject}
   134  	item[attrDestination] = &types.AttributeValueMemberS{Value: string(t.Destination)}
   135  	item[attrScope] = &types.AttributeValueMemberS{Value: t.Scope}
   136  	item[attrInitialCursor] = &types.AttributeValueMemberN{Value: t.InitialCursor.String()}
   137  	item[attrIssuedAt] = &types.AttributeValueMemberS{Value: t.IssuedAt.UTC().Format(time.RFC3339Nano)}
   138  	item[attrExpiresAt] = &types.AttributeValueMemberN{Value: strconv.FormatInt(t.ExpiresAt.Unix(), 10)}
   139  
   140  	return item
   141  }

   143  // fromItem は、item を ticket に戻します。
   144  func fromItem(item map[string]types.AttributeValue) (realtime.StreamTicket, error) {
   145  	cursor, err := dynamodbclient.NumberAttr(item, attrInitialCursor, itemKind)
   146  	if err != nil {
   147  		return realtime.StreamTicket{}, err
   148  	}
   149  
   150  	expires, err := dynamodbclient.NumberAttr(item, attrExpiresAt, itemKind)
   151  	if err != nil {
   152  		return realtime.StreamTicket{}, err
   153  	}
   154  
   155  	issuedAt, err := time.Parse(time.RFC3339Nano, dynamodbclient.StringAttr(item, attrIssuedAt))
   156  	if err != nil {
   157  		return realtime.StreamTicket{}, xerrors.Wrap(apperror.ErrInternal, "stream ticket item: issued_at: "+err.Error())
   158  	}
   159  
   160  	return realtime.StreamTicket{
   161  		Hash:          realtime.TicketHash(dynamodbclient.StringAttr(item, attrTicketHash)),
   162  		Subject:       dynamodbclient.StringAttr(item, attrSubject),
   163  		Destination:   realtime.StreamID(dynamodbclient.StringAttr(item, attrDestination)),
   164  		Scope:         dynamodbclient.StringAttr(item, attrScope),
   165  		InitialCursor: realtime.Sequence(cursor),
   166  		IssuedAt:      issuedAt,
   167  		ExpiresAt:     time.Unix(expires, 0).UTC(),
   168  	}, nil
   169  }
