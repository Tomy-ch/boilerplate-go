    22  const (
    23  	// MaxAttempts は、1 回の API 呼び出しの試行上限（初回 + retry 2 回）です。
    24  	// store が不達のとき無制限に retry せず、呼び出し元へ ErrUnavailable を返して上位の判断
    25  	// （503 + Retry-After、outbox の backoff）に委ねるための固定値です。
    26  	MaxAttempts = 3
    27  	// MaxBackoff は、retry 間の待ち時間の上限です。SSE の write deadline（10 秒）の内側に収めます。
    28  	MaxBackoff = 2 * time.Second
    29  	// CallTimeout は、1 回の API 呼び出し全体（retry を含む）に与える上限です。
    30  	//
    31  	// これが無いと、応答を返さない store に対して呼び出しがいつまでも戻りません。retry は「失敗した
    32  	// 試行」を数える機構なので返ってこない試行には一度も発火せず、SDK にも HTTP クライアントにも
    33  	// 別の上限がありません（DI が渡すクライアントは Timeout を持たず、SSRF ガードのために差し替えた
    34  	// Dialer にも Timeout はありません）。SSE の stream path は request の deadline budget から
    35  	// 外れているため、上位に頼れる上限も無く、ここが唯一の歯止めになります。
    36  	//
    37  	// 値は SSE の write deadline に揃えます。呼び出しが 1 回の書き込みより長く待つ理由がないためです。
    38  	CallTimeout = 10 * time.Second
    39  	// tableWaitTimeout は、table が ACTIVE になるのを待つ上限です。
    40  	tableWaitTimeout = 60 * time.Second

    69  // New は、設定から DynamoDB クライアントを生成します。
    70  // 資格情報を解決できない場合はエラーを返し、認証エラーが最初の操作まで隠れないようにします。
    71  func New(ctx context.Context, cfg Config) (*dynamodb.Client, error) {
    72  	awsCfg, err := awsclient.Resolve(ctx, awsclient.Config{
    73  		Region:          cfg.Region,
    74  		AccessKeyID:     cfg.AccessKeyID,
    75  		SecretAccessKey: cfg.SecretAccessKey,
    76  		HTTPClient:      cfg.HTTPClient,
    77  	})
    78  	if err != nil {
    79  		return nil, err
    80  	}
    81  
    82  	return dynamodb.NewFromConfig(awsCfg, func(o *dynamodb.Options) {
    83  		o.RetryMode = aws.RetryModeStandard
    84  		o.RetryMaxAttempts = MaxAttempts
    85  		o.Retryer = retry.NewStandard(func(so *retry.StandardOptions) {
    86  			so.MaxAttempts = MaxAttempts
    87  			so.MaxBackoff = MaxBackoff
    88  		})
    89  		if cfg.Endpoint != "" {
    90  			o.BaseEndpoint = aws.String(cfg.Endpoint)
    91  		}
    92  		o.APIOptions = append(o.APIOptions, withCallTimeout(CallTimeout))
    93  	}), nil
    94  }

    96  // withCallTimeout は、API 呼び出し 1 回の全体に上限を与える middleware を stack の先頭へ差します。
    97  // retry の外側に置くので、上限は試行ごとではなく呼び出し全体に掛かります。
    98  func withCallTimeout(d time.Duration) func(*smithymiddleware.Stack) error {
    99  	return func(stack *smithymiddleware.Stack) error {
   100  		return stack.Initialize.Add(
   101  			smithymiddleware.InitializeMiddlewareFunc(
   102  				"dynamodbCallTimeout",
   103  				func(
   104  					ctx context.Context, in smithymiddleware.InitializeInput, next smithymiddleware.InitializeHandler,
   105  				) (smithymiddleware.InitializeOutput, smithymiddleware.Metadata, error) {
   106  					ctx, cancel := context.WithTimeout(ctx, d)
   107  					defer cancel()
   108  
   109  					return next.HandleInitialize(ctx, in)
   110  				},
   111  			),
   112  			smithymiddleware.Before,
   113  		)
   114  	}
   115  }

   125  // Normalize は、SDK の失敗を apperror sentinel へ正規化します。context の取り消しは ErrCanceled、
   126  // それ以外は ErrUnavailable です。条件式の不成立は呼び出し側が IsConditionalCheckFailed で先に判定します。
   127  func Normalize(err error, op string) error {
   128  	if xerrors.Is(err, context.Canceled) || xerrors.Is(err, context.DeadlineExceeded) {
   129  		return xerrors.Wrap(apperror.ErrCanceled, op+": "+err.Error())
   130  	}
   131  
   132  	return xerrors.Wrap(apperror.ErrUnavailable, op+": "+err.Error())
   133  }

   135  // EnsureTable は、table が無ければ作り、ACTIVE を待ち、TTL が未設定なら設定します。
   136  // 既にある table に対しては何も変えずに成功するので、何度実行しても同じ状態に収束します
   137  // （application の起動時ではなく one-shot の初期化で呼びます）。
   138  func EnsureTable(ctx context.Context, c *dynamodb.Client, spec TableSpec) error {
   139  	_, err := c.CreateTable(ctx, &dynamodb.CreateTableInput{
   140  		TableName:              aws.String(spec.Name),
   141  		AttributeDefinitions:   spec.Attributes,
   142  		KeySchema:              spec.KeySchema,
   143  		GlobalSecondaryIndexes: spec.GlobalSecondaryIndexes,
   144  		BillingMode:            types.BillingModePayPerRequest,
   145  	})
   146  	if err != nil && !isResourceInUse(err) {
   147  		return Normalize(err, "create table "+spec.Name)
   148  	}
   149  
   150  	waiter := dynamodb.NewTableExistsWaiter(c)
   151  	if err := waiter.Wait(ctx, &dynamodb.DescribeTableInput{TableName: aws.String(spec.Name)}, tableWaitTimeout); err != nil {
   152  		return Normalize(err, "wait table "+spec.Name)
   153  	}
   154  
   155  	if spec.TTLAttribute == "" {
   156  		return nil
   157  	}
   158  
   159  	return ensureTimeToLive(ctx, c, spec.Name, spec.TTLAttribute)
   160  }

   162  // ensureTimeToLive は、TTL が attr で有効（または有効化中）でなければ設定します。
   163  func ensureTimeToLive(ctx context.Context, c *dynamodb.Client, table, attr string) error {
   164  	desc, err := c.DescribeTimeToLive(ctx, &dynamodb.DescribeTimeToLiveInput{TableName: aws.String(table)})
   165  	if err != nil {
   166  		return Normalize(err, "describe ttl "+table)
   167  	}
   168  
   169  	if d := desc.TimeToLiveDescription; d != nil && aws.ToString(d.AttributeName) == attr &&
   170  		(d.TimeToLiveStatus == types.TimeToLiveStatusEnabled || d.TimeToLiveStatus == types.TimeToLiveStatusEnabling) {
   171  		return nil
   172  	}
   173  
   174  	_, err = c.UpdateTimeToLive(ctx, &dynamodb.UpdateTimeToLiveInput{
   175  		TableName:               aws.String(table),
   176  		TimeToLiveSpecification: &types.TimeToLiveSpecification{AttributeName: aws.String(attr), Enabled: aws.Bool(true)},
   177  	})
   178  	if err != nil {
   179  		return Normalize(err, "update ttl "+table)
   180  	}
   181  
   182  	return nil
   183  }

   201  // NumberAttr は、item の N 属性を int64 として返します。無い・N でない・整数に読めない場合は
   202  // ErrInternal です（この adapter が書いた覚えの無い形が store にあることを意味します）。
   203  // kind はエラー文に載せる item の種類（例 "event log"）です。
   204  func NumberAttr(item map[string]types.AttributeValue, name, kind string) (int64, error) {
   205  	n, ok := item[name].(*types.AttributeValueMemberN)
   206  	if !ok {
   207  		return 0, xerrors.Wrap(apperror.ErrInternal, kind+" item: "+name+" is not a number")
   208  	}
   209  
   210  	v, err := strconv.ParseInt(n.Value, 10, 64)
   211  	if err != nil {
   212  		return 0, xerrors.Wrap(apperror.ErrInternal, kind+" item: "+name+": "+err.Error())
   213  	}
   214  
   215  	return v, nil
   216  }
