   377  func (r *repository) UpdateStock(ctx context.Context, p *product.Product) (int, error) {
   378  	ctx, endSpan := r.tracer.Start(ctx)
   379  	defer endSpan()
   380  
   381  	quantity, err := safecast.IntToInt32(p.Quantity())
   382  	if err != nil {
   383  		return 0, xerrors.Wrap(err, "invalid product quantity")
   384  	}
   385  	currentVersion, err := safecast.IntToInt32(p.Version())
   386  	if err != nil {
   387  		return 0, xerrors.Wrap(err, "invalid product version")
   388  	}
   389  
   390  	db := gen.New(driver.New(ctx, r.db))
   391  	lockVersion, err := db.UpdateProductStock(ctx, &gen.UpdateProductStockParams{
   392  		Quantity:       quantity,
   393  		ID:             p.ID(),
   394  		CurrentVersion: currentVersion,
   395  	})
   396  	if err != nil {
   397  		// ロックなしで呼ばれた場合の二重防御です。到達しないからと外してはなりません
   398  		// （詳細: docs/spec/domain/product.md の Product.UpdateStock）。
   399  		if pgerror.IsNoRows(err) {
   400  			return 0, xerrors.Wrap(product.ErrVersionConflict, "product was updated by another transaction")
   401  		}
   402  		return 0, pgerror.NormalizeError(err)
   403  	}
   404  
   405  	return int(lockVersion), nil
   406  }

   623  // rowToProduct は、sqlc が返す商品行（マスタ JOIN 込み）と画像をドメインエンティティへ変換します。
   624  func rowToProduct(row productRow, images []product.Image) (*product.Product, error) {
   625  	price, err := money.NewPrice(row.p.Price)
   626  	if err != nil {
   627  		return nil, pgerror.NormalizeReconstructError(err)
   628  	}
   629  	status, err := product.NewStatusRef(row.p.StatusID, row.statusName)
   630  	if err != nil {
   631  		return nil, pgerror.NormalizeReconstructError(err)
   632  	}
   633  	category, err := product.NewCategoryRef(row.p.CategoryID, row.categoryName)
   634  	if err != nil {
   635  		return nil, pgerror.NormalizeReconstructError(err)
   636  	}
   637  
   638  	entity, err := product.Reconstruct(row.p.ID, product.Attributes{
   639  		Name:                  row.p.Name,
   640  		Description:           row.p.Description,
   641  		Price:                 price,
   642  		Quantity:              int(row.p.Quantity),
   643  		StockWarningThreshold: ptr.Map(row.p.StockWarningThreshold, func(v int32) int { return int(v) }),
   644  		Status:                status,
   645  		Category:              category,
   646  		PublishedAt:           row.p.PublishedAt,
   647  		DiscontinuedAt:        row.p.DiscontinuedAt,
   648  		Images:                images,
   649  	}, int(row.p.LockVersion), row.p.CreatedAt)
   650  	if err != nil {
   651  		return nil, pgerror.NormalizeReconstructError(err)
   652  	}
   653  	return entity, nil
   654  }
