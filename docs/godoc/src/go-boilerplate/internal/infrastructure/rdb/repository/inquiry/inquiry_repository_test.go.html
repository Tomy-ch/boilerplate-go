    21  // takeSeedUser は、seed 済みの利用者を ID 順で index 番目に選びます。
    22  //
    23  // seed の SQL から ID を書き写すと、その利用者が実際に投入されているかは環境によって変わり、
    24  // 外部キー違反で落ちます。並び順で取ることで、どの環境でも実在する利用者だけを使います。
    25  // 問い合わせは利用者ごとに 1 件しか持てないため、並行するケースには別の index を渡します。
    26  func takeSeedUser(ctx context.Context, t *testing.T, testDB driver.DatabaseDriver, index int) uuid.UUID {
    27  	t.Helper()
    28  	var id uuid.UUID
    29  	err := driver.New(ctx, testDB).
    30  		QueryRow(ctx, "SELECT id FROM users ORDER BY id LIMIT 1 OFFSET $1", index).
    31  		Scan(&id)
    32  	require.NoError(t, err, "seed 済みの利用者が %d 人未満です", index+1)
    33  	return id
    34  }
