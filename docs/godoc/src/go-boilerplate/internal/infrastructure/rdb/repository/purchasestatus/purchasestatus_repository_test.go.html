    47  		t.Run("全購入ステータスが9件sortKey昇順で取得できる", func(t *testing.T) {
    48  			t.Parallel()
    49  
    50  			unprocessedID, err := uuid.Parse("a66c996c-86b2-41d8-9bdd-9b685fb7c47d")
    51  			require.NoError(t, err)
    52  
    53  			txm.WithinTx(func(ctx context.Context) {
    54  				expectedUnprocessed, err := status.New(
    55  					unprocessedID, status.Attributes{Name: "未処理", Code: 1, SortKey: 1},
    56  				)
    57  				require.NoError(t, err)
    58  
    59  				actual, err := repo.FindAll(ctx)
    60  				require.NoError(t, err)
    61  				assert.Len(t, actual, 9)
    62  
    63  				// sort_key は UNIQUE のため厳密昇順が正。
    64  				for i := 1; i < len(actual); i++ {
    65  					assert.Less(t, actual[i-1].SortKey(), actual[i].SortKey())
    66  				}
    67  
    68  				// sortKey 昇順の先頭は「未処理」(sortKey=1)。
    69  				assert.Equal(t, expectedUnprocessed, actual[0])
    70  			})

    73  		t.Run("code と sortKey が異なる行を、列の対応どおりに写す", func(t *testing.T) {
    74  			t.Parallel()

    77  			// 先行の一覧検証は通ってしまう。異なる値を持つ行を tx 内で足して対応を固定する
    78  			// （tx はロールバックされる）。

    84  					"INSERT INTO purchase_statuses (id, name, code, sort_key) VALUES ($1,$2,$3,$4)",
    85  					id, "テスト対応検証ステータス", 99, 50,
    86  				)
    87  				require.NoError(t, execErr)

   101  		t.Run("テーブルが空の場合、nilではない空一覧を返す", func(t *testing.T) {
   102  			t.Parallel()
   103  
   104  			// TRUNCATE は依存表ごと ACCESS EXCLUSIVE を取るため、直列化の外側で走る tx と
   105  			// deadlock（40P01）しうる。require で即死させるとトランザクションマネージャーが
   106  			// 戻り値を受け取れず、リトライ可能と宣言済みのエラーが恒久的な失敗になるため、
   107  			// エラーは返して再試行に委ねる（WithinTxE の doc 参照）。
   108  			txm.WithinTxE(func(ctx context.Context) error {
   109  				// purchase_statuses を参照する依存行（purchases など）ごと空にする（tx はロールバックされる）。
   110  				if _, execErr := driver.New(ctx, testDB).Exec(ctx, "TRUNCATE purchase_statuses CASCADE"); execErr != nil {
   111  					return execErr
   112  				}
   113  
   114  				actual, err := repo.FindAll(ctx)
   115  				require.NoError(t, err)
   116  				assert.NotNil(t, actual)
   117  				assert.Empty(t, actual)
   118  
   119  				return nil
   120  			})
   121  		})
   122  	})
