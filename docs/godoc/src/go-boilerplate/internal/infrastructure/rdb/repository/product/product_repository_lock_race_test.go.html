    23  // readCompletionGracePeriod は、ロックを待たされていないことを確認するために読み取りへ与える時間です。
    24  // 高負荷でこの時間内に完了しない場合は「待たされた」側へ倒れるため、偽陰性の方向にだけ振れます。
    25  const (
    26  	readCompletionGracePeriod = 3 * time.Second
    27  
    28  	// raceBlockedGracePeriod は、後続役が先行役のロック解放を待たされていることを確認するために待つ時間です。
    29  	// 待たされた事実は、後続役がロックを取れた時刻が先行役の解放より後であることでも確かめます。
    30  	// 最終状態の一致だけを見ると、ロックが外れていても後続役の往復がこの時間を超えれば同じ観測になり、
    31  	// 退行を見逃します。所要時間そのものを閾値と比べないのは、後続役の計測開始が先行役の待機開始より
    32  	// 遅れるぶん、ロックが効いていても閾値を下回るためです。
    33  	// 高負荷でこの時間内に完了しない場合も「まだ完了していない」側へ倒れるため、負荷は偽陽性を生みません。
    34  	raceBlockedGracePeriod = 300 * time.Millisecond

   125  func Test_lockByIDsSerializesConcurrentUpdates(t *testing.T) {
   126  	t.Parallel()
   127  
   128  	testDB := testkit.NewTestDB(t)
   129  	// 検証用商品の作成から 2 本の tx の完了までを、他パッケージの CASCADE TRUNCATE から守る。
   130  	testkit.HoldSuiteSerialization(t, testDB)

   135  		return driver.NewTransactionManager(
   136  			testDB,
   137  			config.NewDatabaseConfig(config.MockConfigForTest(t)),
   138  			logging.NewTestLogger(t),
   139  			system.NewSleeper(),
   140  		)
   141  	}
   142  
   143  	ctx := context.Background()
   144  	targetID := "cccccccc-0000-4000-8004-000000000001"
   145  	targetUUID := mustParse(t, targetID)
   146  	published := time.Date(2099, time.January, 1, 0, 0, 0, 0, time.UTC)
   147

   149  		probeKeyword+"-LOCKS-RACE", nil, 1999, statusInStock, categoryElectronics, &published)
   150  	t.Cleanup(func() {
   151  		_, cleanupErr := driver.New(ctx, testDB).Exec(ctx, "DELETE FROM products WHERE id = $1", targetID)
   152  		require.NoError(t, cleanupErr)
   153  	})
   154  
   155  	var (
   156  		lockHeld         = make(chan struct{})
   157  		secondDone       = make(chan struct{})
   158  		holderEnd        = make(chan struct{})
   159  		secondAcquiredAt = make(chan time.Time, 1)
   160  		holderReleasedAt = make(chan time.Time, 1)
   161  	)
   162  
   163  	// 後続役: 先行役が商品行を押さえている間にロックを取りに行き、解放まで待たされる。
   164  	go func() {
   165  		defer close(secondDone)
   166  		<-lockHeld
   167  		_ = newTxManager().Do(ctx, func(txCtx context.Context) error {
   168  			_, lockErr := repo.LockByIDs(txCtx, []uuid.UUID{targetUUID})
   169  			secondAcquiredAt <- time.Now()
   170  			if lockErr != nil {
   171  				return xerrors.Join(errRollbackHolderTx, lockErr)
   172  			}
   173  
   174  			return errRollbackHolderTx
   175  		})
   176  	}()

   178  	// 先行役: 商品行を排他ロックし、後続役が待たされていることを確認してから解放する。
   179  	go func() {
   180  		defer close(holderEnd)
   181  		_ = newTxManager().Do(ctx, func(txCtx context.Context) error {
   182  			if _, lockErr := repo.LockByIDs(txCtx, []uuid.UUID{targetUUID}); lockErr != nil {
   183  				return xerrors.Join(errRollbackHolderTx, lockErr)
   184  			}
   185  			close(lockHeld)
   186  
   187  			select {
   188  			case <-secondDone:
   189  				t.Error("後続役が先行役のロックを待たずに完了した")
   190  			case <-time.After(raceBlockedGracePeriod):
   191  			}
   192  			holderReleasedAt <- time.Now()
   193  
   194  			return errRollbackHolderTx
   195  		})
   196  	}()
   197  
   198  	<-holderEnd
   199  	<-secondDone
   200  	// 後続役が完了したのがロック待ちの結果であることを、取得時刻の前後で裏づける。
   201  	assert.True(t, (<-secondAcquiredAt).After(<-holderReleasedAt))
