   231  	t.Parallel()
   232  
   233  	testDB := testkit.NewTestDB(t)
   234  	lt := observability.NewMockInfraLayerTracer(t)
   235  	txm := testkit.NewTestTransactionRunner(t)
   236  	repo := &repository{tracer: lt, db: testDB}
   237  
   238  	const (
   239  		countIdentities = "SELECT COUNT(*) FROM user_identities WHERE user_id = $1"
   240  		countRoles      = "SELECT COUNT(*) FROM user_roles WHERE user_id = $1"
   241  		countUsers      = "SELECT COUNT(*) FROM users WHERE id = $1"
   242  		countCoupons    = "SELECT COUNT(*) FROM coupons WHERE user_id = $1"
   243  	)
   244  
   245  	target := "eeeeeeee-1111-4000-8000-000000000001"
   246  	survivor := "eeeeeeee-1111-4000-8000-000000000002"
   247  	alive := "eeeeeeee-1111-4000-8000-000000000003"
   248  	deletedAt := time.Date(2000, time.January, 1, 0, 0, 0, 0, time.UTC)
   249  
   250  	t.Run("正常系", func(t *testing.T) {
   251  		t.Parallel()
   252  
   253  		t.Run("対象ユーザーを従属データごと削除し削除件数を返す", func(t *testing.T) {
   254  			t.Parallel()
   255  
   256  			txm.WithinTx(func(ctx context.Context) {
   257  				drv := driver.New(ctx, testDB)
   258  				targetID := insertPurgeUser(ctx, t, drv, target, &deletedAt)
   259  				insertPurgeUserDependents(ctx, t, drv, targetID, "eeeeeeee-2222-4000-8000-000000000001")
   260  
   261  				got, err := repo.PurgeByIDs(ctx, []uuid.UUID{targetID})
   262  				require.NoError(t, err)
   263  
   264  				assert.Equal(t, int64(1), got)
   265  				assert.Equal(t, int64(0), countPurgeRows(ctx, t, drv, countIdentities, targetID))
   266  				assert.Equal(t, int64(0), countPurgeRows(ctx, t, drv, countRoles, targetID))
   267  				assert.Equal(t, int64(0), countPurgeRows(ctx, t, drv, countCoupons, targetID))
   268  				assert.Equal(t, int64(0), countPurgeRows(ctx, t, drv, countUsers, targetID))
   269  			})
   270  		})
