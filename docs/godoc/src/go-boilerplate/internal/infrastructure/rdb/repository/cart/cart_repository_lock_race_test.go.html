    22  // raceBlockedGracePeriod は、後続役が先行役のロック解放を待たされていることを確認するために待つ時間です。
    23  // 高負荷でこの時間内に問い合わせが届かない場合も「まだ完了していない」側に倒れるため、負荷は偽陽性を生みません。
    24  // 待たされた事実は、後続役がロックを取れた時刻が先行役の解放より後であることでも確かめます。
    25  // 最終状態の一致だけを見ると、ロックが外れていても後続役の往復がこの時間を超えれば同じ観測になり、
    26  // 退行を見逃します。所要時間そのものを閾値と比べないのは、後続役の計測開始が先行役の待機開始より
    27  // 遅れるぶん、ロックが効いていても閾値を下回るためです。
    28  const raceBlockedGracePeriod = 300 * time.Millisecond

    30  // errRollbackRaceTx は、後続役の tx を成否に関わらずロールバックさせるための番兵です。
    31  var errRollbackRaceTx = xerrors.New("rollback race tx")

    33  // Test_lockSerializesConcurrentCartUpdates は、LockByID が同一カートへの並行更新を直列化することを
    34  // 実 DB の 2 トランザクションで検証します。
    35  //
    36  // tx を 2 本同時に生かす必要があるため testkit.WithinTx（tx 1 本 + ロールバック）では表現できません。
    37  // 先行役をコミットさせる必要から、検証用カートは後始末で物理削除します。
    38  func Test_lockSerializesConcurrentCartUpdates(t *testing.T) {
    39  	t.Parallel()
    40  
    41  	testDB := testkit.NewTestDB(t)
    42  	// 検証用カートの作成から 2 本の tx の完了までを、他パッケージの CASCADE TRUNCATE から守る。
    43  	testkit.HoldSuiteSerialization(t, testDB)
    44  
    45  	repo := &repository{tracer: observability.NewMockInfraLayerTracer(t), db: testDB}
    46  
    47  	newTxManager := func() tx.Manager {
    48  		return driver.NewTransactionManager(
    49  			testDB,
    50  			config.NewDatabaseConfig(config.MockConfigForTest(t)),
    51  			logging.NewTestLogger(t),
    52  			system.NewSleeper(),
    53  		)
    54  	}
    55  
    56  	ctx := context.Background()
    57  	target := newGuestCart(t, "rce")
    58  	require.NoError(t, repo.Create(ctx, target))
    59  	t.Cleanup(func() {
    60  		_, cleanupErr := driver.New(ctx, testDB).Exec(ctx, "DELETE FROM carts WHERE id = $1", target.ID())
    61  		require.NoError(t, cleanupErr)
    62  	})
    63  
    64  	firstLocked := make(chan struct{})
    65  	secondDone := make(chan struct{})
    66  	secondResult := make(chan *domaincart.Cart, 1)
    67  	secondAcquiredAt := make(chan time.Time, 1)
    68  	var firstReleasedAt time.Time
    69  
    70  	// 後続役: 先行役がカート行を押さえている間にロックを取りに行き、確定まで待たされる。
    71  	go func() {
    72  		defer close(secondDone)
    73  		<-firstLocked
    74  		_ = newTxManager().Do(ctx, func(txCtx context.Context) error {
    75  			locked, lockErr := repo.LockByID(txCtx, target.ID())
    76  			secondAcquiredAt <- time.Now()
    77  			if lockErr != nil {
    78  				return xerrors.Join(errRollbackRaceTx, lockErr)
    79  			}
    80  			secondResult <- locked
    81  			return errRollbackRaceTx
    82  		})
    83  	}()
    84  
    85  	extended := baseTime.Add(72 * time.Hour)
    86  
    87  	// 先行役: カート行を排他ロックしてから有効期限の延長を確定させる。
    88  	require.NoError(t, newTxManager().Do(ctx, func(txCtx context.Context) error {
    89  		locked, lockErr := repo.LockByID(txCtx, target.ID())
    90  		if lockErr != nil {
    91  			return lockErr
    92  		}
    93  		close(firstLocked)
    94  
    95  		select {
    96  		case <-secondDone:
    97  			t.Error("後続役が先行役のロックを待たずに完了した")
    98  		case <-time.After(raceBlockedGracePeriod):
    99  		}
   100  		firstReleasedAt = time.Now()
   101  
   102  		locked.Touch(extended, 0)
   103  		return repo.Update(txCtx, locked)
   104  	}))
   105  
   106  	<-secondDone
   107  	// 先行役が確定した以上、待たされていた後続役が読み出すのは延長後のカートになる。
   108  	require.Equal(t, extended.UTC(), (<-secondResult).ExpiresAt().UTC())
   109  	// 延長後を観測できたのがロック待ちの結果であることを、取得時刻の前後で裏づける。
   110  	assert.True(t, (<-secondAcquiredAt).After(firstReleasedAt))
   111  }
