   103  		return nil, pgerror.NormalizeReconstructError(err)

   106  	if err != nil {

   111  		UserID:    row.UserID,
   112  		Discount:  discount,
   113  		Scope:     scope,
   114  		ExpiresAt: row.ExpiresAt,
   115  		IssuedAt:  row.IssuedAt,
   116  	}, row.UsedAt)
   117  	if err != nil {
   118  		return nil, pgerror.NormalizeReconstructError(err)
