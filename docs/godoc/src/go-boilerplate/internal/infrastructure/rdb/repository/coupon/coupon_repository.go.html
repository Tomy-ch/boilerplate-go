   141  // rowToCoupon は、永続化された行からクーポンを再構築します。
   142  func rowToCoupon(row gen.Coupons) (*coupon.Coupon, error) {
   143  	discountKind, err := coupon.NewDiscountKind(int(row.DiscountKind))
   144  	if err != nil {
   145  		return nil, pgerror.NormalizeReconstructError(err)
   146  	}
   147  	discount, err := coupon.NewDiscount(discountKind, row.DiscountValue)
   148  	if err != nil {
   149  		return nil, pgerror.NormalizeReconstructError(err)
   150  	}
   151  
   152  	scopeKind, err := coupon.NewScopeKind(int(row.ScopeKind))
   153  	if err != nil {
   154  		return nil, pgerror.NormalizeReconstructError(err)
   155  	}
   156  	scope, err := coupon.NewScope(scopeKind, row.ScopeTargetID)
   157  	if err != nil {
   158  		return nil, pgerror.NormalizeReconstructError(err)
   159  	}
   160  
   161  	c, err := coupon.Reconstruct(row.ID, coupon.Attributes{
   162  		UserID:    row.UserID,
   163  		Discount:  discount,
   164  		Scope:     scope,
   165  		ExpiresAt: row.ExpiresAt,
   166  		IssuedAt:  row.IssuedAt,
   167  	}, row.UsedAt)
   168  	if err != nil {
   169  		return nil, pgerror.NormalizeReconstructError(err)
   170  	}
   171  
   172  	return c, nil
   173  }
