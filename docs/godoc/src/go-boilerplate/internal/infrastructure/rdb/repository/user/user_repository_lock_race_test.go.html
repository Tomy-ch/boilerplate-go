    23  // raceBlockedGracePeriod は、購入役が退会役のロック解放を待たされていることを確認するために待つ時間です。
    24  // 高負荷でこの時間内に問い合わせが届かない場合も「まだ完了していない」側に倒れるため、負荷は偽陽性を生みません。
    25  const raceBlockedGracePeriod = 300 * time.Millisecond

    34  // Test_lockSerializesWithdrawalAgainstPurchase は、#766 の競合順序を実 DB の 2 トランザクションで再現し、
    35  // 退会（排他ロック）が確定した時点で購入の在籍ガード（共有ロック）が拒否へ転じることを検証します。
    36  //
    37  // tx を 2 本同時に生かす必要があるため testkit.WithinTx（tx 1 本 + ロールバック）では表現できません。
    38  // 退会役をコミットさせる必要から、検証用ユーザーは後始末で物理削除します。
    39  func Test_lockSerializesWithdrawalAgainstPurchase(t *testing.T) {
    40  	t.Parallel()
    41  
    42  	testDB := testkit.NewTestDB(t)
    43  	// 検証用ユーザーの作成から 2 本の tx の完了までを、他パッケージの CASCADE TRUNCATE から守る。
    44  	// 部分的にしか守らないと、作成直後の行が消えてロック競合の手前で落ちる。
    45  	testkit.HoldSuiteSerialization(t, testDB)
    46  
    47  	tracer := observability.NewMockInfraLayerTracer(t)
    48  	repo := &repository{tracer: tracer, db: testDB}
    49  	lockRepo := &lockRepository{tracer: tracer, db: testDB}
    50  
    51  	newTxManager := func() tx.Manager {
    52  		return driver.NewTransactionManager(
    53  			testDB,
    54  			config.NewDatabaseConfig(config.MockConfigForTest(t)),
    55  			logging.NewTestLogger(t),
    56  			system.NewSleeper(),
    57  		)
    58  	}
    59  
    60  	ctx := context.Background()
    61  	now := time.Now().UTC()
    62  	targetID := uuidtestkit.NewTestFromSalt(t, "race-withdrawal-user")
    63  	prefectureID, err := uuid.Parse("a03aaec4-3bd6-4bfb-8e47-2fbfa026d344")
    64  	require.NoError(t, err)
    65  
    66  	target, err := user.New(targetID, user.Attributes{
    67  		FirstName:    "Race",
    68  		LastName:     "Target",
    69  		Email:        "race.target@example.com",
    70  		Phone:        "7777777777",
    71  		PrefectureID: prefectureID,
    72  		City:         "新宿区",
    73  		Street:       "9-9-9",
    74  		PostalCode:   "160-0009",
    75  		CreatedAt:    now,
    76  		UpdatedAt:    now,
    77  	})
    78  	require.NoError(t, err)
    79  
    80  	// 共有シードを退会させると他テストへ波及するため、この検証専用のユーザーを立てて後始末で物理削除する。
    81  	require.NoError(t, repo.Create(ctx, target))
    82  	t.Cleanup(func() {
    83  		_, cleanupErr := driver.New(ctx, testDB).Exec(ctx, "DELETE FROM users WHERE id = $1", targetID)
    84  		require.NoError(t, cleanupErr)
    85  	})
    86  
    87  	withdrawalLocked := make(chan struct{})
    88  	guardDone := make(chan struct{})
    89  	guardResult := make(chan error, 1)
    90  
    91  	// 購入役: 退会役がユーザー行を押さえている間に在籍ガードへ入り、退会の確定まで待たされる。
    92  	go func() {
    93  		defer close(guardDone)
    94  		<-withdrawalLocked
    95  		guardResult <- newTxManager().Do(ctx, func(txCtx context.Context) error {
    96  			purchaser, guardErr := lockRepo.LockShareByID(txCtx, targetID)
    97  			if guardErr != nil {
    98  				return xerrors.Join(errRollbackRaceTx, guardErr)
    99  			}
   100  			// 共有ロックを取れた時点の状態が退会済みであることが、直列化の成立を示す。
   101  			if !purchaser.IsActive() {
   102  				return xerrors.Join(errRollbackRaceTx, errPurchaserWithdrawn)
   103  			}
   104  			return errRollbackRaceTx
   105  		})
   106  	}()
   107  
   108  	// 退会役: ユーザー行を排他ロックしてから論理削除を確定させる。
   109  	require.NoError(t, newTxManager().Do(ctx, func(txCtx context.Context) error {
   110  		withdrawing, lockErr := lockRepo.LockByID(txCtx, targetID)
   111  		if lockErr != nil {
   112  			return lockErr
   113  		}
   114  		close(withdrawalLocked)
   115  
   116  		// ロックを握ったまま、購入役が素通りしないことを確かめる。
   117  		select {
   118  		case <-guardDone:
   119  			t.Error("購入役が退会役のロックを待たずに完了した")
   120  		case <-time.After(raceBlockedGracePeriod):
   121  		}
   122  
   123  		if markErr := withdrawing.MarkAsDeleted(time.Now().UTC()); markErr != nil {
   124  			return markErr
   125  		}
   126  		return repo.Update(txCtx, withdrawing)
   127  	}))
   128  
   129  	<-guardDone
   130  	// 退会が確定した以上、待たされていた購入が読み出す購入者は退会済みになっている。
   131  	require.ErrorIs(t, <-guardResult, errPurchaserWithdrawn)
   132  }
