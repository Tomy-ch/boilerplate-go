    57  	for i, c := range terminal {

    60  			return query.DiscontinueImpactReadModel{}, cerr
    61  		}
    62  		codes[i] = converted
    63  	}
    64  
    65  	purchases, err := db.CountDiscontinueImpactInProgressPurchases(ctx, &gen.CountDiscontinueImpactInProgressPurchasesParams{
    66  		ProductID:           productID,
    67  		TerminalStatusCodes: codes,
    68  	})
    69  	if err != nil {
    70  		return query.DiscontinueImpactReadModel{}, pgerror.NormalizeError(err)
    71  	}
    72  
    73  	return query.DiscontinueImpactReadModel{
    74  		AffectedCartCount:       carts,
    75  		AffectedUserCount:       users,
