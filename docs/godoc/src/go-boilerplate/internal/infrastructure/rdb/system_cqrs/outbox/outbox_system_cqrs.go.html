    36  // OrderingKey が空なら ordering_key / ordering_sequence を NULL で書きます。
    37  func (s *store) Insert(ctx context.Context, p outboxbndry.EmitParams) (uuid.UUID, error) {
    38  	ctx, endSpan := s.tracer.Start(ctx)
    39  	defer endSpan()
    40  
    41  	headers := p.Headers
    42  	if len(headers) == 0 {
    43  		headers = []byte("{}")
    44  	}
    45  
    46  	var orderingKey *string
    47  	var orderingSequence *int64
    48  	if p.OrderingKey != "" {
    49  		orderingKey = &p.OrderingKey
    50  		orderingSequence = &p.OrderingSequence
    51  	}
    52  
    53  	db := gen.New(driver.New(ctx, s.db))
    54  	row, err := db.InsertOutbox(ctx, &gen.InsertOutboxParams{
    55  		AggregateType:    p.AggregateType,
    56  		AggregateID:      p.AggregateID,
    57  		EventType:        p.EventType,
    58  		Payload:          p.Payload,
    59  		Headers:          headers,
    60  		DeliveryChannel:  p.Channel.String(),
    61  		OrderingKey:      orderingKey,
    62  		OrderingSequence: orderingSequence,
    63  	})
    64  	if err != nil {
    65  		return uuid.UUID{}, pgerror.NormalizeError(err)
    66  	}
