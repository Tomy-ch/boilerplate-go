   179  // Test_store_ClaimPending_concurrentSkipLocked は、ADR-0056 (skip-locked-outbox-relay) の核である多インスタンス排他
   180  // （FOR UPDATE SKIP LOCKED により別々の並行 tx が同一 pending 行を二重 claim しない）を、
   181  // 2 コネクション（= 2 tx）を並行させて DB レベルで検証します。
   182  //
   183  // testkit の WithinTx は txLock で全 tx を直列化するため、この並行排他は構造的に再現できません。
   184  // そこで idempotency 側の並行テストと同様に driver.NewTransactionManager を直接使い 2 tx を並行走行させます。
   185  // SKIP LOCKED は両 tx から見える committed 行を必要とするため、本テストのみ非並列にし
   186  // （他テストの「pending が空」アサートと committed 行が衝突しないよう sequential フェーズで完結させる）、
   187  // 専用 aggregate_id の行を投入・後始末します。
   188  //
   189  //nolint:paralleltest // committed fixture が並列テストの全体空アサートと衝突するため非並列
   190  func Test_store_ClaimPending_concurrentSkipLocked(t *testing.T) {
   191  	testDB := testkit.NewTestDB(t)
   192  	lt := observability.NewMockInfraLayerTracer(t)
   193  	s := &store{tracer: lt, db: testDB}
   194  
   195  	dbCfg := config.NewDatabaseConfig(config.MockConfigForTest(t))
   196  	txm := driver.NewTransactionManager(testDB, dbCfg, logging.NewTestLogger(t), system.NewSleeper())
   197  
   198  	const aggregateID = "skiplocked-concurrency-test"
   199  
   200  	// 後始末: 本テストが commit した専用行を必ず削除する（sequential フェーズ内で完結させる）。
   201  	t.Cleanup(func() {
   202  		_ = txm.Do(context.Background(), func(ctx context.Context) error {
   203  			_, err := driver.New(ctx, testDB).Exec(ctx, "DELETE FROM outbox WHERE aggregate_id = $1", aggregateID)
   204  			return err
   205  		})
   206  	})
   207  
   208  	// 2 件の pending 行を commit し、両 tx から見えるようにする。
   209  	params := emitParams()
   210  	params.AggregateID = aggregateID
   211  	require.NoError(t, txm.Do(context.Background(), func(ctx context.Context) error {
   212  		for range 2 {
   213  			if _, err := s.Insert(ctx, params); err != nil {
   214  				return err
   215  			}
   216  		}
   217  		return nil
   218  	}))
   219  
   220  	// 全 skip: 保持側が 2 行すべてをロックする間、競合側は 0 行（すべて skip される）。
   221  	assertSkipLockedContention(t, txm.Do, s, 10, 2, 0)
   222  
   223  	// release 後は行が pending へ戻り、新規 tx から再 claim できる（ロック解放後は取得可能）。
   224  	// SELECT のみのため rollback で pending のまま残し、次の部分 skip フェーズで再利用する。
   225  	reclaimErr := txm.Do(context.Background(), func(ctx context.Context) error {
   226  		claimed, err := s.ClaimPending(ctx, outboxbndry.ChannelHTTP, 10)
   227  		if err != nil {
   228  			return err
   229  		}
   230  		assert.Len(t, claimed, 2, "release 後は保持されていた行を再 claim できる")
   231  		return errRollbackForTest
   232  	})
   233  	require.ErrorIs(t, reclaimErr, errRollbackForTest)
   234  
   235  	// 部分 skip: 保持側が 1 行だけロックし、競合側は残り 1 行を取得できる（ロック済のみ skip）。
   236  	assertSkipLockedContention(t, txm.Do, s, 1, 1, 1)
   237  }
   238  
   239  // assertSkipLockedContention は、保持側 tx が holderLimit 件をロックして保持する間に競合側 tx を走らせ、
   240  // 保持側が wantHolder 件・競合側が wantContender 件を claim することを検証します（SKIP LOCKED の排他/部分 skip）。
   241  // 保持側は行を残さないよう最後に rollback します。do は TransactionManager.Do を渡します。
   242  func assertSkipLockedContention(
   243  	t *testing.T,
   244  	do func(context.Context, func(context.Context) error) error,
   245  	s *store,
   246  	holderLimit int32,
   247  	wantHolder, wantContender int,
   248  ) {
   249  	t.Helper()
   250  
   251  	holderClaimed := make(chan struct{})
   252  	release := make(chan struct{})
   253  	holderDone := make(chan error, 1)
   254  
   255  	var once sync.Once
   256  	rel := func() { once.Do(func() { close(release) }) }
   257  	t.Cleanup(rel) // 失敗時も保持側 goroutine をリークさせない
   258  
   259  	go func() {
   260  		holderDone <- do(context.Background(), func(ctx context.Context) error {
   261  			claimed, err := s.ClaimPending(ctx, outboxbndry.ChannelHTTP, holderLimit)
   262  			if err != nil {
   263  				return err
   264  			}
   265  			if len(claimed) != wantHolder {
   266  				return xerrors.New(fmt.Sprintf("holder: expected to claim %d rows, got %d", wantHolder, len(claimed)))
   267  			}
   268  			close(holderClaimed)
   269  			<-release
   270  			return errRollbackForTest
   271  		})
   272  	}()
   273  
   274  	select {
   275  	case <-holderClaimed:
   276  	case err := <-holderDone:
   277  		require.NoError(t, err, "保持側 tx が claim 前に失敗した")
   278  		return
   279  	}
   280  
   281  	contenderErr := do(context.Background(), func(ctx context.Context) error {
   282  		claimed, err := s.ClaimPending(ctx, outboxbndry.ChannelHTTP, 10)
   283  		if err != nil {
   284  			return err
   285  		}
   286  		assert.Len(t, claimed, wantContender, "SKIP LOCKED: ロック済行は skip し、未ロック行のみ取得する")
   287  		return nil
   288  	})
   289  	require.NoError(t, contenderErr)
   290  
   291  	rel()
   292  	require.ErrorIs(t, <-holderDone, errRollbackForTest)
   293  }
