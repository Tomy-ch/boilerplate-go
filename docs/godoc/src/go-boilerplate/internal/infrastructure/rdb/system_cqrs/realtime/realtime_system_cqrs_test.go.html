   162  // Test_allocator_Allocate_concurrentSerialization は、独立した 2 つのトランザクションが同一ストリームを
   163  // 同時に採番したとき、行ロックにより直列化され gap も重複も生じないことを検証します。
   164  //
   165  // testkit の WithinTx は全 tx を直列化するため、この競合は構造的に再現できません。
   166  // outbox 側の SKIP LOCKED テストと同様に driver.NewTransactionManager を直接使い 2 tx を並行させます。
   167  // 保持側が commit するため専用 stream_id の行を投入し、後始末で必ず削除します。
   168  //
   169  //nolint:paralleltest // commit した fixture が他テストと衝突しないよう非並列にする
   170  func Test_allocator_Allocate_concurrentSerialization(t *testing.T) {
   171  	testDB := testkit.NewTestDB(t)
   172  	a := &allocator{tracer: observability.NewMockInfraLayerTracer(t), db: testDB}
   173  
   174  	dbCfg := config.NewDatabaseConfig(config.MockConfigForTest(t))
   175  	txm := driver.NewTransactionManager(testDB, dbCfg, logging.NewTestLogger(t), system.NewSleeper())
   176  
   177  	const streamID = realtimebndry.StreamID("stream-allocate-concurrent")
   178  
   179  	t.Cleanup(func() {
   180  		_ = txm.Do(context.Background(), func(ctx context.Context) error {
   181  			_, err := driver.New(ctx, testDB).
   182  				Exec(ctx, "DELETE FROM realtime_stream_sequences WHERE stream_id = $1", streamID.String())
   183  			return err
   184  		})
   185  	})
   186  
   187  	allocated := make(chan struct{})
   188  	release := make(chan struct{})
   189  	holderDone := make(chan error, 1)
   190  
   191  	var once sync.Once
   192  	rel := func() { once.Do(func() { close(release) }) }
   193  	t.Cleanup(rel) // 失敗時も保持側 goroutine をリークさせない
   194  
   195  	// 保持側: 採番して commit まで行ロックを保持する。
   196  	go func() {
   197  		holderDone <- txm.Do(context.Background(), func(ctx context.Context) error {
   198  			seq, err := a.Allocate(ctx, streamID)
   199  			if err != nil {
   200  				return err
   201  			}
   202  			if seq != realtimebndry.Sequence(1) {
   203  				return xerrors.New(fmt.Sprintf("holder: expected sequence 1, got %d", seq))
   204  			}
   205  			close(allocated)
   206  			<-release
   207  			return nil
   208  		})
   209  	}()
   210  
   211  	select {
   212  	case <-allocated:
   213  	case err := <-holderDone:
   214  		require.NoError(t, err, "保持側 tx が採番前に失敗した")
   215  		return
   216  	}
   217  
   218  	contenderDone := make(chan realtimebndry.Sequence, 1)
   219  	go func() {
   220  		// 保持側の行ロックが解けるまでブロックし、解けた後に続きの位置を得る。
   221  		var seq realtimebndry.Sequence
   222  		if err := txm.Do(context.Background(), func(ctx context.Context) error {
   223  			var aerr error
   224  			seq, aerr = a.Allocate(ctx, streamID)
   225  			return aerr
   226  		}); err != nil {
   227  			seq = 0
   228  		}
   229  		contenderDone <- seq
   230  	}()
   231  
   232  	rel()
   233  	require.NoError(t, <-holderDone)
   234  
   235  	// 直列化されていれば競合側は 1 を再取得せず 2 を得る（gap も重複も無い）。
   236  	assert.Equal(t, realtimebndry.Sequence(2), <-contenderDone)
   237  }
