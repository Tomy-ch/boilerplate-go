    72  // WithinTx は、テスト用のトランザクションマネージャーでトランザクションを開始し、引数で渡されたfnを実行し、最後にロールバックします。
    73  //
    74  // 使用例:
    75  //
    76  //	txm.WithinTx(func(ctx context.Context) {
    77  //	  // トランザクション内の処理
    78  //	})
    79  func (t *testTxRunner) WithinTx(fn func(ctx context.Context)) {
    80  	t.t.Helper()
    81  
    82  	t.WithinTxE(func(ctx context.Context) error {
    83  		fn(ctx)
    84  		return nil
    85  	})
    86  }

    88  // WithinTxE は、fn がエラーを返せる WithinTx です。fn が返したエラーはトランザクションマネージャーへ
    89  // 伝わるため、deadlock（40P01）や serialization failure（40001）は tx ごと再試行されます。
    90  //
    91  // WithinTx との違いは、fn が失敗をどう表明するかだけです。fn の中で require を使うと testify は
    92  // runtime.Goexit で goroutine を落とし、トランザクションマネージャーは戻り値を受け取れないため、
    93  // リトライ可能と宣言済みのエラーであっても再試行されません。一時障害を再試行させたい文
    94  // （テーブル全体をロックする CASCADE TRUNCATE など）は、require ではなくエラーを返してください。
    95  //
    96  // fn が nil を返した場合もトランザクションはロールバックされます（テストは DB を汚さない）。
    97  func (t *testTxRunner) WithinTxE(fn func(ctx context.Context) error) {
    98  	t.t.Helper()
    99  
   100  	txLock.Lock()
   101  	defer txLock.Unlock()
   102  
   103  	baseCtx := context.Background()
   104  
   105  	err := t.inner.Do(baseCtx, func(txCtx context.Context) error {
   106  		if lockErr := lockSuiteSerialization(txCtx, driver.New(txCtx, t.db)); lockErr != nil {
   107  			return lockErr
   108  		}
   109  		if fnErr := fn(txCtx); fnErr != nil {
   110  			return fnErr
   111  		}
   112  		return errRollbackForTest
   113  	})
   114  
   115  	if xerrors.Is(err, errRollbackForTest) {
   116  		return
   117  	}
   118  	require.NoError(t.t, err)
   119  }

   121  // HoldSuiteSerialization は、呼び出したテストが終わるまでスイート全体の直列化を占有します。
   122  // 占有している間、他パッケージのテスト（別プロセス）が張る CASCADE TRUNCATE は走りません。
   123  // 解放は t.Cleanup で行われます。
   124  //
   125  // 占有は専用のトランザクションで行い、テスト自身のトランザクションはこの直列化に参加しません。
   126  // 参加させると自分の占有を待つことになり、進みません。
   127  //
   128  // 通常のテストは WithinTx が同じ直列化を内部で行うため、これを呼ぶ必要はありません。呼ぶのは、
   129  // トランザクションを 2 本同時に生かす検証（ロック競合の再現など）のように、WithinTx の
   130  // 「1 本 + ロールバック」では表現できないテストに限られます。
   131  func HoldSuiteSerialization(t *testing.T, db driver.DatabaseDriver) {
   132  	t.Helper()
   133  
   134  	release, err := holdSuiteSerialization(context.Background(), db)
   135  	require.NoError(t, err)
   136  	t.Cleanup(release)
   137  }
   138  
   139  // holdSuiteSerialization は、直列化を占有し、解放する関数を返します。
   140  // 占有に失敗した場合は何も保持せずにエラーだけを返します。
   141  //
   142  // 取得と解放をここへ閉じてください。呼び出し側で占有と後始末の登録が別々の文に分かれると、
   143  // その間で失敗したときに解放されない経路ができます。
   144  func holdSuiteSerialization(ctx context.Context, db driver.DatabaseDriver) (func(), error) {
   145  	txLock.Lock()
   146  
   147  	holder, err := db.Begin(ctx)
   148  	if err != nil {
   149  		txLock.Unlock()
   150  		return nil, err
   151  	}
   152  
   153  	if err := lockSuiteSerialization(ctx, holder); err != nil {
   154  		_ = holder.Rollback(ctx)
   155  		txLock.Unlock()
   156  		return nil, err
   157  	}
   158  
   159  	return func() {
   160  		_ = holder.Rollback(ctx)
   161  		txLock.Unlock()
   162  	}, nil
   163  }
   164  
   165  // lockSuiteSerialization は、q が属するトランザクションでスイート直列化用の advisory lock を取ります。
   166  func lockSuiteSerialization(ctx context.Context, q driver.DBTX) error {
   167  	// advisory lock 待ちは lock_timeout(10s) の対象で、高負荷で超過すると 55P03（非リトライ）で
   168  	// 落ちる。取得の間だけ無効化する（直列化ゆえテーブルロック競合は起きず待ち詰まりの懸念はない）。
   169  	if _, err := q.Exec(ctx, "SET LOCAL lock_timeout = 0"); err != nil {
   170  		return err
   171  	}
   172  	if _, err := q.Exec(ctx, "SELECT pg_advisory_xact_lock($1)", txAdvisoryLockKey); err != nil {
   173  		return err
   174  	}
   175  	return nil
   176  }
