   212  // requireTxLockAvailable は、txLock が取得可能になるまで待ちます。
   213  // 同居する並列テストが一時的に握るため即時とは限らず、解放漏れだけが時間切れになります。
   214  //
   215  // 占有の前後どちらでも呼んでください。前で呼ばないと、解放漏れが起きたとき後続が
   216  // txLock.Lock() で止まる窓が広がります。並列サブテストの出力は親が終わるまで出ないため、
   217  // 止まった側に巻き込まれると失敗の内容が消えます。
   218  //
   219  // ロックの強制解除はしません。解放漏れが再発すれば掴みに行った側は止まったままで、
   220  // パッケージは go test のタイムアウトを待ちます。
   221  func requireTxLockAvailable(t *testing.T, msg string) {
   222  	t.Helper()
   223  
   224  	require.Eventually(t, func() bool {
   225  		if !txLock.TryLock() {
   226  			return false
   227  		}
   228  		txLock.Unlock()
   229  		return true
   230  	}, 15*time.Second, 50*time.Millisecond, msg)
   231  }

   251  			assert.False(t, acquired)
   252  		})
   253  
   254  		t.Run("占有したテストが終わると解放される", func(t *testing.T) {
   255  			t.Parallel()
   256  
   257  			// 解放は t.Cleanup で行われるため、占有を子テストへ閉じ込めて完了させ、
   258  			// 親から解放後の状態を観測する。解放されないとスイート全体が止まる。
   259  			//nolint:paralleltest // Cleanup を親の観測より先に走らせるため逐次実行する
   260  			t.Run("占有", func(t *testing.T) {
   261  				HoldSuiteSerialization(t, db)
   262  			})
   263  
   264  			ctx := context.Background()
   265  			probe, err := db.Begin(ctx)
   266  			require.NoError(t, err)
   267  			t.Cleanup(func() { _ = probe.Rollback(ctx) })
   268  
   269  			// 同じキーは他の並列テストの WithinTx も一時的に握るため、取得できるまで待つ。
   270  			// 解放されないなら永久に取得できず、ここで打ち切られる。
   271  			// probe 自身の解放漏れを避けるため、セッション単位ではなく tx 単位で取得する。
   272  			require.Eventually(t, func() bool {
   273  				var acquired bool
   274  				if err := probe.QueryRow(ctx, "SELECT pg_try_advisory_xact_lock($1)", txAdvisoryLockKey).Scan(&acquired); err != nil {
   275  					return false
   276  				}
   277  				return acquired
   278  			}, 15*time.Second, 50*time.Millisecond, "占有したテストが終わっても解放されない")
   279  		})
   280  	})

   283  func Test_lockSuiteSerialization(t *testing.T) {
   284  	t.Parallel()
   285  
   286  	db := NewTestDB(t)
   287  	runner := NewTestTransactionRunner(t)
   288  
   289  	t.Run("正常系", func(t *testing.T) {
   290  		t.Parallel()
   291  
   292  		t.Run("同一トランザクションからの再取得は待たずに成立する", func(t *testing.T) {
   293  			t.Parallel()
   294  			// WithinTx が既に同じキーを保持している。advisory lock は同一トランザクション内で再入可能。
   295  			runner.WithinTx(func(ctx context.Context) {
   296  				require.NoError(t, lockSuiteSerialization(ctx, driver.New(ctx, db)))
   297  			})
   298  		})
   299  
   300  		t.Run("他トランザクションが保持中でも呼び出し側のlock_timeoutで打ち切られない", func(t *testing.T) {
   301  			t.Parallel()
   302  
   303  			ctx := context.Background()
   304  
   305  			// 保持側。WithinTx はこの直列化に参加するため使えず、専用の tx で握る。
   306  			// 解放済みの tx への Rollback は ErrTxClosed を返すだけなので、後始末は無条件で行う。
   307  			holder, err := db.Begin(ctx)
   308  			require.NoError(t, err)
   309  			t.Cleanup(func() { _ = holder.Rollback(ctx) })
   310  
   311  			// このキーは全パッケージが共有するため、保持側自身も接続既定の lock_timeout(10s)
   312  			// で打ち切られうる。検証対象と同じ理由でここでも無効化する。
   313  			_, err = holder.Exec(ctx, "SET LOCAL lock_timeout = 0")
   314  			require.NoError(t, err)
   315  			_, err = holder.Exec(ctx, "SELECT pg_advisory_xact_lock($1)", txAdvisoryLockKey)
   316  			require.NoError(t, err)
   317  
   318  			type waitResult struct {
   319  				err     error
   320  				elapsed time.Duration
   321  			}
   322  			waiterPID := make(chan int32, 1)
   323  			waiterDone := make(chan waitResult, 1)
   324  			go func() {
   325  				waiter, beginErr := db.Begin(ctx)
   326  				if beginErr != nil {
   327  					waiterDone <- waitResult{err: beginErr}
   328  					return
   329  				}
   330  				defer func() { _ = waiter.Rollback(ctx) }()
   331  
   332  				var pid int32
   333  				if scanErr := waiter.QueryRow(ctx, "SELECT pg_backend_pid()").Scan(&pid); scanErr != nil {
   334  					waiterDone <- waitResult{err: scanErr}
   335  					return
   336  				}
   337  				if _, execErr := waiter.Exec(ctx, "SET LOCAL lock_timeout = '50ms'"); execErr != nil {
   338  					waiterDone <- waitResult{err: execErr}
   339  					return
   340  				}
   341  
   342  				waiterPID <- pid
   343  				startedAt := time.Now()
   344  				waiterDone <- waitResult{err: lockSuiteSerialization(ctx, waiter), elapsed: time.Since(startedAt)}
   345  			}()
   346  
   347  			var pid int32
   348  			select {
   349  			case pid = <-waiterPID:
   350  			case got := <-waiterDone:
   351  				require.NoError(t, got.err, "待機側がロック待ちへ入る前に失敗した")
   352  			}
   353  
   354  			// 待機側がロック待ちへ入る前に解放すると、lock_timeout が効いたままでも
   355  			// 素通りして偽陽性で通る。待ちに入ったことを確認してから解放する。
   356  			requireAdvisoryLockWait(t, db, pid)
   357  			time.Sleep(lockTimeoutProbeHold)
   358  			require.NoError(t, holder.Rollback(ctx))
   359  
   360  			got := <-waiterDone
   361  			require.NoError(t, got.err, "無効化されていなければ lock_timeout(50ms) で 55P03 になる")
   362  			assert.Greater(t, got.elapsed, waiterLockTimeout, "解放を待たずに取得しており、待ち合わせを検証できていない")
   363  		})
   364  	})
   365  
   366  	t.Run("異常系", func(t *testing.T) {
   367  		t.Parallel()
   368  
   369  		t.Run("lock_timeoutの無効化に失敗した場合、advisory lockを取りに行かない", func(t *testing.T) {
   370  			t.Parallel()
   371  
   372  			q := mock_driver.NewMockDBTX(gomock.NewController(t))
   373  			// 1 文目が失敗したら 2 文目は発行しない。EXPECT を 1 つだけ置くことがその検証になる。
   374  			q.EXPECT().Exec(gomock.Any(), "SET LOCAL lock_timeout = 0").Return(pgconn.CommandTag{}, errExecFailed)
   375  
   376  			require.ErrorIs(t, lockSuiteSerialization(context.Background(), q), errExecFailed)
   377  		})
   378

   380  			t.Parallel()
   381  
   382  			q := mock_driver.NewMockDBTX(gomock.NewController(t))
   383  			q.EXPECT().Exec(gomock.Any(), "SET LOCAL lock_timeout = 0").Return(pgconn.CommandTag{}, nil)
   384  			q.EXPECT().Exec(gomock.Any(), "SELECT pg_advisory_xact_lock($1)", txAdvisoryLockKey).

   392  // requireAdvisoryLockWait は、pid のセッションがスイート直列化キーの advisory lock 待ちへ
   393  // 入るまで待ちます。このキーは全パッケージが共有するため、pid で絞らないと無関係な
   394  // セッションの待ちを自分のものと取り違えます。
   395  func requireAdvisoryLockWait(t *testing.T, db driver.DatabaseDriver, pid int32) {
   396  	t.Helper()
   397  
   398  	ctx := context.Background()
   399  	require.Eventually(t, func() bool {
   400  		var waiting bool
   401  		if err := driver.New(ctx, db).QueryRow(ctx,
   402  			`SELECT EXISTS (SELECT 1 FROM pg_locks WHERE locktype = 'advisory' AND NOT granted
   403  				AND pid = $1 AND (classid::bigint << 32) + objid::bigint = $2 AND objsubid = 1)`,
   404  			pid, txAdvisoryLockKey).Scan(&waiting); err != nil {
   405  			return false
   406  		}
   407  		return waiting
   408  	}, 10*time.Second, 20*time.Millisecond, "advisory lock 待ちに入らなかった")
   409  }
