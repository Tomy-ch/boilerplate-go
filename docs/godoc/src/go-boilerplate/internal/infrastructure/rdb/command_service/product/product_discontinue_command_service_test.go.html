    87  func newIssueParams(t *testing.T, productID uuid.UUID) command.IssueDiscontinuationCouponsParams {

   101  		ProductID: productID,

   103  		Discount:  discount,
   104  		ExpiresAt: issuedAt.Add(30 * 24 * time.Hour),
   105  		IssuedAt:  issuedAt,
   106  	}
