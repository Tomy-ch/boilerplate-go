    33  // IssueDiscontinuationCoupons は、対象商品の明細を持つカートの所有者へクーポンを一括発行します。
    34  //
    35  // 分割の正当性・往復回数・行構築の契約は docs/spec/usecase/coupon.md の Command Service と
    36  // ADR-0034 (commandservice-atomicity-criterion) の Worked instances を参照。
    37  //
    38  // 空配列を unnest しても 0 行ですが、無駄な往復を避けるため受給者が 0 人なら早期に返します。
    39  func (s *commandService) IssueDiscontinuationCoupons(
    40  	ctx context.Context,
    41  	params command.IssueDiscontinuationCouponsParams,
    42  ) (command.IssueDiscontinuationCouponsResult, error) {
    43  	ctx, endSpan := s.tracer.Start(ctx)
    44  	defer endSpan()
    45  
    46  	db := gen.New(driver.New(ctx, s.db))
    47  
    48  	affectedCarts, err := db.CountDiscontinueAffectedCarts(ctx, params.ProductID)
    49  	if err != nil {
    50  		return command.IssueDiscontinuationCouponsResult{}, pgerror.NormalizeError(err)
    51  	}
    52  
    53  	recipients, err := db.SelectDiscontinueCouponRecipients(ctx, params.ProductID)
    54  	if err != nil {
    55  		return command.IssueDiscontinuationCouponsResult{}, pgerror.NormalizeError(err)
    56  	}
    57  	if len(recipients) == 0 {
    58  		return command.IssueDiscontinuationCouponsResult{AffectedCartCount: affectedCarts}, nil
    59  	}
    60  
    61  	coupons := make(coupon.Coupons, len(recipients))
    62  	ids := make([]uuid.UUID, len(recipients))
    63  	for i, recipient := range recipients {
    64  		id, ierr := uuid.New()
    65  		if ierr != nil {
    66  			return command.IssueDiscontinuationCouponsResult{}, ierr
    67  		}
    68  
    69  		issuedCoupon, cerr := coupon.New(id, coupon.Attributes{
    70  			UserID:    recipient,
    71  			Discount:  params.Discount,
    72  			Scope:     params.Scope,
    73  			ExpiresAt: params.ExpiresAt,
    74  			IssuedAt:  params.IssuedAt,
    75  		})
    76  		if cerr != nil {
    77  			return command.IssueDiscontinuationCouponsResult{}, cerr
    78  		}
    79  
    80  		coupons[i] = issuedCoupon
    81  		ids[i] = issuedCoupon.ID()
    82  	}
    83  
    84  	// 共有列は検証済みの集約から取ります。素の params は使いません
    85  	// （理由は internal/infrastructure/rdb/README.md の command_service を参照）。
    86  	template := coupons[0]
    87  
    88  	discountKind, err := safecast.IntToInt16(template.Discount().Kind().Code())
    89  	if err != nil {
    90  		return command.IssueDiscontinuationCouponsResult{}, err
    91  	}
    92  	scopeKind, err := safecast.IntToInt16(template.Scope().Kind().Code())
    93  	if err != nil {
    94  		return command.IssueDiscontinuationCouponsResult{}, err
    95  	}
    96  
    97  	issued, err := db.InsertCoupons(ctx, &gen.InsertCouponsParams{
    98  		Ids:           ids,
    99  		UserIds:       recipients,
   100  		DiscountKind:  discountKind,
   101  		DiscountValue: template.Discount().Value(),
   102  		ScopeKind:     scopeKind,
   103  		ScopeTargetID: template.Scope().TargetID(),
   104  		ExpiresAt:     template.ExpiresAt(),
   105  		IssuedAt:      template.IssuedAt(),
   106  	})
   107  	if err != nil {
   108  		return command.IssueDiscontinuationCouponsResult{}, pgerror.NormalizeError(err)
   109  	}
   110  
   111  	return command.IssueDiscontinuationCouponsResult{
   112  		AffectedCartCount: affectedCarts,
   113  		AffectedUserCount: int64(len(recipients)),
   114  		IssuedCouponCount: issued,
   115  	}, nil
   116  }
