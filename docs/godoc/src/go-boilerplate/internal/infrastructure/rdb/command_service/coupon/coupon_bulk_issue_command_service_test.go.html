    56  func newBulkIssueParams(t *testing.T) command.IssuePromotionalCouponsParams {
    57  	t.Helper()
    58  	discount, err := domaincoupon.NewRateDiscount(decimaltestkit.MustParse(t, "0.15"))
    59  	require.NoError(t, err)
    60  
    61  	// timestamptz はマイクロ秒までしか保たないため、往復で一致させるにはそこへ丸めて渡す。
    62  	issuedAt := time.Now().Truncate(time.Microsecond)
    63  
    64  	return command.IssuePromotionalCouponsParams{
    65  		Scope:     domaincoupon.NewAllScope(),
    66  		Discount:  discount,
    67  		ExpiresAt: issuedAt.Add(30 * 24 * time.Hour),
    68  		IssuedAt:  issuedAt,
    69  	}
    70  }
