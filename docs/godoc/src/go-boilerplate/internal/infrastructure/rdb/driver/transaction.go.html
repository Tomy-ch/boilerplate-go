    72  // Do は、pkg/retry で tx 全体を包み、指数 backoff + full jitter で有限回まで再試行します。
    73  // 何を再試行の対象とするか、fn に課される冪等性契約、nested が 1 回だけであることは
    74  // tx.Manager（internal/usecase/boundary/tx）の doc が述べています。
    75  func (t *txManager) Do(ctx context.Context, fn func(ctx context.Context) error) error {
    76  	if _, ok := ctx.Value(txKey{}).(pgx.Tx); ok {
    77  		return fn(ctx) // nested: 外側の tx をそのまま使い 1 回だけ実行する（ADR-0035）
    78  	}
    79  
    80  	err := retry.Do(ctx, t.sleeper,
    81  		retry.Policy{MaxAttempts: t.maxAttempts, Backoff: t.backoff.Duration},
    82  		pgerror.IsRetryableTxError,
    83  		func(c context.Context) error { return t.doOnce(c, fn) },
    84  	)
    85  	return normalizeTxResult(err)
    86  }

    88  // normalizeTxResult は、リトライ後の最終エラーを apperror へ正規化します。
    89  // begin / commit 由来の DB エラー（生 pg / 接続）および context キャンセル・期限超過を
    90  // apperror へ写像します。fn が返したエラー（apperror や rollback sentinel 等）は
    91  // そのまま通し、呼出側のエラー判定に干渉しません。
    92  func normalizeTxResult(err error) error {
    93  	if err == nil {
    94  		return nil
    95  	}
    96  	var pgErr *pgconn.PgError
    97  	if xerrors.As(err, &pgErr) || pgerror.IsUnavailable(err) {
    98  		return pgerror.NormalizeError(err)
    99  	}
   100  	if xerrors.Is(err, context.Canceled) || xerrors.Is(err, context.DeadlineExceeded) {
   101  		return pgerror.NormalizeError(err)
   102  	}
   103  	return err
   104  }

   106  // doOnce は、1 回分のトランザクション（begin → fn → commit / rollback）を実行します。
   107  //
   108  // エラーは正規化せず生のまま返します。IsRetryableTxError が生 SQLSTATE を errors.As で
   109  // 参照できる必要があるためで、apperror への写像は呼出元 Do がリトライ後に 1 度だけ行います
   110  // （chain に PgError が残る理由は ADR-0035）。
   111  func (t *txManager) doOnce(ctx context.Context, fn func(ctx context.Context) error) error {
   112  	tx, err := t.db.Begin(ctx)
   113  	if err != nil {
   114  		return err
   115  	}
   116  
   117  	completed := false
   118  	defer func(ctx context.Context) {
   119  		if completed {
   120  			return
   121  		}
   122  		if p := recover(); p != nil {
   123  			t.rollback(ctx, tx, logging.Any("panic", p))
   124  			panic(p)
   125  		}
   126  		// fn が runtime.Goexit（testify の FailNow 等）で中断した場合の後始末。
   127  		t.rollback(ctx, tx)
   128  	}(ctx)
   129  
   130  	ctx = withTx(ctx, tx)
   131  
   132  	if err := fn(ctx); err != nil {
   133  		t.rollback(ctx, tx, logging.Error(logging.OriginalErrorKey, err))
   134  		completed = true
   135  		return err
   136  	}
   137  
   138  	if err := tx.Commit(ctx); err != nil {
   139  		completed = true
   140  		return err
   141  	}
   142  
   143  	completed = true
   144  	return nil
   145  }

   147  // rollback はロールバックし、失敗時に fields を併記してログを残します。
   148  func (t *txManager) rollback(ctx context.Context, tx pgx.Tx, fields ...*logging.Field) {
   149  	cleanupCtx, cancel := context.WithTimeout(context.WithoutCancel(ctx), cleanupTimeout)
   150  	defer cancel()
   151  
   152  	if pgErr := tx.Rollback(cleanupCtx); pgErr != nil {
   153  		logFields := append([]*logging.Field{logging.Error(logging.ErrorKey, pgErr)}, fields...)
   154  		t.logger.CallerSkip(callerSkipCount).Named("TransactionManager").Error(
   155  			ctx, "Failed to rollback transaction", logFields...,
   156  		)
   157  	}
   158  }
