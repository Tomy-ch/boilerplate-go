   200  func Test_queryTracer_TraceQueryStart(t *testing.T) {
   201  	t.Parallel()
   202  
   203  	t.Run("正常系", func(t *testing.T) {
   204  		t.Parallel()
   205  
   206  		t.Run("開始時のspanIDがparentSpanIDとして終了処理経路まで引き継がれる", func(t *testing.T) {
   207  			t.Parallel()
   208  
   209  			ctx, end := observability.NewStubSpanContext(t)
   210  			defer end()
   211  
   212  			wantSpanID := observability.ExtractTraceContext(ctx).SpanID()
   213  			require.NotEmpty(t, wantSpanID)
   214  
   215  			qt, mockLogger := newTestQueryTracer(t)
   216  			qt.slowThreshold = time.Second
   217  			mockLogger.EXPECT().Info(gomock.Any(), "DB query completed", gomock.Any()).Times(1)
   218  
   219  			// TraceQueryStart が開始時点の spanID を queryLogData.parentSpanID に取り込む。
   220  			startedCtx := qt.TraceQueryStart(ctx, nil, pgx.TraceQueryStartData{SQL: "SELECT 1"})
   221  
   222  			ld, ok := startedCtx.Value(queryLogKey{}).(queryLogData)
   223  			require.True(t, ok)
   224  			assert.Equal(t, wantSpanID, ld.parentSpanID)
   225  
   226  			// 手組みの context を使わず TraceQueryStart の出力 context をそのまま
   227  			// TraceQueryEnd へ渡し、parentSpanID が終了処理経路へ引き継がれることを確認する。
   228  			qt.TraceQueryEnd(startedCtx, nil, pgx.TraceQueryEndData{})
   229  		})
   230  
   231  		t.Run("正常終了ログに parent_span_id フィールドが乗る", func(t *testing.T) {
   232  			t.Parallel()
   233  
   234  			ctx, end := observability.NewStubSpanContext(t)
   235  			defer end()
   236  
   237  			// 開始時点の span ID が parent_span_id としてログに乗ることを値レベルで検証する。
   238  			wantParentSpanID := observability.ExtractTraceContext(ctx).SpanID()
   239  			require.NotEmpty(t, wantParentSpanID)
   240  
   241  			logger, observed := logging.NewObservedTestLogger(t)
   242  			qt := newObservedQueryTracer(t, logger)
   243  			qt.slowThreshold = time.Second
   244  
   245  			startedCtx := qt.TraceQueryStart(ctx, nil, pgx.TraceQueryStartData{SQL: "SELECT 1"})
   246  			qt.TraceQueryEnd(startedCtx, nil, pgx.TraceQueryEndData{})
   247  
   248  			entries := observed.FilterMessage("DB query completed").All()
   249  			require.Len(t, entries, 1)
   250  			assert.Equal(t, wantParentSpanID, entries[0].ContextMap()[logging.ParentSpanIDKey])
   251  		})
   252  	})

   256  	t.Parallel()
   257  
   258  	t.Run("正常系", func(t *testing.T) {
   259  		t.Parallel()
   260  
   261  		t.Run("maskArgs が true の場合は args 件数フィールドが付かない", func(t *testing.T) {
   262  			t.Parallel()
   263  
   264  			ld := queryLogData{
   265  				sql:   "SELECT $1",
   266  				args:  []any{"secret"},
   267  				start: time.Now(),
   268  			}
   269  
   270  			maskedLogger, maskedLogs := logging.NewObservedTestLogger(t)
   271  			qtMasked := newObservedQueryTracer(t, maskedLogger)
   272  			qtMasked.maskArgs = true
   273  			qtMasked.slowThreshold = time.Second
   274  			qtMasked.TraceQueryEnd(
   275  				context.WithValue(context.Background(), queryLogKey{}, ld), nil, pgx.TraceQueryEndData{},
   276  			)
   277  
   278  			plainLogger, plainLogs := logging.NewObservedTestLogger(t)
   279  			qtPlain := newObservedQueryTracer(t, plainLogger)
   280  			qtPlain.maskArgs = false
   281  			qtPlain.slowThreshold = time.Second
   282  			qtPlain.TraceQueryEnd(
   283  				context.WithValue(context.Background(), queryLogKey{}, ld), nil, pgx.TraceQueryEndData{},
   284  			)
   285  
   286  			maskedEntries := maskedLogs.FilterMessage("DB query completed").All()
   287  			require.Len(t, maskedEntries, 1)
   288  			plainEntries := plainLogs.FilterMessage("DB query completed").All()
   289  			require.Len(t, plainEntries, 1)
   290  
   291  			// マスク時は args 件数キーが付かず、非マスク時は付く。
   292  			assert.NotContains(t, maskedEntries[0].ContextMap(), logging.QueryArgsCountKey)
   293  			assert.Contains(t, plainEntries[0].ContextMap(), logging.QueryArgsCountKey)
   294  		})
