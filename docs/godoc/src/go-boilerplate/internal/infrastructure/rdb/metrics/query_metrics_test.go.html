    16  // findMetricFamily は、Gather 結果から指定名のメトリクスファミリを返します。
    17  func findMetricFamily(t *testing.T, reg *prometheus.Registry, name string) *dto.MetricFamily {
    18  	t.Helper()
    19  
    20  	families, err := reg.Gather()
    21  	require.NoError(t, err)
    22  
    23  	for _, mf := range families {
    24  		if mf.GetName() == name {
    25  			return mf
    26  		}
    27  	}
    28  	return nil
    29  }
