    27  // NewDownstreamProfile は、outbox publish 向けの resilient プロファイルを返します。

    58  // Publish は、メッセージを受信エンドポイントへ POST します。
    59  // message_id を Idempotency-Key として伝搬します。非 2xx / transport 失敗は substrate が
    60  // apperror へ写像し、本メソッドがさらに永久失敗 / 一時失敗の分類を付けて返します。
    61  func (p *httpPublisher) Publish(ctx context.Context, m boundary.Message) error {
    62  	ctx, endSpan := p.tracer.Start(ctx)
    63  	defer endSpan()
    64  
    65  	header := httpclient.Header{"Content-Type": {"application/json"}}
    66  	for k, v := range m.Headers {
    67  		// emit 側と同じ判定だが、emit を経由しない行への防御として必要。
    68  		if httpheader.IsSensitive(k) {
    69  			continue
    70  		}
    71  		header[k] = []string{v}
    72  	}
    73  
    74  	resp, err := p.client.Do(ctx, httpclient.NewRequest(
    75  		httpclient.MethodPost(), downstream, string(p.endpoint),
    76  		httpclient.WithHeader(header),
    77  		httpclient.WithBody(m.Payload),
    78  		httpclient.WithIdempotencyKey(m.MessageID.String()),
    79  	))
    80  	return classifyOutcome(resp, err)
    81  }
