    10  // 判定は httpclient.RetryableOutcome へ委譲します（理由は README の Design Policy を参照）。
    11  // ctx キャンセルは配送の失敗ではなく停止なので分類せず、そのまま返します。
    12  func classifyOutcome(resp *httpclient.Response, err error) error {
    13  	if err == nil {
    14  		return nil
    15  	}
    16  	if xerrors.Is(err, apperror.ErrCanceled) {
    17  		return err
    18  	}
    19  	if httpclient.RetryableOutcome(resp, err) {
    20  		return xerrors.Join(apperror.ErrRetryable, err)
    21  	}
    22  	return xerrors.Join(apperror.ErrPermanent, err)
    23  }
