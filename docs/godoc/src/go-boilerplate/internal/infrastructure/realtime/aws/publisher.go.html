    45  // Publish は、m の payload を DeliveryEvent として復元し、EventLog へ append してから wakeup を publish します。
    46  // 復元できない payload・eventId の不一致（ErrEventIDMismatch）・位置の衝突（ErrSequenceConflict）は retry で
    47  // 直らないので ErrPermanent で返し、relay がその stream を先頭で止めます。EventLog と SNS の失敗は ErrRetryable
    48  // です（分類表は README「Error classification」）。
    49  func (p *publisher) Publish(ctx context.Context, m publisherbndry.Message) error {
    50  	// 起点の command が headers に載せた trace を継いでから span を開きます。これをしないと
    51  	// append が relay の trace にぶら下がり、command → outbox → relay → EventLog が 1 本になりません。
    52  	ctx = observability.ExtractFromCarrier(ctx, m.Headers)
    53  
    54  	ctx, endSpan := p.tracer.Start(ctx)
    55  	defer endSpan()
    56  
    57  	event, err := decodeEvent(m)
    58  	if err != nil {
    59  		return err
    60  	}
    61  
    62  	event.Origin = observability.TraceContextFromCarrier(m.Headers)
    63  
    64  	if err := p.log.Append(ctx, event); err != nil {
    65  		p.metrics.EventLogAppended(ctx, appendResult(err))
    66  
    67  		return classifyAppend(err)
    68  	}
    69  
    70  	p.metrics.EventLogAppended(ctx, observability.RealtimeResultOK)
    71  
    72  	// outbox に記録されてから EventLog に載るまでが、この経路の遅れです。
    73  	// 起点を持たない封筒（relay 以外の呼び出し元）は、ゼロ値の差で histogram を壊すので数えません。
    74  	if !m.CreatedAt.IsZero() {
    75  		p.metrics.EventLogLag(ctx, float64(time.Since(m.CreatedAt).Milliseconds()))
    76  	}
    77  
    78  	body, attrs, err := encodeWakeup(
    79  		rt.Wakeup{EventID: event.EventID, StreamID: event.StreamID, Sequence: event.Sequence},
    80  	)
    81  	if err != nil {
    82  		return xerrors.Join(apperror.ErrPermanent, err)
    83  	}
    84  
    85  	if _, err := p.sns.Publish(ctx, &sns.PublishInput{
    86  		TopicArn:          awssdk.String(p.topicARN),
    87  		Message:           awssdk.String(body),
    88  		MessageAttributes: attrs,
    89  	}); err != nil {
    90  		p.metrics.WakeupPublishFailed(ctx)
    91  
    92  		return xerrors.Join(apperror.ErrRetryable, normalize(err, "publish wakeup"))
    93  	}
    94  
    95  	return nil
    96  }

    98  // decodeEvent は、payload を DeliveryEvent へ復元します。eventId が空なら outbox の message_id で埋め、
    99  // 入っていて食い違えば ErrEventIDMismatch です。
   100  func decodeEvent(m publisherbndry.Message) (rt.DeliveryEvent, error) {
   101  	event, err := rt.ParseDeliveryEvent(m.Payload)
   102  	if err != nil {
   103  		return rt.DeliveryEvent{}, xerrors.Join(apperror.ErrPermanent, err)
   104  	}
   105  
   106  	messageID := m.MessageID.String()
   107  	switch event.EventID {
   108  	case "":
   109  		event.EventID = messageID
   110  	case messageID:
   111  	default:
   112  		return rt.DeliveryEvent{}, xerrors.Wrap(ErrEventIDMismatch, event.EventID+" != "+messageID)
   113  	}
   114  
   115  	if err := event.Validate(); err != nil {
   116  		return rt.DeliveryEvent{}, xerrors.Join(apperror.ErrPermanent, err)
   117  	}
   118  
   119  	return event, nil
   120  }

   122  // appendResult は、append の失敗を realtime.eventlog.appends の result へ写します。
   123  // 位置の衝突だけは「既に誰かが書いた」ことを表すので、substrate の失敗と分けて数えます。
   124  func appendResult(err error) string {
   125  	if xerrors.Is(err, rt.ErrSequenceConflict) {
   126  		return observability.RealtimeResultConflict
   127  	}
   128  
   129  	return observability.RealtimeResultError
   130  }

   132  // classifyAppend は、append の失敗を relay の分類（Publish の doc）へ写します。
   133  func classifyAppend(err error) error {
   134  	if xerrors.Is(err, rt.ErrSequenceConflict) || xerrors.Is(err, rt.ErrInvalidEvent) ||
   135  		xerrors.Is(err, rt.ErrPayloadTooLarge) {
   136  		return xerrors.Join(apperror.ErrPermanent, err)
   137  	}
   138  
   139  	return xerrors.Join(apperror.ErrRetryable, err)
   140  }
