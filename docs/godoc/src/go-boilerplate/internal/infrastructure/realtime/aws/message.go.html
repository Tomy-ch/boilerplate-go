    64  func decodeNotification(m sqstypes.Message) rt.Notification {
    65  	n := rt.Notification{Receipt: awssdk.ToString(m.ReceiptHandle)}
    66  
    67  	attr, ok := m.MessageAttributes[AttrKind]
    68  	if !ok {
    69  		return n
    70  	}
    71  
    72  	body := []byte(awssdk.ToString(m.Body))
    73  	switch kind := rt.NotificationKind(awssdk.ToString(attr.StringValue)); kind {
    74  	case rt.KindWakeup:
    75  		var w wakeupBody
    76  		if json.Unmarshal(body, &w) != nil {
    77  			return n
    78  		}
    79

    81  		if err != nil {
    82  			return n
    83  		}
    84  
    85  		n.Kind = kind
    86  		n.Wakeup = rt.Wakeup{EventID: w.EventID, StreamID: rt.StreamID(w.StreamID), Sequence: rt.Sequence(seq)}
    87  	case rt.KindRevocation:
    88  		var r revocationBody
    89  		if json.Unmarshal(body, &r) != nil {
    90  			return n
    91  		}
    92  
    93  		n.Kind = kind
    94  		n.Revocation = rt.Revocation{Subject: r.Subject, Destination: rt.StreamID(r.Destination)}
    95  	case rt.KindUnknown:
    96  	}
    97  
    98  	return n
