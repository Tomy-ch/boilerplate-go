    34  func newFanoutFixture(t *testing.T, instances int) *fanoutFixture {
    35  	t.Helper()
    36  
    37  	dynamo := dynamotestkit.NewTestClient(t)
    38  	table := dynamotestkit.TableName(t, "fanout_event_log")
    39  	require.NoError(t, dynamodbclient.EnsureTable(t.Context(), dynamo, eventlogdynamo.TableSpec(table)))
    40  	dynamotestkit.DeleteOnCleanup(t, dynamo, table)
    41  
    42  	clients := testkit.NewTestClients(t)
    43  	f := &fanoutFixture{
    44  		clients:  clients,
    45  		topicARN: testkit.CreateTopic(t, clients, testkit.Name(t, "fanout")),
    46  		log:      eventlog.New(dynamo, table, observability.NewNoopTracerFactory(t)),
    47  	}
    48  
    49  	prefix := testkit.Name(t, "inst")
    50  	for i := range instances {
    51  		sub := aws.NewInstanceSubscription(
    52  			clients.SNS, clients.SQS, aws.SubscriptionTarget{TopicARN: f.topicARN, QueuePrefix: prefix},
    53  			local.NewQueueAttributes(), observability.NewNoopTracerFactory(t),
    54  		)
    55  		require.NoError(t, sub.Provision(t.Context(), rt.InstanceID("i"+strconv.Itoa(i))))
    56  		testkit.TeardownOnCleanup(t, sub)
    57  		f.subs = append(f.subs, sub)
    58  	}
    59  
    60  	return f
    61  }

   104  // TestFanoutContract は、親受入基準 1（1 publish が N 個の serve instance queue へ届く）と 13（emulator で fan-out が
   105  // 成立する）、そして RawMessageDelivery で本文が生の wakeup と一致することを、実 EventLog + 実 emulator で固定します。
   106  func TestFanoutContract(t *testing.T) {
   107  	t.Parallel()
   108  
   109  	t.Run("正常系", func(t *testing.T) {
   110  		t.Parallel()
   111  
   112  		t.Run("1 publish が 3 instance の queue すべてへ 1 件ずつ届き、本文は eventId / streamId / sequence だけ", func(t *testing.T) {
   113  			t.Parallel()
   114  
   115  			f := newFanoutFixture(t, 3)
   116  			m := f.message(t, "stream-1")
   117  			require.NoError(t, f.publisher(t, f.topicARN).Publish(t.Context(), m))
   118  
   119  			for i, sub := range f.subs {
   120  				got := drain(t, sub)
   121  				require.Len(t, got, 1, "instance %d", i)
   122  				assert.Equal(t, rt.Notification{
   123  					Kind: rt.KindWakeup, Wakeup: rt.Wakeup{EventID: m.MessageID.String(), StreamID: "stream-1", Sequence: 1}, Receipt: got[0].Receipt,
   124  				}, got[0])
   125  			}
   126  
   127  			stored, ok, err := f.log.Find(t.Context(), "stream-1", 1)
   128  			require.NoError(t, err)
   129  			require.True(t, ok)
   130  			assert.Equal(t, m.MessageID.String(), stored.EventID)
   131  		})
   132  	})

   135  // TestPublishRetryContract は、親受入基準 3（SNS 成功後の outbox mark 失敗でも event が二重配送されない）を固定します。
   136  // mark の失敗は relay の再 claim を生むだけなので、ここでは「同じ message を再び Publish する」ことで再現し、
   137  // EventLog に 2 件目が入らないこと（重複 wakeup は届くが、client に配られるのは EventLog の 1 件）を確かめます。
   138  func TestPublishRetryContract(t *testing.T) {
   139  	t.Parallel()
   140  
   141  	t.Run("正常系", func(t *testing.T) {
   142  		t.Parallel()
   143  
   144  		t.Run("SNS 成功後に mark が失敗して再 claim されても、EventLog の同じ位置には 1 件しか無い", func(t *testing.T) {
   145  			t.Parallel()
   146  
   147  			f := newFanoutFixture(t, 1)
   148  			p := f.publisher(t, f.topicARN)
   149  			m := f.message(t, "stream-1")
   150  
   151  			require.NoError(t, p.Publish(t.Context(), m))
   152  			require.NoError(t, p.Publish(t.Context(), m), "再 claim: append は冪等に成功し、wakeup が再び publish される")
   153  
   154  			res, err := f.log.ReadAfter(t.Context(), rt.ReadAfterQuery{StreamID: "stream-1", After: 0, Limit: 10})
   155  			require.NoError(t, err)
   156  			require.Len(t, res.Events, 1, "EventLog に 2 件目は入らない")
   157  			assert.Equal(t, m.MessageID.String(), res.Events[0].EventID)
   158  
   159  			got := drain(t, f.subs[0])
   160  			assert.GreaterOrEqual(t, len(got), 1)
   161  			for _, n := range got {
   162  				assert.Equal(t, rt.Sequence(1), n.Wakeup.Sequence, "届く wakeup はすべて同じ位置を指す（読み直しに畳まれる）")
   163  			}
   164  		})
   165  
   166  		t.Run("append 済みで SNS が失敗しても retryable で、再実行で wakeup だけが届く", func(t *testing.T) {
   167  			t.Parallel()
   168  
   169  			f := newFanoutFixture(t, 1)
   170  			m := f.message(t, "stream-2")
   171  
   172  			err := f.publisher(t, "arn:aws:sns:us-east-1:000000000000:does-not-exist").Publish(t.Context(), m)
   173  			require.ErrorIs(t, err, apperror.ErrRetryable)
   174  
   175  			_, ok, err := f.log.Find(t.Context(), "stream-2", 1)
   176  			require.NoError(t, err)
   177  			require.True(t, ok, "append は SNS の前に成立している")
   178  
   179  			require.NoError(t, f.publisher(t, f.topicARN).Publish(t.Context(), m))
   180  
   181  			res, err := f.log.ReadAfter(t.Context(), rt.ReadAfterQuery{StreamID: "stream-2", After: 0, Limit: 10})
   182  			require.NoError(t, err)
   183  			require.Len(t, res.Events, 1)
   184  
   185  			got := drain(t, f.subs[0])
   186  			require.Len(t, got, 1)
   187  			assert.Equal(t, rt.Sequence(1), got[0].Wakeup.Sequence)
   188  		})
   189  	})
   190  
   191  	t.Run("異常系", func(t *testing.T) {
   192  		t.Parallel()
   193  
   194  		t.Run("同じ位置に別の event を publish すると permanent で、EventLog は先勝ちのまま", func(t *testing.T) {
   195  			t.Parallel()
   196  
   197  			f := newFanoutFixture(t, 1)
   198  			p := f.publisher(t, f.topicARN)
   199  			first := f.message(t, "stream-3")
   200  			second := f.message(t, "stream-3")
   201  
   202  			require.NoError(t, p.Publish(t.Context(), first))
   203  
   204  			err := p.Publish(t.Context(), second)
   205  			require.ErrorIs(t, err, apperror.ErrPermanent)
   206  			require.ErrorIs(t, err, rt.ErrSequenceConflict)
   207  
   208  			stored, ok, err := f.log.Find(t.Context(), "stream-3", 1)
   209  			require.NoError(t, err)
   210  			require.True(t, ok)
   211  			assert.Equal(t, first.MessageID.String(), stored.EventID)
   212  		})
   213  	})
