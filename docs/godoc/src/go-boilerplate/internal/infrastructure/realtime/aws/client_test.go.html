    60  func Test_withCallTimeout(t *testing.T) {
    61  	t.Parallel()
    62  
    63  	// middleware が握った deadline を覗くため、stack の続きを差し替えて ctx を取り出す。
    64  	deadlineOf := func(t *testing.T, operation string) time.Duration {
    65  		t.Helper()
    66  
    67  		stack := smithymiddleware.NewStack("test", smithyhttp.NewStackRequest)
    68  		require.NoError(t, withCallTimeout()(stack))
    69  
    70  		var got time.Duration
    71  		require.NoError(t, stack.Initialize.Add(
    72  			smithymiddleware.InitializeMiddlewareFunc("capture",
    73  				func(
    74  					ctx context.Context, _ smithymiddleware.InitializeInput, _ smithymiddleware.InitializeHandler,
    75  				) (smithymiddleware.InitializeOutput, smithymiddleware.Metadata, error) {
    76  					deadline, ok := ctx.Deadline()
    77  					require.True(t, ok, "呼び出しに deadline が設定されていない")
    78  					got = time.Until(deadline).Round(time.Second)
    79  
    80  					return smithymiddleware.InitializeOutput{}, smithymiddleware.Metadata{}, nil
    81  				},
    82  			),
    83  			smithymiddleware.After,
    84  		))
    85  
    86  		ctx := awsmiddleware.SetOperationName(t.Context(), operation)
    87  		_, _, err := smithymiddleware.DecorateHandler(
    88  			smithymiddleware.HandlerFunc(func(context.Context, any) (any, smithymiddleware.Metadata, error) {
    89  				return nil, smithymiddleware.Metadata{}, nil
    90  			}),
    91  			stack,
    92  		).Handle(ctx, nil)
    93  		require.NoError(t, err)
    94  
    95  		return got
    96  	}
    97  
    98  	t.Run("正常系", func(t *testing.T) {
    99  		t.Parallel()
   100  
   101  		t.Run("通常の呼び出しには CallTimeout を与える", func(t *testing.T) {
   102  			t.Parallel()
   103  
   104  			assert.Equal(t, CallTimeout, deadlineOf(t, "DeleteQueue"))
   105  		})
   106  
   107  		t.Run("long polling する受信には待ち時間を超える上限を与える", func(t *testing.T) {
   108  			t.Parallel()
   109  
   110  			// CallTimeout を当てると待ち切る前に落ちて loop が回らない。
   111  			got := deadlineOf(t, receiveOperation)
   112  			assert.Equal(t, receiveCallTimeout, got)
   113  			assert.Greater(t, got, receiveWaitSeconds*time.Second)
