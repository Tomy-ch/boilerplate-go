    38  func subscriptionPage(next string, endpoints ...string) *sns.ListSubscriptionsByTopicOutput {
    39  	out := &sns.ListSubscriptionsByTopicOutput{}
    40  	for i, e := range endpoints {
    41  		out.Subscriptions = append(out.Subscriptions, snstypes.Subscription{
    42  			Endpoint:        awssdk.String(e),
    43  			SubscriptionArn: awssdk.String(testTopicARN + ":sub-" + string(rune('a'+i))),
    44  		})
    45  	}
    46  
    47  	if next != "" {
    48  		out.NextToken = awssdk.String(next)
    49  	}
    50  
    51  	return out
