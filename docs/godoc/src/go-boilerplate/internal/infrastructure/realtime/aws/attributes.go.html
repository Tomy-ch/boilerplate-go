    58  func (a *queueAttributes) Build(queueARN string) (map[string]string, error) {
    59  	attrs := TimingAttributes()
    60  
    61  	policy, err := queuePolicyDocument(queueARN, a.in.TopicARN)
    62  	if err != nil {
    63  		return nil, err
    64  	}
    65  
    66  	attrs[string(sqstypes.QueueAttributeNamePolicy)] = policy
    67  	attrs[string(sqstypes.QueueAttributeNameSqsManagedSseEnabled)] = "true"
    68  
    69  	if a.in.DLQARN != "" {
    70  		redrive, err := redrivePolicyDocument(a.in.DLQARN)
    71  		if err != nil {
    72  			return nil, err
    73  		}
    74  
    75  		attrs[string(sqstypes.QueueAttributeNameRedrivePolicy)] = redrive
    76  	}
    77  
    78  	return attrs, nil
    79  }
