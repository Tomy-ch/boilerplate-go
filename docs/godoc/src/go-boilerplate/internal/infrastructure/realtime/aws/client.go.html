    16  const (
    17  	// CallTimeout は、1 回の API 呼び出し全体（retry を含む）に与える上限です。
    18  	//
    19  	// これが無いと、応答を返さない substrate に対して呼び出しがいつまでも戻りません。SDK にも HTTP
    20  	// クライアントにも別の上限がなく（DI が渡すクライアントは Timeout を持たず、SSRF ガードのために
    21  	// 差し替えた Dialer にも Timeout はありません）、awsclient.Resolve の上限は起動時の資格情報解決
    22  	// 1 回にしか掛かりません。one-shot の job は既定で全体のタイムアウトを持たないため、1 呼び出しが
    23  	// 戻らないとジョブごと戻らなくなります。
    24  	//
    25  	// 値は dynamodbclient.CallTimeout に揃えます。同じ機構が使う 2 つの substrate で、1 呼び出しに
    26  	// 与える猶予を変える理由がないためです。
    27  	CallTimeout = 10 * time.Second
    28  	// receiveCallTimeout は、long polling する受信にだけ与える上限です。受信は receiveWaitSeconds
    29  	// （20 秒）まで意図的に待つので、CallTimeout を当てると毎回 deadline で落ちて loop が回りません。
    30  	// 待ち時間そのものではなく「待ち切ってなお戻らない」ことを捕まえる値にします。
    31  	receiveCallTimeout = (receiveWaitSeconds + receiveTimeoutMargin) * time.Second
    32  	// receiveTimeoutMargin は、long polling の待ち時間に足す余裕（秒）です。
    33  	receiveTimeoutMargin = 10
    34  	// receiveOperation は、long polling する操作の名前です（smithy が middleware へ渡す値）。
    35  	receiveOperation = "ReceiveMessage"
    36  )

    60  // NewClients は、設定から SNS / SQS クライアントを生成します。資格情報の解決は 1 回で、失敗すれば
    61  // エラーを返し、認証エラーが最初の publish / receive まで隠れないようにします。
    62  // NewClients は、設定から SNS / SQS クライアントを生成します。資格情報の解決は 1 回で、失敗すれば
    63  // エラーを返し、認証エラーが最初の publish / receive まで隠れないようにします。
    64  func NewClients(ctx context.Context, cfg ClientConfig) (Clients, error) {
    65  	awsCfg, err := awsclient.Resolve(ctx, awsclient.Config{
    66  		Region:          cfg.Region,
    67  		AccessKeyID:     cfg.AccessKeyID,
    68  		SecretAccessKey: cfg.SecretAccessKey,
    69  		HTTPClient:      cfg.HTTPClient,
    70  	})
    71  	if err != nil {
    72  		return Clients{}, err
    73  	}
    74  
    75  	return Clients{
    76  		SNS: sns.NewFromConfig(awsCfg, func(o *sns.Options) {
    77  			if cfg.Endpoint != "" {
    78  				o.BaseEndpoint = awssdk.String(cfg.Endpoint)
    79  			}
    80  			o.APIOptions = append(o.APIOptions, withCallTimeout())
    81  		}),
    82  		SQS: sqs.NewFromConfig(awsCfg, func(o *sqs.Options) {
    83  			if cfg.Endpoint != "" {
    84  				o.BaseEndpoint = awssdk.String(cfg.Endpoint)
    85  			}
    86  			o.APIOptions = append(o.APIOptions, withCallTimeout())
    87  		}),
    88  	}, nil
    89  }

    91  // withCallTimeout は、API 呼び出し 1 回の全体に上限を与える middleware を stack の先頭へ差します。
    92  // retry の外側に置くので、上限は試行ごとではなく呼び出し全体に掛かります。long polling する受信だけは
    93  // 待ち時間が仕様なので別の上限を当てます。
    94  func withCallTimeout() func(*smithymiddleware.Stack) error {
    95  	return func(stack *smithymiddleware.Stack) error {
    96  		return stack.Initialize.Add(
    97  			smithymiddleware.InitializeMiddlewareFunc(
    98  				"realtimeCallTimeout",
    99  				func(
   100  					ctx context.Context, in smithymiddleware.InitializeInput, next smithymiddleware.InitializeHandler,
   101  				) (smithymiddleware.InitializeOutput, smithymiddleware.Metadata, error) {
   102  					d := CallTimeout
   103  					if awsmiddleware.GetOperationName(ctx) == receiveOperation {
   104  						d = receiveCallTimeout
   105  					}
   106  
   107  					ctx, cancel := context.WithTimeout(ctx, d)
   108  					defer cancel()
   109  
   110  					return next.HandleInitialize(ctx, in)
   111  				},
   112  			),
   113  			smithymiddleware.Before,
   114  		)
   115  	}
   116  }
