    45  	t.Helper()
    46  
    47  	dynamo := dynamotestkit.NewTestClient(t)
    48  	table := dynamotestkit.TableName(t, "cleanup_instance_lease")

    56  		clients:   clients,

   124  func (f *cleanupFixture) queueExists(t *testing.T, id rt.InstanceID) bool {
   125  	t.Helper()
   126  
   127  	name, err := aws.QueueName(f.prefix, id)
   128  	require.NoError(t, err)
   129  
   130  	_, err = f.clients.SQS.GetQueueUrl(t.Context(), &sqs.GetQueueUrlInput{QueueName: awssdk.String(name)})
   131  	if err == nil {
   132  		return true
   133  	}
   134  
   135  	// 判定は production の queueGone と揃える。型付きのエラーだけを見ると、旧来のエラーコードで
   136  	// 「無い」を返す substrate（GoAWS）で、消えたことを確かめられない。
   137  	var gone *sqstypes.QueueDoesNotExist
   138  	if xerrors.As(err, &gone) {
   139  		return false
   140  	}
   141  
   142  	var api smithy.APIError
   143  	require.True(t, xerrors.As(err, &api) && api.ErrorCode() == "AWS.SimpleQueueService.NonExistentQueue",
   144  		"queue の存在確認が想定外の失敗をしました: %v", err)
   145  
   146  	return false
   147  }

   188  // TestOrphanCleanupContract は、crash した instance の受信先と lease が実 substrate 上で回収されること、
   189  // 同時に走った掃除役のうち 1 つだけが引き受けること、繰り返し実行しても同じ状態に収束することを確かめる。
   190  // 親 issue の受入基準「crash 後の queue / subscription を安全に回収する」に対応する。
   191  func TestOrphanCleanupContract(t *testing.T) {
   192  	t.Parallel()
   193  
   194  	t.Run("正常系", func(t *testing.T) {
   195  		t.Parallel()
   196  
   197  		t.Run("crash した instance の subscription・queue・lease を回収する", func(t *testing.T) {
   198  			t.Parallel()
   199  
   200  			f := newCleanupFixture(t)
   201  			f.crash(t, "i-crashed")
   202  			require.True(t, f.queueExists(t, "i-crashed"))
   203  			require.Equal(t, 1, f.subscriptionCount(t, "i-crashed"))
   204  
   205  			got, err := f.sweeper(t, "job-a").Sweep(t.Context())
   206  			require.NoError(t, err)
   207  			assert.Equal(t, ucrealtime.SweepResult{Detected: 1, Claimed: 1, Reclaimed: 1}, got)
   208  
   209  			assert.False(t, f.queueExists(t, "i-crashed"))
   210  			assert.Equal(t, 0, f.subscriptionCount(t, "i-crashed"))
   211  			assert.False(t, f.leaseExists(t, "i-crashed"))
   212  		})
   213  
   214  		t.Run("2 つの掃除役が同時に走っても 1 つしか引き受けない", func(t *testing.T) {
   215  			t.Parallel()
   216  
   217  			f := newCleanupFixture(t)
   218  			f.crash(t, "i-contended")
   219  
   220  			// 先行の掃除役が「引き受けたが、まだ回収を終えていない」状態を実 substrate 上に作る。
   221  			// Sweep を完走させると lease まで消えてしまい、後続は列挙が空になるだけで、
   222  			// 引き受けの条件式による相互排除を一度も通らない。
   223  			cutoff := f.sweepAt.Add(-ucrealtime.LeaseCleanupMargin)
   224  			claimed, err := f.leases.AcquireCleanup(t.Context(), rt.CleanupClaim{
   225  				InstanceID:    "i-contended",
   226  				Owner:         "job-a",
   227  				ExpiredBefore: cutoff,
   228  				Now:           f.sweepAt,
   229  				OwnerUntil:    f.sweepAt.Add(ucrealtime.LeaseCleanupOwnershipTTL),
   230  			})
   231  			require.NoError(t, err)
   232  			require.True(t, claimed)
   233  
   234  			// 後続は同じ lease を見つけるが、引き受けられないので受信先に触らず見送る。
   235  			got, err := f.sweeper(t, "job-b").Sweep(t.Context())
   236  			require.NoError(t, err)
   237  			assert.Equal(t, ucrealtime.SweepResult{Detected: 1, Skipped: 1}, got)
   238  			assert.True(t, f.queueExists(t, "i-contended"))
   239  			assert.True(t, f.leaseExists(t, "i-contended"))
   240  		})
   241  
   242  		t.Run("繰り返し実行しても同じ状態に収束する", func(t *testing.T) {
   243  			t.Parallel()
   244  
   245  			f := newCleanupFixture(t)
   246  			f.crash(t, "i-idempotent")
   247  
   248  			_, err := f.sweeper(t, "job-a").Sweep(t.Context())
   249  			require.NoError(t, err)
   250  
   251  			again, err := f.sweeper(t, "job-a").Sweep(t.Context())
   252  			require.NoError(t, err)
   253  			assert.Equal(t, ucrealtime.SweepResult{}, again)
   254  		})
   255  
   256  		t.Run("期限切れが安全余裕の内側の instance には触らない", func(t *testing.T) {
   257  			t.Parallel()
   258  
   259  			f := newCleanupFixture(t)
   260  			f.crash(t, "i-recent")
   261  			// 期限は切れているが安全余裕を過ぎていない時刻で掃除する。
   262  			f.sweepAt = f.crashedAt.Add(ucrealtime.LeaseExpiry + time.Minute)
   263  
   264  			got, err := f.sweeper(t, "job-a").Sweep(t.Context())
   265  			require.NoError(t, err)
   266  			assert.Equal(t, ucrealtime.SweepResult{}, got)
   267  
   268  			assert.True(t, f.queueExists(t, "i-recent"))
   269  			assert.True(t, f.leaseExists(t, "i-recent"))
   270  		})
   271  	})

   274  // TestReceivingEndGoneContract は、外から受信先を消された instance の Receive / Delete が
   275  // ErrReceivingEndGone を返すことを実 substrate に対して確かめる。
   276  //
   277  // この写像はモックでは検証できない。テストが `&sqstypes.QueueDoesNotExist{}` を注入する限り、
   278  // emulator が別のエラー形を返しても全緑のままで、受信ループは消えた受信先に対して永久に空振りする。
   279  // 復旧経路（作り直し）はこの分類を入口にしているので、入口が実基盤で成立することをここで締める。
   280  func TestReceivingEndGoneContract(t *testing.T) {
   281  	t.Parallel()
   282  
   283  	t.Run("正常系", func(t *testing.T) {
   284  		t.Parallel()
   285  
   286  		t.Run("受信先を外から消されたら ErrReceivingEndGone を返す", func(t *testing.T) {
   287  			t.Parallel()
   288  
   289  			f := newCleanupFixture(t)
   290  			sub := aws.NewInstanceSubscription(
   291  				f.clients.SNS, f.clients.SQS, aws.SubscriptionTarget{TopicARN: f.topicARN, QueuePrefix: f.prefix},
   292  				local.NewQueueAttributes(), observability.NewNoopTracerFactory(t),
   293  			)
   294  			require.NoError(t, sub.Provision(t.Context(), "i-gone"))
   295  
   296  			// 生きているうちは通常どおり受け取れる（空でも成功）。
   297  			_, err := sub.Receive(t.Context(), 1)
   298  			require.NoError(t, err)
   299  
   300  			// orphan cleanup が誤って回収した状態を、実 substrate 上で作る。
   301  			require.NoError(t, f.sweeperReclaimer(t).Reclaim(t.Context(), "i-gone"))
   302  
   303  			_, err = sub.Receive(t.Context(), 1)
   304  			require.ErrorIs(t, err, rt.ErrReceivingEndGone)
   305  
   306  			err = sub.Delete(t.Context(), rt.Notification{Receipt: "r"})
   307  			require.ErrorIs(t, err, rt.ErrReceivingEndGone)
   308  		})
   309  	})
