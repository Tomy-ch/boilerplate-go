    69  // QueueName は、prefix と instance の識別子から instance queue の名前を返します。SQS の制約（使える文字と長さ）を
    70  // 満たさなければ ErrInvalidQueueName を返します。
    71  func QueueName(prefix string, id rt.InstanceID) (string, error) {
    72  	name := prefix + "-" + string(id)
    73  	if len(name) > maxQueueNameLen || !queueNameRe.MatchString(name) {
    74  		return "", xerrors.Wrap(ErrInvalidQueueName, name)
    75  	}
    76  
    77  	return name, nil
    78  }

    80  // Provision は、queue を作り、属性を設定し、topic へ subscribe して RawMessageDelivery を有効にします。
    81  // 途中で失敗したら作った分を片付けてから返します（起動失敗の instance が resource を残さないため）。
    82  func (s *subscription) Provision(ctx context.Context, id rt.InstanceID) error {
    83  	ctx, endSpan := s.tracer.Start(ctx)
    84  	defer endSpan()
    85  
    86  	s.mu.Lock()
    87  	defer s.mu.Unlock()
    88  
    89  	if s.subscriptionARN != "" {
    90  		if s.instanceID == id {
    91  			return nil
    92  		}
    93  
    94  		return xerrors.Wrap(apperror.ErrConflict, "realtime: already provisioned for another instance")
    95  	}
    96  
    97  	name, err := QueueName(s.target.QueuePrefix, id)
    98  	if err != nil {
    99  		return err
   100  	}
   101  
   102  	if err := s.provision(ctx, name); err != nil {
   103  		// 片付けの失敗も返す。握り潰すと「resource は残っていない」と「消せなかった」を呼び出し側が区別できない。
   104  		return xerrors.Join(err, s.teardown(ctx))
   105  	}
   106  
   107  	s.instanceID = id
   108  
   109  	return nil
   110  }

   112  // Receive は、long polling で最大 limit 件（0 以下と 10 超は SQS の上限 10 件）を受け取り、Notification へ復元して返します。
   113  func (s *subscription) Receive(ctx context.Context, limit int) ([]rt.Notification, error) {
   114  	ctx, endSpan := s.tracer.Start(ctx)
   115  	defer endSpan()
   116  
   117  	queueURL, err := s.currentQueueURL()
   118  	if err != nil {
   119  		return nil, err
   120  	}
   121  
   122  	if limit <= 0 || limit > maxReceiveBatch {
   123  		limit = maxReceiveBatch
   124  	}
   125  
   126  	out, err := s.sqs.ReceiveMessage(ctx, &sqs.ReceiveMessageInput{
   127  		QueueUrl:              awssdk.String(queueURL),
   128  		MaxNumberOfMessages:   int32(limit),
   129  		WaitTimeSeconds:       receiveWaitSeconds,
   130  		MessageAttributeNames: []string{"All"},
   131  	})
   132  	if err != nil {
   133  		return nil, s.classifyReceivingEnd(err, "receive notifications")
   134  	}
   135  
   136  	notifications := make([]rt.Notification, 0, len(out.Messages))
   137  	for _, m := range out.Messages {
   138  		notifications = append(notifications, decodeNotification(m))
   139  	}
   140  
   141  	return notifications, nil
   142  }

   175  // classifyReceivingEnd は、受信の失敗を分類します。受信先がもう無いなら、キャッシュした識別子を捨てて
   176  // ErrReceivingEndGone を返します。作り直しはここでは行いません — 順序を持つ呼び出し側の役割です
   177  // （package README の Port mapping / docs/design/realtime-delivery.md §2.5）。
   178  func (s *subscription) classifyReceivingEnd(cause error, op string) error {
   179  	if !queueGone(cause) {
   180  		return normalize(cause, op)
   181  	}
   182  
   183  	s.invalidate()
   184  
   185  	return xerrors.Join(rt.ErrReceivingEndGone, normalize(cause, op))
   186  }

   189  // 早期に返るため、作り直しを受け付ける状態へ戻すにはこれが要ります。捨てるのは受信先の識別子だけで、

   192  	s.mu.Lock()
   193  	defer s.mu.Unlock()
   194  
   195  	s.queueURL, s.queueARN, s.subscriptionARN = "", "", ""
   196  }

   198  func (s *subscription) provision(ctx context.Context, name string) error {
   199  	created, err := s.sqs.CreateQueue(ctx, &sqs.CreateQueueInput{QueueName: awssdk.String(name)})
   200  	if err != nil {
   201  		return normalize(err, "create instance queue")
   202  	}
   203  
   204  	s.queueURL = awssdk.ToString(created.QueueUrl)
   205  
   206  	got, err := s.sqs.GetQueueAttributes(ctx, &sqs.GetQueueAttributesInput{
   207  		QueueUrl:       awssdk.String(s.queueURL),
   208  		AttributeNames: []sqstypes.QueueAttributeName{sqstypes.QueueAttributeNameQueueArn},
   209  	})
   210  	if err != nil {
   211  		return normalize(err, "resolve instance queue arn")
   212  	}
   213  
   214  	s.queueARN = got.Attributes[string(sqstypes.QueueAttributeNameQueueArn)]
   215  	if s.queueARN == "" {
   216  		return xerrors.Wrap(apperror.ErrUnavailable, "resolve instance queue arn: empty QueueArn")
   217  	}
   218  
   219  	attrs, err := s.attrs.Build(s.queueARN)
   220  	if err != nil {
   221  		return err
   222  	}
   223  
   224  	if len(attrs) > 0 {
   225  		if _, err := s.sqs.SetQueueAttributes(ctx, &sqs.SetQueueAttributesInput{
   226  			QueueUrl: awssdk.String(s.queueURL), Attributes: attrs,
   227  		}); err != nil {
   228  			return normalize(err, "set instance queue attributes")
   229  		}
   230  	}
   231  
   232  	sub, err := s.sns.Subscribe(ctx, &sns.SubscribeInput{
   233  		TopicArn:              awssdk.String(s.target.TopicARN),
   234  		Protocol:              awssdk.String(subscriptionProtocol),
   235  		Endpoint:              awssdk.String(s.queueARN),
   236  		ReturnSubscriptionArn: true,
   237  	})
   238  	if err != nil {
   239  		return normalize(err, "subscribe instance queue")
   240  	}
   241  
   242  	s.subscriptionARN = awssdk.ToString(sub.SubscriptionArn)
   243  	if s.subscriptionARN == "" {
   244  		return xerrors.Wrap(apperror.ErrUnavailable, "subscribe instance queue: empty SubscriptionArn")
   245  	}
   246  
   247  	if _, err := s.sns.SetSubscriptionAttributes(ctx, &sns.SetSubscriptionAttributesInput{
   248  		SubscriptionArn: awssdk.String(s.subscriptionARN),
   249  		AttributeName:   awssdk.String(attrRawMessageDelivery),
   250  		AttributeValue:  awssdk.String("true"),
   251  	}); err != nil {
   252  		return normalize(err, "enable raw message delivery")
   253  	}
   254  
   255  	return nil
   256  }

   258  func (s *subscription) teardown(ctx context.Context) error {
   259  	var errs []error
   260  
   261  	if s.subscriptionARN != "" {
   262  		if _, err := s.sns.Unsubscribe(ctx, &sns.UnsubscribeInput{SubscriptionArn: awssdk.String(s.subscriptionARN)}); err != nil {
   263  			errs = append(errs, normalize(err, "unsubscribe instance queue"))
   264  		} else {
   265  			s.subscriptionARN = ""
   266  		}
   267  	}
   268  
   269  	if s.queueURL != "" {
   270  		if _, err := s.sqs.DeleteQueue(ctx, &sqs.DeleteQueueInput{QueueUrl: awssdk.String(s.queueURL)}); err != nil {
   271  			errs = append(errs, normalize(err, "delete instance queue"))
   272  		} else {
   273  			s.queueURL, s.queueARN = "", ""
   274  		}
   275  	}
   276  
   277  	if len(errs) == 0 {
   278  		s.instanceID = ""
   279  	}
   280  
   281  	return xerrors.Join(errs...)
   282  }

   284  // currentQueueURL は、Provision 済みの queue URL を返します。未 provision なら ErrReceivingEndGone です。
   285  func (s *subscription) currentQueueURL() (string, error) {
   286  	s.mu.Lock()
   287  	defer s.mu.Unlock()
   288  
   289  	if s.queueURL == "" {
   290  		// 未 provision も同じ sentinel に畳む — 分けると復旧経路が割れる（README の Port mapping）。

   292  	}
   293  
   294  	return s.queueURL, nil
