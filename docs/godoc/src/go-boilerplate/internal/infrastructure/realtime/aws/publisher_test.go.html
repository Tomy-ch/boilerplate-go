    34  	t.Helper()
    35  
    36  	id, err := uuid.New()

    40  		EventID: eventID, StreamID: "stream-1", Sequence: 7, Type: "inquiry.message.appended.v1",

    42  			2026,

    48  			0,
    49  			time.UTC,
    50  		), SchemaVersion: 1, Payload: []byte(`{"body":"hi"}`),
    51  	}
    52  	payload, err := event.MarshalJSON()

    55  	if eventID == "" {
    56  		event.EventID = id.String()
    57  	}
    58  
    59  	return publisherbndry.Message{MessageID: id, EventType: event.Type, Payload: payload}, event
