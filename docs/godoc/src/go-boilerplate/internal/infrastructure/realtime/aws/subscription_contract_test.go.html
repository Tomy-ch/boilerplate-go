    19  // TestInstanceSubscriptionContract は、実 emulator（GoAWS。環境変数で本番 SNS / SQS へ向け直せる）に対して
    20  // Provision → 通知の受信 → Delete → Teardown の往復を確かめます。queue 属性は emulator が受け付ける集合です。
    21  func TestInstanceSubscriptionContract(t *testing.T) {
    22  	t.Parallel()
    23  
    24  	t.Run("正常系", func(t *testing.T) {
    25  		t.Parallel()
    26  
    27  		t.Run("provision した queue に topic からの wakeup と revocation が届き、削除と片付けまで通る", func(t *testing.T) {
    28  			t.Parallel()
    29  
    30  			c := testkit.NewTestClients(t)
    31  			topicARN := testkit.CreateTopic(t, c, testkit.Name(t, "topic"))
    32  			sub := aws.NewInstanceSubscription(
    33  				c.SNS, c.SQS, aws.SubscriptionTarget{TopicARN: topicARN, QueuePrefix: testkit.Name(t, "q")},
    34  				local.NewQueueAttributes(), observability.NewNoopTracerFactory(t),
    35  			)
    36  
    37  			require.NoError(t, sub.Provision(t.Context(), "inst1"))
    38  			testkit.TeardownOnCleanup(t, sub)
    39  
    40  			notifier := aws.NewRevocationNotifier(c.SNS, topicARN, observability.NewNoopTracerFactory(t))
    41  			require.NoError(t, notifier.NotifyRevoked(t.Context(), "subject-1", "stream-1"))
    42  
    43  			_, err := c.SNS.Publish(t.Context(), &sns.PublishInput{
    44  				TopicArn: awssdk.String(topicARN),
    45  				Message:  awssdk.String(`{"eventId":"e1","streamId":"stream-1","sequence":"5"}`),
    46  				MessageAttributes: map[string]snstypes.MessageAttributeValue{
    47  					aws.AttrKind: {DataType: awssdk.String("String"), StringValue: awssdk.String("wakeup")},
    48  				},
    49  			})
    50  			require.NoError(t, err)
    51  
    52  			var got []rt.Notification
    53  			for range 3 {
    54  				batch, err := sub.Receive(t.Context(), 10)
    55  				require.NoError(t, err)
    56  
    57  				got = append(got, batch...)
    58  				if len(got) >= 2 {
    59  					break
    60  				}
    61  			}
    62  
    63  			require.Len(t, got, 2)
    64  			kinds := map[rt.NotificationKind]rt.Notification{}
    65  			for _, n := range got {
    66  				kinds[n.Kind] = n
    67  				require.NoError(t, sub.Delete(t.Context(), n))
    68  			}
    69  
    70  			assert.Equal(t, rt.Wakeup{EventID: "e1", StreamID: "stream-1", Sequence: 5}, kinds[rt.KindWakeup].Wakeup)
    71  			assert.Equal(
    72  				t,
    73  				rt.Revocation{Subject: "subject-1", Destination: "stream-1"},
    74  				kinds[rt.KindRevocation].Revocation,
    75  			)
    76  
    77  			require.NoError(t, sub.Teardown(t.Context()))
    78  			require.NoError(t, sub.Teardown(t.Context()), "2 度目は何もしない")
    79  		})
    80  	})
