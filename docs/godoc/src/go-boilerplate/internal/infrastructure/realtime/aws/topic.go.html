    16  	out, err := snsAPI.CreateTopic(ctx, &sns.CreateTopicInput{Name: awssdk.String(name)})
    17  	if err != nil {
    18  		return "", normalize(err, "ensure topic")

    20  
    21  	arn := awssdk.ToString(out.TopicArn)
    22  	if arn == "" {
    23  		return "", xerrors.Wrap(apperror.ErrUnavailable, "ensure topic: empty TopicArn")
    24  	}
    25  
    26  	return arn, nil
    27  }
