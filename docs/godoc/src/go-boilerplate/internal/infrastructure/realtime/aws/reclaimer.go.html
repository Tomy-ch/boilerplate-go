    42  // Reclaim は、id の instance が残した登録を解除してから受信先を削除します。順序は固定です
    43  // （先に受信先を消すと、登録が宛先の無いまま topic に残る）。既に無いものは何もしません。
    44  // 片方が失敗しても残りを試み、失敗をまとめて返します。
    45  //
    46  // 削除する名前は topic の登録の endpoint から引き当て、設定から導ける名前も候補に足します。
    47  // 片方だけでは届かない残骸がそれぞれにあります（README の Port mapping / Reclaim）。
    48  func (r *reclaimer) Reclaim(ctx context.Context, id rt.InstanceID) error {
    49  	ctx, endSpan := r.tracer.Start(ctx)
    50  	defer endSpan()
    51  
    52  	configured, err := QueueName(r.target.QueuePrefix, id)
    53  	if err != nil {
    54  		return err
    55  	}
    56  
    57  	var errs []error
    58  
    59  	found, err := r.unsubscribeAll(ctx, id)
    60  	if err != nil {
    61  		errs = append(errs, err)
    62  	}
    63  
    64  	for _, name := range candidateQueues(configured, found) {
    65  		if err := r.deleteQueue(ctx, name); err != nil {
    66  			errs = append(errs, err)
    67  		}
    68  	}
    69  
    70  	return xerrors.Join(errs...)
    71  }
    72  
    73  // unsubscribeAll は、id の instance の受信先を指す登録をすべて解除し、その受信先の名前を返します。
    74  // 同じ受信先に複数の登録がぶら下がる状態（Provision の途中失敗の残骸）も畳みます。
    75  func (r *reclaimer) unsubscribeAll(ctx context.Context, id rt.InstanceID) ([]string, error) {
    76  	var (
    77  		errs  []error
    78  		found []string
    79  		token *string
    80  	)
    81  
    82  	for {
    83  		out, err := r.sns.ListSubscriptionsByTopic(ctx, &sns.ListSubscriptionsByTopicInput{
    84  			TopicArn:  awssdk.String(r.target.TopicARN),
    85  			NextToken: token,
    86  		})
    87  		if err != nil {
    88  			// 途中のページで止まっても、それまでに解除できなかった分は診断に要る。
    89  			return found, xerrors.Join(append(errs, normalize(err, "list subscriptions by topic"))...)
    90  		}
    91  
    92  		for _, s := range out.Subscriptions {
    93  			name := queueNameFromEndpoint(awssdk.ToString(s.Endpoint))
    94  			if !queueBelongsTo(name, id) {
    95  				continue
    96  			}
    97  
    98  			found = append(found, name)
    99  
   100  			arn := awssdk.ToString(s.SubscriptionArn)
   101  			if !isUnsubscribable(arn) {
   102  				continue
   103  			}
   104  
   105  			if _, err := r.sns.Unsubscribe(ctx, &sns.UnsubscribeInput{SubscriptionArn: awssdk.String(arn)}); err != nil {
   106  				errs = append(errs, normalize(err, "unsubscribe orphan queue"))
   107  			}
   108  		}
   109  
   110  		// 続きが無いことを空の token でも表す substrate がある（GoAWS は空文字への非 nil ポインタを返す）。
   111  		// nil だけを終端と見なすと、同じページを取り続けて戻らない。
   112  		token = out.NextToken
   113  		if awssdk.ToString(token) == "" {
   114  			return found, xerrors.Join(errs...)
   115  		}
   116  	}
   117  }

   119  // deleteQueue は、name の受信先を削除します。既に無ければ何もしません。
   120  func (r *reclaimer) deleteQueue(ctx context.Context, name string) error {
   121  	out, err := r.sqs.GetQueueUrl(ctx, &sqs.GetQueueUrlInput{QueueName: awssdk.String(name)})
   122  	if err != nil {
   123  		if queueGone(err) {
   124  			return nil
   125  		}
   126  
   127  		return normalize(err, "resolve orphan queue url")
   128  	}
   129  
   130  	if _, err := r.sqs.DeleteQueue(ctx, &sqs.DeleteQueueInput{QueueUrl: out.QueueUrl}); err != nil {
   131  		if queueGone(err) {
   132  			return nil
   133  		}
   134  
   135  		return normalize(err, "delete orphan queue")
   136  	}
   137  
   138  	return nil
   139  }

   141  // candidateQueues は、削除を試みる受信先の名前を重複なく昇順で返します。
   142  func candidateQueues(configured string, found []string) []string {
   143  	names := append(slices.Clone(found), configured)
   144  	slices.Sort(names)
   145  
   146  	return slices.Compact(names)
   147  }

   149  // queueNameFromEndpoint は、登録の endpoint（queue ARN）から受信先の名前を返します。ARN の末尾要素が
   150  // 名前で、名前自体に `:` は使えません。引けなければ空を返します。
   151  func queueNameFromEndpoint(endpoint string) string {
   152  	i := strings.LastIndex(endpoint, ":")
   153  	if i < 0 || i == len(endpoint)-1 {
   154  		return ""
   155  	}
   156  
   157  	return endpoint[i+1:]
   158  }

   160  // queueBelongsTo は、受信先の名前（`<prefix>-<instance id>`）が id の instance のものかを返します。
   161  // prefix でなく識別子の側で引き当てます（旧 prefix の残骸にも届かせるため。README の Reclaim）。
   162  func queueBelongsTo(name string, id rt.InstanceID) bool {
   163  	return name != "" && strings.HasSuffix(name, "-"+string(id))
   164  }
   165  
   166  // isUnsubscribable は、SubscriptionArn が解除できる値かを返します。確認待ちの登録は ARN の代わりに
   167  // "PendingConfirmation" を返し、それを渡すと Unsubscribe は失敗します。
   168  func isUnsubscribable(arn string) bool {
   169  	return arn != "" && strings.HasPrefix(arn, "arn:")
   170  }

   172  // queueGone は、受信先が既に無いことを示す失敗かを返します。成功と扱うか作り直しの契機と扱うかは呼び出し側が決めます。
   173  //
   174  // 型付きのエラーだけでは足りません。SDK が *QueueDoesNotExist を組み立てるのは応答のエラーコードが
   175  // "QueueDoesNotExist" のときだけで、同じ意味を旧来のコード "AWS.SimpleQueueService.NonExistentQueue" で
   176  // 返す substrate があります（GoAWS が該当）。コードも見ないと、受信先の消失が通常の一時障害に化けます。
   177  func queueGone(err error) bool {
   178  	var notExist *sqstypes.QueueDoesNotExist
   179  	if xerrors.As(err, &notExist) {
   180  		return true
   181  	}
   182  
   183  	var api smithy.APIError
   184  
   185  	return xerrors.As(err, &api) && api.ErrorCode() == nonExistentQueueCode
   186  }
