    57  // CreateTopic は、name の topic を作り、テスト終了時に削除します。戻り値は API が返した ARN です。
    58  func CreateTopic(t *testing.T, c aws.Clients, name string) string {
    59  	t.Helper()
    60  
    61  	arn, err := aws.EnsureTopic(t.Context(), c.SNS, name)
    62  	require.NoError(t, err)
    63  
    64  	t.Cleanup(func() {
    65  		ctx, cancel := context.WithTimeout(context.Background(), deleteTimeout)
    66  		defer cancel()
    67  
    68  		if _, err := c.SNS.DeleteTopic(ctx, &sns.DeleteTopicInput{TopicArn: awssdk.String(arn)}); err != nil {
    69  			t.Logf("[testkit] topic %s の削除に失敗しました: %v", arn, err)
    70  		}
    71  	})
    72  
    73  	return arn
    74  }
