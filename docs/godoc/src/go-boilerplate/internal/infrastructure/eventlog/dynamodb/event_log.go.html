    43  // Append は、event を (StreamID, Sequence) の位置へ条件付きで書きます。位置が埋まっていたら
    44  // 既存 item の EventID を読み比べ、同じなら成功（冪等）、違えば ErrSequenceConflict を返します。
    45  func (s *store) Append(ctx context.Context, event realtime.DeliveryEvent) error {
    46  	ctx, endSpan := s.tracer.Start(ctx)
    47  	defer endSpan()
    48  
    49  	if err := event.Validate(); err != nil {
    50  		return err
    51  	}
    52  
    53  	_, err := s.c.PutItem(ctx, &dynamodb.PutItemInput{
    54  		TableName:           aws.String(s.table),
    55  		Item:                toItem(event),
    56  		ConditionExpression: aws.String("attribute_not_exists(" + attrStreamID + ")"),
    57  	})
    58  	if err == nil {
    59  		return s.advanceWatermark(ctx, event.StreamID, event.Sequence)
    60  	}
    61  
    62  	if !dynamodbclient.IsConditionalCheckFailed(err) {
    63  		return dynamodbclient.Normalize(err, "append event")
    64  	}
    65  
    66  	existing, ok, err := s.Find(ctx, event.StreamID, event.Sequence)
    67  	if err != nil {
    68  		return err
    69  	}
    70  
    71  	if !ok {
    72  		return xerrors.Wrap(apperror.ErrUnavailable, "append event: position was taken but the item could not be read back")
    73  	}
    74  
    75  	if existing.EventID != event.EventID {
    76  		return xerrors.Wrap(realtime.ErrSequenceConflict, "stream "+string(event.StreamID)+" sequence "+event.Sequence.String())
    77  	}
    78  
    79  	return s.advanceWatermark(ctx, event.StreamID, event.Sequence)
    80  }

    82  // AppendedThrough は、watermark item を強い一貫性で 1 件引きます。item が無い（1 度も追記して
    83  // いない）ときは 0 を返します。この item は TTL を持たないので、event が消えても残ります。
    84  func (s *store) AppendedThrough(ctx context.Context, streamID realtime.StreamID) (realtime.Sequence, error) {
    85  	ctx, endSpan := s.tracer.Start(ctx)
    86  	defer endSpan()
    87  
    88  	out, err := s.c.GetItem(ctx, &dynamodb.GetItemInput{
    89  		TableName:      aws.String(s.table),
    90  		Key:            key(streamID, watermarkSequence),
    91  		ConsistentRead: aws.Bool(true),
    92  	})
    93  	if err != nil {
    94  		return 0, dynamodbclient.Normalize(err, "read the append watermark")
    95  	}
    96  
    97  	if len(out.Item) == 0 {
    98  		return 0, nil
    99  	}
   100  
   101  	seq, err := dynamodbclient.NumberAttr(out.Item, attrAppendedThrough, itemKind)
   102  	if err != nil {
   103  		return 0, err
   104  	}
   105  
   106  	return realtime.Sequence(seq), nil
   107  }

   109  // ReadAfter は、cursor より後ろを昇順に強い一貫性で読みます。件数は Limit で切りますが、
   110  // DynamoDB は 1 MiB でも打ち切るため、続きの有無は LastEvaluatedKey で判断します。
   111  func (s *store) ReadAfter(ctx context.Context, q realtime.ReadAfterQuery) (realtime.ReadAfterResult, error) {
   112  	ctx, endSpan := s.tracer.Start(ctx)
   113  	defer endSpan()
   114  
   115  	limit := q.Limit
   116  	if limit <= 0 {
   117  		limit = defaultReadLimit
   118  	}
   119  
   120  	if limit > maxReadLimit {
   121  		limit = maxReadLimit
   122  	}
   123  
   124  	out, err := s.c.Query(ctx, &dynamodb.QueryInput{
   125  		TableName:                aws.String(s.table),
   126  		KeyConditionExpression:   aws.String(attrStreamID + " = :s AND #seq > :after"),
   127  		ExpressionAttributeNames: map[string]string{"#seq": attrSequence},
   128  		ExpressionAttributeValues: map[string]types.AttributeValue{
   129  			":s":     &types.AttributeValueMemberS{Value: string(q.StreamID)},
   130  			":after": &types.AttributeValueMemberN{Value: q.After.String()},
   131  		},
   132  		ConsistentRead:   aws.Bool(true),
   133  		ScanIndexForward: aws.Bool(true),
   134  		Limit:            aws.Int32(limit),
   135  	})
   136  	if err != nil {
   137  		return realtime.ReadAfterResult{}, dynamodbclient.Normalize(err, "read after")
   138  	}
   139  
   140  	events := make([]realtime.DeliveryEvent, 0, len(out.Items))
   141  	for _, item := range out.Items {
   142  		e, err := fromItem(item)
   143  		if err != nil {
   144  			return realtime.ReadAfterResult{}, err
   145  		}
   146  
   147  		events = append(events, e)
   148  	}
   149  
   150  	return realtime.ReadAfterResult{Events: events, HasMore: len(out.LastEvaluatedKey) > 0}, nil
   151  }

   153  // Latest は、stream の最後の event を降順 1 件の読み取りで返します。
   154  // watermark item は event ではないので、位置で除きます。
   155  func (s *store) Latest(ctx context.Context, streamID realtime.StreamID) (realtime.DeliveryEvent, bool, error) {
   156  	ctx, endSpan := s.tracer.Start(ctx)
   157  	defer endSpan()
   158  
   159  	out, err := s.c.Query(ctx, &dynamodb.QueryInput{
   160  		TableName:                aws.String(s.table),
   161  		KeyConditionExpression:   aws.String(attrStreamID + " = :s AND #seq > :watermark"),
   162  		ExpressionAttributeNames: map[string]string{"#seq": attrSequence},
   163  		ExpressionAttributeValues: map[string]types.AttributeValue{
   164  			":s":         &types.AttributeValueMemberS{Value: string(streamID)},
   165  			":watermark": &types.AttributeValueMemberN{Value: strconv.Itoa(watermarkSequence)},
   166  		},
   167  		ConsistentRead:   aws.Bool(true),
   168  		ScanIndexForward: aws.Bool(false),
   169  		Limit:            aws.Int32(1),
   170  	})
   171  	if err != nil {
   172  		return realtime.DeliveryEvent{}, false, dynamodbclient.Normalize(err, "latest")
   173  	}
   174  
   175  	if len(out.Items) == 0 {
   176  		return realtime.DeliveryEvent{}, false, nil
   177  	}
   178  
   179  	e, err := fromItem(out.Items[0])
   180  	if err != nil {
   181  		return realtime.DeliveryEvent{}, false, err
   182  	}
   183  
   184  	return e, true, nil
   185  }

   188  func (s *store) Find(ctx context.Context, streamID realtime.StreamID, seq realtime.Sequence) (realtime.DeliveryEvent, bool, error) {
   189  	ctx, endSpan := s.tracer.Start(ctx)
   190  	defer endSpan()
   191  
   192  	out, err := s.c.GetItem(ctx, &dynamodb.GetItemInput{
   193  		TableName:      aws.String(s.table),
   194  		Key:            key(streamID, seq),
   195  		ConsistentRead: aws.Bool(true),
   196  	})
   197  	if err != nil {
   198  		return realtime.DeliveryEvent{}, false, dynamodbclient.Normalize(err, "find")
   199  	}
   200  
   201  	if len(out.Item) == 0 {
   202  		return realtime.DeliveryEvent{}, false, nil
   203  	}
   204  
   205  	e, err := fromItem(out.Item)
   206  	if err != nil {
   207  		return realtime.DeliveryEvent{}, false, err
   208  	}
   209  
   210  	return e, true, nil
   211  }

   213  // originAttr は、起点 trace の carrier を item から復元します。無ければ nil です。
   214  func originAttr(item map[string]types.AttributeValue) map[string]string {
   215  	m, ok := item[attrOrigin].(*types.AttributeValueMemberM)
   216  	if !ok || len(m.Value) == 0 {
   217  		return nil
   218  	}
   219  
   220  	origin := make(map[string]string, len(m.Value))
   221  	for k, v := range m.Value {
   222  		if s, ok := v.(*types.AttributeValueMemberS); ok {
   223  			origin[k] = s.Value
   224  		}
   225  	}
   226  
   227  	return origin
   228  }

   230  // advanceWatermark は、追記済みの位置を seq まで進めます。後戻りさせないため、既存が seq 以上なら
   231  // 書き換えません。event を書いた後に呼ぶので、ここで失敗しても watermark が event を追い越しません
   232  // （呼出側の retry と Append の冪等性で埋まります）。
   233  func (s *store) advanceWatermark(ctx context.Context, streamID realtime.StreamID, seq realtime.Sequence) error {
   234  	_, err := s.c.UpdateItem(ctx, &dynamodb.UpdateItemInput{
   235  		TableName:           aws.String(s.table),
   236  		Key:                 key(streamID, watermarkSequence),
   237  		UpdateExpression:    aws.String("SET #w = :seq"),
   238  		ConditionExpression: aws.String("attribute_not_exists(#w) OR #w < :seq"),
   239  		ExpressionAttributeNames: map[string]string{
   240  			"#w": attrAppendedThrough,
   241  		},
   242  		ExpressionAttributeValues: map[string]types.AttributeValue{
   243  			":seq": &types.AttributeValueMemberN{Value: seq.String()},
   244  		},
   245  	})
   246  	if err != nil && !dynamodbclient.IsConditionalCheckFailed(err) {
   247  		return dynamodbclient.Normalize(err, "advance the append watermark")
   248  	}
   249  
   250  	return nil
   251  }

   260  // toItem は、封筒を item に写します。
   261  func toItem(e realtime.DeliveryEvent) map[string]types.AttributeValue {
   262  	item := key(e.StreamID, e.Sequence)
   263  	item[attrEventID] = &types.AttributeValueMemberS{Value: e.EventID}
   264  	item[attrType] = &types.AttributeValueMemberS{Value: e.Type}
   265  	item[attrOccurredAt] = &types.AttributeValueMemberS{Value: e.OccurredAt.UTC().Format(time.RFC3339Nano)}
   266  	item[attrSchemaVersion] = &types.AttributeValueMemberN{Value: strconv.Itoa(e.SchemaVersion)}
   267  	item[attrExpiresAt] = &types.AttributeValueMemberN{Value: strconv.FormatInt(e.OccurredAt.Add(realtime.EventLogRetention).Unix(), 10)}
   268  	if len(e.Payload) > 0 {
   269  		item[attrPayload] = &types.AttributeValueMemberB{Value: e.Payload}
   270  	}
   271  
   272  	if len(e.Origin) > 0 {
   273  		origin := make(map[string]types.AttributeValue, len(e.Origin))
   274  		for k, v := range e.Origin {
   275  			origin[k] = &types.AttributeValueMemberS{Value: v}
   276  		}
   277  
   278  		item[attrOrigin] = &types.AttributeValueMemberM{Value: origin}
   279  	}
   280  
   281  	return item
   282  }

   284  // fromItem は、item を封筒に戻します。
   285  func fromItem(item map[string]types.AttributeValue) (realtime.DeliveryEvent, error) {
   286  	seq, err := dynamodbclient.NumberAttr(item, attrSequence, itemKind)
   287  	if err != nil {
   288  		return realtime.DeliveryEvent{}, err
   289  	}
   290  
   291  	version, err := dynamodbclient.NumberAttr(item, attrSchemaVersion, itemKind)
   292  	if err != nil {
   293  		return realtime.DeliveryEvent{}, err
   294  	}
   295  
   296  	if version < 0 || version > math.MaxInt32 {
   297  		return realtime.DeliveryEvent{}, xerrors.Wrap(apperror.ErrInternal, "event log item: schema_version is out of range")
   298  	}
   299  
   300  	occurredAt, err := time.Parse(time.RFC3339Nano, dynamodbclient.StringAttr(item, attrOccurredAt))
   301  	if err != nil {
   302  		return realtime.DeliveryEvent{}, xerrors.Wrap(apperror.ErrInternal, "event log item: occurred_at: "+err.Error())
   303  	}
   304  
   305  	var payload json.RawMessage
   306  	if b, ok := item[attrPayload].(*types.AttributeValueMemberB); ok {
   307  		payload = json.RawMessage(b.Value)
   308  	}
   309  
   310  	return realtime.DeliveryEvent{
   311  		EventID:       dynamodbclient.StringAttr(item, attrEventID),
   312  		StreamID:      realtime.StreamID(dynamodbclient.StringAttr(item, attrStreamID)),
   313  		Sequence:      realtime.Sequence(seq),
   314  		Type:          dynamodbclient.StringAttr(item, attrType),
   315  		OccurredAt:    occurredAt,
   316  		SchemaVersion: int(version),
   317  		Payload:       payload,
   318  		Origin:        originAttr(item),
   319  	}, nil
   320  }
