    74  // Do は、req を送信し Response を返します。retry / backoff / deadline 規律を含みます。
    75  func (c *client) Do(ctx context.Context, req *Request) (*Response, error) {
    76  	if err := req.validate(); err != nil {
    77  		return nil, err
    78  	}
    79  
    80  	profile := c.registry.Profile(req.downstream)
    81  	ds := string(req.downstream)
    82  
    83  	// overall I/O デッドラインは実時間で設定する。context のタイマーは実時刻基準のため、
    84  	// 注入 clock を絶対期限に使うと実時刻とのズレで期限が意図せずずれる。
    85  	ctx, cancel := context.WithTimeout(ctx, profile.OverallTimeout)
    86  	defer cancel()
    87  	// retry 可否は注入 clock のタイムライン上で決定的に判定するため、期限を注入 clock 基準でも保持する。
    88  	overallDeadline := c.clk.Now().Add(profile.OverallTimeout)
    89  
    90  	c.metrics.InFlightAdd(ctx, ds, 1)
    91  	defer c.metrics.InFlightAdd(ctx, ds, -1)
    92  	c.budget.refill(req.downstream, profile.RetryBudgetRatio)
    93  
    94  	start := c.clk.Now()
    95  	resp, err := c.doWithRetry(ctx, req, profile, ds, overallDeadline)
    96  	c.metrics.RecordLatencyMs(ctx, ds, float64(c.clk.Now().Sub(start).Milliseconds()))
    97  
    98  	c.recordOutcome(ctx, ds, resp, err)
    99  	return resp, err
   100  }

   102  // doWithRetry は、breaker / budget / backoff を伴う retry ループを回します。
   103  // overallDeadline は注入 clock 基準の overall デッドラインで、retry 可否判定に用います。
   104  func (c *client) doWithRetry(ctx context.Context, req *Request, profile Profile, ds string, overallDeadline time.Time) (*Response, error) {
   105  	retrySafe := isRetrySafe(req)
   106  	br := c.breakers.get(req.downstream, profile.Breaker)
   107  
   108  	// breaker 状態は level（gauge）なので、最終状態を呼び出し終了時に 1 回だけ記録する。
   109  	defer func() { c.metrics.SetBreakerState(ctx, ds, int64(br.currentState())) }()
   110  
   111  	// MaxAttempts が未設定/不正でも最低 1 回は試行し、(nil, nil) を返さないようにする。
   112  	maxAttempts := max(1, profile.MaxAttempts)
   113  
   114  	var resp *Response
   115  	var err error
   116  	for attempt := 1; attempt <= maxAttempts; attempt++ {
   117  		allowed, generation := br.allow(c.clk.Now())
   118  		if !allowed {
   119  			if resp == nil {
   120  				return nil, xerrors.Wrap(errCircuitOpen, ds)
   121  			}
   122  			return resp, err
   123  		}
   124  
   125  		resp, err = c.attempt(ctx, req, profile)
   126  		// retryable は retry 継続判定に加え breaker への計上も兼ねる（!retryable を成功として記録するため、
   127  		// 429 や応答未取得の transport 失敗は breaker 上の失敗、4xx は成功扱いになる）。
   128  		retryable := RetryableOutcome(resp, err)
   129  		br.record(!retryable, c.clk.Now(), generation)
   130  
   131  		if !retrySafe || !retryable {
   132  			return resp, err
   133  		}
   134  		if attempt == maxAttempts {
   135  			return resp, err
   136  		}
   137  
   138  		// budget は「retry を実際に行うとき」だけ消費する。副作用のない overall デッドライン判定を
   139  		// 先に済ませ、待機で期限超過して retry しない場合にトークンを無駄消費しないようにする。
   140  		wait := retryWait(attempt, profile, resp, c.clk.Now())
   141  		if !c.canRetryWithin(overallDeadline, wait) {
   142  			return resp, err
   143  		}
   144  		if !c.budget.tryConsume(req.downstream) {
   145  			return resp, err
   146  		}
   147  
   148  		c.metrics.RecordRetry(ctx, ds)
   149  		if serr := c.sleeper.Sleep(ctx, wait); serr != nil {
   150  			return nil, normalizeTransportError(serr)
   151  		}
   152  	}
   153  	return resp, err
   154  }
   155  
   156  // canRetryWithin は、backoff 待機後も overall デッドライン内に次の試行を開始できるかを、
   157  // 注入 clock のタイムライン上で判定します（実時間・jitter に依存しない決定的判定）。
   158  func (c *client) canRetryWithin(overallDeadline time.Time, backoff time.Duration) bool {
   159  	return c.clk.Now().Add(backoff).Before(overallDeadline)
   160  }

   162  // attempt は、1 回の HTTP 試行を行います。
   163  func (c *client) attempt(ctx context.Context, req *Request, profile Profile) (*Response, error) {
   164  	attemptCtx, cancel := context.WithTimeout(ctx, profile.PerAttemptTimeout)
   165  	defer cancel()
   166  	attemptCtx = observability.ContextWithTracePropagation(attemptCtx, profile.PropagateTrace)
   167  	attemptCtx = observability.ContextWithAllowPrivateNetwork(attemptCtx, profile.AllowPrivateNetwork)
   168  
   169  	httpReq, err := buildRequest(attemptCtx, req)
   170  	if err != nil {
   171  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, redactErrMessage(err))
   172  	}
   173  
   174  	httpResp, err := c.httpClient.Do(httpReq)
   175  	if err != nil {
   176  		return nil, normalizeTransportError(err)
   177  	}
   178  	defer func() { _ = httpResp.Body.Close() }()
   179  
   180  	body, err := readBody(httpResp.Body, profile.MaxResponseBytes)
   181  	if err != nil {
   182  		// 上限超過は決定的失敗なのでそのまま返す（非 retry 化は RetryableOutcome が担う）。
   183  		// それ以外は応答未完了の transport 失敗として正規化する（Do 失敗経路と一貫）。
   184  		if xerrors.Is(err, errResponseTooLarge) {
   185  			return nil, err
   186  		}
   187  		return nil, normalizeTransportError(err)
   188  	}
   189  
   190  	resp := &Response{
   191  		StatusCode: httpResp.StatusCode,
   192  		Header:     Header(httpResp.Header),
   193  		Body:       body,
   194  	}
   195  
   196  	if appErr := statusToAppError(httpResp.StatusCode); appErr != nil {
   197  		msg := fmt.Sprintf("downstream %s returned status %d", req.downstream, httpResp.StatusCode)
   198  		return resp, xerrors.Wrap(appErr, msg)
   199  	}
   200  	return resp, nil
   201  }

   203  // recordOutcome は、呼び出し結果を metrics に計上します。
   204  func (c *client) recordOutcome(ctx context.Context, ds string, resp *Response, err error) {
   205  	var class string
   206  	if resp != nil {
   207  		class = statusClass(resp.StatusCode)
   208  		c.metrics.RecordRequest(ctx, ds, class)
   209  	}
   210  	if err == nil {
   211  		return
   212  	}
   213  
   214  	switch {
   215  	case resp != nil:
   216  		c.metrics.RecordError(ctx, ds, "http_"+class)
   217  	case xerrors.Is(err, errCircuitOpen):
   218  		c.metrics.RecordError(ctx, ds, "circuit_open")
   219  	case xerrors.Is(err, apperror.ErrCanceled):
   220  		c.metrics.RecordError(ctx, ds, "canceled")
   221  	default:
   222  		c.metrics.RecordError(ctx, ds, "transport")
   223  	}
   224  }

   226  // buildRequest は、自前型の Request から net/http のリクエストを組み立てます。
   227  func buildRequest(ctx context.Context, req *Request) (*http.Request, error) {
   228  	var body io.Reader
   229  	if len(req.body) > 0 {
   230  		body = bytes.NewReader(req.body)
   231  	}
   232  
   233  	httpReq, err := http.NewRequestWithContext(ctx, req.method.String(), req.url, body)
   234  	if err != nil {
   235  		return nil, err
   236  	}
   237  
   238  	for key, values := range req.header {
   239  		for _, v := range values {
   240  			httpReq.Header.Add(key, v)
   241  		}
   242  	}
   243  	if req.idempotencyKey != "" {
   244  		httpReq.Header.Set(idempotencyHeader, req.idempotencyKey)
   245  	}
   246  	return httpReq, nil
   247  }

   249  // readBody は、レスポンスボディを maxBytes まで読み込みます。上限超過時はエラーを返します。
   250  func readBody(body io.Reader, maxBytes int64) ([]byte, error) {
   251  	limited := io.LimitReader(body, maxBytes+1)
   252  	data, err := io.ReadAll(limited)
   253  	if err != nil {
   254  		return nil, err
   255  	}
   256  	if int64(len(data)) > maxBytes {
   257  		return nil, errResponseTooLarge
   258  	}
   259  	return data, nil
   260  }
