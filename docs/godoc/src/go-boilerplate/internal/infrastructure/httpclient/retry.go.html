    25  // isRetrySafe は、req が retry してよいリクエストかを返します。
    26  // 冪等メソッド(GET/PUT/DELETE)は常に安全、非冪等メソッド(POST/PATCH)は AllowRetry 明示時のみ安全です。
    27  func isRetrySafe(req *Request) bool {
    28  	switch req.method {
    29  	case MethodGet(), MethodPut(), MethodDelete():
    30  		return true
    31  	case MethodPost(), MethodPatch():
    32  		return req.allowRetry
    33  	default:
    34  		return false
    35  	}
    36  }

    38  // RetryableOutcome は、試行結果が retry 対象かを返します。
    39  // 5xx / 429 / 応答未取得の transport 失敗は retry 対象、4xx / 成功 / ctx cancel は対象外です。
    40  // この判定表を持つのはこの関数だけです（公開理由は README の Design Policy を参照）。
    41  func RetryableOutcome(resp *Response, err error) bool {
    42  	if err == nil {
    43  		return false
    44  	}
    45  	if xerrors.Is(err, apperror.ErrCanceled) {
    46  		return false
    47  	}
    48  	if resp != nil {
    49  		return resp.StatusCode == retryableStatusTooManyRequests ||
    50  			resp.StatusCode >= retryableStatusServerErrorMin
    51  	}
    52  	// レスポンス上限超過は ErrUnavailable(503) だが再試行しても改善しない決定的失敗なので明示除外する
    53  	// （HTTP ステータス写像は 503 のまま保ち、retry 可否だけをここで分離する）。
    54  	if xerrors.Is(err, errResponseTooLarge) {
    55  		return false
    56  	}
    57  	// 応答未取得は transport 失敗(ErrUnavailable)のみ retry 対象。
    58  	// buildRequest 由来の client エラー(ErrInvalidArgument 等)は決定的なので retry しない。
    59  	return xerrors.Is(err, apperror.ErrUnavailable)
    60  }

    62  // computeBackoff は、attempt（1 起算）に対する指数バックオフ + full jitter の待機時間を返します。
    63  func computeBackoff(attempt int, profile Profile) time.Duration {
    64  	if attempt < 1 {
    65  		attempt = 1
    66  	}
    67  
    68  	exp := backoff.Exponential{
    69  		Initial:    profile.BaseBackoff,
    70  		Max:        profile.MaxBackoff,
    71  		Multiplier: backoffMultiplier,
    72  	}
    73  	return retry.Full(exp.Duration(attempt - 1))
    74  }

    76  // retryWait は、次の retry までの待機時間を決定します。
    77  // レスポンスに Retry-After があればそれを優先し、無ければ指数バックオフ + full jitter を用います。
    78  // Retry-After が過大でも overall deadline 規律（canRetryWithin）が別途打ち切るため、ここでは上限を設けません。
    79  func retryWait(attempt int, profile Profile, resp *Response, now time.Time) time.Duration {
    80  	if resp != nil {
    81  		if d, ok := retryAfter(resp.Header, now); ok {
    82  			return d
    83  		}
    84  	}
    85  	return computeBackoff(attempt, profile)
    86  }

    88  // retryAfter は、Retry-After ヘッダ（delay-seconds 形式 / HTTP-date 形式）を解釈し待機時間を返します。
    89  // 解釈できない / 不在の場合は (0, false) を返し、呼び出し側は通常のバックオフへ fallback します。
    90  func retryAfter(header Header, now time.Time) (time.Duration, bool) {
    91  	if header == nil {
    92  		return 0, false
    93  	}
    94  	values, ok := header[retryAfterHeader]
    95  	if !ok || len(values) == 0 {
    96  		return 0, false
    97  	}
    98  
    99  	raw := strings.TrimSpace(values[0])
   100  	if raw == "" {
   101  		return 0, false
   102  	}
   103  
   104  	// delay-seconds 形式（非負整数秒）。
   105  	if secs, err := strconv.Atoi(raw); err == nil {
   106  		if secs < 0 {
   107  			return 0, false
   108  		}
   109  		return time.Duration(secs) * time.Second, true
   110  	}
   111  
   112  	// HTTP-date 形式。
   113  	if at, err := http.ParseTime(raw); err == nil {
   114  		return max(0, at.Sub(now)), true
   115  	}
   116  	return 0, false
   117  }
