   127  	t.Parallel()
   128  
   129  	profile := Profile{
   130  		BaseBackoff: 10 * time.Millisecond,
   131  		MaxBackoff:  1 * time.Second,
   132  	}
   133  
   134  	t.Run("正常系", func(t *testing.T) {
   135  		t.Parallel()
   136  
   137  		t.Run("各attemptのバックオフが指数cap内のfull_jitter範囲に収まる", func(t *testing.T) {
   138  			t.Parallel()
   139  
   140  			for attempt := 1; attempt <= 8; attempt++ {
   141  				upperBound := profile.BaseBackoff << (attempt - 1)
   142  				if upperBound <= 0 || upperBound > profile.MaxBackoff {
   143  					upperBound = profile.MaxBackoff
   144  				}
   145  
   146  				// jitter は乱数のため複数回試行して常に [0, upperBound] に収まることを確認する。
   147  				for range 50 {
   148  					got := computeBackoff(attempt, profile)
   149  					assert.GreaterOrEqual(t, got, time.Duration(0))
   150  					assert.LessOrEqual(t, got, upperBound)
   151  				}
   152  			}
   153  		})
   154  
   155  		t.Run("attemptが0以下でも1として扱う", func(t *testing.T) {
   156  			t.Parallel()
   157  
   158  			got := computeBackoff(0, profile)
   159  			assert.GreaterOrEqual(t, got, time.Duration(0))
   160  			assert.LessOrEqual(t, got, profile.BaseBackoff)
   161  		})
   162  	})

   166  	t.Parallel()
   167  
   168  	profile := Profile{
   169  		BaseBackoff: 10 * time.Millisecond,
   170  		MaxBackoff:  time.Second,
   171  	}
   172  
   173  	now := time.Unix(1000, 0).UTC()
   174  
   175  	t.Run("正常系", func(t *testing.T) {
   176  		t.Parallel()
   177  
   178  		t.Run("respがnilならバックオフ範囲の待機時間を返す", func(t *testing.T) {
   179  			t.Parallel()
   180  
   181  			// jitter があるため複数回試行し attempt1 の cap 範囲に収まることを確認する。
   182  			for range 50 {
   183  				got := retryWait(1, profile, nil, now)
   184  				assert.GreaterOrEqual(t, got, time.Duration(0))
   185  				assert.LessOrEqual(t, got, profile.BaseBackoff)
   186  			}
   187  		})
   188  
   189  		t.Run("respはあるがRetry-After不在ならバックオフ範囲の待機時間を返す", func(t *testing.T) {
   190  			t.Parallel()
   191  
   192  			resp := &Response{StatusCode: 503, Header: Header{}}
   193  			for range 50 {
   194  				got := retryWait(1, profile, resp, now)
   195  				assert.GreaterOrEqual(t, got, time.Duration(0))
   196  				assert.LessOrEqual(t, got, profile.BaseBackoff)
   197  			}
   198  		})
   199  
   200  		t.Run("Retry-Afterがあればバックオフより優先する", func(t *testing.T) {
   201  			t.Parallel()
   202  
   203  			resp := &Response{StatusCode: 503, Header: Header{retryAfterHeader: {"5"}}}
   204  			got := retryWait(1, profile, resp, now)
   205  			assert.Equal(t, 5*time.Second, got) // バックオフ(<=10ms)ではなく Retry-After(5s)
   206  		})
   207  
   208  		t.Run("HTTP-date形式のRetry-Afterは固定時刻との差分を待機時間とする", func(t *testing.T) {
   209  			t.Parallel()
   210  
   211  			future := now.Add(7 * time.Second).Format(http.TimeFormat)
   212  			resp := &Response{StatusCode: 503, Header: Header{retryAfterHeader: {future}}}
   213  			got := retryWait(1, profile, resp, now)
   214  			assert.Equal(t, 7*time.Second, got) // 渡した now を基準に決定論的に算出する
   215  		})
   216  	})
