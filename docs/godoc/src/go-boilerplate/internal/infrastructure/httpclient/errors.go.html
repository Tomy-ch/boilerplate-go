    39  // statusToAppError は、HTTP ステータスコードを apperror sentinel に写像します。
    40  // 2xx は nil を返します。3xx はリダイレクト非追従のため未解決とみなし ErrUnavailable を返します
    41  // （resp に Location を残すので、追従が必要な呼び出し側はそれを参照できます）。
    42  // 写像は substrate 内部に閉じます。
    43  func statusToAppError(statusCode int) error {
    44  	switch statusCode {
    45  	case http.StatusBadRequest:
    46  		return apperror.ErrInvalidArgument
    47  	case http.StatusUnauthorized:
    48  		return apperror.ErrUnauthenticated
    49  	case http.StatusForbidden:
    50  		return apperror.ErrPermissionDenied
    51  	case http.StatusNotFound:
    52  		return apperror.ErrNotFound
    53  	case http.StatusConflict:
    54  		return apperror.ErrConflict
    55  	case http.StatusUnprocessableEntity:
    56  		return apperror.ErrValidation
    57  	case http.StatusTooManyRequests:
    58  		return apperror.ErrTooManyRequests
    59  	}
    60  
    61  	switch {
    62  	case statusCode >= statusServerErrorMin:
    63  		return apperror.ErrUnavailable
    64  	case statusCode >= statusClientErrorMin:
    65  		return apperror.ErrInvalidArgument
    66  	case statusCode >= statusRedirectMin:
    67  		return apperror.ErrUnavailable // 3xx: リダイレクト非追従のため未解決
    68  	default:
    69  		return nil // 2xx
    70  	}
    71  }

    73  // normalizeTransportError は、応答取得前の transport 事象を apperror sentinel に写像します。
    74  // ctx cancel は ErrCanceled、それ以外（network / DNS / TLS / deadline 超過）は ErrUnavailable です。
    75  func normalizeTransportError(err error) error {
    76  	if xerrors.Is(err, context.Canceled) {
    77  		return xerrors.Wrap(apperror.ErrCanceled, redactErrMessage(err))
    78  	}
    79  	return xerrors.Wrap(apperror.ErrUnavailable, redactErrMessage(err))
    80  }

    82  // redactErrMessage は、エラーメッセージから URL のクエリ等の機密になり得る情報を除去します。
    83  // *url.Error は完全 URL（クエリ込み）を出力するため、クエリ・userinfo・fragment を除去し scheme/host/path を残します。
    84  func redactErrMessage(err error) string {
    85  	var urlErr *url.Error
    86  	if xerrors.As(err, &urlErr) {
    87  		return fmt.Sprintf("%s %s: %v", urlErr.Op, redactURL(urlErr.URL), urlErr.Err)
    88  	}
    89  	return err.Error()
    90  }

    92  // redactURL は、URL から機密になり得るクエリ・userinfo・fragment のみを除去し、
    93  // 診断に有用な scheme/host/path は残します。パース不能時のみ "upstream" を返します。
    94  func redactURL(rawURL string) string {
    95  	parsed, err := url.Parse(rawURL)
    96  	if err != nil {
    97  		return "upstream"
    98  	}
    99  	parsed.RawQuery = ""
   100  	parsed.User = nil
   101  	parsed.Fragment = ""
   102  	return parsed.String()
   103  }
