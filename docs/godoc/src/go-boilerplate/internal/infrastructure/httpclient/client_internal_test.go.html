   357  // Test_client_doWithRetry は、deadline で打ち切られた retry が budget を消費しないことを、外部から
   358  // 観測しにくい内部状態（client.budget.tokens）を直接検査して固定します。
   359  func Test_client_doWithRetry(t *testing.T) {
   360  	t.Parallel()
   361  
   362  	t.Run("異常系", func(t *testing.T) {
   363  		t.Parallel()
   364  
   365  		t.Run("deadline で打ち切られた retry は budget を消費せず残量が実施 retry 数に一致する", func(t *testing.T) {
   366  			t.Parallel()
   367  
   368  			var hits atomic.Int32
   369  			srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, _ *http.Request) {
   370  				hits.Add(1)
   371  				w.WriteHeader(http.StatusServiceUnavailable)
   372  			}))
   373  			t.Cleanup(srv.Close)
   374  
   375  			profile := DefaultProfile()
   376  			profile.MaxAttempts = 20
   377  			// backoff は極小にし、打ち切りは StepClock(20s/step) と OverallTimeout(60s) の関係のみで決める。
   378  			profile.BaseBackoff = time.Millisecond
   379  			profile.MaxBackoff = time.Millisecond
   380  			profile.OverallTimeout = 60 * time.Second
   381  			profile.PerAttemptTimeout = 10 * time.Second // 既定 3s より広く取り、実時間側を打ち切り要因から外す
   382  			profile.RetryBudgetRatio = 100               // budget が deadline より先に絞らないよう十分に確保
   383  			profile.Breaker.MinRequests = 1 << 30        // breaker を開かせない
   384  			const ds Downstream = "acct"
   385  			registry := NewRegistry(map[Downstream]Profile{ds: profile})
   386  
   387  			// Sleep で fake 時刻を 20s/サイクル進める clock を注入し、打ち切りを注入 clock 基準の
   388  			// canRetryWithin（overallDeadline 到達）のみに担わせる（jitter 非依存）。
   389  			// OverallTimeout / PerAttemptTimeout の実時間 ctx タイマーは、fake sleep が実待機せず
   390  			// 往復先も httptest サーバのためテストが ms オーダーで完了し、発火しない。
   391  			fakeClock := clocktestkit.NewStepClock(time.Now().Add(time.Hour), 20*time.Second)
   392  			c, ok := New(
   393  				observability.NewNoopHTTPClientTransport(t),
   394  				fakeClock,
   395  				fakeClock,
   396  				registry,
   397  				observability.NewNoopHTTPClientMetrics(t),
   398  			).(*client)
   399  			require.True(t, ok)
   400  
   401  			_, err := c.Do(context.Background(), NewRequest(MethodGet(), ds, srv.URL))
   402  			require.ErrorIs(t, err, apperror.ErrUnavailable)
   403  
   404  			// 20s/step × 3 backoff で fakeNow がちょうど deadline(60s) に達し、hits=4（= retry 3 回）。
   405  			// 4 回目の attempt は canRetryWithin=false で打ち切られ、budget を消費しない。
   406  			require.Equal(t, int32(4), hits.Load())
   407  
   408  			initial := min(retryBudgetMaxTokens, retryBudgetInitialTokens+profile.RetryBudgetRatio)
   409  			retriesPerformed := hits.Load() - 1
   410  			// 残量 = 初期在庫 - 実施 retry 数。旧実装は打ち切り分も消費して残量が 1 少なくなる。
   411  			assert.InDelta(t, initial-float64(retriesPerformed)*retryBudgetRetryCost, c.budget.tokens[ds], 1e-9)
   412  		})
   413  	})
