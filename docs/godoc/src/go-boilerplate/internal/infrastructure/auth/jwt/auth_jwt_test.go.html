    69  // signToken は指定した署名方式・typ・クレームでトークン文字列を組み立てます。
    70  func signToken(t *testing.T, key *rsa.PrivateKey, method jwtlib.SigningMethod, typ string, claims jwtlib.MapClaims) string {
    71  	t.Helper()
    72  	token := jwtlib.NewWithClaims(method, claims)
    73  	if typ != "" {
    74  		token.Header["typ"] = typ
    75  	}
    76  
    77  	var (
    78  		signed string
    79  		err    error
    80  	)
    81  	switch method {
    82  	case jwtlib.SigningMethodNone:
    83  		signed, err = token.SignedString(jwtlib.UnsafeAllowNoneSignatureType)
    84  	case jwtlib.SigningMethodHS256:
    85  		signed, err = token.SignedString([]byte("shared-secret"))
    86  	default:
    87  		signed, err = token.SignedString(key)
    88  	}
    89  	require.NoError(t, err)
    90  	return signed
    91  }
