    51  func Test_resolver_Resolve(t *testing.T) {
    52  	t.Parallel()
    53  
    54  	testDB := testkit.NewTestDB(t)
    55  	lt := observability.NewMockInfraLayerTracer(t)
    56  	txm := testkit.NewTestTransactionRunner(t)
    57  
    58  	repo := &resolver{
    59  		tracer: lt,
    60  		db:     testDB,
    61  	}
    62  
    63  	newAuthn := func(t *testing.T, issuer, subject string) *authbd.Authn {
    64  		t.Helper()
    65  		authn, err := authbd.New(subject, issuer, nil, nil)
    66  		require.NoError(t, err)
    67  		return authn
    68  	}
    69  
    70  	t.Run("正常系", func(t *testing.T) {
    71  		t.Parallel()
    72  
    73  		t.Run("JWT issuer と subject に対応する未削除ユーザーの UserID を解決する", func(t *testing.T) {
    74  			t.Parallel()
    75  
    76  			txm.WithinTx(func(ctx context.Context) {
    77  				resolved, err := repo.Resolve(ctx, newAuthn(t, config.ResolvedAuthIssuer(t), "user-john-doe"))
    78  				require.NoError(t, err)
    79  				require.True(t, resolved.HasUserID())
    80  				userID, err := resolved.UserID()
    81  				require.NoError(t, err)
    82  				assert.Equal(t, johnUserID, userID.String())
    83  			})
    84  		})
    85  
    86  		t.Run("mock issuer と subject でも同じユーザーの UserID を解決する", func(t *testing.T) {
    87  			t.Parallel()
    88  
    89  			txm.WithinTx(func(ctx context.Context) {
    90  				resolved, err := repo.Resolve(ctx, newAuthn(t, testIssuerMock, johnUserID))
    91  				require.NoError(t, err)
    92  				require.True(t, resolved.HasUserID())
    93  				userID, err := resolved.UserID()
    94  				require.NoError(t, err)
    95  				assert.Equal(t, johnUserID, userID.String())
    96  			})
    97  		})
    98  	})

   141  					INSERT INTO users (id, first_name, last_name, email, phone, prefecture_id, city, street, postal_code)
   142  					VALUES ('00000000-0000-0000-0000-000000000000', 'Zero', 'User', 'zero.user@example.com', '000-0000',
   143  					        (SELECT id FROM prefectures LIMIT 1), '市区町村', '1-1', '000-0000')`)
   144  				require.NoError(t, err)
   145  
   146  				_, err = db.Exec(ctx, `
   147  					INSERT INTO user_identities (id, user_id, issuer, subject)
   148  					VALUES ('3f2a1c4e-0000-4000-8000-0000000000ff', '00000000-0000-0000-0000-000000000000', $1, $2)`,
   149  					testIssuerMock, zeroUserSubject)
   150  				require.NoError(t, err)
   151

   155  			})
