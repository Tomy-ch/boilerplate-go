    38  func Test_registrar_Register(t *testing.T) {
    39  	t.Parallel()
    40  
    41  	testDB := testkit.NewTestDB(t)
    42  	lt := observability.NewMockInfraLayerTracer(t)
    43  	txm := testkit.NewTestTransactionRunner(t)
    44  
    45  	// user_identities は (issuer, subject) に加えて (user_id, issuer) にも一意制約を持つ。
    46  	// seed 済みの利用者は既に本番相当の issuer で 1 行持つため、テストごとに issuer を分ける。
    47  	reg := &registrar{
    48  		tracer: lt,
    49  		db:     testDB,
    50  	}
    51  	res := &resolver{
    52  		tracer: lt,
    53  		db:     testDB,
    54  	}
    55  
    56  	t.Run("正常系", func(t *testing.T) {
    57  		t.Parallel()
    58  
    59  		t.Run("結び付けた issuer と subject が resolver から解決できるようになる", func(t *testing.T) {
    60  			t.Parallel()
    61  
    62  			txm.WithinTx(func(ctx context.Context) {
    63  				issuer := "https://registrar-test.example.com/roundtrip"
    64  				subject := "user-registrar-roundtrip"
    65  
    66  				userID, err := uuid.Parse(johnUserID)
    67  				require.NoError(t, err)
    68  
    69  				require.NoError(t, reg.Register(ctx, userID, issuer, subject))
    70  
    71  				authn, err := authbd.New(subject, issuer, nil, nil)
    72  				require.NoError(t, err)
    73  
    74  				resolved, err := res.Resolve(ctx, authn)
    75  				require.NoError(t, err)
    76  				require.True(t, resolved.HasUserID())
    77  				got, err := resolved.UserID()
    78  				require.NoError(t, err)
    79  				assert.Equal(t, johnUserID, got.String())
    80  			})
    81  		})
    82  	})
    83  
    84  	t.Run("異常系", func(t *testing.T) {
    85  		t.Parallel()
