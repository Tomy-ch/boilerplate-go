     1  // Package discontinuation は、商品の廃番と進行中の購入にまたがる規則を表すドメインサービスです。
     2  //
     3  // 「進行中の購入がある商品は廃番にできない」は、商品と購入のどちらの自然な責務にもなりません。
     4  // 商品集約は自分を含む購入を知らず、購入集約は商品が廃番にできるかを知らないためです。
     5  // 状態を持たず、読み込み済みの状態だけを受け取って可否を返すため、集約の外に置くドメインサービスとして
     6  // 表現します（membership と同じ形）。
     7  //
     8  // 判定に必要な状態の取得（ロック・問い合わせ）と、その順序・トランザクション境界は Usecase の
     9  // 責務です。本パッケージは I/O を持たず、context.Context も受け取りません。

    12  import (
    13  	"go-boilerplate/internal/apperror"
    14  	"go-boilerplate/internal/domain/purchase"
    15  	"go-boilerplate/pkg/xerrors"

    18  // ErrInProgressPurchaseExists は、進行中の購入が残っている商品の廃番を拒否したことを表します。
    19  var ErrInProgressPurchaseExists = xerrors.Wrap(apperror.ErrConflict, "product has in-progress purchases")
    20  
    21  // EnsureDiscontinuable は、商品を廃番にしてよい状態かを判定します。statuses は、その商品を明細に
    22  // 持つ購入が取っているステータスです（重複の有無は問いません）。進行中の購入が 1 件でも残っている
    23  // 場合は ErrInProgressPurchaseExists を返します。
    24  //
    25  // 進行中かどうかの定義は購入集約が持ちます（purchase.Status.IsTerminal の否定）。
    26  func EnsureDiscontinuable(statuses []purchase.Status) error {
    27  	for _, s := range statuses {
    28  		if !s.IsTerminal() {
    29  			return ErrInProgressPurchaseExists
    30  		}
    31  	}
    32  
    33  	return nil
    34  }
