    93  		name:                  attrs.Name,
    94  		description:           ptr.Copy(attrs.Description),

    96  		quantity:              attrs.Quantity,

   102  		images:                sortImagesByDisplaySort(attrs.Images),

   108  // validateAttributes は、商品属性の不変条件を検証します。生成時と更新時で同一の条件を課します。
   109  // 最初の違反で打ち切らず、利用者が直せる項目をすべて検証してから束ねて返します。
   110  // 違反した項目の識別子を付けるのはこの関数だけで、個別の検証関数は項目を名指ししないエラーを
   111  // 返します。複数項目にまたがる違反でどれを名指しするか、採番済みの値の違反を名指ししないことも
   112  // ここで決めます（internal/domain/README.md の Validation を参照）。
   113  func validateAttributes(attrs Attributes) error {
   114  	var errs []error
   115  	var fields []string
   116  
   117  	if ok, msg := stringkit.ValidateInRange(attrs.Name, minNameLength, maxNameLength); !ok {
   118  		errs = append(errs, xerrors.Wrap(ErrInvalidName, msg))
   119  		fields = append(fields, FieldName)
   120  	}
   121  	if err := validateQuantity(int64(attrs.Quantity)); err != nil {
   122  		errs = append(errs, err)
   123  		fields = append(fields, FieldQuantity)
   124  	}
   125  	if err := validateStockWarningThreshold(attrs.StockWarningThreshold); err != nil {
   126  		errs = append(errs, err)
   127  		fields = append(fields, FieldStockWarningThreshold)
   128  	}
   129  	if attrs.Status.id.IsNil() {
   130  		errs = append(errs, xerrors.Wrap(ErrInvalidStatusID, "status is required"))
   131  		fields = append(fields, FieldStatusID)
   132  	}
   133  	if attrs.Category.id.IsNil() {
   134  		errs = append(errs, xerrors.Wrap(ErrInvalidCategoryID, "category is required"))
   135  		fields = append(fields, FieldCategoryID)
   136  	}
   137  	// 名指しするのは publishedAt（理由は docs/spec/domain/product.md の Cross-field Invariants）。
   138  	if err := validateDiscontinuedAt(attrs.DiscontinuedAt, attrs.PublishedAt); err != nil {
   139  		errs = append(errs, err)
   140  		fields = append(fields, FieldPublishedAt)
   141  	}
   142  	if err := validateImages(attrs.Images); err != nil {
   143  		errs = append(errs, err)
   144  		// 画像 ID は採番済み（サーバ内部）で、利用者が直せる項目ではないため名指ししません。
   145  		if !xerrors.Is(err, ErrInvalidID) {
   146  			fields = append(fields, FieldImages)
   147  		}
   148  	}
   149  
   150  	if len(errs) == 0 {
   151  		return nil
   152  	}
   153  	// 名指しする項目が 1 つも無いときは Meta を付けません。空の details を付けると、
   154  	// 「利用者が直せる項目がある」と偽って伝えることになります。
   155  	if len(fields) == 0 {
   156  		return xerrors.Join(errs...)
   157  	}
   158  	return apperror.WithDetails(xerrors.Join(errs...), fields...)
   159  }

   161  // Update は、商品の属性を更新します。生成時と同一の不変条件を課し、違反する場合はエンティティを
   162  // 変更せずに検証エラーを返します。attrs は部分更新を解決した後の確定値であり、据え置く属性には
   163  // 現在値が渡されます。Images は集合ごとの置き換えです。バージョンはここでは進めません
   164  // （詳細は docs/spec/domain/product.md の Behavior Methods を参照）。
   165  func (p *Product) Update(attrs Attributes) error {
   166  	if err := validateAttributes(attrs); err != nil {
   167  		return err
   168  	}
   169  	if err := p.ensureDiscontinuationKept(attrs.DiscontinuedAt); err != nil {
   170  		return err
   171  	}
   172  
   173  	p.name = attrs.Name
   174  	p.description = ptr.Copy(attrs.Description)
   175  	p.price = attrs.Price
   176  	p.quantity = attrs.Quantity
   177  	p.stockWarningThreshold = ptr.Copy(attrs.StockWarningThreshold)
   178  	p.status = attrs.Status
   179  	p.category = attrs.Category
   180  	p.publishedAt = ptr.Copy(attrs.PublishedAt)
   181  	p.discontinuedAt = ptr.Copy(attrs.DiscontinuedAt)
   182  	p.images = sortImagesByDisplaySort(attrs.Images)
   183  
   184  	return nil

   187  // AdjustStock は、在庫数を delta の分だけ増減します。delta は正で補充、負で差し引きを表します。
   188  // 増減後の在庫が負になる場合は、在庫を変更せずに検証エラーを返します。
   189  // バージョンの採番タイミングは Update と同じです。
   190  func (p *Product) AdjustStock(delta int) error {
   191  	// 増減の途中結果は在庫の表現範囲を超えうるため、検証を通すまでは広い幅で保持します。
   192  	adjusted := int64(p.quantity) + int64(delta)
   193  	if err := validateQuantity(adjusted); err != nil {
   194  		return err
   195  	}
   196  
   197  	p.quantity = int(adjusted)
   198  
   199  	return nil
   200  }

   307  	if now.IsZero() {

   317  	return nil
