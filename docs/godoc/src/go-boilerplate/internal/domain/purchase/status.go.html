    62  // TerminalStatusCodes は、終端状態のステータスが取る業務キーを返します。
    63  //
    64  // 「進行中」を SQL の絞り込みで表す経路が、終端の集合をドメインから受け取るために用います。
    65  // 判定そのものは [Status.IsTerminal] が持ち、ここはその集合を列挙するだけです。SQL 側に終端の
    66  // 一覧を書き写すと規則が 2 箇所になり、片方だけが動いたときに黙って食い違います。
    67  func TerminalStatusCodes() []int {
    68  	all := allStatuses()
    69  	codes := make([]int, 0, len(all))
    70  	for _, s := range all {
    71  		if s.IsTerminal() {
    72  			codes = append(codes, s.code)
    73  		}
    74  	}
    75  
    76  	return codes
    77  }

   111  // 集約の状態（発送記録の有無など）に依存する条件は集約側が併せて検証します。
   112  func (s Status) CanTransitionTo(next Status) bool {
   113  	if s.IsTerminal() || s == next {
   114  		return false
   115  	}
   116  	switch next {
   117  	case StatusCanceled:
   118  		// キャンセルは進行中からのみ。発送済みからのキャンセル不可は集約が発送記録で併せて弾く。
   119  		return true
   120  	case StatusPaid:
   121  		// 処理中は支払いを終えた後の段階であり対象外。
   122  		return s == StatusUnprocessed || s == StatusAccepted || s == StatusConfirming
   123  	case StatusShipped:
   124  		return s == StatusPaid
   125  	case StatusDelivered:
   126  		return s == StatusShipped
   127  	default:
   128  		return false
   129  	}
   130  }
