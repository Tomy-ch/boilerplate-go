   128  // validateDetails は、購入集約が保持する明細の集合が満たすべき不変条件を検証します。
   129  // 明細 ID / 商品 ID が未設定、数量が最小値未満の場合はそれぞれ ErrInvalidID / ErrInvalidQuantity を、
   130  // 同一商品 ID が重複する場合は ErrDuplicateProductID を返します。
   131  // 空の集合は呼び出し元が ErrEmptyDetails で扱うため、ここでは許容します。
   132  func validateDetails(details []PurchaseDetail) error {
   133  	seen := make(map[uuid.UUID]struct{}, len(details))
   134  	for _, d := range details {
   135  		if d.id.IsNil() {
   136  			return xerrors.Wrap(ErrInvalidID, "detail id is required")
   137  		}
   138  		if d.productID.IsNil() {
   139  			return xerrors.Wrap(ErrInvalidID, "detail product id is required")
   140  		}
   141  		if d.quantity < minQuantity {
   142  			return xerrors.Wrap(
   143  				ErrInvalidQuantity, fmt.Sprintf("quantity must be %d or greater, got %d", minQuantity, d.quantity),
   144  			)
   145  		}
   146  		if _, dup := seen[d.productID]; dup {
   147  			return xerrors.Wrap(ErrDuplicateProductID, fmt.Sprintf("product %s appears more than once", d.productID))
   148  		}
   149  		seen[d.productID] = struct{}{}
   150  	}
   151  
   152  	return nil
   153  }

   155  // buildDetails は、明細の入力をロック済み在庫と突き合わせて購入明細を組み立て、単価スナップショットに
   156  // よる小計を価格スケールで返します（検証条件は New を参照）。入力の順序を保った明細を返します。
   157  func buildDetails(inputs []DetailInput, locked []LockedProduct) ([]PurchaseDetail, decimal.Decimal, error) {
   158  	lockedByID := make(map[uuid.UUID]LockedProduct, len(locked))
   159  	for _, l := range locked {
   160  		lockedByID[l.id] = l
   161  	}
   162  
   163  	seen := make(map[uuid.UUID]struct{}, len(inputs))
   164  	details := make([]PurchaseDetail, 0, len(inputs))
   165  	subtotalDollars := decimal.FromInt(0)
   166  	for _, in := range inputs {
   167  		l, err := resolveLocked(in, lockedByID, seen)
   168  		if err != nil {
   169  			return nil, decimal.FromInt(0), err
   170  		}
   171  		seen[in.ProductID] = struct{}{}
   172  
   173  		details = append(details, PurchaseDetail{
   174  			id:        in.ID,
   175  			productID: in.ProductID,
   176  			quantity:  in.Quantity,
   177  			unitPrice: l.price,
   178  		})
   179  		subtotalDollars = subtotalDollars.Add(l.price.Decimal().Mul(decimal.FromInt(int64(in.Quantity))))
   180  	}
   181  
   182  	return details, subtotalDollars, nil
   183  }

   185  // resolveLocked は、明細 1 件の入力を検証し、対応するロック済み商品を返します。
   186  // seen は既出の商品 ID の集合で、呼び出し元が返却後に in.ProductID を追加します。
   187  func resolveLocked(
   188  	in DetailInput,
   189  	lockedByID map[uuid.UUID]LockedProduct,
   190  	seen map[uuid.UUID]struct{},
   191  ) (LockedProduct, error) {
   192  	if in.ID.IsNil() {
   193  		return LockedProduct{}, xerrors.Wrap(ErrInvalidID, "detail id is required")
   194  	}
   195  	if in.Quantity < minQuantity {
   196  		return LockedProduct{}, xerrors.Wrap(
   197  			ErrInvalidQuantity, fmt.Sprintf("quantity must be %d or greater, got %d", minQuantity, in.Quantity),
   198  		)
   199  	}
   200  	if _, dup := seen[in.ProductID]; dup {
   201  		return LockedProduct{}, xerrors.Wrap(
   202  			ErrDuplicateProductID, fmt.Sprintf("product %s appears more than once", in.ProductID),
   203  		)
   204  	}
   205  
   206  	l, ok := lockedByID[in.ProductID]
   207  	if !ok {
   208  		return LockedProduct{}, xerrors.Wrap(ErrProductNotFound, fmt.Sprintf("product %s is not available", in.ProductID))
   209  	}
   210  	if in.Quantity > l.quantity {
   211  		return LockedProduct{}, xerrors.Wrap(
   212  			ErrInsufficientStock,
   213  			fmt.Sprintf("product %s: requested %d, in stock %d", in.ProductID, in.Quantity, l.quantity),
   214  		)
   215  	}
   216  
   217  	return l, nil
   218  }

   220  // New は、購入明細の入力とロック済み在庫から購入集約を生成します。
   221  // 明細が空、同一 productID が重複、数量が最小値未満、明細に対応するロック済み商品が無い場合はそれぞれ
   222  // 検証エラー（422）を返し、要求数量がロック済み在庫を超える場合は ErrInsufficientStock（409）を返します。
   223  // 単価は対応するロック済み商品の価格スナップショットとし、金額（小計・税・送料・合計）を整数で計算します
   224  // （税・送料の丸めは切り捨てで本メソッド内 1 箇所に集約）。ステータスは「未処理」で生成します。
   225  func New(
   226  	id uuid.UUID,
   227  	code string,
   228  	userID uuid.UUID,
   229  	inputs []DetailInput,
   230  	locked []LockedProduct,
   231  ) (*Purchase, error) {
   232  	if id.IsNil() {
   233  		return nil, xerrors.Wrap(ErrInvalidID, "id is required")
   234  	}
   235  	if code == "" {
   236  		return nil, xerrors.Wrap(ErrInvalidCode, "code is required")
   237  	}
   238  	if userID.IsNil() {
   239  		return nil, xerrors.Wrap(ErrInvalidUserID, "userID is required")
   240  	}
   241  	if len(inputs) == 0 {
   242  		return nil, ErrEmptyDetails
   243  	}
   244  
   245  	details, subtotalDollars, err := buildDetails(inputs, locked)
   246  	if err != nil {
   247  		return nil, err
   248  	}
   249  
   250  	// 価格スケール→決済スケールの丸め（切り捨て）。以降は整数セントで計算する（ADR-0038 (two-scale-quantity-model)）。
   251  	subtotalCents, err := subtotalDollars.Truncate(minorUnitDigits).ToScaledInt64(minorUnitDigits)
   252  	if err != nil {
   253  		return nil, xerrors.Join(ErrInvalidAmount, xerrors.Wrap(err, "subtotal exceeds the settlement range"))
   254  	}
   255  	if subtotalCents > maxSubtotalCents {
   256  		return nil, xerrors.Wrap(
   257  			ErrInvalidAmount,
   258  			fmt.Sprintf("subtotal %d exceeds the maximum that can carry tax and shipping (%d)", subtotalCents, maxSubtotalCents),
   259  		)
   260  	}
   261  	subtotal := int(subtotalCents)
   262  	tax, shipping, total := settle(settlementBasis{Subtotal: subtotal, Discount: 0})
   263  
   264  	return &Purchase{
   265  		id:             id,
   266  		code:           code,
   267  		userID:         userID,
   268  		status:         StatusUnprocessed,
   269  		subtotalAmount: subtotal,
   270  		taxAmount:      tax,
   271  		shippingFee:    shipping,
   272  		totalAmount:    total,
   273  		details:        details,
   274  	}, nil
   275  }

   277  // settle は、小計と値引き額から税・送料・合計を決めます。課税の基礎は値引き後の額です
   278  // （docs/spec/domain/purchase.md の Cross-field Invariants）。生成時と値引きの適用時が同じ関数を
   279  // 通るので、両者で式がずれません。
   280  func settle(basis settlementBasis) (int, int, int) {
   281  	taxable := basis.Subtotal - basis.Discount
   282  	tax := taxable * taxRatePercent / percentDivisor
   283  	shipping := shippingFeeCents
   284  	total := taxable + tax + shipping
   285  
   286  	return tax, shipping, total
   287  }

   289  // Reconstruct は、永続化済みの購入を再構築します（Repository の読み出し・書き込み後の再検証で使用）。
   290  // attrs.StatusCode は購入ステータスマスタで解決した現在状態、PaidAt / CanceledAt / ShippedAt / DeliveredAt
   291  // は各イベントの発生日時（未発生は nil）です。ID / Code / UserID / StatusID が nil、StatusCode が不正、
   292  // 金額が負、明細が空の場合は検証エラーを返します。StatusCode と timestamps の組み合わせが状態遷移では
   293  // 到達し得ないもの（発送後のキャンセル、発送を伴わない配達 等）の場合も ErrInvalidStatusID を返します。
   294  // 明細は New と同じ不変条件（ID / 商品 ID が設定済み、数量が最小値以上、同一商品 ID が重複しない）を課します。
   295  func Reconstruct(id uuid.UUID, attrs Attributes) (*Purchase, error) {
   296  	if id.IsNil() {
   297  		return nil, xerrors.Wrap(ErrInvalidID, "id is required")
   298  	}
   299  	if attrs.Code == "" {
   300  		return nil, xerrors.Wrap(ErrInvalidCode, "code is required")
   301  	}
   302  	if attrs.UserID.IsNil() {
   303  		return nil, xerrors.Wrap(ErrInvalidUserID, "userID is required")
   304  	}
   305  	if attrs.StatusID.IsNil() {
   306  		return nil, xerrors.Wrap(ErrInvalidStatusID, "statusID is required")
   307  	}
   308  	status, err := NewStatus(attrs.StatusCode)
   309  	if err != nil {
   310  		return nil, err
   311  	}
   312  	if err := validateStatusTimestamps(
   313  		status, attrs.PaidAt, attrs.CanceledAt, attrs.ShippedAt, attrs.DeliveredAt,
   314  	); err != nil {
   315  		return nil, err
   316  	}
   317  	if attrs.SubtotalAmount < 0 || attrs.TaxAmount < 0 || attrs.ShippingFee < 0 || attrs.TotalAmount < 0 {
   318  		return nil, xerrors.Wrap(ErrInvalidAmount, "amounts must not be negative")
   319  	}
   320  	if err := validateDiscount(attrs.CouponID, settlementBasis{Subtotal: attrs.SubtotalAmount, Discount: attrs.DiscountAmount}); err != nil {
   321  		return nil, err
   322  	}
   323  	if len(attrs.Details) == 0 {
   324  		return nil, ErrEmptyDetails
   325  	}
   326  
   327  	copied := make([]PurchaseDetail, len(attrs.Details))
   328  	copy(copied, attrs.Details)
   329  	if err := validateDetails(copied); err != nil {
   330  		return nil, err
   331  	}
   332  
   333  	return &Purchase{
   334  		id:             id,
   335  		code:           attrs.Code,
   336  		userID:         attrs.UserID,
   337  		statusID:       attrs.StatusID,
   338  		status:         status,
   339  		subtotalAmount: attrs.SubtotalAmount,
   340  		couponID:       ptr.Copy(attrs.CouponID),
   341  		discountAmount: attrs.DiscountAmount,
   342  		taxAmount:      attrs.TaxAmount,
   343  		shippingFee:    attrs.ShippingFee,
   344  		totalAmount:    attrs.TotalAmount,
   345  		details:        copied,
   346  		orderedAt:      attrs.OrderedAt,
   347  		paidAt:         ptr.Copy(attrs.PaidAt),
   348  		canceledAt:     ptr.Copy(attrs.CanceledAt),
   349  		shippedAt:      ptr.Copy(attrs.ShippedAt),
   350  		deliveredAt:    ptr.Copy(attrs.DeliveredAt),
   351  	}, nil
   352  }

   354  // validateDiscount は、クーポンの適用と値引き額の対応を検証します。
   355  //
   356  // クーポンを適用したことと値引きが立っていることは常に一致します
   357  // （docs/spec/domain/purchase.md の Cross-field Invariants）。値引きは小計を超えません。
   358  func validateDiscount(couponID *uuid.UUID, basis settlementBasis) error {
   359  	if basis.Discount < 0 {
   360  		return xerrors.Wrap(ErrInvalidAmount, "discountAmount must not be negative")
   361  	}
   362  	if (couponID != nil) != (basis.Discount > 0) {
   363  		return apperror.WithDetails(
   364  			xerrors.Wrap(ErrZeroDiscount, "couponID and a positive discountAmount must be set together"),
   365  			FieldCouponID,
   366  		)
   367  	}
   368  	if basis.Discount > basis.Subtotal {
   369  		return xerrors.Wrap(ErrInvalidAmount, "discountAmount must not exceed the subtotal")
   370  	}
   371  
   372  	return nil
   373  }

   375  // ApplyCoupon は、クーポンの値引きを購入へ適用し、税と合計を計算し直します。
   376  //
   377  // 値引きが 0 以下の場合は ErrZeroDiscount（details に couponID）を返します。クーポンを消費して
   378  // 何も引かれない確定を作らないためで、対象の明細が 1 件も無い場合と、定率の値引きが最小単位に
   379  // 満たない場合がこれに当たります。
   380  // 既に適用済みの購入への再適用は ErrCouponAlreadyApplied を返します。1 回の購入に適用できる
   381  // クーポンは高々 1 枚です。
   382  func (p *Purchase) ApplyCoupon(couponID uuid.UUID, discountAmount int) error {
   383  	if couponID.IsNil() {
   384  		return xerrors.Wrap(ErrInvalidCouponID, "couponID is required")
   385  	}
   386  	if p.couponID != nil {
   387  		return ErrCouponAlreadyApplied
   388  	}
   389  	if discountAmount <= 0 {
   390  		return apperror.WithDetails(
   391  			xerrors.Wrap(ErrZeroDiscount, "the coupon does not reduce this purchase"),
   392  			FieldCouponID,
   393  		)
   394  	}
   395  	if err := validateDiscount(&couponID, settlementBasis{Subtotal: p.subtotalAmount, Discount: discountAmount}); err != nil {
   396  		return err
   397  	}
   398  
   399  	p.couponID = &couponID
   400  	p.discountAmount = discountAmount
   401  	p.taxAmount, p.shippingFee, p.totalAmount = settle(settlementBasis{Subtotal: p.subtotalAmount, Discount: discountAmount})
   402  
   403  	return nil
   404  }

   412  // validateStatusTimestamps は、statusCode と各イベント日時の組み合わせの到達可能性を検証します
   413  // （個々の理由は各分岐のコメント、契約は Reconstruct を参照）。
   414  func validateStatusTimestamps(status Status, paidAt, canceledAt, shippedAt, deliveredAt *time.Time) error {
   415  	// キャンセルと canceledAt は同時セット（双条件。docs/spec/domain/purchase.md Cancel invariants）。
   416  	if (status == StatusCanceled) != (canceledAt != nil) {
   417  		return xerrors.Wrap(ErrInvalidStatusID, "canceled status and canceledAt must be consistent")
   418  	}
   419  	// 支払い済みは paidAt 必須（一方向。docs/spec/domain/purchase.md Pay invariants）。
   420  	if status == StatusPaid && paidAt == nil {
   421  		return xerrors.Wrap(ErrInvalidStatusID, "paid status requires paidAt")
   422  	}
   423  	// 発送済みは shippedAt 必須（一方向。docs/spec/domain/purchase.md Ship invariants）。
   424  	if status == StatusShipped && shippedAt == nil {
   425  		return xerrors.Wrap(ErrInvalidStatusID, "shipped status requires shippedAt")
   426  	}
   427  	// キャンセルは発送前にしか行えない（Cancel が発送済み・配達済みを拒否する）ため、キャンセル status と
   428  	// 発送の記録は同居しない。支払い後のキャンセルは正常なので paidAt は対象外。
   429  	if status == StatusCanceled && shippedAt != nil {
   430  		return xerrors.Wrap(ErrInvalidStatusID, "canceled status must not have shippedAt")
   431  	}
   432  	// 発送済みへは支払い済みからのみ遷移し、支払い済みは paidAt を必須とするため、paidAt を欠く発送済みは
   433  	// 到達不能な状態である。
   434  	if status == StatusShipped && paidAt == nil {
   435  		return xerrors.Wrap(ErrInvalidStatusID, "shipped status requires paidAt")
   436  	}
   437  	// 配達は発送済みからのみ到達するため、shippedAt を欠く deliveredAt は到達不能な状態である。
   438  	if deliveredAt != nil && shippedAt == nil {
   439  		return xerrors.Wrap(ErrInvalidStatusID, "deliveredAt requires shippedAt")
   440  	}
   441  	// 配達済みと deliveredAt は同時セット（双条件。docs/spec/domain/purchase.md Deliver invariants）。
   442  	if (status == StatusDelivered) != (deliveredAt != nil) {
   443  		return xerrors.Wrap(ErrInvalidStatusID, "delivered status and deliveredAt must be consistent")
   444  	}
   445  	return nil
   446  }

   534  // Cancel は、購入をキャンセル状態へ遷移させます。キャンセル可能状態（未処理 / 受付中 / 確認中 / 処理中 /
   535  // 支払い済み）からのみ遷移でき、statusCode をキャンセル（6）へ、canceledAt を now へ同時に更新します。
   536  // 既にキャンセル済みなら ErrAlreadyCanceled、完了・配達済み・発送済み（shippedAt）なら
   537  // ErrCancelNotAllowed をそれぞれ返します（いずれも 409）。
   538  // 遷移に成功したときだけキャンセルの事実（Event）を返します。
   539  func (p *Purchase) Cancel(now time.Time) (Event, error) {
   540  	if p.IsCanceled() {
   541  		return Event{}, ErrAlreadyCanceled
   542  	}
   543  	// 配達済みも deliveredAt と同値なので status で判定する。発送済みは status が配達済みへ進んでも
   544  	// 遷移不可であり続けるため、status ではなく残り続ける shippedAt で判定する。
   545  	if !p.status.CanTransitionTo(StatusCanceled) || p.shippedAt != nil {
   546  		return Event{}, ErrCancelNotAllowed
   547  	}
   548  	p.status = StatusCanceled
   549  	p.canceledAt = &now
   550  	return newEvent(EventCanceled, p.id, now), nil
   551  }

   553  // Pay は、購入を支払い済み状態へ遷移させます。未払い相当（未処理 / 受付中 / 確認中）からのみ遷移でき、
   554  // statusCode を支払い済み（7）へ、paidAt を now へ同時に更新します。支払い済み、または未払い相当でも
   555  // paidAt が記録済みなら ErrAlreadyPaid、処理中・キャンセル済み・完了・配達済み・発送済み（shippedAt）なら
   556  // ErrPayNotAllowed をそれぞれ返します（いずれも 409）。決済 SDK / PSP 連携は行わず、paidAt とステータスの
   557  // 記録のみを担う擬似決済です。遷移に成功したときだけ支払いの事実（Event）を返します。
   558  func (p *Purchase) Pay(now time.Time) (Event, error) {
   559  	if p.status == StatusPaid {
   560  		return Event{}, ErrAlreadyPaid
   561  	}
   562  	// status と shippedAt を併用する理由は Cancel を参照。
   563  	if !p.status.CanTransitionTo(StatusPaid) || p.shippedAt != nil {
   564  		return Event{}, ErrPayNotAllowed
   565  	}
   566  	// 記録済みの paidAt を上書きしないためのガード（二重支払い防止。docs/spec/domain/purchase.md Pay invariants）。
   567  	if p.paidAt != nil {
   568  		return Event{}, ErrAlreadyPaid
   569  	}
   570  	p.status = StatusPaid
   571  	p.paidAt = &now
   572  	return newEvent(EventPaid, p.id, now), nil
   573  }
