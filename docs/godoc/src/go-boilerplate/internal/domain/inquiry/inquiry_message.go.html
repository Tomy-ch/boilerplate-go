    45  func newMessage(id uuid.UUID, attrs MessageAttributes) (*Message, error) {
    46  	if id.IsNil() {
    47  		return nil, xerrors.Wrap(ErrInvalidMessageID, "id is required")
    48  	}
    49  	if err := validateAuthor(attrs.Author.kind, attrs.Author.subjectID); err != nil {
    50  		return nil, err
    51  	}
    52  	if err := validateBody(attrs.Body); err != nil {
    53  		return nil, err
    54  	}
    55  	if attrs.Sequence < 1 {
    56  		return nil, xerrors.Wrap(
    57  			ErrInvalidSequence, fmt.Sprintf("sequence must be 1 or greater, got %d", attrs.Sequence),
    58  		)
    59  	}
    60  
    61  	return &Message{
    62  		id:        id,
    63  		author:    attrs.Author,
    64  		body:      attrs.Body,
    65  		sequence:  attrs.Sequence,
    66  		createdAt: attrs.CreatedAt,
