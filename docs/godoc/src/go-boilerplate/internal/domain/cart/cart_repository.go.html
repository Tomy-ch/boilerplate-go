    36  	// （不存在の検証は呼び出し側の責務です）。
    37  	LockByIDs(ctx context.Context, ids []uuid.UUID) (Carts, error)
    38  
    39  	// Create は、カートを明細込みで新規登録します。
    40  	// 所有者またはセッショントークンが既に使われている場合は Conflict を返します。
    41  	Create(ctx context.Context, c *Cart) error
    42

    45  	// 並行して作成が競合した場合も、勝ったほうのカートが返ります。
    46  	CreateOwnerIfAbsent(ctx context.Context, c *Cart) (*Cart, error)

    51  	//
    52  	// 明細は商品ごとに置き換えられ、最初に入った時刻は保持されます。集合から消えた明細は取り除かれます。
    53  	Update(ctx context.Context, c *Cart) error
    54
