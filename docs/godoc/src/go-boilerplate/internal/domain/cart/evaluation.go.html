    89  // Evaluate は、明細を商品の観測値と突き合わせて再評価結果を返します。
    90  // snapshot が nil の場合は商品を引けなかったことを表し、IssueNotFound だけが立ちます。
    91  func (i CartItem) Evaluate(snapshot *ProductSnapshot) Evaluation {
    92  	if snapshot == nil {
    93  		return Evaluation{issues: []Issue{IssueNotFound}}
    94  	}
    95  
    96  	issues := make([]Issue, 0, maxIssuesPerItem)
    97  	switch {
    98  	case snapshot.discontinued:
    99  		issues = append(issues, IssueDiscontinued)
   100  	case !snapshot.published:
   101  		issues = append(issues, IssueUnpublished)
   102  	}
   103  
   104  	var availableQuantity *int
   105  	switch stock := snapshot.quantity; {
   106  	case stock <= 0:
   107  		issues = append(issues, IssueOutOfStock)
   108  	case stock < i.quantity:
   109  		issues = append(issues, IssueInsufficientStock)
   110  		availableQuantity = &stock
   111  	}
   112  
   113  	if i.lastSeenPrice != nil {
   114  		switch snapshot.price.Decimal().Cmp(i.lastSeenPrice.Decimal()) {
   115  		case 1:
   116  			issues = append(issues, IssuePriceIncreased)
   117  		case -1:
   118  			issues = append(issues, IssuePriceDecreased)
   119  		}
   120  	}
   121  
   122  	return Evaluation{issues: issues, availableQuantity: availableQuantity}
   123  }
