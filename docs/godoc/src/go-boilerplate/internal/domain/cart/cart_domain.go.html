    92  // Reconstruct は、永続化済みのカートを再構築します。
    93  // NewForGuest / NewForOwner と同じ不変条件を課します。保存済みデータのための緩和経路はありません。
    94  // 加えて、監査時刻が設定されている場合は expiresAt が createdAt より後であることを要求します。
    95  func Reconstruct(id uuid.UUID, attrs Attributes) (*Cart, error) {
    96  	c, err := newCart(id, attrs)
    97  	if err != nil {
    98  		return nil, err
    99  	}
   100  	if !attrs.CreatedAt.IsZero() && !attrs.ExpiresAt.After(attrs.CreatedAt) {
   101  		return nil, xerrors.Wrap(ErrInvalidExpiresAt, "expiresAt must be after createdAt")
   102  	}
   103  	return c, nil
   104  }

   106  // newCart は、3 つの入口が共有する検証ゲートです。
   107  // 入口ごとに検証が乖離しないよう、不変条件の判定はここ 1 箇所に閉じます。
   108  func newCart(id uuid.UUID, attrs Attributes) (*Cart, error) {
   109  	if id.IsNil() {
   110  		return nil, xerrors.Wrap(ErrInvalidID, "id is required")
   111  	}
   112  	if err := validateOwnership(attrs.OwnerID, attrs.SessionToken); err != nil {
   113  		return nil, err
   114  	}
   115  	if attrs.ExpiresAt.IsZero() {
   116  		return nil, xerrors.Wrap(ErrInvalidExpiresAt, "expiresAt is required")
   117  	}
   118  
   119  	copied := make([]CartItem, len(attrs.Items))
   120  	copy(copied, attrs.Items)
   121  	if err := validateItems(copied); err != nil {
   122  		return nil, err
   123  	}
   124  
   125  	return &Cart{
   126  		id:           id,
   127  		ownerID:      ptr.Copy(attrs.OwnerID),
   128  		sessionToken: ptr.Copy(attrs.SessionToken),
   129  		items:        copied,
   130  		expiresAt:    attrs.ExpiresAt,
   131  		createdAt:    attrs.CreatedAt,
   132  		updatedAt:    attrs.UpdatedAt,
   133  	}, nil

   136  // validateOwnership は、所有者とセッショントークンがちょうど一方だけ設定されていることを検証します。
   137  // どちらの側も構築を許しません（排他の理由: docs/spec/domain/cart.md の Cross-field Invariants）。
   138  func validateOwnership(ownerID *uuid.UUID, token *SessionToken) error {
   139  	if (ownerID == nil) == (token == nil) {
   140  		return ErrInvalidOwner
   141  	}
   142  	if ownerID != nil && ownerID.IsNil() {
   143  		return xerrors.Wrap(ErrInvalidUserID, "ownerID is required")
   144  	}
   145  	if token != nil && token.IsZero() {
   146  		return xerrors.Wrap(ErrInvalidSessionToken, "sessionToken is required")
   147  	}
   148  	return nil
   149  }

   151  // validateItems は、明細の集合が満たすべき不変条件を検証します。
   152  // 明細 ID / 商品 ID が未設定、数量が範囲外、同一商品 ID の重複、明細数の上限超過をそれぞれ検出します。
   153  func validateItems(items []CartItem) error {
   154  	if len(items) > maxItems {
   155  		return xerrors.Wrap(ErrTooManyItems, fmt.Sprintf("items must be %d or fewer, got %d", maxItems, len(items)))
   156  	}
   157  	seen := make(map[uuid.UUID]struct{}, len(items))
   158  	for _, i := range items {
   159  		if i.id.IsNil() {
   160  			return xerrors.Wrap(ErrInvalidID, "item id is required")
   161  		}
   162  		if i.productID.IsNil() {
   163  			return xerrors.Wrap(ErrInvalidProductID, "item product id is required")
   164  		}
   165  		if err := validateQuantity(i.quantity); err != nil {
   166  			return err
   167  		}
   168  		if _, dup := seen[i.productID]; dup {
   169  			return xerrors.Wrap(ErrDuplicateProductID, fmt.Sprintf("product %s appears more than once", i.productID))
   170  		}
   171  		seen[i.productID] = struct{}{}
   172  	}
   173  	return nil
   174  }

   208  // SetItem は、指定商品の数量を設定します。
   209  // 同一商品の明細が既にあれば数量を置換し（addedAt は保持）、無ければ ItemID で追加します。
   210  // 数量が範囲外なら ErrInvalidQuantity を、追加により明細数が上限を超えるなら ErrTooManyItems を返します。
   211  // 数量 0 は削除ではなくエラーです。削除は RemoveItem が担います。
   212  func (c *Cart) SetItem(attrs SetItemAttributes, now time.Time) error {
   213  	if attrs.ProductID.IsNil() {
   214  		return xerrors.Wrap(ErrInvalidProductID, "productID is required")
   215  	}
   216  	if err := validateQuantity(attrs.Quantity); err != nil {
   217  		return err
   218  	}
   219  
   220  	if idx := c.indexOf(attrs.ProductID); idx >= 0 {
   221  		c.items[idx].quantity = attrs.Quantity
   222  		return nil
   223  	}
   224  
   225  	if attrs.ItemID.IsNil() {
   226  		return xerrors.Wrap(ErrInvalidID, "itemID is required to add a new item")
   227  	}
   228  	if len(c.items)+1 > maxItems {
   229  		return xerrors.Wrap(ErrTooManyItems, fmt.Sprintf("items must be %d or fewer", maxItems))
   230  	}
   231  
   232  	c.items = append(c.items, CartItem{
   233  		id: attrs.ItemID, productID: attrs.ProductID, quantity: attrs.Quantity, addedAt: now,
   234  	})
   235  	return nil
   236  }

   253  // Merge は、別のカートの明細を自身へ取り込みます。
   254  //
   255  // 同一商品は数量を合算し、上限を超える場合は上限へクランプします。自身に無い商品は追加し、明細数の
   256  // 上限を超える分は追加が新しいものから切り捨てます（先に入っていたものを優先して残す）。
   257  // 失われた分は戻り値で報告されます。
   258  //
   259  // error を返しません（理由: docs/spec/domain/cart.md の Behavior Methods の Merge）。
   260  // 取り込み元は変更しません。
   261  func (c *Cart) Merge(other *Cart, now time.Time) MergeResult {
   262  	if other == nil {
   263  		return MergeResult{}
   264  	}
   265  
   266  	result := MergeResult{}
   267  	incoming := make([]CartItem, 0, len(other.items))
   268  	for _, item := range other.items {
   269  		if idx := c.indexOf(item.productID); idx >= 0 {
   270  			sum := c.items[idx].quantity + item.quantity
   271  			if sum > maxQuantityPerItem {
   272  				sum = maxQuantityPerItem
   273  				result.clamped = append(result.clamped, item.productID)
   274  			}
   275  			c.items[idx].quantity = sum
   276  			continue
   277  		}
   278  		incoming = append(incoming, item)
   279  	}
   280  
   281  	// 切り捨ての順序は渡された順ではなく addedAt の値で決める。呼び出し側の並べ方に結果が左右されると、
   282  	// 同じ 2 つのカートをマージしても取り込まれる明細が変わってしまう。
   283  	slices.SortFunc(incoming, compareByAddedAt)
   284  
   285  	for _, item := range incoming {
   286  		if len(c.items) >= maxItems {
   287  			result.dropped = append(result.dropped, item.productID)
   288  			continue
   289  		}
   290  		merged := item
   291  		merged.addedAt = now
   292  		c.items = append(c.items, merged)
   293  	}
   294  
   295  	return result
   296  }

   344  // compareByAddedAt は、明細を追加時刻の昇順へ並べるための比較関数です。
   345  // 同時刻は商品 ID のバイト列で決着させ、入力の並び順が結果に漏れないようにします。
   346  func compareByAddedAt(a, b CartItem) int {
