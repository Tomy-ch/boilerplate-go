    65  // （理由は docs/spec/domain/coupon.md の Cross-field Invariants を参照）。
    66  func validateValidity(issuedAt, expiresAt time.Time) error {
    67  	if issuedAt.IsZero() {
    68  		return xerrors.Wrap(ErrInvalidIssuedAt, "issuedAt is required")
    69  	}
    70  	if expiresAt.IsZero() {
    71  		return xerrors.Wrap(ErrInvalidExpiresAt, "expiresAt is required")
    72  	}
    73  	if !expiresAt.After(issuedAt) {

   139  // DiscountFor は、渡された明細のうち適用範囲に入るものを対象として、差し引く額を決済スケールの
   140  // 整数（USD セント）で返します。対象が 1 件も無い場合と、差し引く額が最小単位に満たない場合は 0 です。
   141  //
   142  // 丸めはこのメソッドでのみ行います（[Discount.Apply] は丸めません。ADR-0038 (two-scale-quantity-model)）。
   143  // 根拠は docs/spec/domain/coupon.md の Behavior Methods > DiscountFor を参照してください。
   144  func (c *Coupon) DiscountFor(lines []Line) (int, error) {
   145  	eligible := decimal.FromInt(0)
   146  	for _, line := range lines {
   147  		if c.scope.Covers(line) {
   148  			eligible = eligible.Add(line.Subtotal())
   149  		}
   150  	}
   151  
   152  	cents, err := c.discount.Apply(eligible).Truncate(minorUnitDigits).ToScaledInt64(minorUnitDigits)
   153  	if err != nil {
   154  		return 0, xerrors.Wrap(ErrInvalidDiscountValue, "discount exceeds the settlement range")
   155  	}
   156  
   157  	return int(cents), nil
   158  }

   160  // Redeem は、クーポンを使用済みにします。使用済みから未使用へ戻すのは [Coupon.Restore] だけです。
   161  //
   162  // 既に使用済みなら ErrAlreadyUsed、渡された時点で失効しているなら ErrExpired を返し、状態を変えません。
   163  func (c *Coupon) Redeem(now time.Time) error {
   164  	if c.IsUsed() {
   165  		return ErrAlreadyUsed
   166  	}
   167  	if c.IsExpired(now) {
   168  		return ErrExpired
   169  	}
   170

   172  
   173  	return nil
   174  }

   176  // Restore は、使用済みのクーポンを未使用へ戻し、戻したかどうかを返します。
   177  //
   178  // 渡された時点で失効している場合は状態を変えず false を返します（理由は

   180  // 未使用のクーポンには ErrNotUsed を返します。
   181  func (c *Coupon) Restore(now time.Time) (bool, error) {
   182  	if !c.IsUsed() {
   183  		return false, ErrNotUsed
   184  	}
   185  	if c.IsExpired(now) {
   186  		return false, nil
   187  	}
   188  
   189  	c.usedAt = nil
   190  
   191  	return true, nil
   192  }
