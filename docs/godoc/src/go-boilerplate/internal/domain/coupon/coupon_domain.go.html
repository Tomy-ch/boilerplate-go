    65  func newCoupon(id uuid.UUID, attrs Attributes, usedAt *time.Time) (*Coupon, error) {
    66  	if id.IsNil() {

    71  	}
    72  	if attrs.Discount.IsZero() {
    73  		return nil, ErrInvalidDiscount
    74  	}
    75  	if attrs.Scope.IsZero() {
    76  		return nil, ErrInvalidScope
    77  	}
    78  	if attrs.IssuedAt.IsZero() {
    79  		return nil, xerrors.Wrap(ErrInvalidIssuedAt, "issuedAt is required")
    80  	}
    81  	if attrs.ExpiresAt.IsZero() {
    82  		return nil, xerrors.Wrap(ErrInvalidExpiresAt, "expiresAt is required")
    83  	}
    84  	if !attrs.ExpiresAt.After(attrs.IssuedAt) {
    85  		return nil, xerrors.Wrap(ErrInvalidExpiresAt, "expiresAt must be after issuedAt")
    86  	}

    88  	used := ptr.Copy(usedAt)

    92  		userID:    attrs.UserID,
    93  		discount:  attrs.Discount,
    94  		scope:     attrs.Scope,

    97  		issuedAt:  attrs.IssuedAt,

   129  // DiscountFor は、渡された明細のうち適用範囲に入るものを対象として、差し引く額を決済スケールの
   130  // 整数（USD セント）で返します。対象が 1 件も無い場合と、差し引く額が最小単位に満たない場合は 0 です。
   131  //
   132  // 丸めはこのメソッドでのみ行います（[Discount.Apply] は丸めません。ADR-0038 (two-scale-quantity-model)）。
   133  // 根拠は docs/spec/domain/coupon.md の Behavior Methods > DiscountFor を参照してください。
   134  func (c *Coupon) DiscountFor(lines []Line) (int, error) {
   135  	eligible := decimal.FromInt(0)
   136  	for _, line := range lines {
   137  		if c.scope.Covers(line) {
   138  			eligible = eligible.Add(line.Subtotal())
   139  		}
   140  	}
   141  
   142  	cents, err := c.discount.Apply(eligible).Truncate(minorUnitDigits).ToScaledInt64(minorUnitDigits)
   143  	if err != nil {
   144  		return 0, xerrors.Wrap(ErrInvalidDiscountValue, "discount exceeds the settlement range")
   145  	}
   146  
   147  	return int(cents), nil
   148  }

   150  // Redeem は、クーポンを使用済みにします。使用済みへの遷移は一度きりで、取り消せません。
   151  //
   152  // 既に使用済みなら ErrAlreadyUsed、渡された時点で失効しているなら ErrExpired を返し、状態を変えません。
   153  // 日時は引数で受け取ります（internal/domain/README.md の Handling time and ID を参照）。
   154  func (c *Coupon) Redeem(now time.Time) error {
   155  	if c.IsUsed() {
   156  		return ErrAlreadyUsed
   157  	}
   158  	if c.IsExpired(now) {
   159  		return ErrExpired
   160  	}
   161  
   162  	c.usedAt = &now
   163  
   164  	return nil
   165  }

   167  // IsExpired は、渡された時点でクーポンが失効しているかどうかを返します。有効期限ちょうども
