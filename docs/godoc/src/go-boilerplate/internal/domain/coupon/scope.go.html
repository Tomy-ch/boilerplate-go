   102  // NewScope は、解決済みの種別と対象 ID から適用範囲を生成します。種別ごとの生成関数へ振り分け、
   103  // 対象 ID の要否だけをここで課します（全体は対象を持ってはなりません）。
   104  // 永続化された行からの復元も、要求が渡した名前からの構築も、種別を解決したあとはこの入口を通ります。
   105  func NewScope(kind ScopeKind, targetID *uuid.UUID) (Scope, error) {
   106  	switch kind {
   107  	case ScopeKindAll:
   108  		if targetID != nil {
   109  			return Scope{}, xerrors.Wrap(ErrInvalidScopeTarget, "an all scope must not have a target")
   110  		}
   111  		return NewAllScope(), nil
   112  	case ScopeKindCategory:
   113  		if targetID == nil {
   114  			return Scope{}, xerrors.Wrap(ErrInvalidScopeTarget, "categoryID is required for a category scope")
   115  		}
   116  		return NewCategoryScope(*targetID)
   117  	case ScopeKindProduct:
   118  		if targetID == nil {
   119  			return Scope{}, xerrors.Wrap(ErrInvalidScopeTarget, "productID is required for a product scope")
   120  		}
   121  		return NewProductScope(*targetID)
   122  	default:
   123  		return Scope{}, xerrors.Wrap(ErrInvalidScopeKind, "scope kind is required")
   124  	}
   125  }

   136  // Covers は、明細がこの適用範囲に入るかどうかを返します。
   137  //
   138  // 「どの明細が対象か」の定義はこのメソッドが持ちます。全体はすべての明細を対象とし、
   139  // カテゴリ限定は明細の商品カテゴリが、商品限定は明細の商品が、それぞれ対象と一致するときだけ対象です。
   140  // 未設定の適用範囲はどの明細も対象にしません。
   141  func (s Scope) Covers(line Line) bool {
   142  	switch s.kind {
   143  	case ScopeKindAll:
   144  		return true
   145  	case ScopeKindCategory:
   146  		return s.targetID != nil && *s.targetID == line.CategoryID()
   147  	case ScopeKindProduct:
   148  		return s.targetID != nil && *s.targetID == line.ProductID()
   149  	default:
   150  		return false
   151  	}
   152  }
