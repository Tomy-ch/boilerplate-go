   123  // Apply は、対象額に対して差し引く額を価格スケールの十進量で返します。
   124  //
   125  // 「いくら引くか」の定義はこのメソッドが持ちます。定額は対象額を上限に切り詰め、定率は対象額に
   126  // 率を掛けます。どちらも対象額を超えないため、請求額が負になることはありません。
   127  // 対象額が 0 以下なら差し引く額も 0 です。
   128  //
   129  // 丸めません。決済スケールへの丸めは [Coupon.DiscountFor] が 1 箇所で行います
   130  // （ADR-0038 (two-scale-quantity-model)）。
   131  func (d Discount) Apply(eligible decimal.Decimal) decimal.Decimal {
   132  	if eligible.Sign() <= 0 {
   133  		return decimal.FromInt(0)
   134  	}
   135  
   136  	switch d.kind {
   137  	case DiscountKindFlat:
   138  		if d.value.Cmp(eligible) > 0 {
   139  			return eligible
   140  		}
   141  		return d.value
   142  	case DiscountKindRate:
   143  		return eligible.Mul(d.value)
   144  	default:
   145  		return decimal.FromInt(0)
   146  	}
   147  }
