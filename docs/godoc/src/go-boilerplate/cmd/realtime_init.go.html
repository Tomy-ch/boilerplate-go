    38  // runRealtimeInit は、設定を読み込み、DynamoDB クライアントを組み立てて初期化を実行します。
    39  // CLI は operator が設定した endpoint にだけ繋ぐため、SSRF ガード付きの HTTP クライアントは使いません。
    40  func runRealtimeInit(ctx context.Context) error {
    41  	cfg, err := newCLIConfig("")
    42  	if err != nil {
    43  		return err
    44  	}
    45  
    46  	rtCfg := config.NewRealtimeConfig(cfg)
    47  	client, err := dynamodbclient.New(ctx, dynamodbclient.Config{
    48  		Endpoint:        config.NewEndpointConfig(cfg).Realtime(),
    49  		Region:          rtCfg.Region(),
    50  		AccessKeyID:     rtCfg.AccessKeyID(),
    51  		SecretAccessKey: rtCfg.SecretAccessKey(),
    52  	})
    53  	if err != nil {
    54  		return xerrors.Wrap(err, "failed to build dynamodb client")
    55  	}
    56  
    57  	// table の定義（キー / index / TTL）は各 adapter package が持つ。
    58  	specs := map[string]dynamodbclient.TableSpec{
    59  		rtCfg.EventLogTable():      eventlogdynamo.TableSpec(rtCfg.EventLogTable()),
    60  		rtCfg.StreamTicketTable():  streamticketdynamo.TableSpec(rtCfg.StreamTicketTable()),
    61  		rtCfg.InstanceLeaseTable(): instanceleasedynamo.TableSpec(rtCfg.InstanceLeaseTable()),
    62  	}
    63  	ensure := func(ctx context.Context, table string) error {
    64  		spec, ok := specs[table]
    65  		if !ok {
    66  			return xerrors.Wrap(errUnknownRealtimeTable, table)
    67  		}
    68  
    69  		return dynamodbclient.EnsureTable(ctx, client, spec)
    70  	}
    71  
    72  	logger := logging.NewJSONLogger(logging.LevelInfo(), logging.LevelError(), nil)
    73  	if err := clirealtimeinit.Run(ctx, rtCfg, ensure, logger); err != nil {
    74  		return err
    75  	}
    76  
    77  	clients, err := realtimeinfra.NewClients(ctx, realtimeinfra.ClientConfig{
    78  		Endpoint:        config.NewEndpointConfig(cfg).RealtimePubSub(),
    79  		Region:          rtCfg.Region(),
    80  		AccessKeyID:     rtCfg.AccessKeyID(),
    81  		SecretAccessKey: rtCfg.SecretAccessKey(),
    82  	})
    83  	if err != nil {
    84  		return xerrors.Wrap(err, "failed to build sns client")
    85  	}
    86  
    87  	ensureTopic := func(ctx context.Context, name string) (string, error) {
    88  		return realtimeinfra.EnsureTopic(ctx, clients, name)
    89  	}
    90  
    91  	return clirealtimeinit.RunTopic(ctx, rtCfg.Topic(), ensureTopic, logger)
    92  }
