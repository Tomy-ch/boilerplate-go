   101  // newSlotPool は、実依存（ホスト実行の registry / pgx admin / docker compose）を配線して Pool を生成します。
   102  func newSlotPool() (*dbslot.Pool, error) {
   103  	root, err := os.Getwd()
   104  	if err != nil {
   105  		return nil, err
   106  	}
   107  
   108  	admin := dbslot.NewPgxAdmin(
   109  		envStr("GOBP_DB_POOL_PGHOST", "localhost"),
   110  		envInt("GOBP_DB_POOL_PGPORT", defaultPoolPGPort),
   111  		envStr("GOBP_DB_POOL_PGUSER", "postgres"),
   112  		envStr("GOBP_DB_POOL_PGPASSWORD", "postgres-password"),
   113  		envStr("GOBP_DB_POOL_PGMAINTDB", "postgres"),
   114  	)
   115  	cfg, err := slotConfig(root)
   116  	if err != nil {
   117  		return nil, err
   118  	}
   119  
   120  	return dbslot.NewPool(newSlotRegistry(root), admin, dbslot.ExecCompose{}, cfg, os.Stdout, os.Stderr), nil
   121  }

   136  func newSlotResolver(out io.Writer) (*dbslot.Resolver, error) {

   152  func slotConfig(root string) (dbslot.Config, error) {
   153  	cfg := dbslot.Config{
   154  		Root:          root,
   155  		SharedProject: envStr("GOBP_DB_SHARED_PROJECT", "gobp-shared"),
   156  		APIBasePort:   envInt("GOBP_API_POOL_BASE", defaultPoolAPIBasePort),
   157  		MockAuthBase:  envInt("GOBP_MOCK_AUTH_POOL_BASE", defaultPoolMockBasePort),
   158  		DlvBase:       envInt("GOBP_DLV_POOL_BASE", defaultPoolDlvBasePort),
   159  		PprofBase:     envInt("GOBP_PPROF_POOL_BASE", defaultPoolPprofBasePort),
   160  		APPEnv:        os.Getenv("APP_ENV"),
   161  	}
   162  
   163  	base, err := dbslot.LoadRealtimeBase()
   164  	if err != nil {
   165  		return dbslot.Config{}, err
   166  	}
   167  
   168  	cfg.Realtime = base
   169  
   170  	return cfg, nil
   171  }

   173  func poolDir() string {
   174  	if v := os.Getenv("GOBP_DB_POOL_DIR"); v != "" {
   175  		return v
   176  	}
   177  	cache := os.Getenv("XDG_CACHE_HOME")
   178  	if cache == "" {
   179  		home, _ := os.UserHomeDir()

   181  	}
   182  	return cache + "/gobp-db-pool"
