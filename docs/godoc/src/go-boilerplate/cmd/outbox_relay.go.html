    16  // newOutboxRelayCommand は、outbox relay コマンドを生成します（replay サブコマンドを含む）。
    17  func newOutboxRelayCommand() *cobra.Command {
    18  	var channel string
    19  
    20  	cmd := &cobra.Command{
    21  		Use:   "outbox-relay",
    22  		Short: "outbox relay を起動します。",
    23  		Long: "outbox-relay コマンドは、outbox テーブルを周期 poll して未 publish メッセージを送る relay を起動し、\n" +
    24  			"SIGTERM まで常駐します。--channel で担当する配送チャネルを 1 つ選び、そのチャネルの行だけを配送します。",
    25  		RunE: func(_ *cobra.Command, _ []string) error { return outboxRelayRun(channel) },
    26  	}
    27  	// 既定値なし: 誤ったチャネルでの起動は担当者不在のレーンを作り、滞留が lag の欠測になる
    28  	// （internal/cli/outbox/README.md の --channel を参照）。
    29  	cmd.Flags().StringVar(&channel, "channel", "", "担当する配送チャネル（http / realtime）")
    30  	_ = cmd.MarkFlagRequired("channel")
    31  	cmd.AddCommand(newOutboxReplayCommand())
    32  	return cmd
    33  }

    35  // outboxRelayRun は outboxcli.RunRelay への薄い委譲殻です。
    36  func outboxRelayRun(channel string) error {
    37  	ch, err := outboxbndry.ParseChannel(channel)
    38  	if err != nil {
    39  		return err
    40  	}
    41  
    42  	cfg, err := config.SetUpConfig()
    43  	if err != nil {
    44  		return err
    45  	}
    46  	appCfg := config.NewApplicationConfig(cfg)
    47  
    48  	ctx, stop := signal.NotifyContext(context.Background(), syscall.SIGINT, syscall.SIGTERM)
    49  	defer stop()
    50  
    51  	app := di.NewOutboxRelayApp(appCfg.ShutdownTimeout(), ch)
    52  	startApp, stopApp := di.NewApplicationServer(app)
    53  
    54  	return outboxcli.RunRelay(ctx, appCfg.ShutdownTimeout(), startApp, stopApp)
    55  }
