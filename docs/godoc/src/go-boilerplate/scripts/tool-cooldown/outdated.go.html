    52  // surveyOutdated は各ツールの上流を調べ、窓を満たす更新先を決めます。
    53  // client は上流への問い合わせ手段、now は齢の基準時刻で、差し替えられるよう引数で受けます。
    54  func surveyOutdated(ctx context.Context, client *http.Client, tools []tool, now time.Time) []candidate {
    55  	out := make([]candidate, len(tools))
    56  	sem := make(chan struct{}, fetchWorkers)
    57  	var wg sync.WaitGroup
    58  	for i, t := range tools {
    59  		wg.Add(1)
    60  		go func(i int, t tool) {
    61  			defer wg.Done()
    62  			sem <- struct{}{}
    63  			defer func() { <-sem }()
    64  			out[i] = evaluate(ctx, client, t, now)
    65  		}(i, t)
    66  	}
    67  	wg.Wait()
    68  
    69  	sort.Slice(out, func(i, j int) bool { return out[i].tool.id() < out[j].tool.id() })
    70  	return out
    71  }

    73  // evaluate は 1 ツールの上流を調べて candidate に畳みます。
    74  func evaluate(ctx context.Context, client *http.Client, t tool, now time.Time) candidate {
    75  	c := candidate{tool: t, window: windowFor(t.backend)}
    76  
    77  	releases, err := listReleases(ctx, client, t)
    78  	if err != nil {
    79  		c.err = err
    80  
    81  		return c
    82  	}
    83  	// 新しい版が先に来る順へ揃える。上流ごとに一覧の順序が違うため、ここで決め直さないと
    84  	// 「最新」も「窓を満たす最新」も上流の都合で変わる。
    85  	sort.Slice(releases, func(i, j int) bool { return lessVersion(releases[j].version, releases[i].version) })
    86  
    87  	if len(releases) > 0 {
    88  		c.latest = releases[0].version
    89  		c.latestAge = ageDays(releases[0].at, now)
    90  	}
    91  
    92  	major := majorOf(t.version)
    93  	for _, r := range releases {
    94  		if majorOf(r.version) != major {
    95  			continue
    96  		}
    97  		if ageDays(r.at, now) < c.window {
    98  			continue
    99  		}
   100  		c.eligible = r.version
   101  		c.eligibleAge = ageDays(r.at, now)
   102  
   103  		break
   104  	}
   105  
   106  	return c
   107  }

   156  func goModuleReleases(ctx context.Context, client *http.Client, pkg string) ([]release, error) {
   157  	versions, base, err := goModuleVersions(ctx, client, pkg)
   158  	if err != nil {

   160  	}
   161  	sort.Slice(versions, func(i, j int) bool { return lessVersion(versions[j], versions[i]) })
   162  	if len(versions) > maxGoProxyProbes {
   163  		versions = versions[:maxGoProxyProbes]

   168  		at, atErr := goModuleAt(ctx, client, base, v)

   170  			continue
   171  		}
   172  		out = append(out, release{version: v, at: at})
   173  	}

   178  // goModuleVersions は @v/list を引き、解決できたモジュールパスも返します。mise の go backend は
   179  // パッケージパスを受けるのに proxy が知るのはモジュールパスなので、末尾要素を落としながら遡ります。
   180  //
   181  // 遡るのは proxy がそのパスをモジュールとして知らなかったときだけです。応答が返った時点でそこが
   182  // モジュールで、安定版を 1 つも持たなくても答えは「無い」であって親のものではありません。
   183  // `golang.org/x/tools/cmd/godoc` は v0.1.0-deprecated しか持たない独立モジュールで、空を理由に
   184  // 遡ると親 `golang.org/x/tools` の版一覧を、この別物のツールの更新先として差し出すことになります。
   185  func goModuleVersions(ctx context.Context, client *http.Client, pkg string) ([]string, string, error) {
   186  	var lastErr error
   187  	for path := pkg; strings.Contains(path, "/"); path = path[:strings.LastIndex(path, "/")] {
   188  		body, err := getText(ctx, client, goProxyBase+escapeModulePath(path)+"/@v/list")
   189  		if err != nil {
   190  			lastErr = err
   191  
   192  			continue
   193  		}
   194  		var versions []string
   195  		for line := range strings.FieldsSeq(body) {
   196  			v := strings.TrimPrefix(line, "v")
   197  			if stableVersion(v) {
   198  				versions = append(versions, v)
   199  			}
   200  		}
   201  
   202  		return versions, path, nil
   203  	}
   204  	if lastErr == nil {
   205  		lastErr = xerrors.Wrap(errNotFound, pkg)
   206  	}
   207  
   208  	return nil, "", lastErr
   209  }

   254  // stableVersion は数値だけの区切りからなる版かどうかを返します。プレリリース識別子やビルド
   255  // メタデータを持つ版を弾くのが目的で、ここを通さないと `1.2.0-rc1` が最新として選ばれます。
   256  func stableVersion(v string) bool {
   257  	if v == "" {
   258  		return false
   259  	}
   260  	for part := range strings.SplitSeq(v, ".") {
   261  		if part == "" {
   262  			return false
   263  		}
   264  		if _, err := strconv.Atoi(part); err != nil {
   265  			return false
   266  		}
   267  	}
   268  
   269  	return true
   270  }
   271  
   272  // lessVersion は数値区切りを左から比べます。区切りの数が違う場合は足りない側を 0 とみなすので、
   273  // `1.2` は `1.2.0` と同じ位置に来ます。
   274  func lessVersion(a, b string) bool {
   275  	as, bs := strings.Split(a, "."), strings.Split(b, ".")
   276  	for i := 0; i < len(as) || i < len(bs); i++ {
   277  		x, y := 0, 0
   278  		if i < len(as) {
   279  			x, _ = strconv.Atoi(as[i])
   280  		}
   281  		if i < len(bs) {
   282  			y, _ = strconv.Atoi(bs[i])
   283  		}
   284  		if x != y {
   285  			return x < y
   286  		}
   287  	}
   288  
   289  	return false
   290  }

   292  // majorOf は先頭の数値区切りを返します。比較ではなく分類に使うので、解釈できない版は
   293  // 元の文字列をそのまま返し、他の major と混ざらないようにします。
   294  func majorOf(v string) string {
   295  	head, _, found := strings.Cut(strings.TrimPrefix(v, "v"), ".")
   296  	if !found {
   297  		return strings.TrimPrefix(v, "v")
   298  	}
   299  
   300  	return head
   301  }

   307  // outdatedReport は棚卸し結果を Markdown にします。issue の本文としてそのまま使うため、

   310  	var actionable, held, failed []candidate
   311  	for _, c := range cands {

   325  		releaseWindowDays, registryWindowDays)

   327  	b.WriteString("## 今すぐ上げられる\n\n")
   328  	if len(actionable) == 0 {
   329  		b.WriteString("なし。\n\n")

   335  		}
   336  		b.WriteString("\n`/tools-upgrade` が同じ窓でこの表を再計算し、宣言と lockfile を更新します。\n\n")
   337  	}
   338  
   339  	b.WriteString("## 窓が明けるのを待っている\n\n")
   340  	if len(held) == 0 {

   343  		b.WriteString("| ツール | 現在 | 上流最新 | 齢 | 窓 |\n| --- | --- | --- | --- | --- |\n")

   363  // reportOutdated は棚卸しを実行し、結果を標準出力と（指定があれば）ファイルへ書きます。
   364  // gate / audit と違い違反という概念が無いため、上流を確認できなかった場合だけ失敗させます。
   365  func reportOutdated(client *http.Client, declared []tool, opt options, now time.Time) error {
   366  	ctx := context.Background()
   367  	resolved, skipped, err := resolveBackends(ctx, declared)
   368  	if err != nil {
   369  		return xerrors.Wrap(err, "❌ backend の解決")
   370  	}
   371  
   372  	cands := surveyOutdated(ctx, client, resolved, now)
   373  	body := outdatedReport(cands, now)
   374  
   375  	var actionable, held, failed int
   376  	for _, c := range cands {
   377  		switch {
   378  		case c.err != nil:
   379  			failed++
   380  		case c.actionable():
   381  			actionable++
   382  		case c.heldByWindow():
   383  			held++
   384  		}
   385  	}
   386  
   387  	log.Printf("ℹ️ tool-cooldown outdated: 対象 %d 件 / ランタイム除外 %d 件 / 窓 GitHub %d 日・レジストリ %d 日",
   388  		len(resolved), len(skipped), releaseWindowDays, registryWindowDays)
   389  	log.Printf("ℹ️ 今すぐ上げられる %d 件 / 窓待ち %d 件 / 上流を確認できず %d 件", actionable, held, failed)
   390  
   391  	if opt.summaryOut != "" {
   392  		if writeErr := os.WriteFile(opt.summaryOut, []byte(body), summaryPerm); writeErr != nil {
   393  			return xerrors.Wrap(writeErr, "❌ write summary")
   394  		}
   395  	}
   396  	// 起票するかどうかは呼び出し側の判断なので、件数だけを渡す。
   397  	if out := os.Getenv("GITHUB_OUTPUT"); out != "" {
   398  		if writeErr := appendOutdatedOutput(out, actionable, held, failed); writeErr != nil {
   399  			return xerrors.Wrap(writeErr, "❌ write GITHUB_OUTPUT")
   400  		}
   401  	}
   402  
   403  	return nil
   404  }
