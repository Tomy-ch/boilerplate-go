   159  // run は mise.toml の宣言を cooldown に照らして報告します。gate で違反が残った場合はエラーを
   160  // 返し、終了コードへの変換は呼び出し側に委ねます。
   161  // client は上流への問い合わせ手段、now は窓と期限の基準時刻で、差し替えられるよう引数で受けます。
   162  func run(args []string, client *http.Client, now time.Time) error {
   163  	sub, opt, err := parseArgs(args)
   164  	if err != nil {
   165  		// ヘルプ要求は失敗ではないので 0 で終える。usage は flag が既に出力している。
   166  		if xerrors.Is(err, flag.ErrHelp) {
   167  			return nil
   168  		}
   169  
   170  		return err
   171  	}
   172  
   173  	// 期限の比較は暦日で行う。時刻を残すと 3 ヶ月の境界が実行時刻とタイムゾーンで動く。
   174  	today := time.Date(now.Year(), now.Month(), now.Day(), 0, 0, 0, 0, time.UTC)
   175  
   176  	paths, err := declarationPaths()
   177  	if err != nil {
   178  		return xerrors.Wrap(err, "❌ 宣言ファイルの列挙")
   179  	}
   180  	decls, err := readDeclarations(paths)
   181  	if err != nil {
   182  		return xerrors.Wrap(err, "❌ 宣言ファイルの読み取り")
   183  	}
   184  	declared, err := parseDeclarations(decls)
   185  	if err != nil {
   186  		return xerrors.Wrap(err, "❌ 宣言ファイルの解析")
   187  	}
   188  
   189  	bypasses, err := readBypasses(bypassFile)
   190  	if err != nil {
   191  		return xerrors.Wrap(err, "❌ "+bypassFile)
   192  	}
   193  	policyViolations, invalidBypasses := validateBypasses(bypasses, declared, today)
   194  	policyViolations = append(policyViolations, verifyLocks(declared)...)
   195  
   196  	targets := declared
   197  	if sub == "gate" {
   198  		targets, err = added(opt.base, paths, declared)
   199  		if err != nil {
   200  			return xerrors.Wrap(err, "❌ base との差分")
   201  		}
   202  	}
   203  
   204  	ctx := context.Background()
   205  	resolved, skipped, err := resolveBackends(ctx, targets)
   206  	if err != nil {
   207  		return xerrors.Wrap(err, "❌ backend の解決")
   208  	}
   209  
   210  	findings, unresolved := inspect(ctx, client, resolved, now)
   211  
   212  	blocking := report(
   213  		sub,
   214  		findings,
   215  		unresolved,
   216  		skipped,
   217  		policyViolations,
   218  		bypasses,
   219  		invalidBypasses,
   220  		resolved,
   221  		opt.github,
   222  	)
   223  
   224  	if opt.summaryOut != "" {
   225  		body := summary(sub, findings, unresolved, skipped, policyViolations, bypasses, invalidBypasses, resolved)
   226  		if writeErr := os.WriteFile(opt.summaryOut, []byte(body), summaryPerm); writeErr != nil {
   227  			return xerrors.Wrap(writeErr, "❌ write summary")
   228  		}
   229  	}
   230  	if out := os.Getenv("GITHUB_OUTPUT"); out != "" {
   231  		if writeErr := appendOutput(out, len(findings), len(resolved), blocking); writeErr != nil {
   232  			return xerrors.Wrap(writeErr, "❌ write GITHUB_OUTPUT")
   233  		}
   234  	}
   235  
   236  	if blocking > 0 {
   237  		return xerrors.Wrap(errBlocking, fmt.Sprintf("❌ tool-cooldown %s: %d 件の違反が残っています", sub, blocking))
   238  	}
   239  
   240  	return nil
   241  }

   243  // parseArgs はサブコマンドとフラグを解釈します。ヘルプ要求は flag.ErrHelp を含むエラーとして
   244  // 返し、失敗として扱うかどうかの判断は呼び出し側に委ねます。
   245  func parseArgs(args []string) (string, options, error) {
   246  	if len(args) == 0 {
   247  		return "", options{}, xerrors.Wrap(errUsage,
   248  			"❌ usage: tool-cooldown <gate|audit> [--base=<git-ref>] [--summary-out=<path>] [--github]")
   249  	}
   250  
   251  	sub := args[0]
   252  	if sub != "gate" && sub != "audit" {
   253  		return "", options{}, xerrors.Wrap(errUsage, fmt.Sprintf("❌ 未知のサブコマンド %q（gate | audit）", sub))
   254  	}
   255  
   256  	fs := flag.NewFlagSet(sub, flag.ContinueOnError)
   257  	base := fs.String("base", "", "gate: この git ref からの差分で追加 / 更新されたツールだけを対象にする")
   258  	summaryOut := fs.String("summary-out", "", "Markdown サマリの出力先")
   259  	github := fs.Bool("github", false, "GitHub Actions のアノテーションを出力する")
   260  	if err := fs.Parse(args[1:]); err != nil {
   261  		return "", options{}, xerrors.Wrap(err, "❌ フラグを解釈できません")
   262  	}
   263  
   264  	// gate は差分を取れないと何も検査しないまま通るため、比較対象の不在を先に落とす。
   265  	if sub == "gate" && *base == "" {
   266  		return "", options{}, xerrors.Wrap(errUsage, "❌ gate には --base が要ります（比較対象が無いと差分を取れません）")
   267  	}
   268  
   269  	return sub, options{base: *base, summaryOut: *summaryOut, github: *github}, nil
   270  }

   272  // parseTools は `[tools]` セクションの宣言だけを読む。`[settings]` の `pipx.uvx` や `[env]` の
   273  // バージョン値をツールとして拾わないよう、セクションを見て範囲を限る。
   274  func parseTools(content []byte) ([]tool, error) {
   275  	var tools []tool
   276  	section := ""
   277  	sc := bufio.NewScanner(strings.NewReader(string(content)))
   278  	for sc.Scan() {
   279  		line := sc.Text()
   280  		trimmed := strings.TrimSpace(line)
   281  		if trimmed == "" || strings.HasPrefix(trimmed, "#") {
   282  			continue
   283  		}
   284  		if m := sectionRe.FindStringSubmatch(trimmed); m != nil {
   285  			section = m[1]
   286  			continue
   287  		}
   288  		if section != "tools" {
   289  			continue
   290  		}
   291  		m := toolLineRe.FindStringSubmatch(trimmed)
   292  		if m == nil {
   293  			m = toolTableLineRe.FindStringSubmatch(trimmed)
   294  		}
   295  		if m == nil {
   296  			continue
   297  		}
   298  		key := m[1]
   299  		if key == "" {
   300  			key = m[2]
   301  		}
   302  		tools = append(tools, tool{key: key, version: m[3]})
   303  	}
   304  	return tools, sc.Err()
   305  }

   307  // parsePyRequirements は requirements の直接宣言を読む。キーは backend を明示した `pypi:` 付きに
   308  // 揃える。mise の宣言と同じ形にしておくと、窓の判定もバイパスのキーも経路で分かれずに済む。
   309  func parsePyRequirements(content []byte) ([]tool, error) {
   310  	var tools []tool
   311  	sc := bufio.NewScanner(strings.NewReader(string(content)))
   312  	for sc.Scan() {
   313  		trimmed := strings.TrimSpace(sc.Text())
   314  		if trimmed == "" || strings.HasPrefix(trimmed, "#") {
   315  			continue
   316  		}
   317  		m := pyRequirementRe.FindStringSubmatch(trimmed)
   318  		if m == nil {
   319  			continue
   320  		}
   321  		tools = append(tools, tool{key: "pypi:" + m[1], version: m[2]})
   322  	}
   323  	return tools, sc.Err()
   324  }

   349  // parseDeclarations は宣言ファイル群をツールの一覧へ畳む。形式はパスで決まる。
   350  func parseDeclarations(decls []declaration) ([]tool, error) {
   351  	var tools []tool
   352  	for _, d := range decls {
   353  		var (
   354  			parsed []tool
   355  			err    error
   356  		)
   357  		if d.path == miseFile {
   358  			parsed, err = parseTools(d.content)
   359  		} else {
   360  			parsed, err = parsePyRequirements(d.content)
   361  		}
   362  		if err != nil {
   363  			return nil, xerrors.Wrap(err, d.path)
   364  		}
   365  		for _, t := range parsed {
   366  			t.file = d.path
   367  			tools = append(tools, t)
   368  		}
   369  	}
   370  	return tools, nil
   371  }

   373  // added は base 時点の宣言に無かった (key, version) の組を返す。
   374  func added(base string, paths []string, current []tool) ([]tool, error) {
   375  	decls, err := baseDeclarations(base, paths)
   376  	if err != nil {
   377  		return nil, err
   378  	}
   379  	return addedFrom(decls, current)

   382  // baseDeclarations は base 時点の宣言ファイルを読む。宣言を増やした pull request では新しい
   383  // ファイルが base に無いのが正常なので、それは空として扱う。ただし mise.toml はどの base にも
   384  // あるはずのファイルで、無いのは base の取り違えを意味する。ref そのものを解決できない場合と
   385  // 併せてエラーにする。差分を取れないまま進むと gate は何も検査せずに通る。
   386  func baseDeclarations(base string, paths []string) ([]declaration, error) {
   387  	ctx, cancel := context.WithTimeout(context.Background(), gitTimeout)
   388  	defer cancel()
   389  
   390  	decls := make([]declaration, 0, len(paths))
   391  	for _, path := range paths {
   392  		//nolint:gosec // base は呼び出し側が与える git ref、path は declarationPaths が決めたもの
   393  		out, err := exec.CommandContext(ctx, "git", "show", base+":"+path).Output()
   394  		if err != nil {
   395  			if path == miseFile {
   396  				return nil, xerrors.Wrap(err, fmt.Sprintf("git show %s:%s（base を取得できていない可能性があります）", base, path))
   397  			}
   398  			if refErr := verifyRef(ctx, base); refErr != nil {
   399  				return nil, xerrors.Wrap(refErr, fmt.Sprintf("git show %s:%s（base を取得できていない可能性があります）", base, path))
   400  			}
   401  			continue
   402  		}
   403  		decls = append(decls, declaration{path: path, content: out})
   404  	}
   405  	return decls, nil
   406  }

   408  // verifyRef は base が commit として解決できるかを見る。
   409  func verifyRef(ctx context.Context, base string) error {
   410  	//nolint:gosec // base は呼び出し側が与える git ref
   411  	if err := exec.CommandContext(ctx, "git", "rev-parse", "--verify", "--quiet", base+"^{commit}").Run(); err != nil {
   412  		return xerrors.Wrap(err, "git rev-parse "+base)
   413  	}
   414  	return nil
   415  }

   417  // addedFrom は base 時点の宣言と現在の宣言を突き合わせる。base を読めなかった場合に空の差分を
   418  // 返すと、gate は何も検査しないまま通る。解析の失敗はエラーとして表に出す。
   419  func addedFrom(baseDecls []declaration, current []tool) ([]tool, error) {
   420  	before, err := parseDeclarations(baseDecls)
   421  	if err != nil {
   422  		return nil, xerrors.Wrap(err, "base の宣言")
   423  	}
   424  	return diffAdded(before, current), nil
   425  }

   427  // diffAdded は before に無い (key, version) の組を返す。
   428  func diffAdded(before, current []tool) []tool {
   429  	known := make(map[string]struct{}, len(before))
   430  	for _, t := range before {
   431  		known[t.id()] = struct{}{}
   432  	}
   433  	var fresh []tool
   434  	for _, t := range current {
   435  		if _, ok := known[t.id()]; !ok {
   436  			fresh = append(fresh, t)
   437  		}
   438  	}

   512  // verifyLocks は *.in が宣言する版を、隣の *.txt が固定しているかどうかで確かめる。ハッシュの
   513  // 照合そのものは install 側の --require-hashes が担うので、ここで見るのは宣言と lockfile が
   514  // 同じ版を指しているかだけ。食い違ったままだと、cooldown を通した版と実際に入る版が別になる。
   515  func verifyLocks(declared []tool) []violation {
   516  	var violations []violation
   517  	for _, t := range declared {
   518  		if !strings.HasSuffix(t.file, pyRequirementSuffix) {
   519  			continue
   520  		}
   521  		lockPath := strings.TrimSuffix(t.file, pyRequirementSuffix) + pyLockSuffix
   522  		content, err := os.ReadFile(lockPath) //nolint:gosec // 宣言ファイルのパスから決まる
   523  		if err != nil {
   524  			violations = append(violations, violation{file: t.file, msg: fmt.Sprintf(
   525  				"%s を読めません（%v）。`make py-lock` で生成してください", lockPath, err,
   526  			)})
   527  			continue
   528  		}
   529  		name := pyPackageName(t.key)
   530  		locked, ok := lockedVersion(content, name)
   531  		if !ok {
   532  			violations = append(violations, violation{file: t.file, msg: fmt.Sprintf(
   533  				"%s が宣言する %s が %s にありません。`make py-lock` で再生成してください", t.file, t.id(), lockPath,
   534  			)})
   535  			continue
   536  		}
   537  		if locked != t.version {
   538  			violations = append(violations, violation{file: t.file, msg: fmt.Sprintf(
   539  				"%s の宣言は %s ですが %s が固定しているのは %s です。`make py-lock` で再生成してください",
   540  				t.file, t.id(), lockPath, name+"=="+locked,
   541  			)})
   542  		}
   543  	}
   544  	return violations
   545  }

   572  // inspect は対象ツールの公開時刻を引き、窓内のものと取得できなかったものを返す。
   573  func inspect(ctx context.Context, client *http.Client, targets []tool, now time.Time) ([]finding, []tool) {
   574  	var (
   575  		mu         sync.Mutex
   576  		findings   []finding
   577  		unresolved []tool
   578  		wg         sync.WaitGroup
   579  	)
   580  	sem := make(chan struct{}, fetchWorkers)
   581  
   582  	for _, t := range targets {
   583  		wg.Add(1)
   584  		go func(t tool) {
   585  			defer wg.Done()
   586  			sem <- struct{}{}
   587  			defer func() { <-sem }()
   588  
   589  			at, err := publishedAt(ctx, client, t)
   590  			mu.Lock()
   591  			defer mu.Unlock()
   592  			if err != nil {
   593  				unresolved = append(unresolved, t)
   594  				return
   595  			}
   596  			window := windowFor(t.backend)
   597  			if age := int(now.Sub(at).Hours() / hoursPerDay); age < window {
   598  				findings = append(findings, finding{tool: t, published: at, ageDays: age, window: window})
   599  			}
   600  		}(t)
   601  	}
   602  	wg.Wait()
   603  
   604  	sort.Slice(findings, func(i, j int) bool { return findings[i].ageDays < findings[j].ageDays })
   605  	sort.Slice(unresolved, func(i, j int) bool { return unresolved[i].id() < unresolved[j].id() })
   606  	return findings, unresolved
   607  }

   626  func getJSON(ctx context.Context, client *http.Client, url string, into any) error {
   627  	req, err := http.NewRequestWithContext(ctx, http.MethodGet, url, nil)
   628  	if err != nil {
   629  		return xerrors.Wrap(err, "new request")
   630  	}
   631  	// 未認証の GitHub API は 60 req/hour（IP 単位）で、このツールの 1 回の実行を賄えない。
   632  	if strings.HasPrefix(url, githubAPI) {
   633  		if token := os.Getenv("GITHUB_TOKEN"); token != "" {
   634  			req.Header.Set("Authorization", "Bearer "+token)
   635  		}
   636  	}
   637  	resp, err := client.Do(req)
   638  	if err != nil {
   639  		return xerrors.Wrap(err, "fetch "+url)
   640  	}
   641  	defer func() { _ = resp.Body.Close() }()
   642  
   643  	if resp.StatusCode == http.StatusNotFound || resp.StatusCode == http.StatusGone {
   644  		return xerrors.Wrap(errNotFound, url)
   645  	}
   646  	if resp.StatusCode != http.StatusOK {
   647  		return xerrors.Wrap(errUpstreamStatus, fmt.Sprintf("%s: %d", url, resp.StatusCode))
   648  	}
   649  	return json.NewDecoder(resp.Body).Decode(into)
   650  }

   652  // githubReleaseAt は Release の published_at を返す。タグに `v` を付ける流儀と付けない流儀が
   653  // あるため、宣言された版そのものと `v` 付きの両方を試す。
   654  func githubReleaseAt(ctx context.Context, client *http.Client, repo, version string) (time.Time, error) {
   655  	var body struct {
   656  		//nolint:tagliatelle // GitHub API の応答フィールド名
   657  		PublishedAt time.Time `json:"published_at"`
   658  	}
   659  	var lastErr error
   660  	for _, tag := range []string{"v" + version, version} {
   661  		err := getJSON(ctx, client, githubAPI+repo+"/releases/tags/"+tag, &body)
   662  		if err == nil {
   663  			return body.PublishedAt, nil
   664  		}
   665  		lastErr = err
   666  	}
   667  	return time.Time{}, lastErr
   668  }

   670  // goModuleAt は module proxy の .info を返す。mise の go backend はパッケージパスを受けるが
   671  // proxy が知るのはモジュールパスなので、解決するまで末尾要素を落としながら遡る。
   672  func goModuleAt(ctx context.Context, client *http.Client, pkg, version string) (time.Time, error) {
   673  	var body struct {
   674  		//nolint:tagliatelle // module proxy の応答フィールド名
   675  		Time time.Time `json:"Time"`
   676  	}
   677  	var lastErr error
   678  	for path := pkg; strings.Contains(path, "/"); path = path[:strings.LastIndex(path, "/")] {
   679  		err := getJSON(ctx, client, goProxyBase+escapeModulePath(path)+"/@v/v"+version+".info", &body)
   680  		if err == nil {
   681  			return body.Time, nil
   682  		}
   683  		lastErr = err
   684  	}
   685  	// pkg に `/` が無いとループが 1 度も回らない。ここで nil を返すと呼び出し側はゼロ値時刻を
   686  	// 「公開から数十年経過」と読み、そのツールが cooldown を無条件で通過する。「問い合わせた結果
   687  	// 見つからない」と「一度も問い合わせていない」は、公開時刻を得られていない点で同じ扱いにする。
   688  	if lastErr == nil {
   689  		return time.Time{}, xerrors.Wrap(errNotFound, pkg+"@"+version)
   690  	}
   691  	return time.Time{}, lastErr
   692  }
   693  
   694  // escapeModulePath は module path を proxy のパス表記へ変換する。proxy は大文字を `!` + 小文字で
   695  // 受けるため、未エスケープのまま問い合わせると 404 になり「存在しないバージョン」に化ける。
   696  func escapeModulePath(path string) string {
   697  	var b strings.Builder
   698  	for _, r := range path {
   699  		if r >= 'A' && r <= 'Z' {
   700  			b.WriteByte('!')
   701  			b.WriteRune(r + ('a' - 'A'))
   702  			continue
   703  		}
   704  		b.WriteRune(r)
   705  	}
   706  	return b.String()
   707  }

   740  // readBypasses はバイパス lockfile を key@version→bypass として読む。空行とコメント行以外で
   741  // 解釈できない行、および既出キーの再定義はエラーにする。読み飛ばしや後勝ちの上書きは、その
   742  // エントリが「存在しない」あるいは「行順で決まる」状態を警告なく作る。
   743  func readBypasses(path string) (map[string]bypass, error) {
   744  	f, err := os.Open(path) //nolint:gosec // path はリテラル
   745  	if os.IsNotExist(err) {
   746  		return map[string]bypass{}, nil
   747  	}
   748  	if err != nil {
   749  		return nil, err
   750  	}
   751  	defer func() { _ = f.Close() }()
   752  
   753  	out := map[string]bypass{}
   754  	sc := bufio.NewScanner(f)
   755  	for lineNo := 1; sc.Scan(); lineNo++ {
   756  		line := strings.TrimSpace(sc.Text())
   757  		if line == "" || strings.HasPrefix(line, "#") {
   758  			continue
   759  		}
   760  		m := bypassLineRe.FindStringSubmatch(line)
   761  		if m == nil {
   762  			return nil, xerrors.Wrap(errBypassInvalidLine, fmt.Sprintf(
   763  				`%d 行目: %q（形式は "<key>@<version>" = { expires = YYYY-MM-DD, issue = <N>, reason = "..." }）`, lineNo, line,
   764  			))
   765  		}
   766  		if _, dup := out[m[1]]; dup {
   767  			return nil, xerrors.Wrap(errBypassDuplicateKey, fmt.Sprintf("%d 行目: %q", lineNo, m[1]))
   768  		}
   769  		expires, err := time.Parse(time.DateOnly, m[2])
   770  		if err != nil {
   771  			return nil, xerrors.Wrap(err, fmt.Sprintf("%d 行目の expires", lineNo))
   772  		}
   773  		issue, err := strconv.Atoi(m[3])
   774  		if err != nil {
   775  			return nil, xerrors.Wrap(err, fmt.Sprintf("%d 行目の issue", lineNo))
   776  		}
   777  		out[m[1]] = bypass{expires: expires, issue: issue, reason: m[4], line: lineNo}
   778  	}
   779  	return out, sc.Err()
   780  }

   782  // validateBypasses はバイパス自身の規約違反と、そのせいで無効になったキーを返す。期限切れを
   783  // 失敗にするのは、外したまま放置されたバイパスが恒久 allowlist と区別できなくなるため。上限を
   784  // 置くのは、期限を遠い未来へ置くだけで同じ状態を作れてしまうため。無効なバイパスは効力も失う。
   785  func validateBypasses(bypasses map[string]bypass, declared []tool, today time.Time) ([]violation, map[string]struct{}) {
   786  	inDeclarations := make(map[string]struct{}, len(declared))
   787  	for _, t := range declared {
   788  		inDeclarations[t.id()] = struct{}{}
   789  	}
   790  	limit := today.AddDate(0, maxBypassMonths, 0)
   791  	invalid := map[string]struct{}{}
   792  	var violations []violation
   793  
   794  	for _, key := range sortedKeys(bypasses) {
   795  		b := bypasses[key]
   796  		switch {
   797  		case b.expires.Before(today):
   798  			invalid[key] = struct{}{}
   799  			violations = append(violations, violation{file: bypassFile, msg: fmt.Sprintf(
   800  				"%s:%d %s の期限 %s が切れています。窓明けの版へ更新してエントリを消すか、期限を延ばす判断を #%d で記録してください",
   801  				bypassFile, b.line, key, b.expires.Format(time.DateOnly), b.issue,
   802  			)})
   803  		case b.expires.After(limit):
   804  			invalid[key] = struct{}{}
   805  			violations = append(violations, violation{file: bypassFile, msg: fmt.Sprintf(
   806  				"%s:%d %s の期限 %s が上限（%s から %d ヶ月 = %s）を越えています。恒久 allowlist にしないための上限です",
   807  				bypassFile, b.line, key, b.expires.Format(time.DateOnly),
   808  				today.Format(time.DateOnly), maxBypassMonths, limit.Format(time.DateOnly),
   809  			)})
   810  		default:
   811  			if _, ok := inDeclarations[key]; !ok {
   812  				violations = append(violations, violation{file: bypassFile, msg: fmt.Sprintf(
   813  					"%s:%d %s は %s / %s のどちらにも存在しません。不要になったエントリは消してください",
   814  					bypassFile, b.line, key, miseFile, pyRequirementsGlob,
   815  				)})
   816  			}
   817  		}
   818  	}
   819  	return violations, invalid
   820  }

   831  // classify は finding を「バイパス済み」「gate を落とすもの」「報告に留めるもの」へ振り分ける。
   832  func classify(sub string, findings []finding, bypasses map[string]bypass, invalid map[string]struct{}) ([]finding, []finding, []finding) {
   833  	var bypassed, blocking, reported []finding
   834  	for _, f := range findings {
   835  		switch {
   836  		case hasBypass(bypasses, invalid, f.tool):
   837  			bypassed = append(bypassed, f)
   838  		case sub == "gate":
   839  			blocking = append(blocking, f)
   840  		default:
   841  			reported = append(reported, f)
   842  		}
   843  	}
   844  	return bypassed, blocking, reported
   845  }

   855  // report は結果を標準出力へ書き、終了コードを非ゼロにすべき件数を返す。audit は窓内の finding では
   856  // 落ちない。ただしバイパス自身の規約違反だけは audit でも失敗させる。期限切れの回収がスケジュール
   857  // 実行に懸かっているため。
   858  func report(
   859  	sub string, findings []finding, unresolved, skipped []tool,
   860  	policyViolations []violation, bypasses map[string]bypass, invalid map[string]struct{},
   861  	targets []tool, github bool,
   862  ) int {
   863  	//nolint:gosec // sub は gate / audit のいずれかに限定済み
   864  	log.Printf("ℹ️ tool-cooldown %s: 対象 %d 件 / ランタイム除外 %d 件 / 窓 GitHub %d 日・レジストリ %d 日",
   865  		sub, len(targets), len(skipped), releaseWindowDays, registryWindowDays)
   866  
   867  	blockingCount := len(policyViolations)
   868  	for _, v := range policyViolations {
   869  		log.Printf("❌ %s", v.msg)
   870  		if github {
   871  			log.Printf("::error file=%s::%s", v.file, v.msg)
   872  		}
   873  	}
   874  
   875  	bypassed, blocked, reported := classify(sub, findings, bypasses, invalid)
   876  
   877  	for _, f := range bypassed {
   878  		b := bypasses[f.tool.id()]
   879  		log.Printf("🔓 %s は公開 %d 日ですが、#%d のバイパスで通します（期限 %s / %s）",
   880  			f.tool.id(), f.ageDays, b.issue, b.expires.Format(time.DateOnly), b.reason)
   881  	}
   882  
   883  	blockingCount += len(blocked)
   884  	for _, f := range blocked {

   918  // fenceFor は text を包むのに足りるフェンスを返す。長さは text 中の最長バッククォート連 + 1。
   919  func fenceFor(text string) string {
   920  	longest, run := 0, 0
   921  	for _, r := range text {
   922  		if r == '`' {
   923  			run++
   924  			if run > longest {
   925  				longest = run
   926  			}
   927  			continue
   928  		}
   929  		run = 0
   930  	}
   931  	return strings.Repeat("`", max(minFenceLen, longest+1))
   932  }
