   160  // run は mise.toml の宣言を cooldown に照らして報告します。gate で違反が残った場合はエラーを
   161  // 返し、終了コードへの変換は呼び出し側に委ねます。
   162  // client は上流への問い合わせ手段、now は窓と期限の基準時刻で、差し替えられるよう引数で受けます。
   163  func run(args []string, client *http.Client, now time.Time) error {
   164  	sub, opt, err := parseArgs(args)
   165  	if err != nil {
   166  		// ヘルプ要求は失敗ではないので 0 で終える。usage は flag が既に出力している。
   167  		if xerrors.Is(err, flag.ErrHelp) {
   168  			return nil
   169  		}
   170  
   171  		return err
   172  	}
   173  
   174  	// 期限の比較は暦日で行う。時刻を残すと 3 ヶ月の境界が実行時刻とタイムゾーンで動く。
   175  	today := time.Date(now.Year(), now.Month(), now.Day(), 0, 0, 0, 0, time.UTC)
   176  
   177  	paths, err := declarationPaths()
   178  	if err != nil {
   179  		return xerrors.Wrap(err, "❌ 宣言ファイルの列挙")
   180  	}
   181  	decls, err := readDeclarations(paths)
   182  	if err != nil {
   183  		return xerrors.Wrap(err, "❌ 宣言ファイルの読み取り")
   184  	}
   185  	declared, err := parseDeclarations(decls)
   186  	if err != nil {
   187  		return xerrors.Wrap(err, "❌ 宣言ファイルの解析")
   188  	}
   189  
   190  	if sub == "outdated" {
   191  		return reportOutdated(client, declared, opt, now)
   192  	}
   193  
   194  	bypasses, err := readBypasses(bypassFile)
   195  	if err != nil {
   196  		return xerrors.Wrap(err, "❌ "+bypassFile)
   197  	}
   198  	policyViolations, invalidBypasses := validateBypasses(bypasses, declared, today)
   199  	policyViolations = append(policyViolations, verifyLocks(declared)...)
   200  
   201  	targets := declared
   202  	if sub == "gate" {
   203  		targets, err = added(opt.base, paths, declared)
   204  		if err != nil {
   205  			return xerrors.Wrap(err, "❌ base との差分")
   206  		}
   207  	}
   208  
   209  	ctx := context.Background()
   210  	resolved, skipped, err := resolveBackends(ctx, targets)
   211  	if err != nil {
   212  		return xerrors.Wrap(err, "❌ backend の解決")
   213  	}
   214  
   215  	findings, unresolved := inspect(ctx, client, resolved, now)
   216  
   217  	blocking := report(
   218  		sub,
   219  		findings,
   220  		unresolved,
   221  		skipped,
   222  		policyViolations,
   223  		bypasses,
   224  		invalidBypasses,
   225  		resolved,
   226  		opt.github,
   227  	)
   228  
   229  	if opt.summaryOut != "" {
   230  		body := summary(sub, findings, unresolved, skipped, policyViolations, bypasses, invalidBypasses, resolved)
   231  		if writeErr := os.WriteFile(opt.summaryOut, []byte(body), summaryPerm); writeErr != nil {
   232  			return xerrors.Wrap(writeErr, "❌ write summary")
   233  		}
   234  	}
   235  	if out := os.Getenv("GITHUB_OUTPUT"); out != "" {
   236  		if writeErr := appendOutput(out, len(findings), len(resolved), blocking); writeErr != nil {
   237  			return xerrors.Wrap(writeErr, "❌ write GITHUB_OUTPUT")
   238  		}
   239  	}
   240  
   241  	if blocking > 0 {
   242  		return xerrors.Wrap(errBlocking, fmt.Sprintf("❌ tool-cooldown %s: %d 件の違反が残っています", sub, blocking))
   243  	}
   244  
   245  	return nil
   246  }

   248  // parseArgs はサブコマンドとフラグを解釈します。ヘルプ要求は flag.ErrHelp を含むエラーとして
   249  // 返し、失敗として扱うかどうかの判断は呼び出し側に委ねます。
   250  func parseArgs(args []string) (string, options, error) {
   251  	if len(args) == 0 {
   252  		return "", options{}, xerrors.Wrap(errUsage,
   253  			"❌ usage: tool-cooldown <gate|audit|outdated> [--base=<git-ref>] [--summary-out=<path>] [--github]")
   254  	}
   255  
   256  	sub := args[0]
   257  	if sub != "gate" && sub != "audit" && sub != "outdated" {
   258  		return "", options{}, xerrors.Wrap(errUsage,
   259  			fmt.Sprintf("❌ 未知のサブコマンド %q（gate | audit | outdated）", sub))
   260  	}
   261  
   262  	fs := flag.NewFlagSet(sub, flag.ContinueOnError)
   263  	base := fs.String("base", "", "gate: この git ref からの差分で追加 / 更新されたツールだけを対象にする")
   264  	summaryOut := fs.String("summary-out", "", "Markdown サマリの出力先")
   265  	github := fs.Bool("github", false, "GitHub Actions のアノテーションを出力する")
   266  	if err := fs.Parse(args[1:]); err != nil {
   267  		return "", options{}, xerrors.Wrap(err, "❌ フラグを解釈できません")
   268  	}
   269  
   270  	// gate は差分を取れないと何も検査しないまま通るため、比較対象の不在を先に落とす。
   271  	if sub == "gate" && *base == "" {
   272  		return "", options{}, xerrors.Wrap(errUsage, "❌ gate には --base が要ります（比較対象が無いと差分を取れません）")
   273  	}
   274  
   275  	return sub, options{base: *base, summaryOut: *summaryOut, github: *github}, nil
   276  }

   278  // parseTools は `[tools]` セクションの宣言だけを読む。`[settings]` の `pipx.uvx` や `[env]` の
   279  // バージョン値をツールとして拾わないよう、セクションを見て範囲を限る。
   280  func parseTools(content []byte) ([]tool, error) {
   281  	var tools []tool
   282  	section := ""
   283  	sc := bufio.NewScanner(strings.NewReader(string(content)))
   284  	for sc.Scan() {
   285  		line := sc.Text()
   286  		trimmed := strings.TrimSpace(line)
   287  		if trimmed == "" || strings.HasPrefix(trimmed, "#") {
   288  			continue
   289  		}
   290  		if m := sectionRe.FindStringSubmatch(trimmed); m != nil {
   291  			section = m[1]
   292  			continue
   293  		}
   294  		if section != "tools" {
   295  			continue
   296  		}
   297  		m := toolLineRe.FindStringSubmatch(trimmed)
   298  		if m == nil {
   299  			m = toolTableLineRe.FindStringSubmatch(trimmed)
   300  		}
   301  		if m == nil {
   302  			continue
   303  		}
   304  		key := m[1]
   305  		if key == "" {
   306  			key = m[2]
   307  		}
   308  		tools = append(tools, tool{key: key, version: m[3]})
   309  	}
   310  	return tools, sc.Err()
   311  }

   313  // parsePyRequirements は requirements の直接宣言を読む。キーは backend を明示した `pypi:` 付きに
   314  // 揃える。mise の宣言と同じ形にしておくと、窓の判定もバイパスのキーも経路で分かれずに済む。
   315  func parsePyRequirements(content []byte) ([]tool, error) {
   316  	var tools []tool
   317  	sc := bufio.NewScanner(strings.NewReader(string(content)))
   318  	for sc.Scan() {
   319  		trimmed := strings.TrimSpace(sc.Text())
   320  		if trimmed == "" || strings.HasPrefix(trimmed, "#") {
   321  			continue
   322  		}
   323  		m := pyRequirementRe.FindStringSubmatch(trimmed)
   324  		if m == nil {
   325  			continue
   326  		}
   327  		tools = append(tools, tool{key: "pypi:" + m[1], version: m[2]})
   328  	}
   329  	return tools, sc.Err()
   330  }

   342  // readDeclarations は作業ツリーの宣言ファイルを読む。
   343  func readDeclarations(paths []string) ([]declaration, error) {
   344  	decls := make([]declaration, 0, len(paths))
   345  	for _, path := range paths {
   346  		content, err := os.ReadFile(path) //nolint:gosec // path は declarationPaths が決めたもの
   347  		if err != nil {
   348  			return nil, xerrors.Wrap(err, path)
   349  		}
   350  		decls = append(decls, declaration{path: path, content: content})
   351  	}
   352  	return decls, nil

   355  // parseDeclarations は宣言ファイル群をツールの一覧へ畳む。形式はパスで決まる。
   356  func parseDeclarations(decls []declaration) ([]tool, error) {
   357  	var tools []tool
   358  	for _, d := range decls {
   359  		var (
   360  			parsed []tool
   361  			err    error
   362  		)
   363  		if d.path == miseFile {
   364  			parsed, err = parseTools(d.content)
   365  		} else {
   366  			parsed, err = parsePyRequirements(d.content)
   367  		}
   368  		if err != nil {
   369  			return nil, xerrors.Wrap(err, d.path)
   370  		}
   371  		for _, t := range parsed {
   372  			t.file = d.path
   373  			tools = append(tools, t)
   374  		}
   375  	}
   376  	return tools, nil
   377  }

   379  // added は base 時点の宣言に無かった (key, version) の組を返す。
   380  func added(base string, paths []string, current []tool) ([]tool, error) {
   381  	decls, err := baseDeclarations(base, paths)
   382  	if err != nil {
   383  		return nil, err
   384  	}
   385  	return addedFrom(decls, current)
   386  }

   388  // baseDeclarations は base 時点の宣言ファイルを読む。宣言を増やした pull request では新しい
   389  // ファイルが base に無いのが正常なので、それは空として扱う。ただし mise.toml はどの base にも
   390  // あるはずのファイルで、無いのは base の取り違えを意味する。ref そのものを解決できない場合と
   391  // 併せてエラーにする。差分を取れないまま進むと gate は何も検査せずに通る。
   392  func baseDeclarations(base string, paths []string) ([]declaration, error) {
   393  	ctx, cancel := context.WithTimeout(context.Background(), gitTimeout)
   394  	defer cancel()
   395  
   396  	decls := make([]declaration, 0, len(paths))
   397  	for _, path := range paths {
   398  		//nolint:gosec // base は呼び出し側が与える git ref、path は declarationPaths が決めたもの
   399  		out, err := exec.CommandContext(ctx, "git", "show", base+":"+path).Output()
   400  		if err != nil {
   401  			if path == miseFile {
   402  				return nil, xerrors.Wrap(err, fmt.Sprintf("git show %s:%s（base を取得できていない可能性があります）", base, path))
   403  			}
   404  			if refErr := verifyRef(ctx, base); refErr != nil {
   405  				return nil, xerrors.Wrap(refErr, fmt.Sprintf("git show %s:%s（base を取得できていない可能性があります）", base, path))
   406  			}
   407  			continue
   408  		}
   409  		decls = append(decls, declaration{path: path, content: out})
   410  	}
   411  	return decls, nil
   412  }

   414  // verifyRef は base が commit として解決できるかを見る。
   415  func verifyRef(ctx context.Context, base string) error {
   416  	//nolint:gosec // base は呼び出し側が与える git ref
   417  	if err := exec.CommandContext(ctx, "git", "rev-parse", "--verify", "--quiet", base+"^{commit}").Run(); err != nil {
   418  		return xerrors.Wrap(err, "git rev-parse "+base)
   419  	}
   420  	return nil
   421  }

   423  // addedFrom は base 時点の宣言と現在の宣言を突き合わせる。base を読めなかった場合に空の差分を
   424  // 返すと、gate は何も検査しないまま通る。解析の失敗はエラーとして表に出す。
   425  func addedFrom(baseDecls []declaration, current []tool) ([]tool, error) {
   426  	before, err := parseDeclarations(baseDecls)
   427  	if err != nil {
   428  		return nil, xerrors.Wrap(err, "base の宣言")
   429  	}
   430  	return diffAdded(before, current), nil
   431  }

   433  // diffAdded は before に無い (key, version) の組を返す。
   434  func diffAdded(before, current []tool) []tool {
   435  	known := make(map[string]struct{}, len(before))
   436  	for _, t := range before {
   437  		known[t.id()] = struct{}{}
   438  	}

   518  // verifyLocks は *.in が宣言する版を、隣の *.txt が固定しているかどうかで確かめる。ハッシュの
   519  // 照合そのものは install 側の --require-hashes が担うので、ここで見るのは宣言と lockfile が
   520  // 同じ版を指しているかだけ。食い違ったままだと、cooldown を通した版と実際に入る版が別になる。
   521  func verifyLocks(declared []tool) []violation {
   522  	var violations []violation
   523  	for _, t := range declared {
   524  		if !strings.HasSuffix(t.file, pyRequirementSuffix) {
   525  			continue
   526  		}
   527  		lockPath := strings.TrimSuffix(t.file, pyRequirementSuffix) + pyLockSuffix
   528  		content, err := os.ReadFile(lockPath) //nolint:gosec // 宣言ファイルのパスから決まる
   529  		if err != nil {
   530  			violations = append(violations, violation{file: t.file, msg: fmt.Sprintf(
   531  				"%s を読めません（%v）。`make py-lock` で生成してください", lockPath, err,
   532  			)})
   533  			continue
   534  		}
   535  		name := pyPackageName(t.key)
   536  		locked, ok := lockedVersion(content, name)
   537  		if !ok {
   538  			violations = append(violations, violation{file: t.file, msg: fmt.Sprintf(
   539  				"%s が宣言する %s が %s にありません。`make py-lock` で再生成してください", t.file, t.id(), lockPath,
   540  			)})
   541  			continue
   542  		}
   543  		if locked != t.version {
   544  			violations = append(violations, violation{file: t.file, msg: fmt.Sprintf(
   545  				"%s の宣言は %s ですが %s が固定しているのは %s です。`make py-lock` で再生成してください",
   546  				t.file, t.id(), lockPath, name+"=="+locked,
   547  			)})
   548  		}
   549  	}
   550  	return violations
   551  }

   578  // inspect は対象ツールの公開時刻を引き、窓内のものと取得できなかったものを返す。
   579  func inspect(ctx context.Context, client *http.Client, targets []tool, now time.Time) ([]finding, []tool) {
   580  	var (
   581  		mu         sync.Mutex
   582  		findings   []finding
   583  		unresolved []tool
   584  		wg         sync.WaitGroup
   585  	)
   586  	sem := make(chan struct{}, fetchWorkers)
   587  
   588  	for _, t := range targets {
   589  		wg.Add(1)
   590  		go func(t tool) {
   591  			defer wg.Done()
   592  			sem <- struct{}{}
   593  			defer func() { <-sem }()
   594  
   595  			at, err := publishedAt(ctx, client, t)
   596  			mu.Lock()
   597  			defer mu.Unlock()
   598  			if err != nil {
   599  				unresolved = append(unresolved, t)
   600  				return
   601  			}
   602  			window := windowFor(t.backend)
   603  			if age := int(now.Sub(at).Hours() / hoursPerDay); age < window {
   604  				findings = append(findings, finding{tool: t, published: at, ageDays: age, window: window})
   605  			}
   606  		}(t)
   607  	}
   608  	wg.Wait()
   609  
   610  	sort.Slice(findings, func(i, j int) bool { return findings[i].ageDays < findings[j].ageDays })
   611  	sort.Slice(unresolved, func(i, j int) bool { return unresolved[i].id() < unresolved[j].id() })
   612  	return findings, unresolved
   613  }

   658  // fetchBody は GET して本文を返す。呼び出し側が Close する。
   659  func fetchBody(ctx context.Context, client *http.Client, url string) (io.ReadCloser, error) {
   660  	req, err := http.NewRequestWithContext(ctx, http.MethodGet, url, nil)
   661  	if err != nil {
   662  		return nil, xerrors.Wrap(err, "new request")
   663  	}
   664  	// 未認証の GitHub API は 60 req/hour（IP 単位）で、このツールの 1 回の実行を賄えない。
   665  	if strings.HasPrefix(url, githubAPI) {
   666  		if token := os.Getenv("GITHUB_TOKEN"); token != "" {
   667  			req.Header.Set("Authorization", "Bearer "+token)
   668  		}
   669  	}
   670  	resp, err := client.Do(req)
   671  	if err != nil {
   672  		return nil, xerrors.Wrap(err, "fetch "+url)
   673  	}
   674  
   675  	if resp.StatusCode == http.StatusNotFound || resp.StatusCode == http.StatusGone {
   676  		_ = resp.Body.Close()
   677  
   678  		return nil, xerrors.Wrap(errNotFound, url)
   679  	}
   680  	if resp.StatusCode != http.StatusOK {
   681  		_ = resp.Body.Close()
   682  
   683  		return nil, xerrors.Wrap(errUpstreamStatus, fmt.Sprintf("%s: %d", url, resp.StatusCode))
   684  	}
   685  
   686  	return resp.Body, nil
   687  }

   689  // githubReleaseAt は Release の published_at を返す。タグに `v` を付ける流儀と付けない流儀が

   691  func githubReleaseAt(ctx context.Context, client *http.Client, repo, version string) (time.Time, error) {
   692  	var body struct {
   693  		//nolint:tagliatelle // GitHub API の応答フィールド名
   694  		PublishedAt time.Time `json:"published_at"`
   695  	}
   696  	var lastErr error
   697  	for _, tag := range []string{"v" + version, version} {
   698  		err := getJSON(ctx, client, githubAPI+repo+"/releases/tags/"+tag, &body)
   699  		if err == nil {

   701  		}
   702  		lastErr = err
   703  	}

   707  // goModuleAt は module proxy の .info を返す。mise の go backend はパッケージパスを受けるが
   708  // proxy が知るのはモジュールパスなので、解決するまで末尾要素を落としながら遡る。
   709  func goModuleAt(ctx context.Context, client *http.Client, pkg, version string) (time.Time, error) {
   710  	var body struct {
   711  		//nolint:tagliatelle // module proxy の応答フィールド名
   712  		Time time.Time `json:"Time"`
   713  	}
   714  	var lastErr error
   715  	for path := pkg; strings.Contains(path, "/"); path = path[:strings.LastIndex(path, "/")] {
   716  		err := getJSON(ctx, client, goProxyBase+escapeModulePath(path)+"/@v/v"+version+".info", &body)
   717  		if err == nil {
   718  			return body.Time, nil
   719  		}
   720  		lastErr = err
   721  	}
   722  	// pkg に `/` が無いとループが 1 度も回らない。ここで nil を返すと呼び出し側はゼロ値時刻を
   723  	// 「公開から数十年経過」と読み、そのツールが cooldown を無条件で通過する。「問い合わせた結果
   724  	// 見つからない」と「一度も問い合わせていない」は、公開時刻を得られていない点で同じ扱いにする。
   725  	if lastErr == nil {
   726  		return time.Time{}, xerrors.Wrap(errNotFound, pkg+"@"+version)
   727  	}
   728  	return time.Time{}, lastErr
   729  }

   731  // escapeModulePath は module path を proxy のパス表記へ変換する。proxy は大文字を `!` + 小文字で
   732  // 受けるため、未エスケープのまま問い合わせると 404 になり「存在しないバージョン」に化ける。
   733  func escapeModulePath(path string) string {
   734  	var b strings.Builder
   735  	for _, r := range path {
   736  		if r >= 'A' && r <= 'Z' {
   737  			b.WriteByte('!')
   738  			b.WriteRune(r + ('a' - 'A'))
   739  			continue
   740  		}
   741  		b.WriteRune(r)
   742  	}
   743  	return b.String()
   744  }

   777  // readBypasses はバイパス lockfile を key@version→bypass として読む。空行とコメント行以外で
   778  // 解釈できない行、および既出キーの再定義はエラーにする。読み飛ばしや後勝ちの上書きは、その
   779  // エントリが「存在しない」あるいは「行順で決まる」状態を警告なく作る。
   780  func readBypasses(path string) (map[string]bypass, error) {
   781  	f, err := os.Open(path) //nolint:gosec // path はリテラル
   782  	if os.IsNotExist(err) {
   783  		return map[string]bypass{}, nil
   784  	}
   785  	if err != nil {
   786  		return nil, err
   787  	}
   788  	defer func() { _ = f.Close() }()
   789  
   790  	out := map[string]bypass{}
   791  	sc := bufio.NewScanner(f)
   792  	for lineNo := 1; sc.Scan(); lineNo++ {
   793  		line := strings.TrimSpace(sc.Text())
   794  		if line == "" || strings.HasPrefix(line, "#") {
   795  			continue
   796  		}
   797  		m := bypassLineRe.FindStringSubmatch(line)
   798  		if m == nil {
   799  			return nil, xerrors.Wrap(errBypassInvalidLine, fmt.Sprintf(
   800  				`%d 行目: %q（形式は "<key>@<version>" = { expires = YYYY-MM-DD, issue = <N>, reason = "..." }）`, lineNo, line,
   801  			))
   802  		}
   803  		if _, dup := out[m[1]]; dup {
   804  			return nil, xerrors.Wrap(errBypassDuplicateKey, fmt.Sprintf("%d 行目: %q", lineNo, m[1]))
   805  		}
   806  		expires, err := time.Parse(time.DateOnly, m[2])
   807  		if err != nil {
   808  			return nil, xerrors.Wrap(err, fmt.Sprintf("%d 行目の expires", lineNo))
   809  		}
   810  		issue, err := strconv.Atoi(m[3])
   811  		if err != nil {
   812  			return nil, xerrors.Wrap(err, fmt.Sprintf("%d 行目の issue", lineNo))
   813  		}
   814  		out[m[1]] = bypass{expires: expires, issue: issue, reason: m[4], line: lineNo}
   815  	}
   816  	return out, sc.Err()
   817  }

   819  // validateBypasses はバイパス自身の規約違反と、そのせいで無効になったキーを返す。期限切れを
   820  // 失敗にするのは、外したまま放置されたバイパスが恒久 allowlist と区別できなくなるため。上限を
   821  // 置くのは、期限を遠い未来へ置くだけで同じ状態を作れてしまうため。無効なバイパスは効力も失う。
   822  func validateBypasses(bypasses map[string]bypass, declared []tool, today time.Time) ([]violation, map[string]struct{}) {
   823  	inDeclarations := make(map[string]struct{}, len(declared))
   824  	for _, t := range declared {
   825  		inDeclarations[t.id()] = struct{}{}
   826  	}
   827  	limit := today.AddDate(0, maxBypassMonths, 0)
   828  	invalid := map[string]struct{}{}
   829  	var violations []violation
   830  
   831  	for _, key := range sortedKeys(bypasses) {
   832  		b := bypasses[key]
   833  		switch {
   834  		case b.expires.Before(today):
   835  			invalid[key] = struct{}{}
   836  			violations = append(violations, violation{file: bypassFile, msg: fmt.Sprintf(
   837  				"%s:%d %s の期限 %s が切れています。窓明けの版へ更新してエントリを消すか、期限を延ばす判断を #%d で記録してください",
   838  				bypassFile, b.line, key, b.expires.Format(time.DateOnly), b.issue,
   839  			)})
   840  		case b.expires.After(limit):
   841  			invalid[key] = struct{}{}
   842  			violations = append(violations, violation{file: bypassFile, msg: fmt.Sprintf(
   843  				"%s:%d %s の期限 %s が上限（%s から %d ヶ月 = %s）を越えています。恒久 allowlist にしないための上限です",
   844  				bypassFile, b.line, key, b.expires.Format(time.DateOnly),
   845  				today.Format(time.DateOnly), maxBypassMonths, limit.Format(time.DateOnly),
   846  			)})
   847  		default:
   848  			if _, ok := inDeclarations[key]; !ok {
   849  				violations = append(violations, violation{file: bypassFile, msg: fmt.Sprintf(
   850  					"%s:%d %s は %s / %s のどちらにも存在しません。不要になったエントリは消してください",
   851  					bypassFile, b.line, key, miseFile, pyRequirementsGlob,
   852  				)})
   853  			}
   854  		}
   855  	}
   856  	return violations, invalid
   857  }

   868  // classify は finding を「バイパス済み」「gate を落とすもの」「報告に留めるもの」へ振り分ける。
   869  func classify(sub string, findings []finding, bypasses map[string]bypass, invalid map[string]struct{}) ([]finding, []finding, []finding) {
   870  	var bypassed, blocking, reported []finding

   879  		}
   880  	}

   892  // report は結果を標準出力へ書き、終了コードを非ゼロにすべき件数を返す。audit は窓内の finding では
   893  // 落ちない。ただしバイパス自身の規約違反だけは audit でも失敗させる。期限切れの回収がスケジュール
   894  // 実行に懸かっているため。
   895  func report(
   896  	sub string, findings []finding, unresolved, skipped []tool,
   897  	policyViolations []violation, bypasses map[string]bypass, invalid map[string]struct{},
   898  	targets []tool, github bool,

   955  // fenceFor は text を包むのに足りるフェンスを返す。長さは text 中の最長バッククォート連 + 1。
   956  func fenceFor(text string) string {
   957  	longest, run := 0, 0
   958  	for _, r := range text {
   959  		if r == '`' {
   960  			run++
   961  			if run > longest {
   962  				longest = run
   963  			}
   964  			continue
   965  		}
   966  		run = 0
   967  	}
   968  	return strings.Repeat("`", max(minFenceLen, longest+1))
   969  }
