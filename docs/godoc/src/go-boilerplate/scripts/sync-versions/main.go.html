    32  var (
    33  	miseSectionRe = regexp.MustCompile(`^\[([^\]]+)\]`)
    34  	miseKeyRe     = regexp.MustCompile(`^([A-Za-z_][A-Za-z0-9_-]*)\s*=\s*"([^"]+)"`)
    35  	goModRe       = regexp.MustCompile(`(?m)^go \d+(?:\.\d+){0,2}$`)
    36  	// `(?m)^[^#\n]*?` でコメント行（先頭 `#`）を除外する。`\n` を除くのは、Go の文字クラスが
    37  	// 改行にもマッチするためで、`[^#]` のままだとマッチが行をまたいで広がる。間に `#` を
    38  	// 含まない 2 つの `FROM` があると、その間の行までマッチに取り込まれ、置換で消える。
    39  	// マッチ件数は変わらないので expectedCount のゲートも通過してしまう。
    40  	golangFromRe = regexp.MustCompile(`(?m)^[^#\n]*?(FROM\s+golang:)\d+(?:\.\d+){0,2}(-[\w.-]+)`)
    41  	nodeFromRe   = regexp.MustCompile(`(?m)^[^#\n]*?(FROM\s+node:)\d+(?:\.\d+){0,2}(-[\w.-]+)`)
    42  	pythonFromRe = regexp.MustCompile(`(?m)^[^#\n]*?(FROM\s+python:)\d+(?:\.\d+){0,2}(-[\w.-]+)`)
    43  	// バッククォートを capture に含めることで置換後も保持する。
    44  	golangImageRe    = regexp.MustCompile("(`golang:)" + `\d+(?:\.\d+){0,2}` + "(-[\\w.-]+`)")
    45  	nodeImageRe      = regexp.MustCompile("(`node:)" + `\d+(?:\.\d+){0,2}` + "(-[\\w.-]+`)")
    46  	pythonImageRe    = regexp.MustCompile("(`python:)" + `\d+(?:\.\d+){0,2}` + "(-[\\w.-]+`)")
    47  	miseDockerfileRe = regexp.MustCompile(`(MISE_VERSION=v)\d+(?:\.\d+){0,2}()`)
    48  	// docker-compose.yaml の otel-lgtm image タグ。suffix は空 capture でタグ末尾を保持する。
    49  	otelLgtmImageRe = regexp.MustCompile("(grafana/otel-lgtm:)" + `\d+(?:\.\d+){0,2}` + "()")

    89  // run は、mise.toml の version を各種ファイルへ反映します。getwd は走査の基点の取得手段です。
    90  func run(getwd func() (string, error)) error {
    91  	root, err := getwd()
    92  	if err != nil {
    93  		return xerrors.Wrap(err, "❌ getwd")
    94  	}
    95  
    96  	v, err := parseMiseTOML(filepath.Join(root, "mise.toml"))
    97  	if err != nil {
    98  		return xerrors.Wrap(err, "❌ mise.toml のパースに失敗")
    99  	}
   100  
   101  	printSource(v)
   102  
   103  	rules := buildRules(v)
   104  
   105  	if errs := validateRules(rules, root); len(errs) > 0 {
   106  		reportProblems("Validation errors（書き換えは行いません）", errs)
   107  
   108  		return errAborted
   109  	}
   110  
   111  	states, errs := computeChanges(rules, root)
   112  	if len(errs) > 0 {
   113  		reportProblems("Match-count errors（書き換えは行いません）", errs)
   114  
   115  		return errAborted
   116  	}
   117  
   118  	return writeChanges(states, root)
   119  }

   121  // parseMiseTOML は mise.toml の [tools] table 配下の go / node / python キーを抽出する。
   122  // 完全な TOML 仕様には準拠しない用途特化の最小 parser。
   123  func parseMiseTOML(path string) (runtimeVersions, error) {
   124  	var v runtimeVersions
   125  	f, err := os.Open(path) //nolint:gosec // path is constructed from cwd + literal filename
   126  	if err != nil {
   127  		return v, xerrors.Wrap(err, "open")
   128  	}
   129  	defer func() { _ = f.Close() }()
   130  
   131  	currentSection := ""
   132  	scanner := bufio.NewScanner(f)
   133  	for scanner.Scan() {
   134  		line := strings.TrimSpace(scanner.Text())
   135  		if line == "" || strings.HasPrefix(line, "#") {
   136  			continue
   137  		}
   138  		if m := miseSectionRe.FindStringSubmatch(line); m != nil {
   139  			currentSection = m[1]
   140  			continue
   141  		}
   142  		if m := miseKeyRe.FindStringSubmatch(line); m != nil {
   143  			applyMiseKV(&v, currentSection, m[1], m[2])
   144  		}
   145  	}
   146  	if err := scanner.Err(); err != nil {
   147  		return v, xerrors.Wrap(err, "scan")
   148  	}
   149  	return v, nil
   150  }

   152  // applyMiseKV は section / key / value を runtimeVersions の該当フィールドへ反映する。
   153  // ルートの min_version、[env] の OTEL_LGTM_VERSION、[tools] の go/node/python のみを対象とする。
   154  func applyMiseKV(v *runtimeVersions, section, key, val string) {
   155  	switch section {
   156  	case "":
   157  		if key == "min_version" {
   158  			v.Mise = val
   159  		}
   160  	case "env":
   161  		if key == "OTEL_LGTM_VERSION" {
   162  			v.OtelLgtm = val
   163  		}
   164  	case "tools":
   165  		switch key {
   166  		case "go":
   167  			v.Go = val
   168  		case "node":
   169  			v.Node = val
   170  		case "python":
   171  			v.Python = val
   172  		}
   173  	}
   174  }

   269  // fromReplacer は 2 capture group regex の version 部分だけを置換し、prefix / suffix を保持する replace 関数を返す。
   270  func fromReplacer(re *regexp.Regexp, version string) func(string) string {
   271  	return func(match string) string {
   272  		sub := re.FindStringSubmatch(match)
   273  		if len(sub) < fromRegexpGroups {
   274  			return match
   275  		}
   276  		return sub[1] + version + sub[2]
   277  	}
   278  }

   280  // validateRules は version 未設定の rule とファイル不在の rule をエラーリストで返す。
   281  func validateRules(rules []rule, root string) []string {
   282  	var errs []string
   283  	for _, r := range rules {
   284  		if r.version == "" {
   285  			errs = append(errs, r.label+": mise.toml に対応する version が未設定")
   286  		}
   287  	}
   288  	for _, r := range rules {
   289  		if _, err := os.Stat(filepath.Join(root, r.file)); err != nil {
   290  			errs = append(errs, fmt.Sprintf(
   291  				"%s: %s が見つかりません", r.label, r.file,
   292  			))
   293  		}
   294  	}
   295  	return errs
   296  }

   298  // computeChanges は各 file の最終内容を計算する（書き出しは行わない）。期待マッチ数未満の rule があればエラーを返す。
   299  func computeChanges(rules []rule, root string) (map[string]*fileState, []string) {
   300  	states := map[string]*fileState{}
   301  	var errs []string
   302  	for _, r := range rules {
   303  		st, ok := states[r.file]
   304  		if !ok {
   305  			data, err := os.ReadFile(filepath.Join(root, r.file)) //nolint:gosec // path from cwd + rule.file
   306  			if err != nil {
   307  				errs = append(errs, fmt.Sprintf("%s: read %s: %v", r.label, r.file, err))
   308  				continue
   309  			}
   310  			st = &fileState{original: string(data), current: string(data)}
   311  			states[r.file] = st
   312  		}
   313  		matches := r.re.FindAllStringIndex(st.current, -1)
   314  		if len(matches) < r.expectedCount {
   315  			errs = append(errs, fmt.Sprintf(
   316  				"%s: マッチ件数 %d が期待値 %d 未満（regex: %s）",
   317  				r.label, len(matches), r.expectedCount, r.re,
   318  			))
   319  			continue
   320  		}
   321  		st.current = r.re.ReplaceAllStringFunc(st.current, r.replace)
   322  		st.applied = append(st.applied, fmt.Sprintf("%s x%d", r.label, len(matches)))
   323  	}
   324  	return states, errs
   325  }

   327  // writeChanges は各 file を 1 回ずつ書き出す。変更が無いファイルは skip する。
   328  func writeChanges(states map[string]*fileState, root string) error {
   329  	anyChange := false
   330  	for file, st := range states {
   331  		if st.original == st.current {
   332  			log.Printf("No change: %s", file)
   333  			continue
   334  		}
   335  		if err := os.WriteFile(
   336  			filepath.Join(root, file), []byte(st.current), filePerm,
   337  		); err != nil {
   338  			return xerrors.Wrap(err, "write "+file)
   339  		}
   340  		log.Printf("Updated: %s [%s]", file, strings.Join(st.applied, ", "))
   341  		anyChange = true
   342  	}
   343  	if anyChange {
   344  		log.Println("Done.")
   345  	} else {
   346  		log.Println("Done (no changes).")
   347  	}
   348  	return nil
   349  }
