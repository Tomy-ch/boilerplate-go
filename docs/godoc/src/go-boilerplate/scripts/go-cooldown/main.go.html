   116  // run は go.mod の require を cooldown に照らして報告します。gate で違反が残った場合はエラーを
   117  // 返し、終了コードへの変換は呼び出し側に委ねます。
   118  // now は窓と期限の基準時刻で、差し替えられるよう引数で受けます。
   119  func run(args []string, now time.Time) error {
   120  	sub, opt, err := parseArgs(args)
   121  	if err != nil {
   122  		// ヘルプ要求は失敗ではないので 0 で終える。usage は flag が既に出力している。
   123  		if xerrors.Is(err, flag.ErrHelp) {
   124  			return nil
   125  		}
   126  
   127  		return err
   128  	}
   129  
   130  	// 期限の比較は暦日で行う。時刻を残すと 3 ヶ月の境界が実行時刻とタイムゾーンで動く。
   131  	today := time.Date(now.Year(), now.Month(), now.Day(), 0, 0, 0, 0, time.UTC)
   132  
   133  	goMod, err := readWorkTreeGoMod()
   134  	if err != nil {
   135  		return xerrors.Wrap(err, "❌ go.mod")
   136  	}
   137  
   138  	current, err := parseGoMod(goMod)
   139  	if err != nil {
   140  		return xerrors.Wrap(err, "❌ go.mod")
   141  	}
   142  
   143  	bypasses, err := readBypasses(bypassFile)
   144  	if err != nil {
   145  		return xerrors.Wrap(err, "❌ "+bypassFile)
   146  	}
   147  
   148  	policyViolations, invalidBypasses := validateBypasses(bypasses, current, today)
   149  
   150  	targets := current
   151  	if sub == "gate" {
   152  		targets, err = added(opt.base, current)
   153  		if err != nil {
   154  			return xerrors.Wrap(err, "❌ base との差分")
   155  		}
   156  	}
   157  
   158  	findings, unresolved, err := inspect(context.Background(), targets, opt.window, now)
   159  	if err != nil {
   160  		return xerrors.Wrap(err, "❌ 公開時刻の取得")
   161  	}
   162  
   163  	blocking := report(sub, findings, unresolved, policyViolations, bypasses, invalidBypasses, targets, opt.window, opt.github)
   164  
   165  	if opt.summaryOut != "" {
   166  		body := summary(sub, findings, unresolved, policyViolations, bypasses, invalidBypasses, targets, opt.window)
   167  		if writeErr := os.WriteFile(opt.summaryOut, []byte(body), summaryPerm); writeErr != nil {
   168  			return xerrors.Wrap(writeErr, "❌ write summary")
   169  		}
   170  	}
   171  	if out := os.Getenv("GITHUB_OUTPUT"); out != "" {
   172  		if writeErr := appendOutput(out, len(findings), len(targets), blocking); writeErr != nil {
   173  			return xerrors.Wrap(writeErr, "❌ write GITHUB_OUTPUT")
   174  		}
   175  	}
   176  
   177  	if blocking > 0 {
   178  		return xerrors.Wrap(errBlocking, fmt.Sprintf("❌ go-cooldown %s: %d 件の違反が残っています", sub, blocking))
   179  	}
   180  
   181  	return nil
   182  }

   184  // parseArgs はサブコマンドとフラグを解釈します。ヘルプ要求は flag.ErrHelp を含むエラーとして
   185  // 返し、失敗として扱うかどうかの判断は呼び出し側に委ねます。
   186  func parseArgs(args []string) (string, options, error) {
   187  	if len(args) == 0 {
   188  		return "", options{}, xerrors.Wrap(errUsage,
   189  			"❌ usage: go-cooldown <gate|audit> [--base=<git-ref>] [--window-days=N] [--summary-out=<path>] [--github]")
   190  	}
   191  
   192  	sub := args[0]
   193  	if sub != "gate" && sub != "audit" {
   194  		return "", options{}, xerrors.Wrap(errUsage, fmt.Sprintf("❌ 未知のサブコマンド %q（gate | audit）", sub))
   195  	}
   196  
   197  	fs := flag.NewFlagSet(sub, flag.ContinueOnError)
   198  	base := fs.String("base", "", "gate: この git ref からの差分で追加 / 更新された require だけを対象にする")
   199  	window := fs.Int("window-days", defaultWindowDays, "公開からこの日数に満たないバージョンを窓内とみなす")
   200  	summaryOut := fs.String("summary-out", "", "Markdown サマリの出力先")
   201  	github := fs.Bool("github", false, "GitHub Actions のアノテーションを出力する")
   202  	if err := fs.Parse(args[1:]); err != nil {
   203  		return "", options{}, xerrors.Wrap(err, "❌ フラグを解釈できません")
   204  	}
   205  
   206  	// gate は差分を取れないと何も検査しないまま通るため、比較対象の不在を先に落とす。
   207  	if sub == "gate" && *base == "" {
   208  		return "", options{}, xerrors.Wrap(errUsage, "❌ gate には --base が要ります（比較対象が無いと差分を取れません）")
   209  	}
   210  
   211  	return sub, options{base: *base, summaryOut: *summaryOut, window: *window, github: *github}, nil
   212  }

   225  // parseGoMod は require ブロックと単一行 require の両方から依存を読む。go.mod は direct と
   226  // indirect で require ブロックが分かれるのが通例だが、`go mod tidy` はどちらの形も書き得る。
   227  func parseGoMod(content []byte) ([]requirement, error) {
   228  	var reqs []requirement
   229  	inBlock := false
   230  	sc := bufio.NewScanner(strings.NewReader(string(content)))
   231  	for sc.Scan() {
   232  		line := sc.Text()
   233  		trimmed := strings.TrimSpace(line)
   234  		if i := strings.Index(trimmed, "//"); i >= 0 && !strings.HasPrefix(strings.TrimSpace(trimmed[i:]), "// indirect") {
   235  			trimmed = strings.TrimSpace(trimmed[:i])
   236  		}
   237  		switch {
   238  		case trimmed == "require (":
   239  			inBlock = true
   240  		case inBlock && trimmed == ")":
   241  			inBlock = false
   242  		case inBlock:
   243  			if m := requireLineRe.FindStringSubmatch(trimmed); m != nil {
   244  				reqs = append(reqs, requirement{module: m[1], version: m[2], indirect: m[3] != ""})
   245  			}
   246  		case strings.HasPrefix(trimmed, "require "):
   247  			if m := requireLineRe.FindStringSubmatch(strings.TrimPrefix(trimmed, "require ")); m != nil {
   248  				reqs = append(reqs, requirement{module: m[1], version: m[2], indirect: m[3] != ""})
   249  			}
   250  		}
   251  	}
   252  	return reqs, sc.Err()
   253  }

   255  // added は base 時点の go.mod に無かった (module, version) の組を返す。バージョンを上げた
   256  // モジュールも新しい組として現れるため、追加と更新をひとつの判定で拾える。
   257  func added(base string, current []requirement) ([]requirement, error) {
   258  	ctx, cancel := context.WithTimeout(context.Background(), gitTimeout)
   259  	defer cancel()
   260  
   261  	out, err := exec.CommandContext(ctx, "git", "show", base+":go.mod").Output() //nolint:gosec // base は呼び出し側が与える git ref
   262  	if err != nil {
   263  		return nil, xerrors.Wrap(err, fmt.Sprintf("git show %s:go.mod（fetch-depth: 0 で base を取得できていない可能性があります）", base))
   264  	}
   265  	before, err := parseGoMod(out)
   266  	if err != nil {
   267  		return nil, xerrors.Wrap(err, "base の go.mod")
   268  	}
   269  	return diffAdded(before, current), nil
   270  }

   272  // diffAdded は before に無い (module, version) の組を返す。
   273  func diffAdded(before, current []requirement) []requirement {
   274  	known := make(map[string]struct{}, len(before))
   275  	for _, r := range before {
   276  		known[r.key()] = struct{}{}
   277  	}
   278  	var fresh []requirement
   279  	for _, r := range current {
   280  		if _, ok := known[r.key()]; !ok {
   281  			fresh = append(fresh, r)
   282  		}
   283  	}
   284  	return fresh
   285  }

   287  // escapePath は module path を proxy のパス表記へ変換する。proxy は大文字を `!` + 小文字で
   288  // 受けるため、未エスケープのまま問い合わせると 404 になり「存在しないバージョン」に化ける。
   289  func escapePath(path string) string {
   290  	var b strings.Builder
   291  	for _, r := range path {
   292  		if r >= 'A' && r <= 'Z' {
   293  			b.WriteByte('!')
   294  			b.WriteRune(r + ('a' - 'A'))
   295  			continue
   296  		}
   297  		b.WriteRune(r)
   298  	}
   299  	return b.String()
   300  }

   302  // publishedAt は module proxy の .info からそのバージョンの公開時刻を取る。GOPROXY プロトコルの
   303  // 一部なので追加の依存を持たずに済む。
   304  func publishedAt(ctx context.Context, client *http.Client, r requirement) (time.Time, error) {
   305  	url := proxyBase + escapePath(r.module) + "/@v/" + r.version + ".info"
   306  	req, err := http.NewRequestWithContext(ctx, http.MethodGet, url, nil)
   307  	if err != nil {
   308  		return time.Time{}, xerrors.Wrap(err, "new request")
   309  	}
   310  	resp, err := client.Do(req)
   311  	if err != nil {
   312  		return time.Time{}, xerrors.Wrap(err, "fetch "+url)
   313  	}
   314  	defer func() { _ = resp.Body.Close() }()
   315  
   316  	if resp.StatusCode == http.StatusNotFound || resp.StatusCode == http.StatusGone {
   317  		return time.Time{}, xerrors.Wrap(errProxyNotFound, r.key())
   318  	}
   319  	if resp.StatusCode != http.StatusOK {
   320  		return time.Time{}, xerrors.Wrap(errProxyStatus, fmt.Sprintf("%s: %d", r.key(), resp.StatusCode))
   321  	}
   322  
   323  	var info struct {
   324  		Time time.Time `json:"Time"` //nolint:tagliatelle // module proxy の応答フィールド名
   325  	}
   326  	if err := json.NewDecoder(resp.Body).Decode(&info); err != nil {
   327  		return time.Time{}, xerrors.Wrap(err, "decode "+url)
   328  	}
   329  	return info.Time, nil
   330  }

   332  // inspect は対象モジュールの公開時刻を引き、窓内のものと取得できなかったものを返す。
   333  func inspect(ctx context.Context, targets []requirement, window int, now time.Time) ([]finding, []requirement, error) {
   334  	client := &http.Client{Timeout: fetchTimeout}
   335  
   336  	var (
   337  		mu         sync.Mutex
   338  		findings   []finding
   339  		unresolved []requirement
   340  		fatal      error
   341  		wg         sync.WaitGroup
   342  	)
   343  	sem := make(chan struct{}, fetchWorkers)
   344  
   345  	for _, r := range targets {
   346  		wg.Add(1)
   347  		go func(r requirement) {
   348  			defer wg.Done()
   349  			sem <- struct{}{}
   350  			defer func() { <-sem }()
   351  
   352  			at, err := publishedAt(ctx, client, r)
   353  			mu.Lock()
   354  			defer mu.Unlock()
   355  			switch {
   356  			case xerrors.Is(err, errProxyNotFound):
   357  				unresolved = append(unresolved, r)
   358  			case err != nil:
   359  				if fatal == nil {
   360  					fatal = err
   361  				}
   362  			default:
   363  				if age := int(now.Sub(at).Hours() / hoursPerDay); age < window {
   364  					findings = append(findings, finding{req: r, published: at, ageDays: age})
   365  				}
   366  			}
   367  		}(r)
   368  	}
   369  	wg.Wait()
   370  
   371  	if fatal != nil {
   372  		return nil, nil, fatal
   373  	}
   374  	sort.Slice(findings, func(i, j int) bool { return findings[i].ageDays < findings[j].ageDays })
   375  	sort.Slice(unresolved, func(i, j int) bool { return unresolved[i].key() < unresolved[j].key() })
   376  	return findings, unresolved, nil
   377  }

   379  // readBypasses はバイパス lockfile を module@version→bypass として読む。空行とコメント行以外で
   380  // 解釈できない行、および既出キーの再定義はエラーにする。読み飛ばしや後勝ちの上書きは、その
   381  // エントリが「存在しない」あるいは「行順で決まる」状態を警告なく作る。
   382  func readBypasses(path string) (map[string]bypass, error) {
   383  	f, err := os.Open(path) //nolint:gosec // path はリテラル
   384  	if os.IsNotExist(err) {
   385  		return map[string]bypass{}, nil
   386  	}
   387  	if err != nil {
   388  		return nil, err
   389  	}
   390  	defer func() { _ = f.Close() }()
   391  
   392  	out := map[string]bypass{}
   393  	sc := bufio.NewScanner(f)
   394  	for lineNo := 1; sc.Scan(); lineNo++ {
   395  		line := strings.TrimSpace(sc.Text())
   396  		if line == "" || strings.HasPrefix(line, "#") {
   397  			continue
   398  		}
   399  		m := bypassLineRe.FindStringSubmatch(line)
   400  		if m == nil {
   401  			return nil, xerrors.Wrap(errBypassInvalidLine, fmt.Sprintf(
   402  				`%d 行目: %q（形式は "<module>@<version>" = { expires = YYYY-MM-DD, issue = <N>, reason = "..." }）`, lineNo, line,
   403  			))
   404  		}
   405  		if _, dup := out[m[1]]; dup {
   406  			return nil, xerrors.Wrap(errBypassDuplicateKey, fmt.Sprintf("%d 行目: %q", lineNo, m[1]))
   407  		}
   408  		expires, err := time.Parse(time.DateOnly, m[2])
   409  		if err != nil {
   410  			return nil, xerrors.Wrap(err, fmt.Sprintf("%d 行目の expires", lineNo))
   411  		}
   412  		issue, err := strconv.Atoi(m[3])
   413  		if err != nil {
   414  			return nil, xerrors.Wrap(err, fmt.Sprintf("%d 行目の issue", lineNo))
   415  		}
   416  		out[m[1]] = bypass{expires: expires, issue: issue, reason: m[4], line: lineNo}
   417  	}
   418  	return out, sc.Err()
   419  }

   421  // validateBypasses はバイパス自身の規約違反と、そのせいで無効になったキーを返す。期限切れを
   422  // 失敗にするのは、外したまま放置されたバイパスが恒久 allowlist と区別できなくなるため。上限を
   423  // 置くのは、期限を遠い未来へ置くだけで同じ状態を作れてしまうため。
   424  //
   425  // 無効なバイパスは効力も失う。実行はどのみち落ちるので素通ししても穴にはならないが、期限切れの
   426  // エントリが「通します」と報告する状態は、そのモジュールが検査を通ったのか外されただけなのかを
   427  // 読者が区別できなくする。
   428  func validateBypasses(bypasses map[string]bypass, current []requirement, today time.Time) ([]string, map[string]struct{}) {
   429  	inGoMod := make(map[string]struct{}, len(current))
   430  	for _, r := range current {
   431  		inGoMod[r.key()] = struct{}{}
   432  	}
   433  	limit := today.AddDate(0, maxBypassMonths, 0)
   434  	invalid := map[string]struct{}{}
   435  	var violations []string
   436  
   437  	for _, key := range sortedKeys(bypasses) {
   438  		b := bypasses[key]
   439  		switch {
   440  		case b.expires.Before(today):
   441  			invalid[key] = struct{}{}
   442  			violations = append(violations, fmt.Sprintf(

   573  // fenceFor は text を包むのに足りるフェンスを返す。長さは text 中の最長バッククォート連 + 1 で、
   574  // 最低 3。text 側がフェンスを閉じられないことが、この計算の目的である。
   575  func fenceFor(text string) string {
   576  	longest, run := 0, 0
   577  	for _, r := range text {
   578  		if r == '`' {
   579  			run++
   580  			if run > longest {
   581  				longest = run
   582  			}
   583  			continue
   584  		}
   585  		run = 0
   586  	}
   587  	return strings.Repeat("`", max(minFenceLen, longest+1))
   588  }

   590  // fenced は見出しと、フェンスで包んだ本体を書く。module path は go.mod 由来で、この検査が
   591  // 動く時点では pull request が中身を決めている。見出しは読ませたいのでテンプレート側に残し、
   592  // 値の側だけをフェンスへ入れる。
   593  func fenced(b *strings.Builder, heading string, lines []string) {
   594  	body := strings.Join(lines, "\n")
   595  	fence := fenceFor(body)
   596  	fmt.Fprintf(b, "## %s (%d)\n\n%stext\n%s\n%s\n\n", heading, len(lines), fence, body, fence)
   597  }

   600  func summary(
   601  	sub string, findings []finding, unresolved []requirement,
   602  	policyViolations []string, bypasses map[string]bypass, invalid map[string]struct{},

   627  		lines := make([]string, 0, len(unresolved))

   630  		}

   635  	if b.Len() == 0 {
   636  		b.WriteString("違反はありません。\n")
   637  	}

   639  }
