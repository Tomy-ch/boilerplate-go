    44  //
    45  // サービスが 0 件のときエラーにするのは、この検査が「何も見ていない」状態を成功と

    47  func loadServices(path string) (map[string]service, error) {

    49  	if err != nil {
    50  		return nil, xerrors.Wrap(err, "failed to read "+path)
    51  	}
    52  
    53  	var parsed composeFile
    54  	if err := yaml.Unmarshal(raw, &parsed); err != nil {

    65  // loadAppServices は、compose.mk の APP_SERVICES 宣言から起動対象サービス名を読み出します。
    66  //
    67  // 対象の一覧をこのツールに持たないのは、両方に置くと片方だけを直したときに黙ってずれるためです。
    68  // 宣言が見つからなければエラーにします。空の一覧で走らせると、検査対象ゼロのまま成功します。
    69  func loadAppServices(path string) ([]string, error) {
    70  	raw, err := os.ReadFile(path) //nolint:gosec // path は -makefile フラグ。既定はリポジトリ内のリテラル
    71  	if err != nil {
    72  		return nil, xerrors.Wrap(err, "failed to read "+path)
    73  	}
    74  
    75  	for line := range strings.Lines(string(raw)) {
    76  		value, ok := appServicesValue(line)
    77  		if !ok {
    78  			continue
    79  		}
    80  
    81  		names := strings.Fields(value)
    82  		if len(names) == 0 {
    83  			return nil, xerrors.Wrap(errNoAppServices, path)
    84  		}
    85  
    86  		return names, nil
    87  	}
    88  
    89  	return nil, xerrors.Wrap(errNoAppServices, path)
    90  }

    92  // appServicesValue は、行が APP_SERVICES の代入なら右辺を返します。
    93  // `?=` と `=` と `:=` を受けるのは、make のどの代入でも宣言としては同じ意味を持つためです。
    94  func appServicesValue(line string) (string, bool) {
    95  	trimmed := strings.TrimSpace(line)
    96  
    97  	rest, found := strings.CutPrefix(trimmed, appServicesKey)
    98  	if !found {
    99  		return "", false
   100  	}
   101  
   102  	rest = strings.TrimLeft(rest, " \t")
   103  	for _, op := range []string{"?=", ":=", "="} {
   104  		if value, ok := strings.CutPrefix(rest, op); ok {
   105  			return strings.TrimSpace(value), true
   106  		}
   107  	}
   108  
   109  	return "", false
   110  }

   112  // checkHealthcheck は、起動対象サービスの healthcheck 宣言を検査し、違反の報告文を返します
   113  // （違反が無ければ空文字）。
   114  //
   115  // 起動対象に挙がっているサービスが compose に無い場合は違反ではなくエラーにします。名前が
   116  // ずれているのに「healthcheck を持つサービスは全部持っていた」と報告するのは、見ていないのと同じです。
   117  func checkHealthcheck(appServices []string, services map[string]service) (string, error) {
   118  	missing := make([]string, 0, len(appServices))
   119  
   120  	for _, name := range appServices {
   121  		svc, ok := services[name]
   122  		if !ok {
   123  			return "", xerrors.Wrap(errUnknownService, name)
   124  		}
   125  
   126  		if svc.Healthcheck == nil || svc.Healthcheck.Test.IsZero() {
   127  			missing = append(missing, name)
   128  		}
   129  	}
   130  
   131  	if len(missing) == 0 {
   132  		return "", nil
   133  	}
   134  
   135  	sort.Strings(missing)
   136  
   137  	return fmt.Sprintf(
   138  		"app 層のサービスに healthcheck が宣言されていません: %s\n"+
   139  			"  make serve は healthcheck を --wait で待って起動の成否を判定します"+
   140  			"（.makefiles/app/server.mk の app-up）。宣言が無いサービスは、"+
   141  			"コンテナが起動したことだけで成功と見なされます。",
   142  		strings.Join(missing, " "),
   143  	), nil
   144  }
