     1  // Package main は、カバレッジプロファイルの総カバレッジがしきい値以上かを検査するツール。
     2  //
     3  //	-profile:   `go test -coverprofile` が出力したプロファイル（既定 coverage.out）
     4  //	-threshold: 下回ってはならない総カバレッジのパーセント（既定 90）
     5  //	-warn:      下限割れを失敗させず警告に留める（scripts/ 配下の計測で使う）
     6  //	-github:    警告を GitHub Actions の ::warning:: アノテーションで出す
     7  //
     8  // 守る対象の違う 2 つの計測がこのツールを共有する。boilerplate 本体（internal / pkg）は
     9  // 下限割れを失敗にするが、scripts/ 配下の開発ツールは警告に留める。後者の劣化が前者の
    10  // ゲートを止めてしまうと、出荷物と無関係な理由でマージがブロックされるためである。
    11  //
    12  // 総カバレッジは `go tool cover -func` の `total:` 行から取る。プロファイルの
    13  // 集計規則（-covermode ごとの重み付けなど）を写し取らずに済ませるためで、
    14  // ここが持つのは「取り出す」ことと「比較する」ことだけ。
    15  //
    16  // このツールは CI のカバレッジゲートから呼ばれる。壊れ方が「何も検査しなくなる」
    17  // 方向に出るため、判定ロジックはシェルの中ではなくテストの当たる Go 側に置く。

    53  // run は、プロファイルの総カバレッジをしきい値に照らして報告します。
    54  // total は総カバレッジの取得手段で、差し替えられるよう引数で受けます。
    55  func run(args []string, total func(profile string) (float64, error)) error {
    56  	fs := flag.NewFlagSet("cover-gate", flag.ContinueOnError)
    57  	profile := fs.String("profile", "coverage.out", "検査するカバレッジプロファイル")
    58  	threshold := fs.Int("threshold", defaultThreshold, "総カバレッジの下限（パーセント）")
    59  	warn := fs.Bool("warn", false, "下限を割っても失敗させず警告に留める")
    60  	github := fs.Bool("github", false, "GitHub Actions の ::warning:: アノテーションを出力する")
    61  
    62  	if err := fs.Parse(args); err != nil {
    63  		// ヘルプ要求は失敗ではないので 0 で終える。usage は flag が既に出力している。
    64  		if xerrors.Is(err, flag.ErrHelp) {
    65  			return nil
    66  		}
    67  
    68  		return xerrors.Wrap(err, "failed to parse flags")
    69  	}
    70  
    71  	if _, err := os.Stat(*profile); err != nil {
    72  		return xerrors.Wrap(err, "❌ "+*profile+" がありません（先に make go-test-cover-ci を実行）")
    73  	}
    74  
    75  	value, err := total(*profile)
    76  	if err != nil {
    77  		return xerrors.Wrap(err, "❌ 総カバレッジを取得できません")
    78  	}
    79  
    80  	message, ok := judge(value, *threshold)
    81  	if ok {
    82  		log.Print(message)
    83  
    84  		return nil
    85  	}
    86  
    87  	// 下限割れを失敗にするか警告に留めるかの理由は package doc を参照。ここはフラグの指示に従う。
    88  	if *warn {
    89  		log.Print(annotate(message, *github))
    90  
    91  		return nil
    92  	}
    93  
    94  	return xerrors.Wrap(errBelowThreshold, message)
    95  }

    97  // coverTotal は、`go tool cover -func` を起動して総カバレッジを取り出します。
    98  func coverTotal(profile string) (float64, error) {
    99  	cover := exec.CommandContext(context.Background(), "go", "tool", "cover", "-func="+profile) //nolint:gosec // 引数は profile のみで、コマンド自体は固定
   100  
   101  	out, err := cover.Output()
   102  	if err != nil {
   103  		return 0, xerrors.Wrap(err, "go tool cover")
   104  	}
   105  
   106  	return parseTotal(string(out))
   107  }

   109  // annotate は、GitHub Actions で読ませる場合に警告アノテーションの接頭辞を付けます。
   110  // 付けないときは元の文言をそのまま返します。
   111  func annotate(message string, github bool) string {
   112  	if !github {
   113  		return message
   114  	}
   115  
   116  	return "::warning::" + message
   117  }

   119  // parseTotal は、`go tool cover -func` の出力から総カバレッジのパーセント値を取り出します。
   120  // `total:` で始まる行の最終フィールド（`87.5%` 形式）を読み、総カバレッジ行が無い場合や
   121  // パーセント表記が数値でない場合はエラーを返します。
   122  func parseTotal(coverOutput string) (float64, error) {
   123  	for line := range strings.Lines(coverOutput) {
   124  		if !strings.HasPrefix(line, "total:") {
   125  			continue
   126  		}
   127  
   128  		fields := strings.Fields(line)
   129  
   130  		last := fields[len(fields)-1]
   131  		if !strings.HasSuffix(last, "%") {
   132  			return 0, xerrors.Wrap(errNoTotalLine, "total is not a percentage: "+last)
   133  		}
   134  
   135  		total, err := strconv.ParseFloat(strings.TrimSuffix(last, "%"), 64)
   136  		if err != nil {
   137  			return 0, xerrors.Wrap(err, "invalid total coverage: "+last)
   138  		}
   139  
   140  		return total, nil
   141  	}
   142  
   143  	return 0, errNoTotalLine
   144  }

   146  // judge は、総カバレッジがしきい値以上かを判定し、表示する報告文と合否を返します。
   147  // しきい値ちょうどは合格です。
   148  func judge(total float64, threshold int) (string, bool) {
   149  	if total < float64(threshold) {
   150  		return fmt.Sprintf("❌ 総カバレッジ %.1f%% がしきい値 %d%% を下回っています", total, threshold), false
   151  	}
   152  
   153  	return fmt.Sprintf("✅ 総カバレッジ %.1f%% (しきい値 %d%%)", total, threshold), true
   154  }
