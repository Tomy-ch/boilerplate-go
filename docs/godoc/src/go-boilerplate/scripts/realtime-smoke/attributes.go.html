    61  // setAndReadBack は、queue に attrs を設定し、1 つずつ読み戻して同じ値かを確かめます。
    62  // 受理されたのに読み戻せない属性は silent drop として非互換にします。
    63  func (s *pubSubSmoke) setAndReadBack(ctx context.Context, url string, attrs map[sqstypes.QueueAttributeName]string) error {
    64  	set := make(map[string]string, len(attrs))
    65  	for name, value := range attrs {
    66  		set[string(name)] = value
    67  	}
    68  
    69  	if _, err := s.sqs.SetQueueAttributes(ctx, &sqs.SetQueueAttributesInput{QueueUrl: aws.String(url), Attributes: set}); err != nil {
    70  		return err
    71  	}
    72  
    73  	for name, want := range attrs {
    74  		got, err := s.queueAttribute(ctx, url, name)
    75  		if err != nil {
    76  			return err
    77  		}
    78  
    79  		if got == "" {
    80  			return incompatible(string(name) + " は受理されたが読み戻せない（silent drop）")
    81  		}
    82  
    83  		if got != want && !sameJSON(want, got) {
    84  			return incompatible("読み戻した " + string(name) + " が設定値と異なる: " + got)
    85  		}
    86  	}
    87  
    88  	return nil
    89  }

   183  // listSubscriptions は、topic の subscription 一覧から先頭 queue の ARN を endpoint に持つものを引き、
   184  // Subscribe が返した ARN と一致することを確かめます（orphan cleanup が lease だけから resource を辿る前提）。
   185  func (s *pubSubSmoke) listSubscriptions(ctx context.Context) (string, error) {
   186  	if len(s.queueArns) == 0 || len(s.subArns) == 0 {
   187  		return "", incompatible("先行検査 G3 / G5 で queue ARN と subscription を得られていない")
   188  	}
   189  
   190  	var found *snstypes.Subscription
   191  
   192  	p := sns.NewListSubscriptionsByTopicPaginator(s.sns, &sns.ListSubscriptionsByTopicInput{TopicArn: aws.String(s.topicArn)})
   193  	for p.HasMorePages() && found == nil {
   194  		page, err := p.NextPage(ctx)
   195  		if err != nil {
   196  			return "", err
   197  		}
   198  
   199  		for i := range page.Subscriptions {
   200  			if aws.ToString(page.Subscriptions[i].Endpoint) == s.queueArns[0] {
   201  				found = &page.Subscriptions[i]
   202  
   203  				break
   204  			}
   205  		}
   206  	}
   207  
   208  	if found == nil {
   209  		return "", incompatible("queue ARN " + s.queueArns[0] + " を endpoint に持つ subscription が一覧に無い")
   210  	}
   211  
   212  	if aws.ToString(found.SubscriptionArn) != s.subArns[0] {
   213  		return "", incompatible("一覧の SubscriptionArn が Subscribe の戻り値と異なる: " + aws.ToString(found.SubscriptionArn))
   214  	}
   215  
   216  	return "queue ARN から subscription を引けた（" + s.subArns[0] + "）", nil
   217  }

   219  // notificationTypes は、wakeup と失効通知を type 属性だけ変えて publish し、受信側で属性から振り分けられることを確かめます。
   220  func (s *pubSubSmoke) notificationTypes(ctx context.Context) (string, error) {
   221  	url, err := s.firstQueue()
   222  	if err != nil {
   223  		return "", err
   224  	}
   225  
   226  	wakeup, err := json.Marshal(notification{EventID: "evt-" + s.names.runID + "-2", StreamID: streamID, Sequence: "2"})
   227  	if err != nil {
   228  		return "", xerrors.Wrap(err, "marshal wakeup")
   229  	}
   230  
   231  	revoked, err := json.Marshal(revocation{Subject: "subject-" + s.names.runID, Destination: streamID})
   232  	if err != nil {
   233  		return "", xerrors.Wrap(err, "marshal revocation")
   234  	}
   235  
   236  	for kind, body := range map[string][]byte{typeWakeup: wakeup, typeRevocation: revoked} {
   237  		if _, err := s.sns.Publish(ctx, &sns.PublishInput{
   238  			TopicArn: aws.String(s.topicArn),
   239  			Message:  aws.String(string(body)),
   240  			MessageAttributes: map[string]snstypes.MessageAttributeValue{
   241  				attrNotificationType: {DataType: aws.String("String"), StringValue: aws.String(kind)},
   242  			},
   243  		}); err != nil {
   244  			return "", err
   245  		}
   246  	}
   247  
   248  	msgs, err := s.receive(ctx, url)
   249  	if err != nil {
   250  		return "", err
   251  	}
   252  
   253  	seen, err := dispatchByType(msgs)
   254  	if err != nil {
   255  		return "", err
   256  	}
   257  
   258  	for _, m := range msgs {
   259  		if _, err := s.sqs.DeleteMessage(ctx, &sqs.DeleteMessageInput{QueueUrl: aws.String(url), ReceiptHandle: m.ReceiptHandle}); err != nil {
   260  			return "", err
   261  		}
   262  	}
   263  
   264  	return strconv.Itoa(len(msgs)) + " 件を type 属性で振り分けた（" + seen + "）", nil
   265  }

   267  // receiveAfterAttributes は、直前の属性設定の後に 1 件 publish し、先頭 queue に 1 件だけ届いて
   268  // それを削除できることを確かめます。属性の受理と配送・削除の振る舞いが連動する emulator があるためです。
   269  func (s *pubSubSmoke) receiveAfterAttributes(ctx context.Context) (string, error) {
   270  	url, err := s.firstQueue()
   271  	if err != nil {
   272  		return "", err
   273  	}
   274  
   275  	body, err := json.Marshal(notification{EventID: "evt-" + s.names.runID + "-3", StreamID: streamID, Sequence: "3"})
   276  	if err != nil {
   277  		return "", xerrors.Wrap(err, "marshal wakeup")
   278  	}
   279  
   280  	if _, err := s.sns.Publish(ctx, &sns.PublishInput{
   281  		TopicArn: aws.String(s.topicArn),
   282  		Message:  aws.String(string(body)),
   283  		MessageAttributes: map[string]snstypes.MessageAttributeValue{
   284  			attrNotificationType: {DataType: aws.String("String"), StringValue: aws.String(typeWakeup)},
   285  		},
   286  	}); err != nil {
   287  		return "", err
   288  	}
   289  
   290  	msgs, err := s.receive(ctx, url)
   291  	if err != nil {
   292  		return "", err
   293  	}
   294  
   295  	// 属性設定後に受信と削除の対応が崩れる emulator があるため、削除の失敗は transport ではなく事後条件の不成立として扱う。
   296  	for _, m := range msgs {
   297  		if _, err := s.sqs.DeleteMessage(ctx, &sqs.DeleteMessageInput{QueueUrl: aws.String(url), ReceiptHandle: m.ReceiptHandle}); err != nil {
   298  			return "", incompatible("受信した message を削除できない: " + err.Error())
   299  		}
   300  	}
   301  
   302  	switch len(msgs) {
   303  	case 1:
   304  	case 0:
   305  		return "", incompatible("1 件 publish して 1 件も届かない（属性設定後の配送停止）")
   306  	default:
   307  		return "", incompatible("1 件 publish して " + strconv.Itoa(len(msgs)) + " 件届いた（属性設定後の重複配送）")
   308  	}
   309  
   310  	return "属性設定後も 1 件だけ届いた", nil
   311  }

   313  // dispatchByType は、受信した message を type 属性で振り分け、wakeup と revocation が 1 件ずつ揃っているかを確かめます。
   314  func dispatchByType(msgs []sqstypes.Message) (string, error) {
   315  	counts := map[string]int{}
   316  	for _, m := range msgs {
   317  		attr, ok := m.MessageAttributes[attrNotificationType]
   318  		if !ok {
   319  			return "", incompatible("type 属性の無い message が届いた: " + aws.ToString(m.Body))
   320  		}
   321  
   322  		kind := aws.ToString(attr.StringValue)
   323  		switch kind {
   324  		case typeWakeup:
   325  			if !isNotification(aws.ToString(m.Body)) {
   326  				return "", incompatible("wakeup の body が通知の形でない: " + aws.ToString(m.Body))
   327  			}
   328  		case typeRevocation:
   329  		default:
   330  			return "", incompatible("未知の type 属性: " + kind)
   331  		}
   332  
   333  		counts[kind]++
   334  	}
   335  
   336  	if counts[typeWakeup] != 1 || counts[typeRevocation] != 1 {
   337  		return "", incompatible("wakeup " + strconv.Itoa(counts[typeWakeup]) + " 件 / revocation " +
   338  			strconv.Itoa(counts[typeRevocation]) + " 件（期待は 1 件ずつ）")
   339  	}
   340  
   341  	return typeWakeup + " 1 / " + typeRevocation + " 1", nil
   342  }
