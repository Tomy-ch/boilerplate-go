    98  func (f *awsFake) serve(w http.ResponseWriter, r *http.Request) {
    99  	body, err := io.ReadAll(r.Body)
   100  	if err != nil {
   101  		if r.Context().Err() != nil {
   102  			return // 呼び出し側が窓切れで接続を閉じた。fake の失敗ではない
   103  		}
   104  
   105  		f.t.Errorf("fake: body の読み取りに失敗: %v", err)
   106  	}
   107  
   108  	req := parseFakeRequest(r, body)
   109  
   110  	f.mu.Lock()
   111  	h := f.handlers[req.op]
   112  	f.calls = append(f.calls, req.op)
   113  	f.mu.Unlock()
   114  
   115  	var resp fakeResponse
   116  	switch {
   117  	case h != nil:
   118  		resp = h(req)
   119  	case req.form != nil:
   120  		f.t.Errorf("fake: 未登録の操作 %q", req.op)
   121  		resp = xmlErr("InvalidAction", "no fake handler")
   122  	default:
   123  		f.t.Errorf("fake: 未登録の操作 %q", req.op)
   124  		resp = jsonErr("InvalidAction", "no fake handler")
   125  	}
   126  
   127  	w.Header().Set("Content-Type", resp.contentType)
   128  	w.WriteHeader(resp.status)
   129  	_, _ = w.Write([]byte(resp.body))
   130  }

   275  func fakeQuery(req fakeRequest) fakeResponse {
   276  	kce, _ := req.json["KeyConditionExpression"].(string)
   277  	forward := true
   278  	if v, ok := req.json["ScanIndexForward"].(bool); ok {
   279  		forward = v
   280  	}
   281  
   282  	limit, _ := req.json["Limit"].(float64)
   283  	start, _ := jsonPath(req.json, "ExclusiveStartKey", attrSequence, "N").(string)
   284  
   285  	switch {
   286  	case strings.Contains(kce, "#seq"):
   287  		return ddbPage([]int{3, 4, 5}, 0)
   288  	case !forward:
   289  		return ddbPage([]int{5}, 0)
   290  	case limit == pageLimit && start == "":
   291  		return ddbPage([]int{1, 2}, 2)
   292  	case limit == pageLimit && start == "2":
   293  		return ddbPage([]int{3, 4}, 4)
   294  	case limit == pageLimit:
   295  		return ddbPage([]int{5}, 0)
   296  	default:
   297  		return ddbPage([]int{1, 2, 3, 4, 5}, 0)
   298  	}
   299  }

   484  func (st *goawsState) publish(req fakeRequest) fakeResponse {
   485  	message := req.form.Get("Message")

   487  	for i := 1; ; i++ {
   488  		prefix := "MessageAttributes.entry." + strconv.Itoa(i)
   489  		name := req.form.Get(prefix + ".Name")
   490  		if name == "" {
   491  			break
   492  		}

   495  			"DataType":    req.form.Get(prefix + ".Value.DataType"),
   496  			"StringValue": req.form.Get(prefix + ".Value.StringValue"),
   497  		}
   498  	}
   499  
   500  	st.mu.Lock()
   501  	defer st.mu.Unlock()
   502  
   503  	for subArn, name := range st.subs {
   504  		body := message
   505  		if !st.raw[subArn] {
   506  			body = `{"Type":"Notification","Message":` + strconv.Quote(message) + `}`
   507  		}
   508  
   509  		st.inbox[st.queueURL(name)] = append(st.inbox[st.queueURL(name)], map[string]any{
   510  			"MessageId":         "m-" + name,
   511  			"ReceiptHandle":     "r-" + name,
   512  			"Body":              body,
   513  			"MessageAttributes": attrs,
   514  		})
   515  	}
   516  
   517  	return xmlOK("Publish", "<MessageId>m-1</MessageId>")
   518  }
