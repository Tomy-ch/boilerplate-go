    92  func run(ctx context.Context, args []string, out io.Writer, random io.Reader, dial dialFunc) (int, error) {
    93  	opts, err := parseOptions(args)

    96  			return 0, nil
    97  		}
    98  
    99  		return 1, err

   104

   106  		return 1, err

   110  	if err != nil {

   112  	}
   113  
   114  	clients, err := newClients(ctx, opts)
   115  	if err != nil {
   116  		return 1, err
   117  	}
   118  
   119  	rec := &recorder{}

   121  	log.Printf("ℹ️ run id: %s（table %s / topic %s）", runID, n.table(), n.topic())
   122  
   123  	runDynamoDB(ctx, clients.dynamoDB, n.table(), opts.keep, rec)
   124  	runPubSub(ctx, clients.sns, clients.sqs, n, opts.subscribers, opts.keep, rec)
   125  
   126  	if err := write(out, opts.format, rec.results); err != nil {

   166  // waitReady は、各 endpoint へ TCP 接続できるまで待ちます。compose の healthcheck に依存しないのは、
   167  // `infra-up --wait` が healthcheck の無いサービスを running の時点で ready と見なすためです。
   168  func waitReady(ctx context.Context, dial dialFunc, timeout time.Duration, endpoints ...string) error {
   169  	ctx, cancel := context.WithTimeout(ctx, timeout)
   170  	defer cancel()
   171  
   172  	for _, ep := range endpoints {
   173  		addr, err := hostPort(ep)
   174  		if err != nil {
   175  			return err
   176  		}
   177  
   178  		if err := waitTCP(ctx, dial, addr); err != nil {
   179  			return xerrors.Join(errNotReady, xerrors.Wrap(err, ep))
   180  		}
   181  	}
   182  
   183  	return nil
   184  }

   186  func waitTCP(ctx context.Context, dial dialFunc, addr string) error {
   187  	for {
   188  		conn, err := dial(ctx, "tcp", addr)
   189  		if err == nil {
   190  			return conn.Close()
   191  		}
   192  
   193  		select {
   194  		case <-ctx.Done():
   195  			return xerrors.Wrap(err, "last dial error")
   196  		case <-time.After(readyPollInterval):
   197  		}
   198  	}
   199  }

   201  // hostPort は、endpoint URL から接続先 host:port を返します。port 省略時は scheme の既定を使います。
   202  func hostPort(endpoint string) (string, error) {
   203  	u, err := url.Parse(endpoint)
   204  	if err != nil {
   205  		return "", xerrors.Wrap(err, "parse endpoint")
   206  	}
   207  
   208  	// "localhost:8000" は scheme=localhost / host="" に解釈されるので、host 空は入力ミスとして止める。
   209  	if u.Hostname() == "" || (u.Scheme != "http" && u.Scheme != "https") {
   210  		return "", xerrors.Wrap(errEndpoint, endpoint)
   211  	}
   212  
   213  	port := u.Port()
   214  	if port == "" {
   215  		port = "80"
   216  		if u.Scheme == "https" {
   217  			port = "443"
   218  		}
   219  	}
   220  
   221  	return net.JoinHostPort(u.Hostname(), port), nil
   222  }

   224  // newClients は、静的資格情報と endpoint 上書きでクライアントを組み立てます。application の
   225  // awsclient と同じ SDK 経路（LoadDefaultConfig + BaseEndpoint）を通し、smoke の結論が
   226  // application の接続形に対して成り立つようにします。
   227  func newClients(ctx context.Context, opts options) (clients, error) {
   228  	cfg, err := awsconfig.LoadDefaultConfig(ctx,
   229  		awsconfig.WithRegion(opts.region),
   230  		awsconfig.WithCredentialsProvider(awscreds.NewStaticCredentialsProvider(smokeCredential, smokeCredential, "")),
   231  	)
   232  	if err != nil {
   233  		return clients{}, xerrors.Wrap(err, "load aws config")
   234  	}
   235  
   236  	return clients{
   237  		dynamoDB: dynamodb.NewFromConfig(cfg, func(o *dynamodb.Options) {
   238  			o.BaseEndpoint = aws.String(opts.dynamoDBEndpoint)
   239  		}),
   240  		sns: sns.NewFromConfig(cfg, func(o *sns.Options) {
   241  			o.BaseEndpoint = aws.String(opts.goAWSEndpoint)
   242  		}),
   243  		sqs: sqs.NewFromConfig(cfg, func(o *sqs.Options) {
   244  			o.BaseEndpoint = aws.String(opts.goAWSEndpoint)
   245  		}),
   246  	}, nil
   247  }
