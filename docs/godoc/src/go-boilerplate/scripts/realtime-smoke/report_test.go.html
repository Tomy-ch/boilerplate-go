    35  	t.Parallel()
    36  
    37  	var _ net.Error = fakeNetError{}
    38  
    39  	tests := map[string]struct {
    40  		err  error
    41  		want Verdict
    42  	}{
    43  		"nil は互換":                            {err: nil, want: VerdictCompatible},
    44  		"事後条件の不成立は非互換":                       {err: incompatible("envelope が届いた"), want: VerdictIncompatible},
    45  		"context の期限切れは検証不能":                 {err: xerrors.Wrap(context.DeadlineExceeded, "receive"), want: VerdictUnverifiable},
    46  		"net.Error は検証不能":                    {err: xerrors.Wrap(fakeNetError{}, "dial"), want: VerdictUnverifiable},
    47  		"InvalidAction は未対応":                 {err: &fakeAPIError{code: "InvalidAction", msg: "no"}, want: VerdictUnsupported},
    48  		"NotImplemented は未対応":                {err: &fakeAPIError{code: "NotImplemented", msg: "no"}, want: VerdictUnsupported},
    49  		"deserialization failed（XML 応答）は未対応": {err: xerrors.New("operation error SQS: ListQueues, deserialization failed"), want: VerdictUnsupported},
    50  		"404 は未対応":                           {err: xerrors.New("https response error StatusCode: 404, RequestID: x"), want: VerdictUnsupported},
    51  		"ConditionalCheckFailedException は非互換（受理された拒否）": {
    52  			err:  &fakeAPIError{code: "ConditionalCheckFailedException", msg: "x"},
    53  			want: VerdictIncompatible,
    54  		},
    55  		"分類できない応答は非互換": {err: xerrors.New("something else"), want: VerdictIncompatible},
    56  	}
    57  
    58  	for name, tt := range tests {
    59  		t.Run(name, func(t *testing.T) {
    60  			t.Parallel()
    61  
    62  			assert.Equal(t, tt.want, classify(tt.err))
    63  		})
    64  	}

   172  func Test_runChain(t *testing.T) {
   173  	t.Parallel()
   174  
   175  	pass := func(context.Context) (string, error) { return "ok", nil }
   176  	fail := func(context.Context) (string, error) { return "", &fakeAPIError{code: "InvalidAction", msg: "no"} }
   177  
   178  	t.Run("halt な検査の失敗は後続を検証不能として残す", func(t *testing.T) {
   179  		t.Parallel()
   180  
   181  		rec := &recorder{}
   182  		first := runChain(t.Context(), "s", []step{
   183  			{id: "1", check: "a", halt: true, fn: fail},
   184  			{id: "2", check: "b", fn: pass},
   185  			{id: "3", check: "c", fn: pass},
   186  		}, rec)
   187  
   188  		require.Len(t, rec.results, 3)
   189  		assert.False(t, first)
   190  		assert.Equal(t, VerdictUnsupported, rec.results[0].Verdict)
   191  		assert.Equal(t, VerdictUnverifiable, rec.results[1].Verdict)
   192  		assert.Equal(t, VerdictUnverifiable, rec.results[2].Verdict)
   193  		assert.Contains(t, rec.results[2].Detail, "先行検査 1")
   194  	})
   195  
   196  	t.Run("halt でない検査の失敗は後続を止めない", func(t *testing.T) {
   197  		t.Parallel()
   198  
   199  		rec := &recorder{}
   200  		first := runChain(t.Context(), "s", []step{
   201  			{id: "1", check: "a", fn: pass},
   202  			{id: "2", check: "b", fn: fail},
   203  			{id: "3", check: "c", fn: pass},
   204  		}, rec)
   205  
   206  		require.Len(t, rec.results, 3)
   207  		assert.True(t, first)
   208  		assert.Equal(t, VerdictCompatible, rec.results[2].Verdict)
   209  		assert.Equal(t, "ok", rec.results[2].Detail)
   210  	})
   211  
   212  	t.Run("先頭が非 halt で失敗しても後続は走り、戻り値は false", func(t *testing.T) {
   213  		t.Parallel()
   214  
   215  		rec := &recorder{}
   216  		first := runChain(t.Context(), "s", []step{{id: "1", check: "a", fn: fail}, {id: "2", check: "b", fn: pass}}, rec)
   217  
   218  		assert.False(t, first)
   219  		assert.Equal(t, VerdictCompatible, rec.results[1].Verdict)
   220  	})

   249  func Test_exitCode(t *testing.T) {
   250  	t.Parallel()
   251  
   252  	compatible := Result{ID: "1", Verdict: VerdictCompatible}
   253  	incompatibleResult := Result{ID: "2", Verdict: VerdictIncompatible}
   254  	unsupported := Result{ID: "3", Verdict: VerdictUnsupported}
   255  	unverifiable := Result{ID: "4", Verdict: VerdictUnverifiable}
   256  
   257  	t.Run("結果が 0 件なら失敗（検査していないことをクリーンにしない）", func(t *testing.T) {
   258  		t.Parallel()
   259  
   260  		code, err := exitCode(nil, false)
   261  		require.ErrorIs(t, err, errNoResults)
   262  		assert.Equal(t, 1, code)
   263  	})
   264  
   265  	tests := map[string]struct {
   266  		results []Result
   267  		strict  bool
   268  		want    int
   269  	}{
   270  		"全て互換なら 0":             {results: []Result{compatible}, want: 0},
   271  		"検証不能が 1 件でもあれば 1":     {results: []Result{compatible, unverifiable}, want: 1},
   272  		"非互換は strict でなければ 0":  {results: []Result{compatible, incompatibleResult}, want: 0},
   273  		"非互換は strict なら 1":     {results: []Result{compatible, incompatibleResult}, strict: true, want: 1},
   274  		"未対応は strict なら 1":     {results: []Result{unsupported}, strict: true, want: 1},
   275  		"未対応は strict でなければ 0":  {results: []Result{unsupported}, want: 0},
   276  		"strict でも検証不能は変わらず 1": {results: []Result{unverifiable}, strict: true, want: 1},
   277  	}
   278  
   279  	for name, tt := range tests {
   280  		t.Run(name, func(t *testing.T) {
   281  			t.Parallel()
   282  
   283  			code, err := exitCode(tt.results, tt.strict)
   284  			require.NoError(t, err)
   285  			assert.Equal(t, tt.want, code)
   286  		})
   287  	}
   288  }
