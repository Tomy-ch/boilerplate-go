   115  
   116  		attempts := 0
   117  		dial := func(ctx context.Context, network, addr string) (net.Conn, error) {

   122  
   123  			return realDial(ctx, network, addr)
   124  		}
   125  
   126  		ctx, cancel := context.WithTimeout(t.Context(), 5*time.Second)
   127  		defer cancel()
   128  
   129  		require.NoError(t, waitTCP(ctx, dial, ln.Addr().String()))
   130  		assert.Equal(t, 2, attempts)
   131  	})
