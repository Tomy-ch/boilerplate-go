   173  	doc, err := json.Marshal(queuePolicy{
   174  		Version: "2012-10-17",
   175  		Statement: []policyStatement{{
   176  			Sid:       "AllowTopic",
   177  			Effect:    "Allow",
   178  			Principal: map[string]string{"Service": "sns.amazonaws.com"},
   179  			Action:    "sqs:SendMessage",
   180  			Resource:  queueArn,
   181  			Condition: map[string]any{"ArnEquals": map[string]string{"aws:SourceArn": s.topicArn}},
   182  		}},
   183  	})
   184  	if err != nil {
   185  		return "", xerrors.Wrap(err, "marshal policy")

   192  func (s *pubSubSmoke) verifyPolicy(ctx context.Context, url, doc string) error {

   276  func canonicalJSON(s string) (string, error) {
   277  	var v any
   278  	if err := json.Unmarshal([]byte(s), &v); err != nil {
   279  		return "", xerrors.Wrap(err, "unmarshal")
   280  	}
   281

   305  			SubscriptionArn: aws.String(subArn),

   309  			return "", err
   310  		}
   311  
   312  		got, err := s.sns.GetSubscriptionAttributes(ctx, &sns.GetSubscriptionAttributesInput{SubscriptionArn: aws.String(subArn)})
   313  		if err != nil {
   314  			return "", err
   315  		}
   316  
   317  		if got.Attributes[attrRawMessageDelivery] != "true" {
   318  			return "", incompatible("RawMessageDelivery が読み戻せない: " + got.Attributes[attrRawMessageDelivery])
   319  		}
   320  
   321  		s.subArns = append(s.subArns, subArn)
   322  	}
   323  
   324  	return strconv.Itoa(len(s.subArns)) + " subscription で RawMessageDelivery=true が読み戻せた", nil

   358  // receive は、1 queue で配送を待ち、届いた message をすべて返します。窓の中に届かなければ空を返し
   359  // （非互換の判定は verifyDelivery が行う）、実行全体の期限切れは transport 失敗としてそのまま返します。
   360  // 最初の message を受けた後にもう 1 回短く poll し、別 batch に着地した重複配送も拾います。
   361  func (s *pubSubSmoke) receive(parent context.Context, url string) ([]sqstypes.Message, error) {
   362  	ctx, cancel := context.WithTimeout(parent, s.window)
   363  	defer cancel()
   364  
   365  	for {
   366  		msgs, err := s.pollOnce(ctx, url, receiveWait)
   367  		if err != nil {
   368  			if parent.Err() != nil {
   369  				return nil, err // 実行全体の期限。窓切れではないので検証不能として classify へ渡す
   370  			}
   371  
   372  			if ctx.Err() != nil {
   373  				return nil, nil // 窓を使い切った
   374  			}
   375  
   376  			return nil, err
   377  		}
   378  
   379  		if len(msgs) == 0 {
   380  			continue
   381  		}
   382  
   383  		extra, err := s.pollOnce(ctx, url, drainWait)
   384  		if err != nil && ctx.Err() == nil {
   385  			return nil, err
   386  		}
   387  
   388  		return append(msgs, extra...), nil
   389  	}
   390  }

   406  func (s *pubSubSmoke) verifyDelivery() (string, error) {
   407  	delivered := 0
   408  	for i, msgs := range s.received {
   409  		switch {
   410  		case len(msgs) == 0:
   411  			continue
   412  		case len(msgs) > 1:
   413  			return "", incompatible("queue " + strconv.Itoa(i) + " に " + strconv.Itoa(len(msgs)) + " 件届いた（重複配送）")
   414  		}
   415  
   416  		body := aws.ToString(msgs[0].Body)
   417  		if isSNSEnvelope(body) {
   418  			return "", incompatible("RawMessageDelivery が効かず SNS envelope（Type: Notification）が届いた")
   419  		}
   420  
   421  		if body != s.payload {
   422  			return "", incompatible("body が raw payload と一致しない: " + body)
   423  		}
   424  
   425  		delivered++
   426  	}
   427  
   428  	if delivered != len(s.received) {
   429  		return "", incompatible(strconv.Itoa(len(s.received)) + " queue 中 " + strconv.Itoa(delivered) + " queue にしか届かない")
   430  	}
   431  
   432  	return strconv.Itoa(delivered) + " queue すべてに raw payload が 1 件ずつ届いた", nil
   433  }

   442  }

   444  func (s *pubSubSmoke) messageAttributes(_ context.Context) (string, error) {
   445  	if s.received == nil {
   446  		return "", incompatible("先行検査 G6 で message を受け取れていない")
   447  	}
   448  
   449  	for i, msgs := range s.received {
   450  		if len(msgs) == 0 {
   451  			return "", incompatible("queue " + strconv.Itoa(i) + " に message が無い")

   461  		}
   462  	}

   464  	return "String 属性が raw delivery でも透過した", nil

   489  // cleanup は、unsubscribe → queue 削除 → topic 削除 の順（orphan cleanup と同じ順）で後片付けします。
   490  // 作られなかった resource は飛ばし、失敗は行として残します。
   491  func (s *pubSubSmoke) cleanup(ctx context.Context, keep bool, rec *recorder) {
   492  	const id, check = "G9", "Unsubscribe → DeleteQueue → DeleteTopic（後片付け）"
   493  
   494  	switch {
   495  	case s.topicArn == "" && len(s.queueURLs) == 0:
   496  		rec.skip(id, pubSubSubject, check, "先行検査により作成された resource が無い")
   497  	case keep:
   498  		rec.skip(id, pubSubSubject, check, "-keep により未実施（topic "+s.topicArn+" が残っている）")
   499  	default:
