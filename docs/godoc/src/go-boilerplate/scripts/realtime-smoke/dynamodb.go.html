    47  func runDynamoDB(ctx context.Context, c *dynamodb.Client, table string, keep bool, rec *recorder) {
    48  	s := &ddbSmoke{c: c, table: table}
    49  
    50  	created := runChain(ctx, ddbSubject, s.steps(), rec)
    51  
    52  	cctx, cancel := cleanupContext(ctx)
    53  	defer cancel()
    54  
    55  	s.cleanup(cctx, created, keep, rec)

    58  func (s *ddbSmoke) steps() []step {
    59  	return []step{
    60  		{id: "D1", check: "CreateTable + TableExists waiter", halt: true, fn: s.createTable},
    61  		{id: "D2", check: "conditional PutItem（attribute_not_exists）", halt: true, fn: s.conditionalPut},
    62  		{id: "D3", check: "同一 key の再 PutItem → ConditionalCheckFailedException", fn: s.duplicatePut},
    63  		{id: "D4", check: "Query ConsistentRead=true, sequence > cursor（昇順・欠番なし）", fn: s.queryAfterCursor},
    64  		{id: "D5", check: "Query ScanIndexForward=false Limit=1（最新 1 件）", fn: s.queryLatest},
    65  		{id: "D6", check: "Query Limit + ExclusiveStartKey の pagination", fn: s.queryPaginated},
    66  		{id: "D7", check: "UpdateTimeToLive + DescribeTimeToLive", fn: s.timeToLive},
    67  	}
    68  }

    70  func (s *ddbSmoke) createTable(ctx context.Context) (string, error) {
    71  	_, err := s.c.CreateTable(ctx, &dynamodb.CreateTableInput{
    72  		TableName: aws.String(s.table),
    73  		AttributeDefinitions: []types.AttributeDefinition{
    74  			{AttributeName: aws.String(attrStream), AttributeType: types.ScalarAttributeTypeS},
    75  			{AttributeName: aws.String(attrSequence), AttributeType: types.ScalarAttributeTypeN},
    76  		},
    77  		KeySchema: []types.KeySchemaElement{
    78  			{AttributeName: aws.String(attrStream), KeyType: types.KeyTypeHash},
    79  			{AttributeName: aws.String(attrSequence), KeyType: types.KeyTypeRange},
    80  		},
    81  		BillingMode: types.BillingModePayPerRequest,
    82  	})
    83  	if err != nil {
    84  		return "", err
    85  	}
    86  
    87  	waiter := dynamodb.NewTableExistsWaiter(s.c)
    88  	if err := waiter.Wait(ctx, &dynamodb.DescribeTableInput{TableName: aws.String(s.table)}, tableWaitTimeout); err != nil {
    89  		return "", xerrors.Wrap(err, "wait table exists")
    90  	}
    91  
    92  	return "table " + s.table, nil
    93  }

    96  func (s *ddbSmoke) putSequence(ctx context.Context, seq int) error {
    97  	_, err := s.c.PutItem(ctx, &dynamodb.PutItemInput{
    98  		TableName: aws.String(s.table),
    99  		Item: map[string]types.AttributeValue{
   100  			attrStream:   &types.AttributeValueMemberS{Value: streamID},
   101  			attrSequence: &types.AttributeValueMemberN{Value: strconv.Itoa(seq)},
   102  			attrPayload:  &types.AttributeValueMemberS{Value: "payload-" + strconv.Itoa(seq)},
   103  		},
   104  		ConditionExpression: aws.String("attribute_not_exists(" + attrStream + ")"),
   105  	})
   106  
   107  	return err
   108  }

   142  func (s *ddbSmoke) query(ctx context.Context, in *dynamodb.QueryInput) ([]int, map[string]types.AttributeValue, error) {
   143  	in.TableName = aws.String(s.table)
   144  	// 式で参照しない名前を渡すと production の DynamoDB も ValidationException を返すため、使う式にだけ付ける。
   145  	if strings.Contains(aws.ToString(in.KeyConditionExpression), "#seq") {
   146  		in.ExpressionAttributeNames = map[string]string{"#seq": attrSequence}
   147  	}
   148  
   149  	out, err := s.c.Query(ctx, in)
   150  	if err != nil {
   151  		return nil, nil, err
   152  	}
   153  
   154  	seqs := make([]int, 0, len(out.Items))
   155  	for _, item := range out.Items {
   156  		n, ok := item[attrSequence].(*types.AttributeValueMemberN)
   157  		if !ok {
   158  			return nil, nil, incompatible("sequence が N 型で返らない")
   159  		}
   160  
   161  		v, err := strconv.Atoi(n.Value)
   162  		if err != nil {
   163  			return nil, nil, incompatible("sequence が整数で返らない: " + n.Value)
   164  		}
   165  
   166  		seqs = append(seqs, v)
   167  	}
   168  
   169  	return seqs, out.LastEvaluatedKey, nil
   170  }

   172  func (s *ddbSmoke) queryAfterCursor(ctx context.Context) (string, error) {
   173  	if err := s.seed(ctx); err != nil {
   174  		return "", err
   175  	}
   176  
   177  	seqs, _, err := s.query(ctx, &dynamodb.QueryInput{
   178  		KeyConditionExpression: aws.String(attrStream + " = :s AND #seq > :c"),
   179  		ExpressionAttributeValues: map[string]types.AttributeValue{
   180  			":s": &types.AttributeValueMemberS{Value: streamID},
   181  			":c": &types.AttributeValueMemberN{Value: strconv.Itoa(cursorAfter)},
   182  		},
   183  		ConsistentRead: aws.Bool(true),
   184  	})
   185  	if err != nil {
   186  		return "", err
   187  	}
   188  
   189  	want := make([]int, 0, seedCount-cursorAfter)
   190  	for seq := cursorAfter + 1; seq <= seedCount; seq++ {
   191  		want = append(want, seq)
   192  	}
   193  
   194  	if !equalInts(seqs, want) {
   195  		return "", incompatible("期待 " + joinInts(want) + " に対し " + joinInts(seqs) + " が返った")
   196  	}
   197  
   198  	return joinInts(seqs) + " が昇順で返った", nil
   199  }

   201  func (s *ddbSmoke) queryLatest(ctx context.Context) (string, error) {
   202  	seqs, _, err := s.query(ctx, &dynamodb.QueryInput{
   203  		KeyConditionExpression:    aws.String(attrStream + " = :s"),
   204  		ExpressionAttributeValues: map[string]types.AttributeValue{":s": &types.AttributeValueMemberS{Value: streamID}},
   205  		ScanIndexForward:          aws.Bool(false),
   206  		Limit:                     aws.Int32(1),
   207  		ConsistentRead:            aws.Bool(true),
   208  	})
   209  	if err != nil {
   210  		return "", err
   211  	}
   212  
   213  	if !equalInts(seqs, []int{seedCount}) {
   214  		return "", incompatible("最新 1 件として " + joinInts(seqs) + " が返った")
   215  	}
   216  
   217  	return "sequence " + strconv.Itoa(seedCount) + " が返った", nil

   220  func (s *ddbSmoke) queryPaginated(ctx context.Context) (string, error) {
   221  	var (
   222  		got   []int
   223  		pages int
   224  		start map[string]types.AttributeValue
   225  	)
   226  
   227  	for {
   228  		seqs, last, err := s.query(ctx, &dynamodb.QueryInput{
   229  			KeyConditionExpression:    aws.String(attrStream + " = :s"),
   230  			ExpressionAttributeValues: map[string]types.AttributeValue{":s": &types.AttributeValueMemberS{Value: streamID}},
   231  			Limit:                     aws.Int32(pageLimit),
   232  			ExclusiveStartKey:         start,
   233  			ConsistentRead:            aws.Bool(true),
   234  		})
   235  		if err != nil {
   236  			return "", err
   237  		}
   238  
   239  		pages++
   240  		got = append(got, seqs...)
   241  
   242  		if last == nil || pages > seedCount {
   243  			break
   244  		}
   245  
   246  		start = last
   247  	}
   248  
   249  	want := make([]int, 0, seedCount)
   250  	for seq := 1; seq <= seedCount; seq++ {
   251  		want = append(want, seq)
   252  	}
   253  
   254  	if !equalInts(got, want) {
   255  		return "", incompatible("pagination の結果が " + joinInts(got) + "（" + strconv.Itoa(pages) + " ページ）")
   256  	}
   257  
   258  	return strconv.Itoa(pages) + " ページで " + strconv.Itoa(len(got)) + " 件（LastEvaluatedKey で継続）", nil
   259  }

   261  func (s *ddbSmoke) timeToLive(ctx context.Context) (string, error) {
   262  	_, err := s.c.UpdateTimeToLive(ctx, &dynamodb.UpdateTimeToLiveInput{
   263  		TableName: aws.String(s.table),
   264  		TimeToLiveSpecification: &types.TimeToLiveSpecification{
   265  			AttributeName: aws.String(attrExpires),
   266  			Enabled:       aws.Bool(true),
   267  		},
   268  	})
   269  	if err != nil {
   270  		return "", err
   271  	}
   272  
   273  	out, err := s.c.DescribeTimeToLive(ctx, &dynamodb.DescribeTimeToLiveInput{TableName: aws.String(s.table)})
   274  	if err != nil {
   275  		return "", err
   276  	}
   277  
   278  	desc := out.TimeToLiveDescription
   279  	if desc == nil || aws.ToString(desc.AttributeName) != attrExpires {
   280  		return "", incompatible("UpdateTimeToLive は受理されたが DescribeTimeToLive に反映されない")
   281  	}
   282  
   283  	switch desc.TimeToLiveStatus {
   284  	case types.TimeToLiveStatusEnabled, types.TimeToLiveStatusEnabling:
   285  		return string(desc.TimeToLiveStatus) + " が読み戻った（DynamoDB Local は期限切れ item を削除しない — 既知の caveat）", nil
   286  	case types.TimeToLiveStatusDisabled, types.TimeToLiveStatusDisabling:
   287  		return "", incompatible("TTL の状態が " + string(desc.TimeToLiveStatus))
   288  	default:
   289  		return "", incompatible("TTL の状態が不明: " + string(desc.TimeToLiveStatus))
   290  	}
   291  }

   293  // cleanup は、作成した table を削除します。-keep のときは行として残し、黙って省きません。
   294  func (s *ddbSmoke) cleanup(ctx context.Context, created, keep bool, rec *recorder) {
   295  	const id, check = "D8", "DeleteTable（後片付け）"
   296  
   297  	switch {
   298  	case !created:
   299  		rec.skip(id, ddbSubject, check, "先行検査 D1 により実行不能")
   300  	case keep:
   301  		rec.skip(id, ddbSubject, check, "-keep により未実施（table "+s.table+" が残っている）")
   302  	default:
   303  		_, err := s.c.DeleteTable(ctx, &dynamodb.DeleteTableInput{TableName: aws.String(s.table)})
   304  		rec.record(id, ddbSubject, check, "table "+s.table+" を削除", err)
   305  	}
   306  }
