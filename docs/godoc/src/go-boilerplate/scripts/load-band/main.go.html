     1  // Package main は、ホスト負荷の配分（負荷帯・CPU シェア・各ツールへ渡す並列度）を解決するツール。
     2  //
     3  //	env:    解決結果を KEY=VALUE で出力する。make のレシピが eval して読む。
     4  //	status: 解決結果と帯ごとの助言を人間向けに表示する（make load-status）。
     5  //
     6  // 帯の意味と閾値の意図は .makefiles/README.md の `.makefiles/load` group が持つ。ここが持つのは
     7  // 「窓の数と CPU 数から帯・シェア・フラグを導く」計算だけ。
     8  //
     9  // このツールはコミット毎・push 毎に走るゲートの並列度を決める。壊れ方が「黙って
    10  // 既定の帯へ縮退し、ホストを飽和させる／検査を飛ばす」方向に出るため、判定ロジックは
    11  // シェルの中ではなくテストの当たる Go 側に置く。窓の数え方も同じ理由でここが持つ。

    29  // 負荷帯。auto は窓の数から決めるため、解決後の値には現れない。
    30  const (
    31  	bandAuto    = "auto"
    32  	bandFull    = "full"
    33  	bandLow     = "low"
    34  	bandCIFirst = "ci-first"
    35  )
    36  
    37  const (
    38  	// defaultLowThreshold / defaultCIFirstThreshold は帯域の境目の既定値。
    39  	// make 側は上書き値だけを渡すため、既定値の在処はここ 1 箇所。
    40  	defaultLowThreshold     = 3
    41  	defaultCIFirstThreshold = 5
    42  
    43  	// niceValue は絞る帯で重いゲートに付ける優先度。
    44  	niceValue = "nice -n 10"
    45  
    46  	// minShare は 1 窓あたりの CPU share の下限。0 を渡すとツールによっては「無制限」と
    47  	// 解釈され、絞る意図が反転する。
    48  	minShare = 1
    49  
    50  	// minWindows は窓の数の下限。share は CPU 数を窓の数で割って求めるため、0 が入ると
    51  	// 0 除算になる。countWindows も同じ下限を保証するが、resolve はそれを経ずに呼べる。
    52  	minWindows = 1
    53  )

    75  // throttled は、絞る側の帯（low / ci-first）かを返します。ci-first でもローカルに残す軽い
    76  // ゲートは走るため、絞り自体は両方に効かせます。
    77  func (b band) throttled() bool {
    78  	return b.resolved == bandLow || b.resolved == bandCIFirst

    81  // main は 1:1 テスト規約の対象外で分岐を検査できないため、判断は run に置きます。
    82  func main() {
    83  	log.SetFlags(0)
    84  
    85  	if err := run(os.Args[1:], os.Stdout, countWindows(worktreeList()), runtime.NumCPU()); err != nil {
    86  		log.Fatalf("%v", err)
    87  	}
    88  }

    90  // run は、サブコマンドとフラグから負荷帯を解決し、解決結果を out へ書き出します。
    91  // 窓の数と CPU 数はホストの実測値で、差し替えられるよう引数で受けます。
    92  func run(args []string, out io.Writer, windows, cpus int) error {
    93  	if len(args) == 0 {
    94  		return errUsage
    95  	}
    96  
    97  	subcommand := args[0]
    98  
    99  	fs := flag.NewFlagSet(subcommand, flag.ContinueOnError)
   100  	load := fs.String("load", "", "負荷帯 (auto / full / low / ci-first。空なら auto)")
   101  	low := fs.Int("low", defaultLowThreshold, "low へ落とす窓の数")
   102  	ciFirst := fs.Int("ci-first", defaultCIFirstThreshold, "ci-first へ落とす窓の数")
   103  
   104  	if err := fs.Parse(args[1:]); err != nil {
   105  		// ヘルプ要求は失敗ではないので 0 で終える。usage は flag が既に出力している。
   106  		if xerrors.Is(err, flag.ErrHelp) {
   107  			return nil
   108  		}
   109  
   110  		return xerrors.Wrap(err, "failed to parse flags")
   111  	}
   112  
   113  	resolved, err := resolve(*load, windows, cpus, *low, *ciFirst)
   114  	if err != nil {
   115  		return xerrors.Wrap(err, "❌ 負荷帯を解決できません")
   116  	}
   117  
   118  	switch subcommand {
   119  	case "env":
   120  		_, _ = fmt.Fprint(out, renderEnv(resolved))
   121  	case "status":
   122  		_, _ = fmt.Fprint(out, renderStatus(resolved))
   123  	default:
   124  		return errUnknownSubcommand
   125  	}
   126  
   127  	return nil
   128  }

   130  // worktreeList は `git worktree list` の出力を返します。git が使えない・リポジトリでない等で
   131  // 数えられない場合は空文字を返します（呼び出し側で 1 窓として扱われます）。
   132  func worktreeList() string {
   133  	out, err := exec.CommandContext(context.Background(), "git", "worktree", "list").Output()
   134  	if err != nil {
   135  		return ""
   136  	}
   137  
   138  	return string(out)
   139  }

   141  // countWindows は、`git worktree list` の出力から窓（worktree）の数を数えます。
   142  // 数えられない場合は 1 を返します。0 を返すと「窓が無い」という有り得ない状態が
   143  // 帯の判定へ流れ込むため、下限は常に 1 です。
   144  //
   145  // スロットのリースではなく worktree を数えるのは、スロット取得が opt-in で「窓はあるが
   146  // スロットは取っていない」状態が普通にあるため。CPU を食うのは窓のほうです。
   147  func countWindows(gitOutput string) int {
   148  	count := 0
   149  
   150  	for line := range strings.Lines(gitOutput) {
   151  		if strings.TrimSuffix(line, "\n") != "" {
   152  			count++
   153  		}
   154  	}
   155  
   156  	if count < 1 {
   157  		return 1
   158  	}
   159  
   160  	return count
   161  }

   163  // resolve は、指定された帯（空 / auto なら窓の数）から負荷帯と CPU シェアを決めます。
   164  // 未知の帯名はエラーにします（黙って full へ落とすと、絞る指定の書き損じに気づけません）。
   165  func resolve(requested string, windows, cpus, lowThreshold, ciFirstThreshold int) (band, error) {
   166  	if requested == "" {
   167  		requested = bandAuto
   168  	}
   169  
   170  	// 下限は帯の判定より前に効かせる。判定に渡す数と報告する数が食い違うと、
   171  	// 表示された窓の数から帯を再現できなくなる。
   172  	windows = max(windows, minWindows)
   173  	cpus = max(cpus, minShare)
   174  
   175  	b := band{
   176  		requested:        requested,
   177  		windows:          windows,
   178  		cpus:             cpus,
   179  		lowThreshold:     lowThreshold,
   180  		ciFirstThreshold: ciFirstThreshold,
   181  	}
   182  
   183  	switch requested {
   184  	case bandAuto:
   185  		b.resolved = bandFor(windows, lowThreshold, ciFirstThreshold)
   186  	case bandFull, bandLow, bandCIFirst:
   187  		b.resolved = requested
   188  	default:
   189  		return band{}, xerrors.Wrap(errUnknownBand, "GOBP_LOAD="+requested)
   190  	}
   191  
   192  	b.share = b.cpus
   193  	if b.throttled() {
   194  		b.share = max(b.cpus/b.windows, minShare)
   195  	}
   196  
   197  	return b, nil
   198  }

   200  // bandFor は、窓の数から負荷帯を決めます。
   201  func bandFor(windows, lowThreshold, ciFirstThreshold int) string {
   202  	switch {
   203  	case windows >= ciFirstThreshold:
   204  		return bandCIFirst
   205  	case windows >= lowThreshold:
   206  		return bandLow
   207  	default:
   208  		return bandFull
   209  	}
   210  }

   212  // renderEnv は、レシピが eval して読む KEY=VALUE を組み立てます。
   213  // 絞らない帯では各フラグを空にし、ツールの既定（golangci-lint は設定ファイルの
   214  // concurrency、go test は GOMAXPROCS）へ委ねます。
   215  func renderEnv(b band) string {
   216  	values := [][2]string{
   217  		{"GOBP_LOAD_RESOLVED", b.resolved},
   218  		{"GOBP_WINDOWS", strconv.Itoa(b.windows)},
   219  		{"GOBP_CPUS", strconv.Itoa(b.cpus)},
   220  		{"GOBP_SHARE", strconv.Itoa(b.share)},
   221  		{"GOBP_THROTTLED", throttledFlag(b)},
   222  		{"GOBP_NICE", nice(b)},
   223  		{"GOLANGCI_CONCURRENCY_FLAG", golangciConcurrencyFlag(b)},
   224  		{"GO_TEST_P_FLAG", goTestPFlag(b)},
   225  		{"GO_TEST_LOAD_ENV", goTestLoadEnv(b)},
   226  	}
   227  
   228  	var sb strings.Builder
   229  	for _, kv := range values {
   230  		fmt.Fprintf(&sb, "%s=%s\n", kv[0], shellQuote(kv[1]))
   231  	}
   232  
   233  	return sb.String()
   234  }

   245  // nice は、重いゲートに付ける優先度指定を返します。絞る帯では他の窓の対話操作

   247  func nice(b band) string {
   248  	if b.throttled() {
   249  		return niceValue
   250  	}
   251  
   252  	return ""

   264  // goTestPFlag は、go test へ渡すパッケージ同時実行数フラグを返します。
   265  func goTestPFlag(b band) string {
   266  	if b.throttled() {
   267  		return "-p " + strconv.Itoa(b.share)
   268  	}
   269  
   270  	return ""
   271  }

   273  // goTestLoadEnv は、go test に渡す環境変数の代入形を返します。-p はパッケージの同時実行数しか
   274  // 絞れず、各テストバイナリ内部の並列度（t.Parallel() の上限＝GOMAXPROCS）は絞れません。
   275  // -race はここが特に重いため両方を絞ります。
   276  func goTestLoadEnv(b band) string {
   277  	if b.throttled() {
   278  		return "GOMAXPROCS=" + strconv.Itoa(b.share)
   279  	}
   280  
   281  	return ""
   282  }

   284  // shellQuote は、値をシングルクォートで囲んで eval 可能にします。
   285  func shellQuote(v string) string {
   286  	return "'" + strings.ReplaceAll(v, "'", `'\''`) + "'"

   289  // renderStatus は、`make load-status` が表示する人間向けの解決結果を組み立てます。
   290  // 負荷帯まわりで人間がデバッグに使う唯一の窓口なので、導出に使った値を残らず印字します。
   291  func renderStatus(b band) string {
   292  	var sb strings.Builder
   293  
   294  	fmt.Fprintf(&sb, "load      : %s  (GOBP_LOAD=%s)\n", b.resolved, b.requested)
   295  	fmt.Fprintf(&sb, "windows   : %d worktree  (low >= %d, ci-first >= %d)\n", b.windows, b.lowThreshold, b.ciFirstThreshold)
   296  	fmt.Fprintf(&sb, "cpus      : %d  ->  share %d / 窓\n", b.cpus, b.share)
   297  	fmt.Fprintf(&sb, "golangci  : %s\n", orElse(golangciConcurrencyFlag(b), "設定ファイルの concurrency に委譲"))
   298  	fmt.Fprintf(&sb, "go test   : %s\n", orElse(strings.TrimSpace(goTestPFlag(b)+" "+goTestLoadEnv(b)), "既定（ホスト全体）"))
   299  	fmt.Fprintf(&sb, "nice      : %s\n", orElse(nice(b), "なし"))
   300  	sb.WriteString("\n")
   301  	sb.WriteString(advice(b))
   302  
   303  	return sb.String()
   304  }

   306  // advice は、帯ごとの助言（💡）を返します。
   307  func advice(b band) string {
   308  	switch b.resolved {
   309  	case bandCIFirst:
   310  		return "💡 窓が多いため CI-first です。重いゲート（lint / test）はローカルで走らせず、\n" +
   311  			"   push して CI で検証します。手元に残るのは commitlint / secret-scan / pin 検査など、\n" +
   312  			"   push 後では手遅れになる軽いゲートだけです。\n" +
   313  			"   一時的に手元で回すなら: make lint GOBP_LOAD=low\n"
   314  	case bandLow:
   315  		return fmt.Sprintf(
   316  			"💡 窓が多いため低負荷モードです。重いゲートは CPU share %d に絞り、逐次で走ります。\n"+
   317  				"   さらに窓を増やすなら CI へ委ねる方が速く確実です: GOBP_LOAD=ci-first\n", b.share,
   318  		)
   319  	default:
   320  		return "💡 窓が少ないためホスト全体を使います（従来どおり）。\n"
   321  	}
   322  }
