   114  func (f failOpenFS) Open(name string) (fs.File, error) {
   115  	if name == f.failPath {
   116  		return nil, &fs.PathError{Op: "open", Path: name, Err: fs.ErrPermission}
   117  	}
   118  	return f.fsys.Open(name)

   132  	t.Helper()
   133  	if _, err := exec.LookPath(shellcheck.Binary); err == nil {
   134  		return
   135  	}
   136  	if os.Getenv(requireShellcheckEnv) != "" {
   137  		t.Fatalf("shellcheck が PATH にありません（%s 指定時は skip しません）", requireShellcheckEnv)
   138  	}
   139  	t.Skip("shellcheck が PATH にありません")
   140  }
