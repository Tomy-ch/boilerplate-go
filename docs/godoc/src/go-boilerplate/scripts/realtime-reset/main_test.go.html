   115  func Test_configuredTables(t *testing.T) {
   116  	t.Run("正常系", func(t *testing.T) {
   117  		t.Run("埋め込みenvのsuffixを付けた3つのtable名を返す", func(t *testing.T) {
   118  			// 被験体が設定を通らず OS env を読むだけの退行を捉えるため、期待値は埋め込み
   119  			// ファイルから直に作り、プロセス側の同名変数を外してから呼ぶ。
   120  			// t.Setenv では未設定へ戻せない。空文字にすると config.Load が「設定済み」と見て
   121  			// 埋め込みの値で上書きせず、正しい被験体まで空 suffix になってしまう。
   122  			if v, ok := os.LookupEnv("REALTIME_TABLE_SUFFIX"); ok {
   123  				t.Cleanup(func() { _ = os.Setenv("REALTIME_TABLE_SUFFIX", v) }) //nolint:usetesting // 復元のため
   124  				require.NoError(t, os.Unsetenv("REALTIME_TABLE_SUFFIX"))
   125  			}
   126  
   127  			b, err := root.FS.ReadFile("env/.env")
   128  			require.NoError(t, err)
   129  			kv, err := godotenv.Parse(bytes.NewReader(b))
   130  			require.NoError(t, err)
   131  			suffix := kv["REALTIME_TABLE_SUFFIX"]
   132  			require.NotEmpty(t, suffix)
   133  
   134  			got, err := configuredTables()
   135  			require.NoError(t, err)
   136  			assert.Equal(t, []string{
   137  				"realtime_event_log_" + suffix,
   138  				"realtime_stream_ticket_" + suffix,
   139  				"realtime_instance_lease_" + suffix,
   140  			}, got, "別の checkout の table や Realtime 以外の table を消さない")
   141  		})
   142  	})
   143  
   144  	t.Run("異常系", func(t *testing.T) {
   145  		t.Run("設定を読めなければ削除対象を返さない", func(t *testing.T) {
   146  			t.Setenv("APP_MODE", "invalid-mode")
   147  
   148  			got, err := configuredTables()
   149  			require.ErrorIs(t, err, config.ErrInvalidAppMode)
   150  			assert.Nil(t, got, "対象が定まらないまま削除へ進ませない")
   151  		})
   152  	})
