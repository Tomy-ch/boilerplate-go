   138  func parseOptions(args []string) (options, error) {
   139  	var opts options
   140  
   141  	fs := flag.NewFlagSet("realtime-reset", flag.ContinueOnError)
   142  	fs.StringVar(&opts.endpoint, "endpoint", defaultEndpoint, "DynamoDB 互換 store の endpoint")
   143  	fs.StringVar(&opts.region, "region", defaultRegion, "署名に使う region")
   144  	fs.DurationVar(&opts.timeout, "timeout", defaultTimeout, "実行全体の上限時間")
   145  
   146  	if err := fs.Parse(args); err != nil {
   147  		if xerrors.Is(err, flag.ErrHelp) {
   148  			return opts, err
   149  		}
   150

   152  	}
   153  
   154  	if err := validateEndpoint(opts.endpoint); err != nil {
   155  		return opts, err
   156  	}
   157  
   158  	return opts, nil

   161  // validateEndpoint は、endpoint が emulator を指す形であることを確かめます。空や host の無い
   162  // endpoint は入力ミスとして止め、実 AWS の host は拒みます。自前ホストの emulator を使う構成が
   163  // あるので loopback には限定しません。
   164  func validateEndpoint(endpoint string) error {
   165  	u, err := url.Parse(endpoint)
   166  	if err != nil {
   167  		return xerrors.Wrap(errEndpoint, endpoint)
   168  	}
   169  
   170  	// "localhost:8000" は scheme=localhost / host="" に解釈されるので、host 空は入力ミスとして止める。
   171  	if u.Hostname() == "" || (u.Scheme != "http" && u.Scheme != "https") {
   172  		return xerrors.Wrap(errEndpoint, endpoint)
   173  	}
   174  
   175  	// 末尾のドットは FQDN の書き方の違いでしかなく、TLS 検証も剥がして通す。落としてから比べる。
   176  	host := strings.TrimRight(strings.ToLower(u.Hostname()), ".")
   177  	for suffix := range strings.SplitSeq(awsHostSuffixes, ",") {
   178  		if host == strings.TrimPrefix(suffix, ".") || strings.HasSuffix(host, suffix) {
   179  			return xerrors.Wrap(errRealAWS, endpoint)
   180  		}
   181  	}
   182  
   183  	return nil
   184  }

   187  func newClient(ctx context.Context, opts options) (*dynamodb.Client, error) {
   188  	cfg, err := awsconfig.LoadDefaultConfig(ctx,
   189  		awsconfig.WithRegion(opts.region),
   190  		awsconfig.WithCredentialsProvider(awscreds.NewStaticCredentialsProvider(resetCredential, resetCredential, "")),

   192  	if err != nil {
   193  		return nil, xerrors.Wrap(err, "load aws config")
   194  	}
   195  
   196  	return dynamodb.NewFromConfig(cfg, func(o *dynamodb.Options) {
   197  		o.BaseEndpoint = aws.String(opts.endpoint)
   198  	}), nil
   199  }
