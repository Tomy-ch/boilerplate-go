    16  // deleteTables は、渡された table を順に削除し、消え切るまで待ちます。
    17  // 既に無い table は、この関数の目的（無い状態にすること）から見れば成功なので読み飛ばします。
    18  func deleteTables(ctx context.Context, api tableAPI, tables []string, out io.Writer) error {
    19  	for _, table := range tables {
    20  		deleted, err := deleteTable(ctx, api, table)
    21  		if err != nil {
    22  			return xerrors.Wrap(err, "realtime-reset: "+table)
    23  		}
    24  
    25  		if !deleted {
    26  			_, _ = fmt.Fprintf(out, "・%s は既にありません\n", table)
    27  
    28  			continue
    29  		}
    30  
    31  		if err := waitGone(ctx, api, table); err != nil {
    32  			return xerrors.Wrap(err, "realtime-reset: "+table)
    33  		}
    34  
    35  		_, _ = fmt.Fprintf(out, "・%s を削除しました\n", table)
    36  	}
    37  
    38  	return nil

    42  func deleteTable(ctx context.Context, api tableAPI, table string) (bool, error) {
    43  	_, err := api.DeleteTable(ctx, &dynamodb.DeleteTableInput{TableName: aws.String(table)})
    44  	if err == nil {
    45  		return true, nil
    46  	}
    47  
    48  	var notFound *dynamodbtypes.ResourceNotFoundException
    49  	if xerrors.As(err, &notFound) {
    50  		return false, nil
    51  	}
    52  
    53  	return false, xerrors.Wrap(err, "delete table")
    54  }

    56  // waitGone は、table が引けなくなるまで待ちます。DeleteTable は非同期に返るため、待たずに
    57  // realtime-init へ進むと作り直しが ResourceInUseException で落ちます。
    58  //
    59  // 打ち切りは待っている間にも問い合わせの最中にも起こるので、errGone はその両方に付けます。
    60  // 片方だけだと、同じ「消え切らなかった」が呼び出し側から 2 通りの error に見えます。
    61  func waitGone(ctx context.Context, api tableAPI, table string) error {
    62  	for {
    63  		_, err := api.DescribeTable(ctx, &dynamodb.DescribeTableInput{TableName: aws.String(table)})
    64  
    65  		var notFound *dynamodbtypes.ResourceNotFoundException
    66  		if xerrors.As(err, &notFound) {
    67  			return nil
    68  		}
    69  
    70  		if err != nil {
    71  			if ctx.Err() != nil {
    72  				return xerrors.Join(errGone, xerrors.Wrap(err, "describe table"))
    73  			}
    74  
    75  			return xerrors.Wrap(err, "describe table")
    76  		}
    77  
    78  		select {
    79  		case <-ctx.Done():
    80  			return xerrors.Join(errGone, ctx.Err())
    81  		case <-time.After(gonePollInterval):
    82  		}
    83  	}
    84  }
