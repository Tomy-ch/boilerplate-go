    37  // notFound は、SDK が返すのと同じ包み方で ResourceNotFoundException を返します。
    38  // 実際の呼び出しでは smithy が例外を OperationError の中へ入れて返すため、裸で返す fake では
    39  // 判定を `xerrors.As` から型アサーションへ落とす退行を見逃します。
    40  func notFound(op string) error {
    41  	return &smithy.OperationError{
    42  		ServiceID:     "DynamoDB",
    43  		OperationName: op,
    44  		Err:           &dynamodbtypes.ResourceNotFoundException{},
    45  	}
    46  }

    48  func newFakeTableAPI(existing ...string) *fakeTableAPI {
    49  	f := &fakeTableAPI{existing: make(map[string]bool, len(existing))}
    50  	for _, name := range existing {

    52  	}
    53  
    54  	return f

    57  func (f *fakeTableAPI) DeleteTable(
    58  	_ context.Context, in *dynamodb.DeleteTableInput, _ ...func(*dynamodb.Options),
    59  ) (*dynamodb.DeleteTableOutput, error) {
    60  	name := aws.ToString(in.TableName)
    61  	if f.deleteErr != nil && (f.deleteErrOn == "" || f.deleteErrOn == name) {
    62  		return nil, f.deleteErr
    63  	}
    64  
    65  	if !f.existing[name] {
    66  		return nil, notFound("DeleteTable")
    67  	}
    68  
    69  	delete(f.existing, name)
    70  	f.deleted = append(f.deleted, name)
    71  
    72  	return &dynamodb.DeleteTableOutput{}, nil
    73  }

    75  func (f *fakeTableAPI) DescribeTable(
    76  	ctx context.Context, in *dynamodb.DescribeTableInput, _ ...func(*dynamodb.Options),
    77  ) (*dynamodb.DescribeTableOutput, error) {
    78  	f.described = append(f.described, aws.ToString(in.TableName))
    79  
    80  	// 実 SDK は取り消し済みの ctx で呼ばれると転送前に締切のエラーを返す。
    81  	if err := ctx.Err(); err != nil {
    82  		return nil, xerrors.Wrap(err, "DescribeTable")
    83  	}
    84  
    85  	if f.describeErr != nil {
    86  		return nil, f.describeErr
    87  	}
    88  
    89  	if f.describeLeft > 0 {
    90  		f.describeLeft--
    91  
    92  		return &dynamodb.DescribeTableOutput{}, nil
    93  	}
    94  
    95  	return nil, notFound("DescribeTable")
    96  }
