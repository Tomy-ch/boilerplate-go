     1  // Package main は Dockerfile の `FROM` base image、docker compose の `image:`、そして
     2  // workflow / composite action の `uses: docker://` を不変の digest へ固定するツール。
     3  //
     4  // registry の image を指す参照は、書かれた場所によらずこの機構が持つ。`uses:` の行であっても
     5  // 参照先は GitHub のリポジトリではないため、tag を git ls-remote で commit へ解決する
     6  // pin-actions では扱えない（docs/design/security.md）。
     7  //
     8  //	resolve: docker/*/Dockerfile の FROM、docker-compose*.yaml の image:、.github の
     9  //	         uses: docker:// を走査し、
    10  //	         image:tag を registry の現在 digest へ解決して lockfile (docker/images-pin.toml) へ書き出す。
    11  //	apply:   lockfile を SSOT に各参照を `image:tag@sha256:...` へ固定する（FROM は AS stage を保持）。
    12  //	check:   apply と同じ判定を書き換えなしで行い、未固定/未登録/drift があれば非ゼロ終了する（CI / hook 用）。
    13  //
    14  // tag は版の SSOT として FROM 行に残し、digest を lock 側で管理する（tag 再割り当て攻撃の遮断）。
    15  //
    16  // supply-chain cooldown: --min-age-days 未満（＝公開直後）の digest は採用しない。
    17  // mutable tag は「N 日前に何を指していたか」を registry へ問えないため、step-back 先は
    18  // git tag のような版履歴ではなく自分の前回 lock。前回 lock が無い初回は退行先が無く、
    19  // 出来立ての未検証 digest を掴む危険があるため、resolve は tag のまま残さず非ゼロ終了する
    20  // （緊急ブートストラップのみ --min-age-days=0 で明示採用）。
    21  //
    22  // apply/check は fail-closed：管理対象の FROM は image:tag@sha256:... へ固定され、かつ lock(SSOT)
    23  // に登録済みでなければならない。lock 未登録の image は「未登録」、digest 無し（tag のみ）は
    24  // 「未固定」として非ゼロ終了する（tag のみ運用は一切許容しない）。

    54  var (
    55  	// FROM [--platform=...] <ref> [AS <stage>]
    56  	// クオートを ref から締め出すのは、含めると `FROM "alpine:3.24"` がクオートごと一致し、
    57  	// `"alpine` を image 名として固定対象に載せてしまうため。締め出せば一致しなくなり、
    58  	// detectLooseRefs が対応記法の外として拾う。
    59  	fromRe      = regexp.MustCompile(`(?m)^(FROM[ \t]+)(?:--platform=\S+[ \t]+)?([^\s'"]+)([ \t]+(?i:AS)[ \t]+\S+)?[ \t]*$`)
    60  	fromLooseRe = regexp.MustCompile(`(?i)^[ \t]*FROM[ \t]+\S`)
    61  	// docker compose service の `image: <ref>`。FROM と同じ prefix/ref/suffix の 3 グループ構成に
    62  	// して rewritePins を共用する。suffix は末尾空白と行末コメントを取り込み保持する。
    63  	// クオートを締め出す理由は fromRe と同じ。
    64  	composeImageRe    = regexp.MustCompile(`(?m)^([ \t]+image:[ \t]+)([^\s'"]+)([ \t]*(?:#.*)?)$`)
    65  	composeImageLoose = regexp.MustCompile(`^[ \t]+image[ \t]*:[ \t]*\S`)
    66  	// `FROM <ref> AS <stage>` の接尾辞から宣言されたステージ名を取り出す。
    67  	fromStageNameRe = regexp.MustCompile(`[ \t]+(?i:AS)[ \t]+(\S+)`)
    68  	// workflow / composite action の `uses: [-] docker://<ref>`。GitHub Actions が registry の
    69  	// image を直接実行するステップの記法で、参照先は GitHub のリポジトリではない。pin-actions は
    70  	// tag を git ls-remote で commit へ解決する機構なので registry には効かず、digest を扱う
    71  	// こちらが持つ（docs/design/security.md）。
    72  	//
    73  	// ref に tag を必須とするのは、省略が :latest を意味するため。tag 無しを通すと parseRef が
    74  	// false を返して固定対象から静かに外れ、可動タグのまま CI で走る。一致しない形は
    75  	// detectLooseRefs が拾って fail-close する。
    76  	usesDockerRe = regexp.MustCompile(
    77  		`(?m)^([ \t]*(?:-[ \t]*)?uses:[ \t]*docker://)((?:[^\s@]+/)?[^\s@/:]+:[^\s@/]+(?:@\S+)?)([ \t]*(?:#.*)?)$`,
    78  	)
    79  	// usesDockerRe の取りこぼしを拾う緩いパターン。引用符付き・flow mapping・tag 無しのいずれも
    80  	// ここへ落ちる。owner/repo 形式の uses: には反応しない（pin-actions の担当）。
    81  	looseDockerUsesRe = regexp.MustCompile(`\buses[ \t]*:[ \t]*["']?docker://`)
    82  	// workflow / composite action の service container の `image: <ref>`。ジョブと同じランナー上で
    83  	// 走り CI の判定に直接効くので、FROM や compose と同じく固定する（docs/design/security.md）。
    84  	//
    85  	// compose の image: と同形だが composeImageRe をそのまま当てられない。workflow の image: は
    86  	// step の with: 配下にも現れ、そちらは ${{ }} で組み立てられて固定のしようがないためである。
    87  	// ref に ${{ }} を含む行は、この正規表現にも下の loose にも一致させない。
    88  	serviceImageRe = regexp.MustCompile(
    89  		`(?m)^([ 	]+image:[ 	]+)((?:[^\s'"$]|\$[^{])+)([ 	]*(?:#.*)?)$`,
    90  	)
    91  	// serviceImageRe の取りこぼしを拾う緩いパターン。${{ }} を含む行だけは固定対象でないので外す。
    92  	serviceImageLoose = regexp.MustCompile(`^[ 	]+image[ 	]*:[ 	]*(?:[^\s$]|\$[^{])`)
    93  	// lockfile 行: "image:tag" = "sha256:..."
    94  	lockRe   = regexp.MustCompile(`^"([^"]+)"\s*=\s*"(sha256:[0-9a-f]+)"`)
    95  	digestRe = regexp.MustCompile(`(?m)^Digest:[ \t]+(sha256:[0-9a-f]+)`)

   150  // run は、サブコマンドを解釈して固定処理へ振り分けます。
   151  // wd は走査の基点となるディレクトリの取得手段で、差し替えられるよう引数で受けます。
   152  func run(args []string, wd func() (string, error)) error {
   153  	if len(args) == 0 {
   154  		return errUsage
   155  	}
   156  
   157  	root, err := wd()
   158  	if err != nil {
   159  		return xerrors.Wrap(err, "getwd")
   160  	}
   161  
   162  	targets, err := targetFiles(root)
   163  	if err != nil {
   164  		return err
   165  	}
   166  
   167  	switch args[0] {
   168  	case "resolve":
   169  		fs := flag.NewFlagSet("resolve", flag.ContinueOnError)
   170  		minAge := fs.Int("min-age-days", 0, "N 日未満の新しすぎる digest は quarantine（0 で無効）")
   171  		if err := fs.Parse(args[1:]); err != nil {
   172  			// ヘルプ要求は失敗ではないので 0 で終える。usage は flag が既に出力している。
   173  			if xerrors.Is(err, flag.ErrHelp) {
   174  				return nil
   175  			}
   176  
   177  			return xerrors.Wrap(err, "failed to parse flags")
   178  		}
   179  
   180  		return resolve(root, targets, *minAge)
   181  	case "apply":
   182  		return applyOrCheck(root, targets, false)
   183  	case "check":
   184  		return applyOrCheck(root, targets, true)
   185  	default:
   186  		return errUsage
   187  	}
   188  }

   190  // targetFiles は走査対象を返す。Dockerfile の FROM、compose の image:、workflow / composite action の
   191  // uses: docker:// と services の image: の 4 種で、registry の image を指す参照は書かれた場所に
   192  // よらずこの機構が固定する（docs/design/security.md）。
   193  //
   194  // workflow は 2 つの target として登録する。target が 1 ファイルにつき 1 つの正規表現しか持たない
   195  // ためで、uses: docker:// と service の image: は書式が違う。同じファイルを 2 度走査するが掴む行は
   196  // 重ならない。uses: owner/repo@<sha> は pin-actions の担当で、ファイル集合そのものは ghfiles が
   197  // 両者へ与える。
   198  func targetFiles(root string) ([]target, error) {
   199  	dockerfiles, err := globFiles(root, "docker/*/Dockerfile")
   200  	if err != nil {
   201  		return nil, err
   202  	}
   203  	compose, err := globFiles(root, "docker-compose*.yaml")
   204  	if err != nil {
   205  		return nil, err
   206  	}
   207  	workflows, err := ghfiles.Collect(root)
   208  	if err != nil {
   209  		return nil, err
   210  	}
   211  	var targets []target
   212  	for _, f := range dockerfiles {
   213  		targets = append(targets, target{
   214  			path: f, re: fromRe, loose: fromLooseRe, exemptTagless: dockerfileExemptTagless,
   215  		})
   216  	}
   217  	for _, f := range compose {
   218  		targets = append(targets, target{path: f, re: composeImageRe, loose: composeImageLoose})
   219  	}
   220  	for _, f := range workflows {
   221  		targets = append(targets, target{path: f, re: usesDockerRe, loose: looseDockerUsesRe})
   222  		targets = append(targets, target{path: f, re: serviceImageRe, loose: serviceImageLoose})
   223  	}
   224  	sort.Slice(targets, func(i, j int) bool { return targets[i].path < targets[j].path })
   225  	return targets, nil
   226  }

   228  // detectLooseRefs は固定対象として解釈できない参照の行番号を返す。
   229  //
   230  // usesDockerRe は行頭にアンカーした 1 行 1 ステップのブロック記法かつ tag 付きにしか一致しない
   231  // が、YAML は同じ内容を flow mapping や引用符でも書けるし、tag を省いた形も書ける。いずれも
   232  // 一致ゼロになり、その状態は「固定漏れ無し」と区別が付かない。緩いパターンで補い、残った行は
   233  // 呼び出し元が fail-close する。
   234  //
   235  // ブロックスカラーの中身は YAML の構造ではなく単なるテキストなので走査から外す。外さないと
   236  // `run:` スクリプトが uses: を含む文字列を出力するだけで検出が誤爆する。
   237  func detectLooseRefs(data string, t target) []int {
   238  	blanked := t.re.ReplaceAllStringFunc(data, func(line string) string {
   239  		return strings.Repeat(" ", len(line))
   240  	})
   241  	// 範囲の判定は潰す前の内容で行う。潰した行は字下げごと空白になり、ブロックの終わりに見える。
   242  	inBlockScalar := yamlblock.ContentLines(data)
   243  	var lines []int
   244  	for i, line := range strings.Split(blanked, "\n") {
   245  		if inBlockScalar[i+1] {
   246  			continue
   247  		}
   248  		if strings.HasPrefix(strings.TrimSpace(line), "#") {
   249  			continue
   250  		}
   251  		if t.loose.MatchString(line) {
   252  			lines = append(lines, i+1)
   253  		}
   254  	}
   255  	lines = append(lines, taglessLines(data, t)...)
   256  	sort.Ints(lines)
   257  
   258  	return lines
   259  }

   261  // validateLoose は全対象を走査し、解釈できない参照行があれば fail-close する。resolve と
   262  // applyOrCheck の両方から呼ぶ——lockfile へ載らない参照を作らないため、走査の前に検査する。
   263  func validateLoose(root string, targets []target) error {
   264  	var found []string
   265  	for _, t := range targets {
   266  		data, err := os.ReadFile(t.path)
   267  		if err != nil {
   268  			return xerrors.Wrap(err, "read "+rel(root, t.path))
   269  		}
   270  		for _, line := range detectLooseRefs(string(data), t) {
   271  			found = append(found, fmt.Sprintf("%s:%d", rel(root, t.path), line))
   272  		}
   273  	}
   274  	if len(found) == 0 {
   275  		return nil
   276  	}
   277  	sort.Strings(found)
   278  
   279  	return xerrors.Wrap(errLooseRef,
   280  		strings.Join(found, ", ")+"（引用符を外し、1 行 1 参照・tag 明示の形へ直してください）")
   281  }

   283  // dockerfileExemptTagless は registry の image を指さない FROM の値を返す。ビルドステージ参照は
   284  // Docker が大文字小文字を区別しないため、比較は小文字へ揃える。
   285  func dockerfileExemptTagless(data string) map[string]bool {
   286  	exempt := map[string]bool{"scratch": true}
   287  	for _, m := range fromRe.FindAllStringSubmatch(data, -1) {
   288  		if stage := fromStageNameRe.FindStringSubmatch(m[3]); stage != nil {
   289  			exempt[strings.ToLower(stage[1])] = true
   290  		}
   291  	}
   292  
   293  	return exempt
   294  }

   296  // taglessLines は厳格なパターンには一致したが tag を持たない参照の行番号を返す。
   297  //
   298  // この形は detectLooseRefs の走査では拾えない——厳格なパターンに一致した時点で空白へ潰され、
   299  // 残らないためである。一方 parseRef は tag が無ければ false を返すので固定対象にも載らず、
   300  // どの報告にも現れないまま :latest が実行される。固定の網から参照が黙って消える向きの穴。
   301  func taglessLines(data string, t target) []int {
   302  	var exempt map[string]bool
   303  	if t.exemptTagless != nil {
   304  		exempt = t.exemptTagless(data)
   305  	}
   306  	var lines []int
   307  	for _, loc := range t.re.FindAllStringSubmatchIndex(data, -1) {
   308  		ref := data[loc[4]:loc[5]]
   309  		if _, ok := parseRef(ref); ok || exempt[strings.ToLower(ref)] {
   310  			continue
   311  		}
   312  		lines = append(lines, strings.Count(data[:loc[0]], "\n")+1)
   313  	}
   314  
   315  	return lines
   316  }

   318  // globFiles は root からの相対パターン pat に一致するパスを返す。
   319  // パターンが不正なら 0 件へ縮退させずエラーを返す。走査対象が静かに空になると、そこに書かれた
   320  // image 参照が検疫・固定・drift 検査のいずれからも外れたまま「全て固定済み」と report される。
   321  func globFiles(root, pat string) ([]string, error) {
   322  	m, err := filepath.Glob(filepath.Join(root, pat))
   323  	if err != nil {
   324  		return nil, xerrors.Wrap(err, "glob "+pat)
   325  	}
   326  	return m, nil
   327  }

   329  // parseRef は FROM / compose image の ref を image:tag へ分解する。第2戻り値が false なら対象外
   330  // （tag 無し＝ビルドステージ参照 / scratch、あるいは registry port を tag と誤認する形）。
   331  func parseRef(ref string) (imageRef, bool) {
   332  	name, _, _ := strings.Cut(ref, "@") // 既存の @digest を捨てる
   333  	image, tag, ok := lastColonSplit(name)
   334  	if !ok || tag == "" || strings.Contains(tag, "/") {
   335  		return imageRef{}, false // tag が無い / 最後の ':' が registry:port だった
   336  	}
   337  	return imageRef{image: image, tag: tag}, true
   338  }

   340  // lastColonSplit は最後の ':' で分割する（tag に ':' は含まれない前提）。
   341  func lastColonSplit(s string) (string, string, bool) {
   342  	i := strings.LastIndex(s, ":")
   343  	if i < 0 {
   344  		return "", "", false
   345  	}
   346  	return s[:i], s[i+1:], true
   347  }

   349  func resolve(root string, targets []target, minAgeDays int) error {
   350  	if err := validateLoose(root, targets); err != nil {
   351  		return err
   352  	}
   353  	keys, err := collectKeys(targets)
   354  	if err != nil {
   355  		return err
   356  	}
   357  	// lockfile 不在（初回）は空マップで続行するが、それ以外の読み込み失敗は握り潰さず fail-close する
   358  	// （既存ピンが lock から脱落して quarantine の退行先が消え、出来立ての未検証 digest を掴むか
   359  	// errNoStepBack で落ちるかの二択になるのを防ぐ。applyOrCheck と対称）。
   360  	existing, err := readLock(filepath.Join(root, lockFile))
   361  	if !isIgnorableLockErr(err) {
   362  		return xerrors.Wrap(err, "read lockfile")
   363  	}
   364  
   365  	ctx := context.Background()
   366  	lock := map[string]string{}
   367  	var notes, skipped []string
   368  	for k, r := range keys {
   369  		digest, err := resolveDigest(ctx, r.key())
   370  		if err != nil {
   371  			return xerrors.Wrap(err, "resolve "+k)
   372  		}
   373  		use, note, err := quarantine(ctx, r, k, digest, minAgeDays, existing)
   374  		if err != nil {
   375  			return xerrors.Wrap(err, "age "+k)
   376  		}
   377  		if note != "" {
   378  			notes = append(notes, note)
   379  		}
   380  		if use == "" {
   381  			skipped = append(skipped, k)
   382  			continue
   383  		}
   384  		lock[k] = use
   385  		log.Printf("  %s -> %s", k, use)
   386  	}
   387  	sort.Strings(notes)
   388  	for _, n := range notes {
   389  		log.Printf("  ⚠️ %s", n)
   390  	}
   391  
   392  	if err := writeLock(filepath.Join(root, lockFile), lock); err != nil {
   393  		return xerrors.Wrap(err, "write lockfile")
   394  	}
   395  	log.Printf("✅ %s に %d 件を書き出しました", lockFile, len(lock))
   396  
   397  	// 退行先の無い出来立て image（fresh + 既存ピン無し）は tag のまま残さず失敗させる。
   398  	// tag のみ運用を許すと未検証 digest を掴むため、check も同状態を弾く（fail-closed）。
   399  	if len(skipped) > 0 {
   400  		sort.Strings(skipped)
   401  
   402  		return xerrors.Wrap(errNoStepBack, fmt.Sprintf(
   403  			"%d 日未満・既存ピン無し。aged 後に再実行するか、緊急時のみ --min-age-days=0 で明示採用してください: %s",
   404  			minAgeDays, strings.Join(skipped, ", "),
   405  		))
   406  	}
   407  
   408  	return nil
   409  }

   427  // quarantine は minAgeDays 未満の新しすぎる digest を採用しない。第1戻り値 "" は skip。
   428  // 経過日数の取得に失敗した場合は err で呼び出し元へ伝播する。
   429  func quarantine(ctx context.Context, r imageRef, key, candidate string, minAgeDays int, existing map[string]string) (string, string, error) {
   430  	if minAgeDays <= 0 {
   431  		return candidate, "", nil
   432  	}
   433  	age, err := digestAgeDays(ctx, r.key())
   434  	if err != nil {
   435  		return "", "", err
   436  	}
   437  	if age >= minAgeDays {
   438  		return candidate, "", nil
   439  	}
   440  	if prev, ok := existing[key]; ok {
   441  		return prev, fmt.Sprintf("%s: 現 digest が %d 日 (<%d) のため既存ピンを維持", key, age, minAgeDays), nil
   442  	}
   443  	return "", fmt.Sprintf("%s: 現 digest が %d 日 (<%d)・既存ピン無しのため tag のまま skip", key, age, minAgeDays), nil
   444  }

   469  func earliestCreated(ctx context.Context, ref string) (time.Time, error) {
   470  	// マルチアーキ (index) は .Image が map、単一アーキは struct。前者→後者でフォールバックする。
   471  	// inspect 自体のエラー（429 等）は握り潰さず伝播し、テンプレート不一致のときだけ次を試す。
   472  	var lastErr error
   473  	for _, tmpl := range []string{
   474  		`{{ range .Image }}{{ println .Created }}{{ end }}`,
   475  		`{{ println .Image.Created }}`,
   476  	} {
   477  		out, err := inspect(ctx, ref, "--format", tmpl)
   478  		if err != nil {
   479  			lastErr = err
   480  			continue
   481  		}
   482  		if t, ok := minCreated(out); ok {
   483  			return t, nil
   484  		}
   485  	}
   486  	if lastErr != nil {
   487  		return time.Time{}, lastErr
   488  	}
   489  	return time.Time{}, xerrors.Wrap(errCreatedUnparsable, ref)
   490  }

   492  // minCreated は inspect 出力の各行（"2006-01-02 15:04:05.9 +0000 UTC"）から最古の時刻を返す。
   493  func minCreated(out string) (time.Time, bool) {
   494  	var earliest time.Time
   495  	found := false
   496  	for line := range strings.SplitSeq(out, "\n") {
   497  		fields := strings.Fields(line)
   498  		if len(fields) < createdTimeCols {
   499  			continue
   500  		}
   501  		// "date time zone" の 3 フィールドだけを RFC 化して parse（末尾 "UTC" 等は捨てる）。
   502  		t, err := time.Parse("2006-01-02 15:04:05.999999999 -0700",
   503  			strings.Join(fields[:createdTimeCols], " "))
   504  		if err != nil {
   505  			continue
   506  		}
   507  		if !found || t.Before(earliest) {
   508  			earliest, found = t, true
   509  		}
   510  	}
   511  	return earliest, found
   512  }

   528  // rewritePins は lock を元に FROM を digest 固定した内容と、lock 未登録の image キー一覧を返す。
   529  // lock に無い image は「未登録」として報告し、行は書き換えない（digest は剥がさない）。
   530  func rewritePins(data string, re *regexp.Regexp, lock map[string]string) (string, []string) {
   531  	var missing []string
   532  	out := re.ReplaceAllStringFunc(data, func(line string) string {
   533  		m := re.FindStringSubmatch(line)
   534  		r, ok := parseRef(m[2])
   535  		if !ok {
   536  			return line
   537  		}
   538  		digest, found := lock[r.key()]
   539  		if !found {
   540  			missing = append(missing, r.key())
   541  			return line
   542  		}
   543  		return m[1] + r.key() + "@" + digest + m[3]
   544  	})
   545  	return out, missing
   546  }

   548  // applyOrCheck は lockfile を SSOT に FROM を digest 固定する。dryRun=true は書き換えず
   549  // 未固定/未登録/drift を非ゼロ終了で報告する。tag のみへ戻す正規化はしない（fail-closed）。
   550  //
   551  // 全ファイルを読み切って未登録の有無を確定させてから書き込む。1 ファイルずつ書きながら進むと、
   552  // 後続ファイルの未登録参照で中断したときに「exit 1 なのに作業ツリーは書き換え済み」という
   553  // 中途半端な状態が残る。
   554  func applyOrCheck(root string, targets []target, dryRun bool) error {
   555  	if err := validateLoose(root, targets); err != nil {
   556  		return err
   557  	}
   558  	lock, err := readLock(filepath.Join(root, lockFile))
   559  	if err != nil {
   560  		return xerrors.Wrap(err, "read lockfile（先に make pin-images-resolve を実行してください）")
   561  	}
   562  	var missing, drifted, pending []string
   563  	rewritten := map[string]string{}
   564  	for _, t := range targets {
   565  		data, err := os.ReadFile(t.path)
   566  		if err != nil {
   567  			return xerrors.Wrap(err, "read "+rel(root, t.path))
   568  		}
   569  		out, miss := rewritePins(string(data), t.re, lock)
   570  		missing = append(missing, miss...)
   571  		if out == string(data) {
   572  			continue
   573  		}
   574  		if dryRun {
   575  			drifted = append(drifted, rel(root, t.path))
   576  			continue
   577  		}
   578  		pending = append(pending, t.path)
   579  		rewritten[t.path] = out
   580  	}
   581  
   582  	if err := validateMissing(missing); err != nil {
   583  		return err
   584  	}
   585  
   586  	for _, path := range pending {
   587  		if err := os.WriteFile(path, []byte(rewritten[path]), filePerm); err != nil {
   588  			return xerrors.Wrap(err, "write "+rel(root, path))
   589  		}
   590  		log.Printf("  updated %s", rel(root, path))
   591  	}
   592  
   593  	return report(drifted, dryRun, len(pending))
   594  }

   642  // isIgnorableLockErr は、lockfile 読み込みエラーのうち無視して続行してよいものを判定する。
   643  // nil（成功）と「ファイル不在」（初回 resolve）のみ true。それ以外（解釈できない行・キー重複・
   644  // 権限エラー等）は fail-close 対象。

   646  	return err == nil || xerrors.Is(err, os.ErrNotExist)

   649  // readLock は lockfile を image:tag→digest として読む。空行とコメント行以外で代入として解釈
   650  // できない行、および既出キーの再定義はエラーにする。読み飛ばしや後勝ちの上書きは、そのエントリが
   651  // 「存在しない」あるいは「行順で決まる」状態を警告なく作り、lockfile が SSOT として機能しなくなる。
   652  func readLock(path string) (map[string]string, error) {
   653  	f, err := os.Open(path) //nolint:gosec // path is constructed from cwd + literal filename
   654  	if err != nil {
   655  		return nil, err
   656  	}
   657  	defer func() { _ = f.Close() }()
   658  	lock := map[string]string{}
   659  	sc := bufio.NewScanner(f)
   660  	for lineNo := 1; sc.Scan(); lineNo++ {
   661  		line := strings.TrimSpace(sc.Text())
   662  		if line == "" || strings.HasPrefix(line, "#") {
   663  			continue
   664  		}
   665  		m := lockRe.FindStringSubmatch(line)
   666  		if m == nil {
   667  			return nil, xerrors.Wrap(errLockInvalidLine,
   668  				fmt.Sprintf("%d 行目: %q（make pin-images-resolve を実行するか該当行を削除してください）", lineNo, line))
   669  		}
   670  		if _, dup := lock[m[1]]; dup {
   671  			return nil, xerrors.Wrap(errLockDuplicateKey,
   672  				fmt.Sprintf("%d 行目: %q（make pin-images-resolve を実行するか重複行を削除してください）", lineNo, m[1]))
   673  		}
   674  		lock[m[1]] = m[2]
   675  	}
   676  	return lock, sc.Err()
   677  }
