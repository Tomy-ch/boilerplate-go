     1  // Package main は pnpm の minimumReleaseAgeExclude に期限を強制する検査ツール。
     2  //
     3  // 兄弟の go-cooldown / tool-cooldown とは守る対象が違う。あちらは窓そのものを強制する本体で、
     4  // go.mod や mise.toml には解決時に窓を効かせる機構が無いために存在する。pnpm はこの repo で
     5  // 唯一、resolver 自身が窓を強制する生態系で、しかも解決時だけでなく install のたびに lockfile
     6  // 全体を照らし直す（--frozen-lockfile の再生も含む）。窓の番人はもう居る。
     7  //
     8  // 番人が居ないのは逃げ道のほうにある。minimumReleaseAgeExclude は特定バージョンだけ窓を外す
     9  // 例外で、追加時は tracked かつ CODEOWNERS 配下のファイルへ書くのでレビューに乗るが、そこから
    10  // 先を見る仕組みが無かった。docs/design/security.md が「A bypass is dated debt」と述べている
    11  // dated を、pnpm でも機械が読めるようにするのがこのツールの仕事にあたる。
    12  //
    13  // 期限の置き場は宣言のコメントではなく .github/pnpm-cooldown-bypass.toml。兄弟 2 つと同じ形に
    14  // したのは、期限が pnpm にとって意味を持たない情報で、pnpm が読む宣言に相乗りさせる必然性が
    15  // 無いため。構造化された TOML を読むことで、YAML のコメントを機械可読な宣言として扱うことに
    16  // 伴う取りこぼしも消える。
    17  //
    18  // 宣言側は yaml.Node として読む。行走査でシーケンスを拾うと、YAML として妥当で pnpm も honor
    19  // する書き方——キーと同じ桁のブロックシーケンス、フロー形式 `[a@1]`、コロン前の空白——を
    20  // 取りこぼし、期限の無い例外が「例外ゼロ」として通る。同じ機構は scripts/actions-shellcheck が
    21  // 先に使っている。
    22  //
    23  // 期限は pnpm-workspace.yaml が変わらなくても訪れるので、pull request の trigger だけでは
    24  // 取りこぼす。兄弟と同じくスケジュール実行にも載せる必要がある。

    62  	errBypassDuplicateKey = xerrors.New("duplicate key in the bypass lockfile")

    64  	// bypassLineRe は `"key" = { expires = ..., issue = ..., reason = "..." }` を読む。
    65  	// 兄弟 2 つと同じ形式にしてある。
    66  	bypassLineRe = regexp.MustCompile(
    67  		`^"([^"]+)"\s*=\s*\{\s*expires\s*=\s*(\d{4}-\d{2}-\d{2})\s*,\s*issue\s*=\s*(\d+)\s*,\s*reason\s*=\s*"([^"]*)"\s*\}$`,
    68  	)

    70  	// `'@scope/name@1.0.0':` の形で現れるため。

   108  // run は root 配下の pnpm-workspace.yaml とバイパス lockfile を突き合わせ、違反があれば
   109  // エラーを返します。
   110  func run(root string, out io.Writer, today time.Time) error {
   111  	files, err := findWorkspaces(root)
   112  	if err != nil {
   113  		return err
   114  	}
   115  
   116  	var exclusions []exclusion
   117  
   118  	for _, f := range files {
   119  		got, parseErr := parseWorkspace(root, f)
   120  		if parseErr != nil {
   121  			return parseErr
   122  		}
   123  
   124  		exclusions = append(exclusions, got...)
   125  	}
   126  
   127  	bypasses, err := parseBypasses(root)
   128  	if err != nil {
   129  		return err
   130  	}
   131  
   132  	violations := validate(root, exclusions, bypasses, today)
   133  
   134  	_, _ = fmt.Fprintf(out, "ℹ️ pnpm-cooldown: workspace %d 件 / 例外 %d 件 / バイパス %d 件 / 上限 %d ヶ月\n",
   135  		len(files), len(exclusions), len(bypasses), maxBypassMonths)
   136  
   137  	if len(violations) == 0 {
   138  		_, _ = fmt.Fprintf(out, "✅ pnpm-cooldown: 違反なし\n")
   139  
   140  		return nil
   141  	}
   142  
   143  	for _, v := range violations {
   144  		_, _ = fmt.Fprintf(out, "❌ %s\n", v)
   145  	}
   146  
   147  	return xerrors.Wrap(errViolations, strconv.Itoa(len(violations))+" 件")
   148  }

   150  // findWorkspaces は root 配下の pnpm-workspace.yaml を集めます。node_modules と vendor は
   151  // 依存が持ち込んだ他人の宣言なので除外します。
   152  func findWorkspaces(root string) ([]string, error) {
   153  	var out []string
   154  
   155  	err := filepath.WalkDir(root, func(path string, d fs.DirEntry, err error) error {
   156  		if err != nil {
   157  			return err
   158  		}
   159  
   160  		if d.IsDir() {
   161  			switch d.Name() {
   162  			case "node_modules", "vendor", ".git":
   163  				return fs.SkipDir
   164  			}
   165  
   166  			return nil
   167  		}
   168  
   169  		if d.Name() != workspaceFile {
   170  			return nil
   171  		}
   172  
   173  		rel, relErr := filepath.Rel(root, path)
   174  		if relErr != nil {
   175  			return relErr
   176  		}
   177  
   178  		out = append(out, filepath.ToSlash(rel))
   179  
   180  		return nil
   181  	})
   182  	if err != nil {
   183  		return nil, xerrors.Wrap(err, "walk")
   184  	}
   185  
   186  	sort.Strings(out)
   187  
   188  	return out, nil
   189  }

   191  // parseWorkspace は 1 つの pnpm-workspace.yaml から例外エントリを読み出します。行走査ではなく
   192  // ノードとして読むのは、YAML が同じ意味をいくつもの書き方で表せるためです。
   193  func parseWorkspace(root, rel string) ([]exclusion, error) {
   194  	raw, err := os.ReadFile(filepath.Join(root, rel)) //nolint:gosec // 走査で見つけた repo 内の宣言ファイル
   195  	if err != nil {
   196  		return nil, xerrors.Wrap(err, rel)
   197  	}
   198  
   199  	var doc yaml.Node
   200  	if err := yaml.Unmarshal(raw, &doc); err != nil {
   201  		return nil, xerrors.Wrap(err, rel)
   202  	}
   203  
   204  	seq := findExcludeSequence(&doc)
   205  	if seq == nil {
   206  		return nil, nil
   207  	}
   208  
   209  	out := make([]exclusion, 0, len(seq.Content))
   210  	for _, item := range seq.Content {
   211  		out = append(out, exclusion{file: rel, line: item.Line, spec: item.Value})
   212  	}
   213  
   214  	return out, nil
   215  }

   217  // findExcludeSequence は文書直下のマッピングから minimumReleaseAgeExclude のシーケンスを
   218  // 探します。見つからない、あるいはシーケンスでない場合は nil を返します。
   219  func findExcludeSequence(doc *yaml.Node) *yaml.Node {
   220  	if len(doc.Content) == 0 {
   221  		return nil
   222  	}
   223  
   224  	root := doc.Content[0]
   225  	if root.Kind != yaml.MappingNode {
   226  		return nil
   227  	}
   228  
   229  	// マッピングの Content はキーと値が交互に並ぶ。
   230  	for i := 0; i+1 < len(root.Content); i += 2 {
   231  		if root.Content[i].Value != excludeKey {
   232  			continue
   233  		}
   234  
   235  		if v := root.Content[i+1]; v.Kind == yaml.SequenceNode {
   236  			return v
   237  		}
   238  
   239  		return nil
   240  	}
   241  
   242  	return nil
   243  }

   245  // parseBypasses はバイパス lockfile を読みます。解釈できない行と重複キーはエラーにします。
   246  // 読み飛ばすと、書き損じた期限が「期限なし」ではなく「無検査」になってしまうためです。
   247  func parseBypasses(root string) (map[string]bypass, error) {
   248  	f, err := os.Open(filepath.Join(root, bypassFile)) //nolint:gosec // repo 内の固定パス
   249  	if err != nil {
   250  		if os.IsNotExist(err) {
   251  			return map[string]bypass{}, nil
   252  		}
   253  
   254  		return nil, xerrors.Wrap(err, bypassFile)
   255  	}
   256  	defer func() { _ = f.Close() }()
   257  
   258  	out := map[string]bypass{}
   259  	lineNo := 0
   260  
   261  	sc := bufio.NewScanner(f)
   262  	for sc.Scan() {
   263  		lineNo++
   264  
   265  		line := strings.TrimSpace(sc.Text())
   266  		if line == "" || strings.HasPrefix(line, "#") {
   267  			continue
   268  		}
   269  
   270  		entry, key, err := parseBypassLine(line, lineNo)
   271  		if err != nil {
   272  			return nil, err
   273  		}
   274  
   275  		if _, dup := out[key]; dup {
   276  			return nil, xerrors.Wrap(errBypassDuplicateKey, fmt.Sprintf("%s:%d %q", bypassFile, lineNo, key))
   277  		}
   278  
   279  		out[key] = entry
   280  	}
   281  
   282  	if err := sc.Err(); err != nil {
   283  		return nil, xerrors.Wrap(err, bypassFile)
   284  	}
   285  
   286  	return out, nil
   287  }

   289  // parseBypassLine は 1 行を 1 エントリへ解釈します。
   290  func parseBypassLine(line string, lineNo int) (bypass, string, error) {
   291  	m := bypassLineRe.FindStringSubmatch(line)
   292  	if m == nil {
   293  		return bypass{}, "", xerrors.Wrap(errBypassInvalidLine, fmt.Sprintf(
   294  			`%s:%d %q（形式は "<name>@<version>" = { expires = YYYY-MM-DD, issue = <N>, reason = "..." }）`,
   295  			bypassFile, lineNo, line,
   296  		))
   297  	}
   298  
   299  	expires, err := time.Parse(time.DateOnly, m[2])
   300  	if err != nil {
   301  		return bypass{}, "", xerrors.Wrap(err, fmt.Sprintf("%s:%d の expires", bypassFile, lineNo))
   302  	}
   303  
   304  	issue, err := strconv.Atoi(m[3])
   305  	if err != nil {
   306  		return bypass{}, "", xerrors.Wrap(err, fmt.Sprintf("%s:%d の issue", bypassFile, lineNo))
   307  	}
   308  
   309  	return bypass{line: lineNo, expires: expires, issue: issue, reason: m[4]}, m[1], nil
   310  }

   312  // validate は例外とバイパスを突き合わせ、違反を集めます。期限切れを失敗にするのは、外したまま
   313  // 放置された例外が恒久 allowlist と区別できなくなるためです。
   314  func validate(root string, exclusions []exclusion, bypasses map[string]bypass, today time.Time) []string {
   315  	limit := today.AddDate(0, maxBypassMonths, 0)
   316  	seen := map[string]struct{}{}
   317  
   318  	var violations []string
   319  
   320  	for _, e := range exclusions {
   321  		seen[e.spec] = struct{}{}
   322  		violations = append(violations, checkExclusion(root, e, bypasses, today, limit)...)
   323  	}
   324  
   325  	for _, key := range sortedKeys(bypasses) {
   326  		if _, ok := seen[key]; ok {
   327  			continue
   328  		}
   329  
   330  		violations = append(violations, fmt.Sprintf(
   331  			"%s:%d %s はどの %s の %s にもありません。不要になったエントリは消してください",
   332  			bypassFile, bypasses[key].line, key, workspaceFile, excludeKey,
   333  		))
   334  	}
   335  
   336  	return violations
   337  }

   339  // checkExclusion は例外 1 件分の違反を返します。期限の違反で打ち切らず残骸も併せて見るのは、
   340  // 片方ずつ直させると往復が増え、期限を延ばした次の実行で初めて残骸だと分かるためです。
   341  func checkExclusion(
   342  	root string, e exclusion, bypasses map[string]bypass, today, limit time.Time,
   343  ) []string {
   344  	at := fmt.Sprintf("%s:%d %s", e.file, e.line, e.spec)
   345  
   346  	b, ok := bypasses[e.spec]
   347  	if !ok {
   348  		return []string{fmt.Sprintf(
   349  			"%s に期限がありません。%s へ "+`"%s" = { expires = YYYY-MM-DD, issue = <N>, reason = "..." }`+" を足してください",
   350  			at, bypassFile, e.spec,
   351  		)}
   352  	}
   353  
   354  	var out []string
   355  
   356  	switch {
   357  	case b.expires.Before(today):
   358  		out = append(out, fmt.Sprintf(
   359  			"%s の期限 %s（%s:%d）が切れています。窓明けの版へ解決し直して例外を消すか、期限を延ばす判断を #%d で記録してください",
   360  			at, b.expires.Format(time.DateOnly), bypassFile, b.line, b.issue,
   361  		))
   362  	case b.expires.After(limit):
   363  		out = append(out, fmt.Sprintf(
   364  			"%s の期限 %s（%s:%d）が上限（%s から %d ヶ月 = %s）を越えています。恒久 allowlist にしないための上限です",
   365  			at, b.expires.Format(time.DateOnly), bypassFile, b.line,
   366  			today.Format(time.DateOnly), maxBypassMonths, limit.Format(time.DateOnly),
   367  		))
   368  	}
   369  
   370  	if msg := checkResolved(root, e); msg != "" {
   371  		out = append(out, msg)
   372  	}
   373  
   374  	return out
   375  }

   377  // checkResolved は、その版を lockfile が実際に解決しているかを検査します。違反があればその
   378  // 文言を、無ければ空文字列を返します。lockfile が読めない場合も違反として報告します。読めない
   379  // ことを「解決済み」に倒すと、lockfile を欠落させるだけで残骸の検知を無効化できるためです。
   380  func checkResolved(root string, e exclusion) string {
   381  	lock := filepath.Join(root, filepath.Dir(e.file), lockFile)
   382  
   383  	specs, err := lockedSpecs(lock)
   384  	if err != nil {
   385  		return fmt.Sprintf("%s:%d %s の照合先 %s が読めません: %v", e.file, e.line, e.spec, lockFile, err)
   386  	}
   387  
   388  	if _, ok := specs[e.spec]; ok {
   389  		return ""
   390  	}
   391  
   392  	return fmt.Sprintf(
   393  		"%s:%d %s は同じディレクトリの %s が解決していません。不要になった例外は消してください",
   394  		e.file, e.line, e.spec, lockFile,
   395  	)
   396  }
   397  
   398  // lockedSpecs は lockfile が解決している <name>@<version> の集合を返します。引用符と peer
   399  // サフィックスを剥がすのは、実物が `'@scope/name@1.0.0':` や `name@1.0.0(peer@2.0.0):` の形で
   400  // 現れるためで、素の完全一致では scoped パッケージが常に未解決に見えてしまいます。
   401  func lockedSpecs(path string) (map[string]struct{}, error) {
   402  	f, err := os.Open(path) //nolint:gosec // 宣言ファイルと同じディレクトリの lockfile
   403  	if err != nil {
   404  		return nil, xerrors.Wrap(err, path)
   405  	}
   406  	defer func() { _ = f.Close() }()
   407  
   408  	out := map[string]struct{}{}
   409  
   410  	sc := bufio.NewScanner(f)
   411  	sc.Buffer(make([]byte, 0, bufio.MaxScanTokenSize), bufio.MaxScanTokenSize)
   412  
   413  	for sc.Scan() {
   414  		m := lockKeyRe.FindStringSubmatch(sc.Text())
   415  		if m == nil {
   416  			continue
   417  		}
   418  
   419  		key := peerSuffixRe.ReplaceAllString(m[1]+m[2]+m[3], "") // 引用符の種類ごとに 1 つだけ埋まる
   420  		if strings.Contains(key, "@") {
   421  			out[key] = struct{}{}
   422  		}
   423  	}
   424  
   425  	if err := sc.Err(); err != nil {
   426  		return nil, xerrors.Wrap(err, path)
   427  	}
   428  
   429  	return out, nil
   430  }
