     2  //
     3  //	-dir:       graphify の出力ディレクトリ（既定 graphify-out）

     5  //	-threshold: この行数以上たまっていたら未処理ありとして報告する（既定 0 = 報告のみ）

     8  // graphify のグラフは 2 つの半分でできている。コードを読む決定的な半分は tree-sitter だけで
     9  // 完結するので無条件に回せるが、ドキュメントを読む意味論の半分はモデルを要するので、変更の
    10  // たびに回すわけにいかない。そこで「どれだけ溜まったか」を測り、まとめて回す判断の材料にする。

    12  // 溜まり具合をファイル数で測らないのは、1 行の誤字修正と ADR の全面改稿が同じ 1 件になって
    13  // しまうためである。代わりに変更行数を累積する。基準点は
    14  // `graphify-out/cache/semantic/` の最終変更コミット — 意味論抽出が走ったときにだけ書き込まれ、
    15  // かつ追跡下にあるので、新しい状態を持ち込まずに「最後に回した地点」を指せる。決定的な半分が

    94  func run(args []string, read func(name string) ([]byte, error), git func(args ...string) (string, error)) error {
    95  	fs := flag.NewFlagSet("graphify-pending", flag.ContinueOnError)
    96  	dir := fs.String("dir", defaultDir, "graphify の出力ディレクトリ")
    97  	ignore := fs.String("ignore", defaultIgnore, "解析対象から外すパスの宣言")
    98  	threshold := fs.Int("threshold", 0, "この行数以上たまっていたら未処理ありとして報告する")
    99  	summary := fs.Bool("github", false, "GitHub Actions の job summary へ書き出す")
   100  
   101  	if err := fs.Parse(args); err != nil {
   102  		if xerrors.Is(err, flag.ErrHelp) {
   103  			return nil
   104  		}
   105  
   106  		return xerrors.Wrap(err, "failed to parse flags")
   107  	}
   108  
   109  	stale, err := staleDocuments(path.Join(*dir, "manifest.json"), *ignore, read)
   110  	if err != nil {
   111  		return err
   112  	}
   113  	if len(stale) == 0 {
   114  		log.Print("graphify: 意味論抽出の未処理はありません")
   115  
   116  		return report(*summary, 0, 0, *threshold, nil)
   117  	}
   118  
   119  	baseline, err := lastSemanticCommit(path.Join(*dir, semanticCache), git)
   120  	if err != nil {
   121  		// 基準が無いのは意味論抽出をまだ一度もコミットしていない状態なので、落とさず件数だけ報告する。
   122  		if xerrors.Is(err, errNoBaseline) {
   123  			log.Printf("graphify: 意味論抽出の未処理 %d ファイル（基準コミットが無いため行数は測れません）", len(stale))
   124  			for _, file := range stale[:min(len(stale), maxListedFiles)] {
   125  				log.Printf("     %s", file)
   126  			}
   127  
   128  			return report(*summary, len(stale), 0, *threshold, nil)
   129  		}
   130  
   131  		return err
   132  	}
   133  
   134  	items, total, err := measure(baseline, stale, git)
   135  	if err != nil {
   136  		return err
   137  	}
   138  
   139  	log.Printf("graphify: 意味論抽出の未処理 %d ファイル / %d 行（基準 %s）", len(items), total, short(baseline))
   140  	for _, item := range items[:min(len(items), maxListedFiles)] {
   141  		log.Printf("     %6d  %s", item.changed, item.path)
   142  	}
   143  	if len(items) > maxListedFiles {
   144  		log.Printf("     ... 他 %d ファイル", len(items)-maxListedFiles)
   145  	}
   146  
   147  	return report(*summary, len(items), total, *threshold, items)
   148  }

   150  // staleDocuments は、manifest 上で意味論抽出が追いついていないドキュメントを返します。
   151  // 判定は保存された semantic_hash といまのファイル内容の MD5 の比較です。manifest 内の
   152  // ast_hash との突き合わせでは、未処理が永久に 0 件に見えます。
   153  func staleDocuments(manifestPath, ignorePath string, read func(name string) ([]byte, error)) ([]string, error) {
   154  	body, err := read(manifestPath)
   155  	if err != nil {
   156  		return nil, xerrors.Wrap(errManifestUnreadable, manifestPath+": "+err.Error())
   157  	}
   158  
   159  	var entries map[string]manifestEntry
   160  	if err := json.Unmarshal(body, &entries); err != nil {
   161  		return nil, xerrors.Wrap(errManifestUnreadable, manifestPath+": "+err.Error())
   162  	}
   163  
   164  	excluded, err := loadIgnore(ignorePath, read)
   165  	if err != nil {
   166  		return nil, err
   167  	}
   168  
   169  	var stale []string
   170  	for file, entry := range entries {
   171  		if !isDocument(file) || excluded(file) {
   172  			continue
   173  		}
   174  		body, err := read(file)
   175  		if err != nil {
   176  			// manifest に載っているのに読めないのは、そのファイルが消えた場合。抽出し直す対象では
   177  			// ないので数えません。
   178  			continue
   179  		}
   180  		// nosemgrep: go.lang.security.audit.crypto.use_of_weak_crypto.use-of-md5
   181  		if entry.SemanticHash != fmt.Sprintf("%x", md5.Sum(body)) { //nolint:gosec // 変更検出のみで、graphify の manifest と同じ選択 // DevSkim: ignore DS126858
   182  			stale = append(stale, file)
   183  		}
   184  	}
   185  	sort.Strings(stale)
   186  
   187  	return stale, nil
   188  }

   190  // loadIgnore は .graphifyignore を読んで、除外されるかを判定する述語を返します。
   191  // 宣言が無いことは異常ではない（除外なしの構成）ため、その場合は何も除外しません。
   192  func loadIgnore(ignorePath string, read func(name string) ([]byte, error)) (func(string) bool, error) {
   193  	body, err := read(ignorePath)
   194  	if err != nil {
   195  		if xerrors.Is(err, os.ErrNotExist) {
   196  			return func(string) bool { return false }, nil
   197  		}
   198  
   199  		return nil, xerrors.Wrap(err, "❌ "+ignorePath+" を読み込めません")
   200  	}
   201  
   202  	var patterns []string
   203  	scanner := bufio.NewScanner(bytes.NewReader(body))
   204  	for scanner.Scan() {
   205  		line := strings.TrimSpace(scanner.Text())
   206  		if line != "" && !strings.HasPrefix(line, "#") {
   207  			patterns = append(patterns, line)
   208  		}
   209  	}
   210  	if err := scanner.Err(); err != nil {
   211  		return nil, xerrors.Wrap(err, "❌ "+ignorePath+" を読めません")
   212  	}
   213  
   214  	return func(file string) bool {
   215  		for _, pattern := range patterns {
   216  			if matchesIgnore(file, pattern) {
   217  				return true
   218  			}
   219  		}
   220  
   221  		return false
   222  	}, nil
   223  }

   225  // matchesIgnore は、gitignore 風の 1 パターンがリポジトリ相対パスに当たるかを判定します。
   226  // 汎用の gitignore 実装ではなく、`.graphifyignore` が実際に使う形だけを扱います。
   227  //
   228  //	docs/godoc/     ルートからのディレクトリ
   229  //	**/gen/         どの階層にあってもよいディレクトリ
   230  //	*.ja.md         ファイル名のパターン
   231  //	**/*.gen.go     どの階層にあってもよいファイル名のパターン
   232  //	openapi/x.yaml  素のパス
   233  //
   234  // 解釈が graphify 本体と一致する保証は無く、食い違いは閾値が早く鳴る向きにだけ倒れます。
   235  func matchesIgnore(file, pattern string) bool {
   236  	pattern = strings.TrimPrefix(pattern, "./")
   237  	pattern, anywhere := strings.CutPrefix(pattern, "**/")
   238  
   239  	if directory, isDirectory := strings.CutSuffix(pattern, "/"); isDirectory {
   240  		if anywhere {
   241  			return slices.Contains(strings.Split(path.Dir(file), "/"), directory)
   242  		}
   243  
   244  		return strings.HasPrefix(file, directory+"/")
   245  	}
   246  
   247  	if strings.HasPrefix(pattern, "*") {
   248  		return matchesGlob(path.Base(file), pattern)
   249  	}
   250  	if anywhere {
   251  		return path.Base(file) == pattern
   252  	}
   253  
   254  	return file == pattern || strings.HasPrefix(file, pattern+"/")
   255  }

   257  // matchesGlob は `*.ja.md` のような、先頭の `*` 一つだけを持つパターンを判定します。
   258  func matchesGlob(name, pattern string) bool {
   259  	suffix, ok := strings.CutPrefix(pattern, "*")
   260  	if !ok {
   261  		return name == pattern
   262  	}
   263  
   264  	return strings.HasSuffix(name, suffix)
   265  }

   274  // lastSemanticCommit は、意味論キャッシュを最後に変更したコミットを返します。
   275  func lastSemanticCommit(cachePath string, git func(args ...string) (string, error)) (string, error) {
   276  	out, err := git("log", "-1", "--format=%H", "--", cachePath)

   291  func measure(baseline string, files []string, git func(args ...string) (string, error)) ([]pending, int, error) {
   292  	args := append([]string{"diff", "--numstat", baseline + "..HEAD", "--"}, files...)
   293  	out, err := git(args...)
   294  	if err != nil {
   295  		return nil, 0, xerrors.Wrap(err, "❌ "+short(baseline)+" からの差分を測れません")
   296  	}
   297  
   298  	return parseNumstat(out)
   299  }

   301  // parseNumstat は `git diff --numstat` の出力を積み上げます。バイナリ扱いの行は `-` になるため
   302  // 行数としては 0 と数えますが、ファイル自体は未処理として残します。
   303  func parseNumstat(out string) ([]pending, int, error) {
   304  	var items []pending
   305  	total := 0
   306  	for line := range strings.SplitSeq(strings.TrimSpace(out), "\n") {
   307  		if line == "" {
   308  			continue
   309  		}
   310  		fields := strings.SplitN(line, "\t", numstatFields)
   311  		if len(fields) != numstatFields {
   312  			return nil, 0, xerrors.Wrap(errManifestUnreadable, "numstat の行を解釈できません: "+strconv.Quote(line))
   313  		}
   314  		changed := atoiOrZero(fields[0]) + atoiOrZero(fields[1])
   315  		items = append(items, pending{path: fields[2], changed: changed})
   316  		total += changed
   317  	}
   318  	sort.Slice(items, func(i, j int) bool {
   319  		if items[i].changed != items[j].changed {
   320  			return items[i].changed > items[j].changed
   321  		}
   322  
   323  		return items[i].path < items[j].path
   324  	})
   325  
   326  	return items, total, nil
   327  }

   339  // report は結果を job summary と、後続のステップが読む出力へ書き出します。閾値は判断の材料で
   340  // あって門ではないので、超過しても終了コードは 0 のままにします。

   342  	if !summary {
   343  		return nil
   344  	}
   345

   347  		return err

   349  
   350  	destination := os.Getenv("GITHUB_STEP_SUMMARY")
   351  	if destination == "" {
   352  		return nil
   353  	}
   354  
   355  	var body strings.Builder
   356  	body.WriteString("## graphify semantic extraction\n\n")
   357  	if total >= threshold && threshold > 0 {
   358  		fmt.Fprintf(&body,
   359  			"⚠️ %d files / %d changed lines are waiting (threshold %d). Run `/graphify --update`.\n\n",
   360  			files, total, threshold)
   361  	} else {
   362  		fmt.Fprintf(&body, "%d files / %d changed lines are waiting (threshold %d).\n\n", files, total, threshold)
   363  	}
   364  	for _, item := range items[:min(len(items), maxListedFiles)] {
   365  		fmt.Fprintf(&body, "- `%s` (%d)\n", item.path, item.changed)
   366  	}
   367  	if len(items) > maxListedFiles {
   368  		fmt.Fprintf(&body, "- ... %d more\n", len(items)-maxListedFiles)
   369  	}
   370  
   371  	file, err := os.OpenFile(destination, os.O_APPEND|os.O_WRONLY|os.O_CREATE, summaryFileMode) //nolint:gosec // 宛先は Actions が渡す固定の環境変数
   372  	if err != nil {
   373  		return xerrors.Wrap(err, "❌ job summary を開けません")
   374  	}
   375  	defer func() { _ = file.Close() }()
   376  	if _, err := file.WriteString(body.String()); err != nil {
   377  		return xerrors.Wrap(err, "❌ job summary へ書けません")
   378  	}
   379  
   380  	return nil
   381  }
