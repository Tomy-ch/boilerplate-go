type ControlEvent ¶

ControlEvent レスポンス確定後にサーバから client へ送る制御指示（SSE の `event: control`）。SSE `id` を持たないため `Last-Event-ID` を汚しません。 client は `action` だけで分岐し、`reason` は記録用の安定値として扱います。`STOP` / `REAUTHENTICATE` / `RESYNC` は サーバの切断を待たず client 側で接続を閉じてから遷移します。 

type ControlEvent struct {
    // Action client がとる動作。`RECONNECT` は通常の再接続、`RETRY_LATER` は `retryAfterMs` 後の再接続、
    // `REAUTHENTICATE` は新しい ticket を取得しての再接続、`RESYNC` は正本の再取得後の再接続、`STOP` は再接続しない。
    //
    //
    // Example: RECONNECT
    Action ControlEventAction `json:"action"`

    // Reason 指示の理由を表す安定した値。人が読む文言は含みません。
    //
    // Example: SERVER_DRAINING
    Reason ControlEventReason `json:"reason"`

    // RetryAfterMs 再接続までの待ち時間（ミリ秒）。目安であり、client は jitter を加えてよい。レスポンス確定前の `Retry-After` ヘッダは秒単位です。
    //
    // Example: 5000
    RetryAfterMs *int `json:"retryAfterMs,omitempty"`
}

type DeliveryEvent ¶

DeliveryEvent SSE で配送される business event の封筒。feature に依らず同じ形で、`payload` の形は `type` ごとに feature が定めます。 SSE の `id` には `sequence` がそのまま入ります。 

type DeliveryEvent struct {
    // EventId event の一意な識別子。同じ event が再送されても値は変わりません。
    //
    // Example: 123e4567-e89b-12d3-a456-426614174000
    EventId openapi_types.UUID `json:"eventId"`

    // OccurredAt event が発生した時刻（UTC）。
    //
    // Example: 2026-08-15T12:00:00Z
    OccurredAt time.Time `json:"occurredAt"`

    // Payload event の本体。形は `type` ごとに feature が定めます。
    //
    // Example: {"messageId":"123e4567-e89b-12d3-a456-426614174000"}
    Payload map[string]interface{} `json:"payload"`

    // SchemaVersion payload の schema 世代。`type` の `.vN` と同じ値を保ちます。
    //
    // Example: 1
    SchemaVersion int `json:"schemaVersion"`

    // Sequence stream 内の位置（stream-local な単調増加 sequence）。JavaScript の数値精度に依存しないよう 10 進文字列で表します。
    // business event の SSE `id` と同じ値で、`0` は stream の先頭を意味します。
    //
    //
    // Example: 1001
    Sequence StreamCursor `json:"sequence"`

    // StreamId この event が属する stream の識別子。接続先の destination と一致します。
    //
    // Example: resource-8f3c
    StreamId string `json:"streamId"`

    // Type feature が定める event 種別。末尾の `.vN` で契約の世代を表します。
    //
    // Example: resource.updated.v1
    Type string `json:"type"`
}
