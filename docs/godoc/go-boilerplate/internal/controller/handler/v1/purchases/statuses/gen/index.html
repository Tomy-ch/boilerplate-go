type PurchaseStatusRef ¶

PurchaseStatusRef 購入に紐づくステータス（ID・業務キー・名称）。name は事前解決済みで、別途の名称解決 API 呼び出しは不要です。 表示には name を、分岐にはドメインの業務キーである code を用います。 

type PurchaseStatusRef struct {
    // Code 購入ステータスの業務キー。値は到達順序を意味しません（完了より支払い済みのほうが大きい）。
    //
    // Example: 7
    Code int64 `json:"code"`

    // Id 購入ステータスID
    //
    // Example: 0190b0d4-7b1a-7c2e-9f3a-1b2c3d4e5f60
    Id openapi_types.UUID `json:"id"`

    // Name 購入ステータス名
    //
    // Example: 完了
    Name string `json:"name"`
}
