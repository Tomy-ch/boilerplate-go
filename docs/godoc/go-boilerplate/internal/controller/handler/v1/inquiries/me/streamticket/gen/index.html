type InquiryStreamTicketResponse ¶

InquiryStreamTicketResponse 購読に使う 1 回限りの ticket。生値はこの応答にしか現れず、サーバは hash しか保存しない。 接続は GET /v1/streams/{destination} に streamId を渡して開始する。 

type InquiryStreamTicketResponse struct {
    // ExpiresAt この ticket で新しい接続を始められる期限。確立済みの接続は別に最大接続時間で区切られる。
    //
    // Example: 2026-09-01T12:39:56Z
    ExpiresAt time.Time `json:"expiresAt"`

    // StreamId 接続先の stream。GET /v1/streams/{destination} の destination に渡す。
    //
    // Example: 0198a1b2-c3d4-7e5f-8a9b-0c1d2e3f4a60
    StreamId string `json:"streamId"`

    // Ticket ticket の生値。log や URL に残さないこと。
    //
    // Example: 0oXk1c5t8Yb1yq3aB7pQwR2sT4uV6wX8
    Ticket string `json:"ticket"`
}

type RegisterHandlersOptions struct {
    // BaseURL is prepended to every registered path so the API can be served
    // under a prefix.
    BaseURL string
    // OperationMiddlewares lets the caller attach per-operation middleware at
    // registration time. The map key is the OpenAPI `operationId` value as it
    // appears in the spec (the raw, un-normalized form). Operations that have
    // no entry are registered with no extra middleware. A nil map disables
    // per-operation middleware entirely.
    OperationMiddlewares map[string][]echo.MiddlewareFunc
}
